  77→def _format_sandbox_result(result: dict) -> str:
  78→    """将沙箱执行结果封装为结构化文本（stdout/stderr/exitCode/超时）。"""
  79→    parts: list[str] = []
  80→    if result.get("timedOut"):
  81→        parts.append(result.get("stderr", ""))
  82→    else:
  83→        if result.get("stdout"):
  84→            parts.append(f"exitCode: {result.get('exitCode')}\nstdout:\n{result['stdout']}")
  85→        if result.get("stderr"):
  86→            parts.append(f"stderr:\n{result['stderr']}")
  87→    if result.get("truncated", {}).get("stdout") or result.get("truncated", {}).get("stderr"):
  88→        parts.append("（输出已按长度上限截断）")
  89→    if not parts:
  90→        parts.append(f"exitCode: {result.get('exitCode')}（无输出）")
  91→    return "\n".join(parts)

 149→    async def _async_batch_process(image_urls: list[str], model: str) -> str:
 150→        """大批量异步处理：提交后台任务 → 进入 async_wait 中断，任务完成回调自动 resume。
 151→
 152→        中断点保存 async_wait 数据（taskId/taskType/预计耗时）供断线恢复与回调反查；
 153→        resume 时 interrupt() 返回任务结果摘要，据此更新任务状态并返回给 LLM。
 154→        """
 155→        ctx = current_run_ctx(template)
 156→        thread_id = f"{ctx['conversation_id']}:{ctx['message_id']}"
 157→        task_id = submit_batch_task(
 158→            conv_id=ctx["conversation_id"],
 159→            msg_id=ctx["message_id"],
 160→            user_id=ctx["user_id"],
 161→            image_urls=image_urls,
 162→            model=model,
 163→            stream_session_id=ctx["stream_session_id"],
 164→            thread_id=thread_id,
 165→        )
 166→        # assistant 消息 task_id 写异步任务 ID（供前端进度追踪/回调反查）
 167→        try:
 168→            async with get_db_session() as db:
 169→                from app.repository.ai_message_repository import ai_message_repository
 170→
 171→                await ai_message_repository.update_task_id(db, ctx["message_id"], task_id)
 172→                await db.commit()
 173→        except Exception:
 174→            logger.warning("异步任务 ID 落库失败: task_id=%s", task_id, exc_info=True)
 175→        interrupt_data = {
 176→            "type": "async_wait",
 177→            "stream_session_id": ctx["stream_session_id"],
 178→            "data": {
 179→                "taskId": task_id,
 180→                "taskType": "batch_process",
 181→                "estDuration": f"约 {len(image_urls) * 5} 秒",  # 粗估，供前端进度展示
 182→                "imageCount": len(image_urls),
 183→            },
 184→        }
 185→        await interrupt_handler.save_interrupt(thread_id, "async_wait", interrupt_data)
 186→        # 暂停图等待异步任务完成；resume 时 interrupt() 返回 {async_task: summary}
 187→        resume = interrupt(interrupt_data)
 188→        summary = (resume or {}).get("async_task") or {}
 189→        ctx.update(
 190→            {
 191→                "task_type": "dehaze",
 192→                "task_model": str(model),
 193→                "task_params": {"image_urls": image_urls},
 194→                "task_status": "completed" if summary.get("failed") == 0 else "failed",
 195→                "task_id": task_id,
 196→                "task_artifacts": [
 197→                    {
 198→                        "id": r["pred_log_id"],
 199→                        "type": "image_result",
 200→                        "summary": {"modelId": model},
 201→                    }
 202→                    for r in summary.get("results", [])
 203→                    if r.get("pred_log_id")
 204→                ],
 205→            }
 206→        )
 207→        total = summary.get("total", len(image_urls))
 208→        success = summary.get("success", 0)
 209→        failed = summary.get("failed", 0)
 210→        return f"批量处理完成：共 {total} 张，成功 {success} 张，失败 {failed} 张"

 281→        """网络搜索：配额超限或服务不可用时自动降级为知识库检索（§5.1）。"""

 305→    async def _execute_code(code: str, language: str = "python", timeout: int = 60) -> str:
 306→        """受限沙箱执行 Python 脚本或 Shell 命令；高风险能力需用户确认（§2.2）。
 307→
 308→        Shell 一律确认；Python 仅在命中网络访问/子进程/沙箱外文件读写时确认——
 309→        这两类能力与 Shell 同级，否则 python 模式可绕过确认访问宿主网络与文件。
 310→        """
 311→        language = (language or "python").lower()
 312→        if language == "shell":
 313→            rejection = code_sandbox.check_command_policy(code)
 314→            if rejection:
 315→                return rejection
 316→            action = "execute_shell_command"
 317→            impact = "将在受限沙箱中执行该 Shell 命令，请确认命令内容与影响范围。"
 318→        else:
 319→            risk = code_sandbox.check_python_risk(code) if language == "python" else None
 320→            if risk is None:
 321→                result = await code_sandbox.execute_code(code, language, timeout)
 322→                return _format_sandbox_result(result)
 323→            action = "execute_python_code"
 324→            impact = f"该代码包含{risk}（与 Shell 同级的高风险操作），请确认影响范围。"
 325→
 326→        ctx = current_run_ctx(template)
 327→        interrupt_data = {
 328→            "type": "confirm",
 329→            "stream_session_id": ctx.get("stream_session_id"),
 330→            "data": {
 331→                "confirmKind": ConfirmKind.DANGEROUS_OP,
 332→                "action": action,
 333→                "command": code,
 334→                "impact": impact,
 335→            },
 336→        }
 337→        # 危险操作确认与算法推荐一致：持久化中断点，否则 resume 端点查不到中断、用户无法确认
 338→        await interrupt_handler.save_interrupt(
 339→            f"{ctx['conversation_id']}:{ctx['message_id']}", "confirm", interrupt_data
 340→        )
 341→        resume = interrupt(interrupt_data)
 342→        if isinstance(resume, dict) and resume.get("confirmed") is False:
 343→            if language == "shell":
 344→                return "用户拒绝了该 Shell 命令的执行"
 345→            return "用户拒绝了该代码的执行"
 346→        result = await code_sandbox.execute_code(code, language, timeout)
 347→        return _format_sandbox_result(result)
