 285→    async def _handle_messages(self, data: Any) -> None:
 286→        """messages 事件：data 形如 [AIMessageChunk, metadata]，提取文本/工具调用增量。
 287→
 288→        文本内容块在首个 text delta 前推送 content_block.start；工具调用增量同时
 289→        累积到 _tool_calls（首个 chunk 携带 id/name，后续 chunk 仅携带 args 增量），
 290→        ToolMessage 返回时按 tool_call_id 匹配还原完整入参。
 291→        """

 293→            return

 306→                    {

 310→                )
