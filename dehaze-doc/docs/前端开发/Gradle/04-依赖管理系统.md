# 第4章：依赖管理系统

## 4.1 依赖配置基础

### 4.1.1 依赖声明语法

**基础依赖声明：**

Gradle的依赖声明语法相比Maven更加灵活和强大。与Maven的`<dependency>`标签相比，Gradle使用编程式的DSL语法，支持更复杂的依赖配置。

```groovy
// build.gradle - 基础依赖声明语法

// 1. 基本语法格式
dependencies {
    // configuration 'group:name:version:classifier@extension'
    implementation 'org.springframework.boot:spring-boot-starter-web:3.2.0'

    // 支持简写（如果group:name唯一）
    implementation 'spring-boot-starter-web'  // 需要在BOM中定义版本

    // 带分类器的依赖
    implementation 'org.springframework.boot:spring-boot-starter-web:3.2.0:jar'

    // 带扩展名的依赖
    implementation 'com.example:custom-library:1.0.0@zip'
}

// 2. 常用依赖配置类型
dependencies {
    // implementation - 编译时需要，运行时需要，不传递给依赖方
    implementation 'org.springframework.boot:spring-boot-starter-web'

    // api - 编译时需要，运行时需要，会传递给依赖方
    api 'com.google.guava:guava:32.1.3-jre'

    // compileOnly - 仅编译时需要
    compileOnly 'javax.servlet:javax.servlet-api:4.0.1'

    // runtimeOnly - 仅运行时需要
    runtimeOnly 'mysql:mysql-connector-java:8.2.0'

    // testImplementation - 测试编译时需要
    testImplementation 'org.junit.jupiter:junit-jupiter:5.10.0'

    // testRuntimeOnly - 测试运行时需要
    testRuntimeOnly 'org.junit.platform:junit-platform-launcher'

    // annotationProcessor - 注解处理器
    annotationProcessor 'org.projectlombok:lombok:1.18.30'

    // developmentOnly - 开发时需要，生产环境不需要
    developmentOnly 'org.springframework.boot:spring-boot-devtools'
}

// 3. 复杂依赖配置示例
dependencies {
    // 带条件判断的依赖
    if (project.hasProperty('enableActuator')) {
        implementation 'org.springframework.boot:spring-boot-starter-actuator'
    }

    // 基于Java版本的依赖
    if (JavaVersion.current().isCompatibleWith(JavaVersion.VERSION_17)) {
        implementation 'org.example:java17-feature:1.0.0'
    }

    // 可选依赖
    implementation('org.example:optional-library:1.0.0') {
        changing = true  // 标记为可变版本
    }
}
```

**依赖声明的高级特性：**

```groovy
// build.gradle - 高级依赖声明特性

// 1. 闭包配置依赖
dependencies {
    implementation('org.springframework.boot:spring-boot-starter-web') {
        // 强制特定版本
        force = true

        // 排除传递依赖
        exclude group: 'org.springframework.boot', module: 'spring-boot-starter-tomcat'

        // 设置transitive属性
        transitive = true

        // 设置changing属性（SNAPSHOT版本）
        changing = false

        // 自定义属性
        attributes {
            attribute(Attribute.of('org.springframework.boot', String), 'web')
        }
    }
}

// 2. 文件依赖
dependencies {
    // 依赖本地文件
    implementation files('libs/custom-library.jar')

    // 依赖文件集合
    implementation fileTree('libs') {
        include '*.jar'
        exclude 'test-*.jar'
    }

    // 依赖目录
    implementation fileTree(dir: 'libs', include: ['*.jar'])
}

// 3. 项目依赖（多模块项目）
dependencies {
    // 依赖同项目的其他模块
    implementation project(':common:utils')
    implementation project(':infrastructure:persistence')

    // 依赖项目并指定配置
    testImplementation project(path: ':common:test-utils', configuration: 'testArtifacts')

    // 条件化项目依赖
    if (project.hasProperty('includeApiModule')) {
        implementation project(':application:api')
    }
}

// 4. 仓库特定依赖
dependencies {
    // 从特定仓库获取依赖
    implementation('com.company:internal-library:1.0.0') {
        artifact {
            name = 'internal-library'
            type = 'jar'
            extension = 'jar'
            classifier = ''
        }
    }
}

// 5. 动态版本依赖
dependencies {
    // 最新版本（不推荐生产使用）
    implementation 'org.example:library:latest.release'

    // 最新集成版本
    implementation 'org.example:library:latest.integration'

    // 版本范围（谨慎使用）
    implementation 'org.example:library:[1.0.0,2.0.0)'

    // +号语法（1.0.+ 表示 1.0.0 <= version < 1.1.0）
    implementation 'org.example:library:1.0.+'
}
```

### 4.1.2 依赖配置类型

**Java插件提供的依赖配置：**

```groovy
// build.gradle - Java插件依赖配置详解

// 应用Java插件
plugins {
    id 'java'
    id 'java-library'  // 提供额外的API配置
}

// Java插件提供的依赖配置
dependencies {
    // === 编译相关配置 ===

    // compileOnly - 仅编译时需要，不会出现在运行时classpath中
    compileOnly 'javax.annotation:javax.annotation-api:1.3.2'

    // implementation - 编译和运行时需要，不暴露给依赖方
    implementation 'org.slf4j:slf4j-api:2.0.9'

    // api - 编译和运行时需要，会暴露给依赖方（Java Library插件）
    api 'com.google.guava:guava:32.1.3-jre'

    // compileOnlyApi - 仅编译时需要，但会暴露给依赖方（Java Library插件）
    compileOnlyApi 'org.springframework.boot:spring-boot-autoconfigure:3.2.0'

    // === 运行时相关配置 ===

    // runtimeOnly - 仅运行时需要，编译时不需要
    runtimeOnly 'ch.qos.logback:logback-classic:1.4.11'

    // runtimeClasspath - runtimeOnly + implementation（只读配置）
    // runtimeClasspath 'com.example:runtime-dep:1.0.0'  // 不能直接配置

    // === 测试相关配置 ===

    // testCompileOnly - 测试编译时需要
    testCompileOnly 'javax.servlet:javax.servlet-api:4.0.1'

    // testImplementation - 测试编译和运行时需要
    testImplementation 'org.junit.jupiter:junit-jupiter:5.10.0'

    // testRuntimeOnly - 测试运行时需要
    testRuntimeOnly 'org.junit.platform:junit-platform-launcher:1.10.0'

    // testApi - 测试API依赖（Java Library插件）
    testApi 'org.mockito:mockito-core:5.6.0'

    // === 注解处理器相关配置 ===

    // annotationProcessor - 主代码的注解处理器
    annotationProcessor 'org.projectlombok:lombok:1.18.30'
    annotationProcessor 'org.mapstruct:mapstruct-processor:1.5.5.Final'

    // testAnnotationProcessor - 测试代码的注解处理器
    testAnnotationProcessor 'org.projectlombok:lombok:1.18.30'

    // === 其他配置 ===

    // archives - 归档配置（默认）
    // archives 'com.example:library:1.0.0'  // 由archives任务自动配置

    // default - 默认配置
    // default 'com.example:library:1.0.0'  // 包含archives和runtime

    // compile - 编译配置（已废弃，使用implementation替代）
    // compile 'com.example:library:1.0.0'  // 不推荐使用

    // runtime - 运行时配置（已废弃，使用runtimeOnly替代）
    // runtime 'com.example:library:1.0.0'  // 不推荐使用

    // testCompile - 测试编译配置（已废弃）
    // testCompile 'com.example:library:1.0.0'  // 不推荐使用

    // testRuntime - 测试运行时配置（已废弃）
    // testRuntime 'com.example:library:1.0.0'  // 不推荐使用
}

// 自定义依赖配置
configurations {
    // 创建自定义配置
    provided

    // 配置继承关系
    compileClasspath.extendsFrom provided

    // 配置属性
    implementation {
        // 可以解析
        canBeResolved = true

        // 可以消费
        canBeConsumed = true

        // 描述
        description = 'Implementation dependencies'
    }

    // 依赖排除配置
    testImplementation {
        exclude group: 'org.junit.vintage', module: 'junit-vintage-engine'
    }
}

// 使用自定义配置
dependencies {
    provided 'javax.servlet:javax.servlet-api:4.0.1'
}
```

### 4.1.3 依赖范围与传递性

**依赖传递性详解：**

```groovy
// build.gradle - 依赖传递性详解

// 1. 传递性依赖示例
dependencies {
    // 添加Spring Boot Web Starter
    implementation 'org.springframework.boot:spring-boot-starter-web:3.2.0'

    // 这会自动传递引入以下依赖：
    // - org.springframework.boot:spring-boot-starter
    // - org.springframework.boot:spring-boot-starter-json
    // - org.springframework.boot:spring-boot-starter-tomcat
    // - org.springframework:spring-web
    // - org.springframework:spring-webmvc
    // - com.fasterxml.jackson.core:jackson-databind
    // - com.fasterxml.jackson.datatype:jackson-datatype-jsr310
    // - 等等...
}

// 2. 不同配置的传递性
dependencies {
    // implementation - 传递给运行时，不传递给编译时
    implementation 'org.springframework.boot:spring-boot-starter-web'

    // api - 传递给编译时和运行时
    api 'com.google.guava:guava:32.1.3-jre'

    // compileOnly - 不传递
    compileOnly 'javax.servlet:javax.servlet-api:4.0.1'

    // runtimeOnly - 传递给运行时
    runtimeOnly 'mysql:mysql-connector-java:8.2.0'

    // testImplementation - 不传递
    testImplementation 'org.junit.jupiter:junit-jupiter:5.10.0'
}

// 3. 依赖传递性控制
dependencies {
    // 禁用传递依赖
    implementation('org.example:library:1.0.0') {
        transitive = false
    }

    // 选择性排除传递依赖
    implementation('org.springframework.boot:spring-boot-starter-web') {
        exclude group: 'org.springframework.boot', module: 'spring-boot-starter-tomcat'
        exclude module: 'spring-webmvc'  // 可以省略group
    }

    // 使用BOM管理传递依赖版本
    implementation platform('org.springframework.boot:spring-boot-dependencies:3.2.0')
}

// 4. 查看依赖传递性
task showTransitiveDependencies {
    doLast {
        println "=== Implementation Dependencies ==="
        configurations.implementation.resolvedConfiguration.firstLevelModuleDependencies.each { dep ->
            println "${dep.moduleGroup}:${dep.moduleName}:${dep.moduleVersion}"
            dep.children.each { child ->
                println "  └── ${child.moduleGroup}:${child.moduleName}:${child.moduleVersion}"
            }
        }
    }
}

// 5. 依赖范围对比表
task dependencyScopeComparison {
    doLast {
        println """
        === 依赖范围对比表 ===

        配置类型         编译时   运行时   传递性   用途
        ==============================================
        implementation     ✓      ✓      部分     主要实现依赖
        api               ✓      ✓      ✓       公共API依赖
        compileOnly       ✓      ✗      ✗       仅编译时依赖
        runtimeOnly       ✗      ✓      ✓       仅运行时依赖
        testImplementation ✓      ✓      ✗       测试实现依赖
        testCompileOnly   ✓      ✗      ✗       测试编译依赖
        testRuntimeOnly   ✗      ✓      ✗       测试运行依赖
        annotationProcessor ✓     ✗      ✗       注解处理器
        developmentOnly   ✗      ✓      部分     开发时依赖

        说明：
        - ✓ 表示包含
        - ✗ 表示不包含
        - 部分表示在特定情况下包含
        """
    }
}
```

**依赖冲突解决策略：**

```groovy
// build.gradle - 依赖冲突解决

dependencies {
    // Gradle默认使用最新版本策略解决冲突
    // 例如：如果依赖A要求guava:28.0，依赖B要求guava:30.0，会选择30.0

    // 1. 强制指定版本
    configurations.all {
        resolutionStrategy {
            force 'com.google.guava:guava:32.1.3-jre'
            force 'org.slf4j:slf4j-api:2.0.9'
        }
    }

    // 2. 依赖替换
    configurations.all {
        resolutionStrategy {
            dependencySubstitution {
                // 用slf4j替换commons-logging
                substitute module('commons-logging:commons-logging') with module('org.slf4j:jcl-over-slf4j')

                // 用项目依赖替换外部依赖
                substitute project(':common:utils') with module('com.example:common-utils:1.0.0')
            }
        }
    }

    // 3. 版本选择策略
    configurations.all {
        resolutionStrategy {
            // 选择最高版本
            failOnVersionConflict()

            // 或者自定义选择策略
            eachDependency { DependencyResolveDetails details ->
                if (details.requested.group == 'org.springframework') {
                    details.useVersion '6.1.0'
                }
            }
        }
    }

    // 4. 冲突查看
    configurations.compileClasspath {
        incoming.afterResolve { ResolvableDependencies result ->
            println "=== Dependency Conflicts ==="
            result.resolutionResult.allDependencies.each { dep ->
                if (dep instanceof ResolvedDependencyResult) {
                    println "${dep.selected.moduleGroup}:${dep.selected.moduleName}:${dep.selected.moduleVersion}"
                }
            }
        }
    }
}
```

### 4.1.4 Maven POM到Gradle的映射

**Maven依赖映射对照表：**

| Maven Scope | Gradle Configuration | 说明 |
|-------------|---------------------|------|
| compile | implementation | 编译和运行时都需要 |
| provided | compileOnly | 仅编译时需要 |
| runtime | runtimeOnly | 仅运行时需要 |
| test | testImplementation | 测试编译和运行时 |
| system | compileOnly files(...) | 系统提供的依赖 |
| import | platform(...) | BOM导入 |

**Maven项目转换示例：**

```xml
<!-- pom.xml - Maven依赖配置 -->
<dependencies>
    <dependency>
        <groupId>org.springframework.boot</groupId>
        <artifactId>spring-boot-starter-web</artifactId>
        <version>3.2.0</version>
        <scope>compile</scope>
    </dependency>

    <dependency>
        <groupId>javax.servlet</groupId>
        <artifactId>javax.servlet-api</artifactId>
        <version>4.0.1</version>
        <scope>provided</scope>
    </dependency>

    <dependency>
        <groupId>mysql</groupId>
        <artifactId>mysql-connector-java</artifactId>
        <version>8.2.0</version>
        <scope>runtime</scope>
    </dependency>

    <dependency>
        <groupId>org.projectlombok</groupId>
        <artifactId>lombok</artifactId>
        <version>1.18.30</version>
        <scope>provided</scope>
        <optional>true</optional>
    </dependency>

    <dependency>
        <groupId>org.junit.jupiter</groupId>
        <artifactId>junit-jupiter</artifactId>
        <version>5.10.0</version>
        <scope>test</scope>
    </dependency>

    <dependency>
        <groupId>com.example</groupId>
        <artifactId>internal-library</artifactId>
        <version>1.0.0</version>
        <scope>system</scope>
        <systemPath>${project.basedir}/libs/internal-library.jar</systemPath>
    </dependency>
</dependencies>
```

```groovy
// build.gradle - 对应的Gradle配置
dependencies {
    // Maven compile scope -> Gradle implementation
    implementation 'org.springframework.boot:spring-boot-starter-web:3.2.0'

    // Maven provided scope -> Gradle compileOnly
    compileOnly 'javax.servlet:javax.servlet-api:4.0.1'

    // Maven runtime scope -> Gradle runtimeOnly
    runtimeOnly 'mysql:mysql-connector-java:8.2.0'

    // Maven provided + optional -> Gradle compileOnly
    compileOnly 'org.projectlombok:lombok:1.18.30'
    annotationProcessor 'org.projectlombok:lombok:1.18.30'

    // Maven test scope -> Gradle testImplementation
    testImplementation 'org.junit.jupiter:junit-jupiter:5.10.0'

    // Maven system scope -> Gradle files
    compileOnly files('libs/internal-library.jar')

    // Maven exclusions
    implementation('org.springframework.boot:spring-boot-starter-web') {
        exclude group: 'org.springframework.boot', module: 'spring-boot-starter-tomcat'
    }
}
```

**仓库配置的高级特性：**

```groovy
// build.gradle - 高级仓库配置

repositories {
    // 1. 条件化仓库配置
    if (project.hasProperty('useInternalRepo')) {
        maven {
            name = 'Internal'
            url = 'https://repo.internal.company.com/maven2'
            credentials {
                username = project.findProperty('internalRepoUser')
                password = project.findProperty('internalRepoPassword')
            }
        }
    }

    // 2. 仓库优先级和内容过滤
    maven {
        name = 'PrimaryRepo'
        url = 'https://repo.primary.com/maven2'

        mavenContent {
            // 只接受release版本
            releasesOnly()

            // 包含特定组
            includeGroup 'com.company'
            includeGroup 'org.springframework'

            // 排除特定组
            excludeGroup 'com.unwanted'
        }
    }

    // 3. 仓库元数据配置
    maven {
        name = 'ConfigurableRepo'
        url = 'https://repo.configurable.com/maven2'

        // 仓库行为配置
        artifactUrls {
            // 备用下载URL
            'https://mirror1.configurable.com/maven2'
            'https://mirror2.configurable.com/maven2'
        }

        // 元数据验证
        metadataSources {
            mavenPom()
            ignoreGradleMetadataRedirection()
            artifact()
        }
    }

    // 4. 镜像配置（类似Maven的mirror配置）
    maven {
        name = 'MirrorCentral'
        url = 'https://mirror.company.com/maven2'

        // 镜像特定仓库
        artifactUrls 'https://repo.maven.apache.org/maven2'

        mavenContent {
```
