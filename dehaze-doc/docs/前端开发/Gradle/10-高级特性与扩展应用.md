# 第十章：高级特性与扩展应用

## 10.1 自定义扩展开发

### 10.1.1 扩展接口实现

**Gradle扩展开发详解：**

Gradle扩展是扩展Gradle功能的重要方式，允许我们为项目添加自定义的配置属性和方法。相比Maven的Profile和Property，Gradle扩展提供了更强的类型安全和编程能力。

```groovy
// buildSrc/src/main/groovy/com/example/extensions/ApplicationExtension.groovy
package com.example.extensions

import org.gradle.api.model.ObjectFactory
import org.gradle.api.provider.Property
import org.gradle.api.provider.ListProperty
import org.gradle.api.provider.MapProperty
import org.gradle.api.provider.ProviderFactory
import javax.inject.Inject

/**
 * 应用程序扩展配置
 * 用于配置Spring Boot应用程序的各种参数
 */
abstract class ApplicationExtension {

    // 应用程序基本信息
    @Inject
    ApplicationExtension(ObjectFactory objects) {
        // 初始化默认值
        appName.convention("MyApplication")
        version.convention("1.0.0")
        description.convention("Enterprise Spring Boot Application")
        author.convention("Enterprise Team")

        // 环境配置
        environments.convention(["dev", "test", "prod"])
        activeEnvironment.convention("dev")

        // 服务器配置
        serverPort.convention(8080)
        managementPort.convention(8081)
        contextPath.convention("/")

        // 数据库配置
        database {
            url.convention("jdbc:mysql://localhost:3306/myapp")
            username.convention("root")
            password.convention("password")
            driverClassName.convention("com.mysql.cj.jdbc.Driver")
            hikari {
                maximumPoolSize.convention(20)
                minimumIdle.convention(5)
                connectionTimeout.convention(30000)
                idleTimeout.convention(600000)
                maxLifetime.convention(1800000)
            }
        }

        // Redis配置
        redis {
            host.convention("localhost")
            port.convention(6379)
            password.convention("")
            database.convention(0)
            timeout.convention(5000)
        }

        // 日志配置
        logging {
            level.convention("INFO")
            file.enabled.convention(true)
            file.path.convention("logs/application.log")
            file.maxSize.convention("100MB")
            file.maxHistory.convention(30)
            console.enabled.convention(true)
        }

        // 安全配置
        security {
            enabled.convention(true)
            jwt {
                secret.convention("mySecretKey")
                expiration.convention(86400) // 24小时
            }
            cors {
                allowedOrigins.convention(["http://localhost:3000"])
                allowedMethods.convention(["GET", "POST", "PUT", "DELETE"])
                allowedHeaders.convention(["*"])
            }
        }

        // 监控配置
        monitoring {
            enabled.convention(true)
            metrics {
                enabled.convention(true)
                jvm.enabled.convention(true)
                web.enabled.convention(true)
            }
            health {
                enabled.convention(true)
                endpoints.convention(["health", "info", "metrics"])
            }
        }
    }

    // 应用程序名称
    abstract Property<String> getAppName()

    // 应用程序版本
    abstract Property<String> getVersion()

    // 应用程序描述
    abstract Property<String> getDescription()

    // 作者信息
    abstract Property<String> getAuthor()

    // 支持的环境列表
    abstract ListProperty<String> getEnvironments()

    // 当前激活的环境
    abstract Property<String> getActiveEnvironment()

    // 服务器配置
    abstract ServerConfig getServer()

    interface ServerConfig {
        abstract Property<Integer> getPort()
        abstract Property<Integer> getManagementPort()
        abstract Property<String> getContextPath()
    }

    // 数据库配置
    abstract DatabaseConfig getDatabase()

    interface DatabaseConfig {
        abstract Property<String> getUrl()
        abstract Property<String> getUsername()
        abstract Property<String> getPassword()
        abstract Property<String> getDriverClassName()
        abstract HikariConfig getHikari()

        interface HikariConfig {
            abstract Property<Integer> getMaximumPoolSize()
            abstract Property<Integer> getMinimumIdle()
            abstract Property<Integer> getConnectionTimeout()
            abstract Property<Integer> getIdleTimeout()
            abstract Property<Integer> getMaxLifetime()
        }
    }

    // Redis配置
    abstract RedisConfig getRedis()

    interface RedisConfig {
        abstract Property<String> getHost()
        abstract Property<Integer> getPort()
        abstract Property<String> getPassword()
        abstract Property<Integer> getDatabase()
        abstract Property<Integer> getTimeout()
    }

    // 日志配置
    abstract LoggingConfig getLogging()

    interface LoggingConfig {
        abstract Property<String> getLevel()
        abstract FileConfig getFile()
        abstract Property<Boolean> getConsole()

        interface FileConfig {
            abstract Property<Boolean> getEnabled()
            abstract Property<String> getPath()
            abstract Property<String> getMaxSize()
            abstract Property<Integer> getMaxHistory()
        }
    }

    // 安全配置
    abstract SecurityConfig getSecurity()

    interface SecurityConfig {
        abstract Property<Boolean> getEnabled()
        abstract JwtConfig getJwt()
        abstract CorsConfig getCors()

        interface JwtConfig {
            abstract Property<String> getSecret()
            abstract Property<Integer> getExpiration()
        }

        interface CorsConfig {
            abstract ListProperty<String> getAllowedOrigins()
            abstract ListProperty<String> getAllowedMethods()
            abstract ListProperty<String> getAllowedHeaders()
        }
    }

    // 监控配置
    abstract MonitoringConfig getMonitoring()

    interface MonitoringConfig {
        abstract Property<Boolean> getEnabled()
        abstract MetricsConfig getMetrics()
        abstract HealthConfig getHealth()

        interface MetricsConfig {
            abstract Property<Boolean> getEnabled()
            abstract JvmMetricsConfig getJvm()
            abstract WebMetricsConfig getWeb()

            interface JvmMetricsConfig {
                abstract Property<Boolean> getEnabled()
            }

            interface WebMetricsConfig {
                abstract Property<Boolean> getEnabled()
            }
        }

        interface HealthConfig {
            abstract Property<Boolean> getEnabled()
            abstract ListProperty<String> getEndpoints()
        }
    }

    // 验证配置
    void validate() {
        def errors = []

        // 验证必需字段
        if (!appName.present) {
            errors << "应用程序名称不能为空"
        }

        if (!version.present) {
            errors << "版本号不能为空"
        }

        // 验证端口配置
        if (server.port.get() < 1 || server.port.get() > 65535) {
            errors << "服务器端口必须在1-65535之间"
        }

        // 验证环境配置
        def activeEnv = activeEnvironment.get()
        if (!environments.get().contains(activeEnv)) {
            errors << "激活环境 '$activeEnv' 不在支持的环境列表中: ${environments.get()}"
        }

        // 验证数据库配置
        if (!database.url.present) {
            errors << "数据库URL不能为空"
        }

        if (!database.username.present) {
            errors << "数据库用户名不能为空"
        }

        // 验证Redis配置
        if (redis.port.get() < 1 || redis.port.get() > 65535) {
            errors << "Redis端口必须在1-65535之间"
        }

        if (!errors.isEmpty()) {
            throw new IllegalArgumentException("配置验证失败:\n" + errors.join("\n"))
        }

        println "✅ 应用程序配置验证通过"
    }

    // 生成配置文件
    void generateConfigurationFiles() {
        def outputDir = file("src/main/resources")
        outputDir.mkdirs()

        // 生成application.yml
        def applicationYml = generateApplicationYml()
        file("$outputDir/application.yml").text = applicationYml

        // 生成application-${activeEnvironment.get()}.yml
        def envYml = generateEnvironmentYml()
        file("$outputDir/application-${activeEnvironment.get()}.yml").text = envYml

        // 生成logback-spring.xml
        def logbackXml = generateLogbackXml()
        file("$outputDir/logback-spring.xml").text = logbackXml

        println "✅ 配置文件生成完成"
    }

    // 生成application.yml
    private String generateApplicationYml() {
        return """
        server:
          port: ${server.port.get()}
          servlet:
            context-path: '${server.contextPath.get()}'

        spring:
          application:
            name: ${appName.get()}
          profiles:
            active: ${activeEnvironment.get()}

        # 数据库配置
        spring.datasource:
          url: \${DATABASE_URL:${database.url.get()}}
          username: \${DATABASE_USERNAME:${database.username.get()}}
          password: \${DATABASE_PASSWORD:${database.password.get()}}
          driver-class-name: ${database.driverClassName.get()}
          hikari:
            maximum-pool-size: ${database.hikari.maximumPoolSize.get()}
            minimum-idle: ${database.hikari.minimumIdle.get()}
            connection-timeout: ${database.hikari.connectionTimeout.get()}
            idle-timeout: ${database.hikari.idleTimeout.get()}
            max-lifetime: ${database.hikari.maxLifetime.get()}

        # Redis配置
        spring.redis:
          host: \${REDIS_HOST:${redis.host.get()}}
          port: \${REDIS_PORT:${redis.port.get()}}
          password: \${REDIS_PASSWORD:${redis.password.get()}}
          database: ${redis.database.get()}
          timeout: ${redis.timeout.get()}ms
          lettuce:
            pool:
              max-active: 8
              max-idle: 8
              min-idle: 0

        # 监控配置
        management:
          server:
            port: ${server.managementPort.get()}
          endpoints:
            web:
              exposure:
                include: ${monitoring.health.endpoints.get()}
          endpoint:
            health:
              show-details: always
        """.stripIndent()
    }

    // 生成环境特定配置
    private String generateEnvironmentYml() {
        def env = activeEnvironment.get()

        switch (env) {
            case 'dev':
                return """
                # 开发环境配置
                logging:
                  level:
                    com.example: DEBUG
                    org.springframework.web: DEBUG
                  pattern:
                    console: "%d{yyyy-MM-dd HH:mm:ss} [%thread] %-5level %logger{36} - %msg%n"

                spring:
                  devtools:
                    restart:
                      enabled: true
                    livereload:
                      enabled: true
                """.stripIndent()

            case 'test':
                return """
                # 测试环境配置
                spring:
                  datasource:
                    url: jdbc:h2:mem:testdb
                    driver-class-name: org.h2.Driver
                    username: sa
                    password:
                  jpa:
                    hibernate:
                      ddl-auto: create-drop
                    show-sql: true
                """.stripIndent()

            case 'prod':
                return """
                # 生产环境配置
                logging:
                  level:
                    root: WARN
                    com.example: INFO
                  file:
                    name: ${logging.file.path.get()}
                    max-size: ${logging.file.maxSize.get()}
                    max-history: ${logging.file.maxHistory.get()}

                spring:
                  jpa:
                    hibernate:
                      ddl-auto: validate
                    show-sql: false
                """.stripIndent()

            default:
                return "# 未知环境配置"
        }
    }

    // 生成logback配置
    private String generateLogbackXml() {
        return """
        <?xml version="1.0" encoding="UTF-8"?>
        <configuration>
            <springProfile name="!prod">
                <include resource="org/springframework/boot/logging/logback/defaults.xml"/>
                <property name="LOG_FILE" value="\${LOG_FILE:-\${LOG_PATH:-\${LOG_TEMP:-\${java.io.tmpdir:-/tmp}}/}spring.log}"/>
                <include resource="org/springframework/boot/logging/logback/console-appender.xml" />
                <include resource="org/springframework/boot/logging/logback/file-appender.xml" />
                <root level="INFO">
                    <appender-ref ref="CONSOLE"/>
                    <appender-ref ref="FILE"/>
                </root>
            </springProfile>

            <springProfile name="prod">
                <include resource="org/springframework/boot/logging/logback/defaults.xml"/>
                <property name="LOG_FILE" value="\${LOG_FILE}"/>
                <include resource="org/springframework/boot/logging/logback/file-appender.xml" />
                <root level="${logging.level.get()}">
                    <appender-ref ref="FILE"/>
                </root>
            </springProfile>
        </configuration>
        """.stripIndent()
    }

    // 显示配置摘要
    void printConfigurationSummary() {
        println """
        ========================================
        应用程序配置摘要
        ========================================
        应用名称: ${appName.get()}
        版本: ${version.get()}
        描述: ${description.get()}
        作者: ${author.get()}
        激活环境: ${activeEnvironment.get()}

        服务器配置:
        - 端口: ${server.port.get()}
        - 管理端口: ${server.managementPort.get()}
        - 上下文路径: ${server.contextPath.get()}

        数据库配置:
        - URL: ${database.url.get()}
        - 用户名: ${database.username.get()}

        - 主机: ${redis.host.get()}

    }
}
```

### 10.1.2 DSL设计模式

**类型安全DSL设计：**

```groovy
// buildSrc/src/main/groovy/com/example/dsl/ConfigurationDsl.groovy
package com.example.dsl

import org.gradle.api.Action
import org.gradle.api.model.ObjectFactory
import javax.inject.Inject

/**
 * 配置DSL构建器
 * 提供流畅的API来配置应用程序
 */
abstract class ConfigurationDsl {

    @Inject
    ConfigurationDsl(ObjectFactory objects) {
        applicationExtension = objects.newInstance(ApplicationExtension)
        webExtension = objects.newInstance(WebExtension)
        databaseExtension = objects.newInstance(DatabaseExtension)
        securityExtension = objects.newInstance(SecurityExtension)
    }

    private final ApplicationExtension applicationExtension
    private final WebExtension webExtension
    private final DatabaseExtension databaseExtension
    private final SecurityExtension securityExtension

    // 应用程序配置DSL
    void application(Action<ApplicationExtension> action) {
        action.execute(applicationExtension)
    }

    // Web配置DSL
    void web(Action<WebExtension> action) {
        action.execute(webExtension)
    }

    // 数据库配置DSL
    void database(Action<DatabaseExtension> action) {
        action.execute(databaseExtension)
    }

    // 安全配置DSL
    void security(Action<SecurityExtension> action) {
        action.execute(securityExtension)
    }

    // 构建配置
    void build() {
        // 验证所有配置
        applicationExtension.validate()
        webExtension.validate()
        databaseExtension.validate()
        securityExtension.validate()

        // 生成配置文件
        applicationExtension.generateConfigurationFiles()
        webExtension.generateConfiguration()
        databaseExtension.generateConfiguration()
        securityExtension.generateConfiguration()

        // 打印配置摘要
        applicationExtension.printConfigurationSummary()
        webExtension.printConfigurationSummary()
        databaseExtension.printConfigurationSummary()
        securityExtension.printConfigurationSummary()

        println "✅ 所有配置构建完成"
    }

    // 获取配置扩展
    ApplicationExtension getApplicationExtension() { applicationExtension }
    WebExtension getWebExtension() { webExtension }
    DatabaseExtension getDatabaseExtension() { databaseExtension }
    SecurityExtension getSecurityExtension() { securityExtension }
}

// Web扩展配置
abstract class WebExtension {
    @Inject
    WebExtension(ObjectFactory objects) {
        server.convention(objects.newInstance(ServerConfig))
        cors.convention(objects.newInstance(CorsConfig))
        compression.convention(objects.newInstance(CompressionConfig))
        ssl.convention(objects.newInstance(SSLConfig))
    }

    abstract ServerConfig getServer()
    abstract CorsConfig getCors()
    abstract CompressionConfig getCompression()
    abstract SSLConfig getSsl()

    interface ServerConfig {
        abstract Property<String> getHost()
        abstract Property<Integer> getPort()
        abstract Property<Integer> getIoThreads()
        abstract Property<Integer> getWorkerThreads()
    }

    interface CorsConfig {
        abstract Property<Boolean> getEnabled()
        abstract ListProperty<String> getAllowedOrigins()
        abstract ListProperty<String> getAllowedMethods()
        abstract ListProperty<String> getAllowedHeaders()
    }

    interface CompressionConfig {
        abstract Property<Boolean> getEnabled()
        abstract Property<String> getMimeTypes()
        abstract Property<Integer> getMinResponseSize()
    }

    interface SSLConfig {
        abstract Property<Boolean> getEnabled()
        abstract Property<String> getKeyStore()
        abstract Property<String> getKeyStorePassword()
        abstract Property<String> getKeyStoreType()
    }

    void validate() {
        if (server.port.get() < 1 || server.port.get() > 65535) {
            throw new IllegalArgumentException("Web服务器端口无效: ${server.port.get()}")
        }
        println "✅ Web配置验证通过"
    }

    void generateConfiguration() {
        def config = """
        # Web服务器配置
        server:
          host: ${server.host.get()}
          port: ${server.port.get()}
          tomcat:
            threads:
              max: ${server.workerThreads.get()}
              min-spare: ${server.ioThreads.get()}

        # CORS配置
        cors:
          enabled: ${cors.enabled.get()}
          allowed-origins: ${cors.allowedOrigins.get()}
          allowed-methods: ${cors.allowedMethods.get()}
          allowed-headers: ${cors.allowedHeaders.get()}

        # 压缩配置
        server:
          compression:
            enabled: ${compression.enabled.get()}
            mime-types: ${compression.mimeTypes.get()}
            min-response-size: ${compression.minResponseSize.get()}
        """.stripIndent()

        println "Web配置生成完成"
    }

    void printConfigurationSummary() {
        println """
        Web配置摘要:
        - 服务器: ${server.host.get()}:${server.port.get()}
        - IO线程: ${server.ioThreads.get()}
        - 工作线程: ${server.workerThreads.get()}
        - CORS: ${cors.enabled.get() ? '启用' : '禁用'}
        - 压缩: ${compression.enabled.get() ? '启用' : '禁用'}
        - SSL: ${ssl.enabled.get() ? '启用' : '禁用'}
        """
    }
}

abstract class SecurityExtension {
    @Inject
    SecurityExtension(ObjectFactory objects) {
        jwt.convention(objects.newInstance(JWTConfig))
        oauth2.convention(objects.newInstance(OAuth2Config))
        csrf.convention(objects.newInstance(CSRFConfig))
        session.convention(objects.newInstance(SessionConfig))
    }

    abstract JWTConfig getJwt()
    abstract OAuth2Config getOauth2()
    abstract CSRFConfig getCsrf()
    abstract SessionConfig getSession()

    interface JWTConfig {
        abstract Property<String> getSecret()
        abstract Property<Integer> getExpiration()
        abstract Property<String> getIssuer()
    }

    interface OAuth2Config {
        abstract Property<Boolean> getEnabled()
        abstract Property<String> getClientId()
        abstract Property<String> getClientSecret()
        abstract ListProperty<String> getScopes()
    }

    interface CSRFConfig {
        abstract Property<Boolean> getEnabled()
        abstract ListProperty<String> getIgnoredMethods()
        abstract ListProperty<String> getIgnoredPatterns()
    }

    interface SessionConfig {
        abstract Property<Integer> getTimeout()
        abstract Property<String> getStoreType()
        abstract Property<Boolean> getRedisEnabled()
    }

    void validate() {
        if (!jwt.secret.present || jwt.secret.get().length() < 32) {
            throw new IllegalArgumentException("JWT密钥长度必须至少32个字符")
        }
        println "✅ 安全配置验证通过"
    }

    void generateConfiguration() {
        def config = """
        # JWT配置
        security.jwt:
          secret: \${JWT_SECRET:${jwt.secret.get()}}
          expiration: ${jwt.expiration.get()}
          issuer: ${jwt.issuer.get()}

        # OAuth2配置
        spring.security.oauth2:
          enabled: ${oauth2.enabled.get()}
          client:
            client-id: \${OAUTH2_CLIENT_ID:${oauth2.clientId.get()}}
            client-secret: \${OAUTH2_CLIENT_SECRET:${oauth2.clientSecret.get()}}

        # CSRF配置
        security.csrf:
          enabled: ${csrf.enabled.get()}
          ignored-methods: ${csrf.ignoredMethods.get()}
          ignored-patterns: ${csrf.ignoredPatterns.get()}

        # Session配置
        server:
          servlet:
            session:
              timeout: ${session.timeout.get()}s
        """.stripIndent()

        println "安全配置生成完成"
    }

    void printConfigurationSummary() {
        println """
        安全配置摘要:
        - JWT: 启用 (过期时间: ${jwt.expiration.get()}秒)
        - OAuth2: ${oauth2.enabled.get() ? '启用' : '禁用'}
        - CSRF: ${csrf.enabled.get() ? '启用' : '禁用'}
        - Session超时: ${session.timeout.get()}秒
        """
    }
}
```

```groovy
class DynamicConfigLoader {
    static Map<String, Object> loadConfiguration() {
        def config = [:]

        // 从环境变量加载
        System.getenv().each { key, value ->
            if (key.startsWith('APP_')) {
                config[key.replace('APP_', '').toLowerCase()] = value
            }
        }

        // 从配置文件加载
        def configFile = file('app-config.json')
        if (configFile.exists()) {
            def jsonSlurper = new JsonSlurper()
            def fileConfig = jsonSlurper.parse(configFile)
            config.putAll(fileConfig)
        }

        // 从系统属性加载
        System.properties.each { key, value ->
            if (key.startsWith('app.')) {
                config[key.replace('app.', '')] = value
            }
        }

        return config
    }

    static void saveConfiguration(Map<String, Object> config) {
        def jsonBuilder = new JsonBuilder(config)
        def configFile = file('app-config.json')
        configFile.text = jsonBuilder.toPrettyString()
        println "配置已保存到: ${configFile.absolutePath}"
    }
}

// 3. 构建时序分析器
class BuildSequenceAnalyzer {
    static void analyzeBuildSequence() {
        def taskGraph = gradle.taskGraph
        def executionPlan = taskGraph.executionPlan

        println "=== 构建时序分析 ==="
        println "总任务数: ${taskGraph.allTasks.size()}"
        println "执行计划节点数: ${executionPlan.size()}"

        // 分析关键路径
        def criticalPath = findCriticalPath(executionPlan)
        println "关键路径长度: ${criticalPath.size()}"
        println "关键路径: ${criticalPath.collect { it.task.name }.join(' -> ')}"

        // 分析并行度
        def parallelism = calculateParallelism(executionPlan)
        println "最大并行度: ${parallelism.max}"
        println "平均并行度: ${parallelism.average.round(2)}"

        // 分析任务类型分布
        def taskTypes = analyzeTaskTypes(taskGraph.allTasks)
        println "任务类型分布:"
        taskTypes.each { type, count ->
            println "  $type: $count"
        }
    }

    private static List<TaskExecutionPlan> findCriticalPath(List<TaskExecutionPlan> plan) {
        // 简化的关键路径算法
        def path = []
        def visited = [] as Set

        def findPath = { TaskExecutionPlan node ->
            if (node in visited) return
            visited << node

            if (node.dependencies.isEmpty()) {
                path << node
            } else {
                node.dependencies.each { dep ->
                    findPath(dep)
                }
                path << node
            }
        }

        plan.findAll { it.successors.isEmpty() }.each { endNode ->
            findPath(endNode)
        }

        return path.unique { it.task.name }
    }

    private static Map<String, Integer> calculateParallelism(List<TaskExecutionPlan> plan) {
        def parallelismLevels = [:]
        def visited = [:]

        def calculateLevel = { TaskExecutionPlan node ->
            if (visited.containsKey(node)) return visited[node]

            def maxLevel = node.dependencies.isEmpty() ? 0 :
                node.dependencies.collect { calculateLevel(it) }.max() + 1

            visited[node] = maxLevel
            parallelismLevels[maxLevel] = (parallelismLevels[maxLevel] ?: 0) + 1

            return maxLevel
        }

        plan.each { calculateLevel(it) }

        def maxLevel = parallelismLevels.keySet().max()
        def totalNodes = plan.size()

        return [
            max: parallelismLevels.values().max(),
            average: totalNodes / (maxLevel + 1)
        ]
    }

    private static Map<String, Integer> analyzeTaskTypes(Set<Task> tasks) {
        def types = [:]
        tasks.each { task ->
            def type = determineTaskType(task)
            types[type] = (types[type] ?: 0) + 1
        }
        return types
    }

    private static String determineTaskType(Task task) {
        def className = task.class.simpleName

        if (className.contains('JavaCompile')) return '编译'
        if (className.contains('Test')) return '测试'
        if (className.contains('Jar')) return '打包'
        if (className.contains('Copy')) return '文件操作'
        if (className.contains('Delete')) return '清理'

        return '其他'
    }
}

// 4. 智能依赖管理器
class IntelligentDependencyManager {
    static void optimizeDependencies() {
        println "=== 智能依赖优化 ==="

        def conflicts = detectDependencyConflicts()
        if (!conflicts.isEmpty()) {
            println "发现依赖冲突:"
            conflicts.each { conflict ->
                println "  ${conflict.module}: ${conflict.versions.join(' vs ')}"
            }

            resolveConflicts(conflicts)
        }

        def duplicates = detectDuplicateDependencies()
        if (!duplicates.isEmpty()) {
            println "发现重复依赖:"
            duplicates.each { dep ->
                println "  ${dep.module}: ${dep.count}次"
            }
        }

        def unused = detectUnusedDependencies()
        if (!unused.isEmpty()) {
            println "发现未使用依赖:"
            unused.each { dep ->
                println "  ${dep.group}:${dep.name}:${dep.version}"
            }
        }

        generateOptimizationReport(conflicts, duplicates, unused)
    }

    private static List<Map<String, Object>> detectDependencyConflicts() {
        def conflicts = []
        def versionMap = [:]

        configurations.implementation.resolvedConfiguration.lenientConfiguration.allDependencies.each { dep ->
            if (dep instanceof ResolvedDependencyResult) {
                def key = "${dep.selected.moduleGroup}:${dep.selected.moduleName}"
                def version = dep.selected.moduleVersion

                if (versionMap.containsKey(key) && versionMap[key] != version) {
                    conflicts << [
                        module: key,
                        versions: [versionMap[key], version].unique()
                    ]
                } else {
                    versionMap[key] = version
                }
            }
        }

        return conflicts
    }

    private static List<Map<String, Object>> detectDuplicateDependencies() {
        def duplicates = []
        def depCount = [:]

        configurations.implementation.dependencies.each { dep ->
            def key = "${dep.group}:${dep.name}"
            depCount[key] = (depCount[key] ?: 0) + 1
        }

        depCount.each { key, count ->
            if (count > 1) {
                def parts = key.split(':')
                duplicates << [
                    module: key,
                    count: count,
                    group: parts[0],
                    name: parts[1]
                ]
            }
        }

        return duplicates
    }

    private static List<Dependency> detectUnusedDependencies() {
        def unused = []
        def usedDependencies = [] as Set

        // 分析源码中的import语句
        sourceSets.main.allJava.each { sourceFile ->
            sourceFile.text.eachLine { line ->
                if (line.contains('import ')) {
                    def importMatch = line =~ /import\s+([\w.]+)\.*/
                    if (importMatch.find()) {
                        def pkg = importMatch[0][1]

                        configurations.implementation.dependencies.each { dep ->
                            if (dep.group == pkg || dep.group.startsWith("$pkg.")) {
                                usedDependencies << "${dep.group}:${dep.name}"
                            }
                        }
                    }
                }
            }
        }

        configurations.implementation.dependencies.each { dep ->
            def depKey = "${dep.group}:${dep.name}"
            if (!usedDependencies.contains(depKey)) {
                unused << dep
            }
        }

        return unused
    }

    private static void resolveConflicts(List<Map<String, Object>> conflicts) {
        println "解决依赖冲突..."

        conflicts.each { conflict ->
            def versions = conflict.versions as List<String>
            def resolvedVersion = versions.max { a, b ->
                compareVersions(a, b)
            }

            println "  ${conflict.module}: 选择版本 $resolvedVersion"

            // 强制使用解析的版本
            configurations.all {
                resolutionStrategy {
                    force "${conflict.module}:${resolvedVersion}"
                }
            }
        }
    }

    private static int compareVersions(String v1, String v2) {
        def v1Parts = v1.split('\\.').collect { it as int }
        def v2Parts = v2.split('\\.').collect { it as int }

        for (int i = 0; i < Math.min(v1Parts.size(), v2Parts.size()); i++) {
            def comparison = v1Parts[i] <=> v2Parts[i]
            if (comparison != 0) return comparison
        }

        return v1Parts.size() <=> v2Parts.size()
    }

class MetaprogrammingBuilder {
    static void generateDynamicTasks() {
        println "=== 动态任务生成 ==="

        // 从配置文件读取任务定义
        def taskConfig = loadTaskConfiguration()

        taskConfig.tasks.each { config ->
            def taskName = config.name
            def taskType = config.type
            def taskDependencies = config.dependencies ?: []

            println "生成动态任务: $taskName (类型: $taskType)"

            switch (taskType) {
                case 'JavaCompile':
                    createDynamicJavaCompileTask(taskName, taskDependencies, config)
                    break
                case 'Test':
                    createDynamicTestTask(taskName, taskDependencies, config)
                    break
                case 'Copy':
                    createDynamicCopyTask(taskName, taskDependencies, config)
                    break
                case 'Exec':
                    createDynamicExecTask(taskName, taskDependencies, config)
                    break
                default:
                    createGenericDynamicTask(taskName, taskDependencies, config)
            }
        }
    }

    private static Map loadTaskConfiguration() {
        def configFile = file('dynamic-tasks.json')
        if (!configFile.exists()) {
            // 创建默认配置

                tasks: [
                    [
                        name: 'dynamicCompile',

                        config: [
                            source: 'src/dynamic/java',
                            classpath: 'configurations.compileClasspath',

                println "动态测试任务 $taskName 执行完成"
```

```groovy
    group = 'python'

        if (environmentType == 'virtualenv') {

        installDevDependencies()

        println "✅ Python环境设置完成"

void createVirtualEnvironment() {
    if (!virtualEnvLocation.exists()) {
        println "创建Python虚拟环境..."

        exec {
            commandLine 'python', '-m', 'venv', virtualEnvLocation.absolutePath
        }
    }

    // 激活虚拟环境
    if (targetPlatform.operatingSystem.windows) {
        // Windows
        exec {
            commandLine "${virtualEnvLocation}\\Scripts\\activate.bat"
        }
    } else {
        // Linux/Mac
        exec {
            commandLine "source ${virtualEnvLocation}/bin/activate"
        }
    }
}

void installDependencies() {
    if (pip.requirementsFile.exists()) {
        println "安装Python依赖..."

        exec {
            def pipCommand = targetPlatform.operatingSystem.windows ?
                "${virtualEnvLocation}\\Scripts\\pip.exe" :
                "${virtualEnvLocation}/bin/pip"

            commandLine pipCommand, 'install', '-r', pip.requirementsFile.absolutePath
        }
    }
}

void installDevDependencies() {
    def devRequirementsFile = file('requirements-dev.txt')
    if (devRequirementsFile.exists()) {
        println "安装开发依赖..."

        exec {
            def pipCommand = targetPlatform.operatingSystem.windows ?
                "${virtualEnvLocation}\\Scripts\\pip.exe" :
                "${virtualEnvLocation}/bin/pip"

            commandLine pipCommand, 'install', '-r', devRequirementsFile.absolutePath
        }
    }
}
```
