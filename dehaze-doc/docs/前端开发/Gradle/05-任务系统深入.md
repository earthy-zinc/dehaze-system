# 第5章：任务系统深入

## 5.1 Task类型体系

### 5.1.1 DefaultTask基础任务

**DefaultTask核心特性：**

DefaultTask是所有Gradle任务的基类，提供了任务的基本功能和生命周期管理。理解DefaultTask对于掌握Gradle任务系统至关重要。

```groovy
// build.gradle - DefaultTask基础特性演示

// 1. 基础任务创建
task basicTask {
    description 'A basic task demonstrating DefaultTask features'
    group 'demo'

    // 任务属性配置
    enabled = true
    onlyIf { project.hasProperty('enableBasic') }

    // 生命周期回调
    doFirst {
        println "First action in basic task"
    }

    doLast {
        println "Last action in basic task"
        println "Task executed at: ${new Date()}"
    }
}

// 2. 任务属性配置
task configuredTask {
    description 'Task with extensive configuration'
    group 'configuration'

    // 启用/禁用任务
    enabled = true

    // 条件执行
    onlyIf {
        def condition = project.hasProperty('runConfigured')
        println "Checking condition for configuredTask: $condition"
        return condition
    }

    // 依赖任务
    dependsOn basicTask

    // 最终化任务
    finalizedBy 'cleanupTask'

    // 多个doFirst和doLast
    doFirst {
        println "Configured task - First action 1"
    }

    doFirst {
        println "Configured task - First action 2"
    }

    doLast {
        println "Configured task - Last action 1"
    }

    doLast {
        println "Configured task - Last action 2"
    }
}

// 3. 动态任务配置
task dynamicConfiguration {
    description 'Task with dynamic configuration'
    group 'dynamic'

    // 动态配置块
    configure {
        // 在配置阶段执行
        println "Dynamic configuration being applied"
        description = "Dynamically configured task"
    }

    // 动态添加行为
    doLast {
        println "Dynamic task execution"
        println "Project name: ${project.name}"
        println "Gradle version: ${gradle.gradleVersion}"
    }
}

// 4. 任务扩展属性
task extendedTask {
    description 'Task with custom properties'
    group 'extended'

    // 扩展属性
    ext {
        customProperty = 'Custom Value'
        configMap = [
            timeout: 30,
            retries: 3,
            verbose: true
        ]
    }

    doLast {
        println "Custom property: ${customProperty}"
        println "Config map: ${configMap}"
        println "Timeout setting: ${configMap.timeout}"
    }
}

// 5. 任务状态检查
task statusCheck {
    description 'Check task status and properties'
    group 'monitoring'

    doLast {
        def tasksToCheck = [basicTask, configuredTask, dynamicConfiguration]

        tasksToCheck.each { task ->
            println "Task: ${task.name}"
            println "  Enabled: ${task.enabled}"
            println "  Group: ${task.group}"
            println "  Description: ${task.description}"
            println "  State: ${task.state}"
            println "  Did work: ${task.didWork}"
            println "  Executed: ${task.state.executed}"
            println "  Up-to-date: ${task.state.upToDate}"
            println "  Skipped: ${task.state.skipped}"
            println "  Failed: ${task.state.failure}"
            println ""
        }
    }
}

// 6. 清理任务
task cleanupTask {
    description 'Cleanup task'
    group 'cleanup'

    doLast {
        println "Performing cleanup operations"
        // 清理逻辑
    }
}
```

**DefaultTask高级特性：**

```groovy
// buildSrc/src/main/groovy/com/example/tasks/AdvancedDefaultTask.groovy
package com.example.tasks

import org.gradle.api.DefaultTask
import org.gradle.api.tasks.Input
import org.gradle.api.tasks.Internal
import org.gradle.api.tasks.Optional
import org.gradle.api.tasks.TaskAction

/**
 * 扩展DefaultTask实现高级功能
 */
abstract class AdvancedDefaultTask extends DefaultTask {

    @Input
    @Optional
    abstract String getInputMessage()

    @Input
    @Optional
    abstract Boolean getVerboseMode()

    @Internal
    String getTimestamp() {
        new Date().format('yyyy-MM-dd HH:mm:ss')
    }

    AdvancedDefaultTask() {
        group = 'advanced'
        description = 'Advanced DefaultTask demonstration'

        // 构造函数中的配置
        outputs.upToDateWhen { false } // 总是执行，不进行up-to-date检查

        // 添加扩展属性
        ext.executionCount = 0
        ext.lastExecutionTime = null
    }

    @TaskAction
    void execute() {
        ext.executionCount++
        ext.lastExecutionTime = timestamp

        if (verboseMode) {
            println "=== Advanced Task Execution ==="
            println "Task name: ${name}"
            println "Execution count: ${executionCount}"
            println "Timestamp: ${timestamp}"
            println "Input message: ${inputMessage ?: 'No input message'}"
            println "================================"
        } else {
            println "Advanced task executed (count: ${executionCount})"
        }
    }
}

// build.gradle - 使用高级DefaultTask
task advancedDemo(type: AdvancedDefaultTask) {
    description 'Demonstrate advanced DefaultTask features'
    group 'advanced'

    inputMessage = 'Hello from advanced task'
    verboseMode = true

    doFirst {
        println "Pre-execution setup for advanced task"
    }

    doLast {
        println "Post-execution cleanup for advanced task"
    }
}
```

### 5.1.2 Java相关任务类型

**JavaCompile任务详解：**

```groovy
// build.gradle - JavaCompile任务配置

// 1. 基础Java编译配置
task customJavaCompile(type: JavaCompile) {
    description 'Custom Java compilation task'
    group 'java'

    // 源码配置
    source = fileTree('src/custom/java')
    include '**/*.java'
    exclude '**/generated/**'

    // 类路径配置
    classpath = sourceSets.main.compileClasspath
    destinationDirectory = file("$buildDir/classes/custom")

    // 编译选项
    options.encoding = 'UTF-8'
    options.compilerArgs += [
        '-Xlint:unchecked',
        '-Xlint:deprecation',
        '-parameters',
        '-Werror'
    ]

    // Java版本
    sourceCompatibility = JavaVersion.VERSION_17
    targetCompatibility = JavaVersion.VERSION_17

    // 增量编译
    options.incremental = true
}

// 2. 多源码目录编译
task multiSourceCompile(type: JavaCompile) {
    description 'Compile from multiple source directories'
    group 'java'

    // 多个源码目录
    source = files(
        fileTree('src/main/java'),
        fileTree('src/generated/java'),
        fileTree('src/legacy/java')
    )

    classpath = sourceSets.main.compileClasspath
    destinationDirectory = file("$buildDir/classes/multi-source")

    // 不同源码目录的不同处理
    doFirst {
        println "Compiling multiple source directories:"
        source.each { file ->
            println "  - ${file.absolutePath}"
        }
    }
}

// 3. 条件化编译
task conditionalCompile(type: JavaCompile) {
    description 'Conditional compilation based on properties'
    group 'java'

    // 根据条件选择源码
    if (project.hasProperty('enableExperimental')) {
        source = fileTree('src/experimental/java')
        destinationDirectory = file("$buildDir/classes/experimental")
    } else {
        source = fileTree('src/standard/java')
        destinationDirectory = file("$buildDir/classes/standard")
    }

    classpath = sourceSets.main.compileClasspath

    // 条件化编译选项
    if (project.hasProperty('strictMode')) {
        options.compilerArgs += '-Werror'
    }

    options.encoding = 'UTF-8'
}

// 4. 注解处理器配置
task annotationProcessorCompile(type: JavaCompile) {
    description 'Compilation with annotation processors'
    group 'java'

    source = fileTree('src/annotation-processor/java')
    classpath = sourceSets.main.compileClasspath
    destinationDirectory = file("$buildDir/classes/annotation-processed")

    // 注解处理器配置
    options.annotationProcessorPath = configurations.annotationProcessor
    options.compilerArgs += [
        '-processor', 'com.example.processor.CustomProcessor',
        '-Aprocessor.option=value'
    ]

    // 生成源码目录
    options.generatedSourceOutputDirectory = file("$buildDir/generated/sources/annotationProcessor/java/main")
}

// 5. 编译错误处理
task errorHandlingCompile(type: JavaCompile) {
    description 'Compilation with custom error handling'
    group 'java'

    source = fileTree('src/main/java')
    classpath = sourceSets.main.compileClasspath
    destinationDirectory = file("$buildDir/classes/error-handling")

    // 自定义编译错误处理
    options.listener = { JavaCompile.CompileMessage message ->
        if (message.kind == 'error') {
            logger.error("Compilation error in ${message.source}: ${message.message}")
        } else if (message.kind == 'warning') {
            logger.warn("Compilation warning in ${message.source}: ${message.message}")
        }
    }

    // 失败时继续编译其他文件
    options.failOnError = false

    doLast {
        // 检查编译结果
        if (state.failure) {
            println "Compilation completed with errors"
        } else {
            println "Compilation completed successfully"
        }
    }
}
```

**Copy任务深度应用：**

```groovy
// build.gradle - Copy任务高级配置

// 1. 基础文件复制
task basicCopy(type: Copy) {
    description 'Basic file copy task'
    group 'file operations'

    from 'src/main/resources'
    into "$buildDir/resources"

    // 包含和排除模式
    include '**/*.properties'
    include '**/*.yml'
    exclude '**/*template*'

    doLast {
        println "Basic copy completed"
        println "Copied files: ${source.files.size()}"
    }
}

// 2. 带文件处理的复制
task processAndCopy(type: Copy) {
    description 'Copy files with processing'
    group 'file operations'

    from 'src/main/resources'
    into "$buildDir/processed-resources"

    // 文件过滤和替换
    filter { line ->
        line
            .replace('\${project.name}', project.name)
            .replace('\${project.version}', project.version.toString())
            .replace('\${build.timestamp}', new Date().format('yyyy-MM-dd HH:mm:ss'))
    }

    // 条件化复制
    onlyIf { !project.hasProperty('skipResourceProcessing') }

    doLast {
        println "Processed and copied resources"
    }
}

// 3. 多源复制
task multiSourceCopy(type: Copy) {
    description 'Copy from multiple sources'
    group 'file operations'

    // 多个源目录
    from('src/main/resources') {
        into 'static'
        include '**/*.html', '**/*.css', '**/*.js'
    }

    from('src/main/templates') {
        into 'templates'
        include '**/*.html'
    }

    from('src/main/config') {
        into 'config'
        include '**/*.properties', '**/*.yml'
    }

    into "$buildDir/merged-resources"

    doLast {
        println "Multi-source copy completed"
    }
}

// 4. 增量复制
task incrementalCopy(type: Copy) {
    description 'Incremental copy operation'
    group 'file operations'

    from 'src/main/resources'
    into "$buildDir/incremental-resources"

    // 增量处理
    outputs.upToDateWhen {
        // 自定义up-to-date条件
        def lastModified = new File("$buildDir/.last-copy-time").text?.trim()
        if (lastModified) {
            def lastTime = Long.parseLong(lastModified)
            def sourceModTime = source.files.collect { it.lastModified() }.max()
            return sourceModTime <= lastTime
        }
        return false
    }

    doLast {
        // 记录复制时间
        new File("$buildDir/.last-copy-time").text = System.currentTimeMillis().toString()
        println "Incremental copy completed at ${new Date()}"
    }
}

// 5. 压缩和归档复制
task compressCopy(type: Copy) {
    description 'Copy and compress files'
    group 'file operations'

    from 'src/main/resources'
    into "$buildDir/compressed-resources"

    // 文件压缩
    eachFile { file ->
        if (file.name.endsWith('.json')) {
            // 压缩JSON文件
            def content = file.file.text
            def compressed = content.replaceAll(/\s+/, ' ').trim()
            file.filter { line -> compressed }
        }
    }

    // 文件重命名
    rename { fileName ->
        if (fileName.endsWith('.template')) {
            fileName.replace('.template', '.processed')
        } else {
            fileName
        }
    }

    doLast {
        println "Compressed copy completed"
    }
}
```

```groovy
task taskWithProperties {
    description 'Demonstrate task properties'
    group 'properties'

    // 基础属性
    enabled = true
    group = 'properties' // 覆盖上面定义的group
    description = 'Task with various properties configured'

        println "Group: ${group}"
        println "Description: ${description}"
        println "Enabled: ${enabled}"
        println "Custom timeout: ${customTimeout}"
        println "Retry count: ${retryCount}"
        println "Verbose: ${verbose}"
        println "Config map: ${configMap}"
    }
}

// 2. 动态属性配置
task dynamicProperties {
    description 'Task with dynamic property configuration'
    group 'properties'

    // 根据条件配置属性

        debugMode = project.hasProperty('debug')

        println "Build number: ${buildNumber}"
    }

// 3. 条件属性配置
task conditionalProperties {
    description 'Task with conditional property configuration'
    group 'properties'

    // 条件化启用
    onlyIf {
        def shouldRun = project.hasProperty('enableConditional')
        println "Conditional task will run: $shouldRun"
        return shouldRun
    }

    // 条件化配置
    if (project.hasProperty('fastMode')) {
        ext.maxRetries = 1
        ext.timeout = 5
    } else {
        ext.maxRetries = 5
        ext.timeout = 60
    }
```

```groovy
// 3. 重试机制
task retryMechanism {
    description 'Task with retry mechanism'
    group 'failure-handling'

    ext {
        maxRetries = 3
        retryCount = 0
    }

    doLast {
        def success = false
        def attempts = 0

        while (attempts <= maxRetries && !success) {
            attempts++
            retryCount = attempts

            try {
                println "Attempt $attempts of $maxRetries"

                // 模拟可能失败的操作
                if (project.hasProperty('retryFailure') && attempts < 3) {
                    throw new RuntimeException("Attempt $attempts failed")
                }

                println "Operation succeeded on attempt $attempts"
                success = true

            } catch (Exception e) {
                println "Attempt $attempts failed: ${e.message}"

                if (attempts <= maxRetries) {
                    println "Retrying in 1 second..."
                    Thread.sleep(1000)
                } else {
                    println "Max retries exceeded, giving up"
                    throw e
                }
            }
        }
    }
}

// 5. 优雅关闭
task gracefulShutdown {
    description 'Task with graceful shutdown handling'
    group 'failure-handling'

    ext {
        shutdownRequested = false
        cleanupResources = []
    }

    doFirst {
        // 注册关闭钩子
        Runtime.runtime.addShutdownHook {
            println "Shutdown hook triggered"
            shutdownRequested = true

            // 执行清理
            cleanupResources.each { resource ->
                println "Cleaning up: $resource"
            }
        }

        println "Task started, shutdown hook registered"
    }

    doLast {
        try {
            // 模拟长时间运行的任务
            (1..10).each { i ->
                if (shutdownRequested) {
                    println "Shutdown requested, stopping gracefully"
                    return
                }

                println "Processing step $i"
                cleanupResources << "resource-$i"
                Thread.sleep(500)
            }

            if (!shutdownRequested) {
                println "Task completed normally"
            }

        } catch (InterruptedException e) {
            println "Task interrupted, shutting down gracefully"
        } finally {
            println "Final cleanup completed"
        }
    }
}
```
