## 🌐 HTTP协议基础

### HTTP协议概述

HTTP（HyperText Transfer Protocol，超文本传输协议）是Web通信的基础协议，也是Android应用网络通信的核心。Android应用主要通过HTTP/1.1和HTTP/2协议与服务器进行数据交换。

#### HTTP/1.1 vs HTTP/2

```java
// HTTP/1.1 特点
// - 每个请求需要独立的TCP连接
// - 请求头冗余
// - 队头阻塞问题

// HTTP/2 特点
// - 多路复用：单个TCP连接可以同时处理多个请求
// - 头部压缩：HPACK算法减少冗余数据
// - 二进制分帧：更高效的数据传输
// - 服务器推送：服务器可以主动推送资源
```

## 📊 网络状态检测

### ConnectivityManager工具类

```java
// NetworkUtils.java
import android.content.Context;
import android.net.ConnectivityManager;
import android.net.Network;
import android.net.NetworkCapabilities;
import android.net.NetworkInfo;
import android.net.NetworkRequest;
import android.os.Build;

import androidx.annotation.NonNull;
import androidx.annotation.RequiresApi;

import io.reactivex.rxjava3.core.Observable;
import io.reactivex.rxjava3.subjects.BehaviorSubject;

public class NetworkUtils {
    private final Context context;
    private final ConnectivityManager connectivityManager;
    private final BehaviorSubject<NetworkState> networkStateSubject;

    public NetworkUtils(Context context) {
        this.context = context.getApplicationContext();
        this.connectivityManager =
            (ConnectivityManager) this.context.getSystemService(Context.CONNECTIVITY_SERVICE);
        this.networkStateSubject = BehaviorSubject.create();

        if (connectivityManager != null) {
            registerNetworkCallback();
        }
    }

    // 网络状态枚举
    public enum NetworkState {
        CONNECTED_WIFI("WiFi连接"),
        CONNECTED_MOBILE("移动网络连接"),
        CONNECTED_ETHERNET("以太网连接"),
        CONNECTING("连接中"),
        DISCONNECTED("未连接"),
        UNKNOWN("未知状态");

        private final String description;

        NetworkState(String description) {
            this.description = description;
        }

        public String getDescription() {
            return description;
        }
    }

    // 检查网络是否连接
    public boolean isNetworkConnected() {
        if (connectivityManager == null) {
            return false;
        }

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            Network network = connectivityManager.getActiveNetwork();
            if (network == null) {
                return false;
            }

            NetworkCapabilities capabilities =
                connectivityManager.getNetworkCapabilities(network);
            return capabilities != null &&
                   (capabilities.hasTransport(NetworkCapabilities.TRANSPORT_WIFI) ||
                    capabilities.hasTransport(NetworkCapabilities.TRANSPORT_CELLULAR) ||
                    capabilities.hasTransport(NetworkCapabilities.TRANSPORT_ETHERNET));
        } else {
            // 兼容旧版本
            NetworkInfo networkInfo = connectivityManager.getActiveNetworkInfo();
            return networkInfo != null && networkInfo.isConnected();
        }
    }

    // 获取当前网络状态
    public NetworkState getCurrentNetworkState() {
        if (!isNetworkConnected()) {
            return NetworkState.DISCONNECTED;
        }

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            Network network = connectivityManager.getActiveNetwork();
            if (network != null) {
                NetworkCapabilities capabilities =
                    connectivityManager.getNetworkCapabilities(network);
                if (capabilities != null) {
                    if (capabilities.hasTransport(NetworkCapabilities.TRANSPORT_WIFI)) {
                        return NetworkState.CONNECTED_WIFI;
                    } else if (capabilities.hasTransport(NetworkCapabilities.TRANSPORT_CELLULAR)) {
                        return NetworkState.CONNECTED_MOBILE;
                    } else if (capabilities.hasTransport(NetworkCapabilities.TRANSPORT_ETHERNET)) {
                        return NetworkState.CONNECTED_ETHERNET;
                    }
                }
            }
        } else {
            // 兼容旧版本
            NetworkInfo networkInfo = connectivityManager.getActiveNetworkInfo();
            if (networkInfo != null && networkInfo.isConnected()) {
                switch (networkInfo.getType()) {
                    case ConnectivityManager.TYPE_WIFI:
                        return NetworkState.CONNECTED_WIFI;
                    case ConnectivityManager.TYPE_MOBILE:
                        return NetworkState.CONNECTED_MOBILE;
                    case ConnectivityManager.TYPE_ETHERNET:
                        return NetworkState.CONNECTED_ETHERNET;
                }
            }
        }

        return NetworkState.UNKNOWN;
    }

    // 网络状态观察
    public Observable<NetworkState> observeNetworkState() {
        return networkStateSubject.distinctUntilChanged();
    }

    // 注册网络状态回调
    @RequiresApi(api = Build.VERSION_CODES.N)
    private void registerNetworkCallback() {
        NetworkRequest request = new NetworkRequest.Builder()
            .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
            .addTransportType(NetworkCapabilities.TRANSPORT_WIFI)
            .addTransportType(NetworkCapabilities.TRANSPORT_CELLULAR)
            .addTransportType(NetworkCapabilities.TRANSPORT_ETHERNET)
            .build();

        ConnectivityManager.NetworkCallback callback =
            new ConnectivityManager.NetworkCallback() {
            @Override
            public void onAvailable(@NonNull Network network) {
                networkStateSubject.onNext(getCurrentNetworkState());
            }

            @Override
            public void onLost(@NonNull Network network) {
                networkStateSubject.onNext(NetworkState.DISCONNECTED);
            }

            @Override
            public void onCapabilitiesChanged(@NonNull Network network,
                                             @NonNull NetworkCapabilities capabilities) {
                networkStateSubject.onNext(getCurrentNetworkState());
            }
        };

        connectivityManager.registerNetworkCallback(request, callback);
    }

    // 检查网络是否可用（能够访问互联网）
    public boolean isNetworkAvailable() {
        return isNetworkConnected() && hasInternetAccess();
    }

    // 检查是否有互联网访问权限
    private boolean hasInternetAccess() {
        try {
            // 使用ping或其他方式检查
            Process process = Runtime.getRuntime().exec("ping -c 1 8.8.8.8");
            return process.waitFor() == 0;
        } catch (Exception e) {
            return false;
        }
    }

    // 获取网络信息
    public NetworkInfo getNetworkInfo() {
        if (connectivityManager == null) {
            return null;
        }

        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.M) {
            return connectivityManager.getActiveNetworkInfo();
        }

        return null; // Android M及以上版本推荐使用NetworkCapabilities
    }
}
```

## 🗂️ 缓存策略实现

### HTTP缓存配置

```java
// CacheManager.java
import android.content.Context;
import android.util.Log;

import okhttp3.Cache;
import okhttp3.Interceptor;
import okhttp3.OkHttpClient;
import okhttp3.Request;
import okhttp3.Response;

import java.io.File;
import java.io.IOException;
import java.util.concurrent.TimeUnit;

public class CacheManager {
    private static final String TAG = "CacheManager";
    private static final long CACHE_SIZE = 10 * 1024 * 1024; // 10MB缓存
    private static final long CACHE_MAX_AGE = 60 * 60; // 1小时缓存
    private static final long CACHE_MAX_STALE = 60 * 60 * 24; // 24小时陈旧数据

    private final Context context;
    private final OkHttpClient httpClient;

    public CacheManager(Context context) {
        this.context = context.getApplicationContext();
        this.httpClient = createOkHttpClientWithCache();
    }

    // 创建带缓存的OkHttpClient
    private OkHttpClient createOkHttpClientWithCache() {
        // 创建缓存目录
        File cacheDir = new File(context.getCacheDir(), "http_cache");
        if (!cacheDir.exists()) {
            cacheDir.mkdirs();
        }

        Cache cache = new Cache(cacheDir, CACHE_SIZE);

        return new OkHttpClient.Builder()
            .cache(cache)
            .addInterceptor(cacheControlInterceptor)
            .addNetworkInterceptor(networkCacheInterceptor)
            .connectTimeout(10, TimeUnit.SECONDS)
            .readTimeout(30, TimeUnit.SECONDS)
            .writeTimeout(10, TimeUnit.SECONDS)
            .build();
    }

    // 应用层缓存控制拦截器
    private final Interceptor cacheControlInterceptor = chain -> {
        Request request = chain.request();

        // 如果网络不可用，强制使用缓存
        if (!isNetworkConnected()) {
            Log.d(TAG, "网络不可用，使用缓存");
            request = request.newBuilder()
                .header("Cache-Control", "public, only-if-cached, max-stale=" + CACHE_MAX_STALE)
                .build();
        }

        Response response = chain.proceed(request);

        // 缓存控制
        if (isNetworkConnected()) {
            int maxAge = 60; // 在线时缓存1分钟
            response = response.newBuilder()
                .header("Cache-Control", "public, max-age=" + maxAge)
                .removeHeader("Pragma")
                .build();
        } else {
            // 离线时使用缓存
            int maxStale = 60 * 60 * 24 * 7; // 离线时缓存7天
            response = response.newBuilder()
                .header("Cache-Control", "public, only-if-cached, max-stale=" + maxStale)
                .removeHeader("Pragma")
                .build();
        }

        return response;
    };

    // 网络层缓存控制拦截器
    private final Interceptor networkCacheInterceptor = chain -> {
        Response response = chain.proceed(chain.request());

        // 从服务器响应头读取缓存策略
        String cacheControl = response.header("Cache-Control");
        if (cacheControl == null || cacheControl.contains("no-store") ||
            cacheControl.contains("no-cache") || cacheControl.contains("must-revalidate")) {
            // 不缓存
            return response.newBuilder()
                .header("Cache-Control", "no-store")
                .build();
        }

        return response;
    };

    // 清除缓存
    public void clearCache() {
        try {
            httpClient.cache().evictAll();
            Log.d(TAG, "缓存已清除");
        } catch (IOException e) {
            Log.e(TAG, "清除缓存失败", e);
        }
    }

    // 获取缓存大小
    public long getCacheSize() {
        return httpClient.cache().size();
    }

    // 获取OkHttpClient实例
    public OkHttpClient getHttpClient() {
        return httpClient;
    }

    // 检查网络连接
    private boolean isNetworkConnected() {
        NetworkUtils networkUtils = new NetworkUtils(context);
        return networkUtils.isNetworkConnected();
    }
}
```

### HTTPS安全配置

```java
// SecurityManager.java
import android.content.Context;
import android.util.Log;

import java.security.KeyManagementException;
import java.security.NoSuchAlgorithmException;
import java.security.cert.CertificateException;
import java.security.cert.X509Certificate;
import java.util.Arrays;
import java.util.concurrent.TimeUnit;

import javax.net.ssl.HostnameVerifier;
import javax.net.ssl.SSLContext;
import javax.net.ssl.SSLSession;
import javax.net.ssl.SSLSocketFactory;
import javax.net.ssl.TrustManager;
import javax.net.ssl.X509TrustManager;

import okhttp3.OkHttpClient;
import okhttp3.TlsVersion;

public class SecurityManager {
    private static final String TAG = "SecurityManager";
    private final Context context;

    public SecurityManager(Context context) {
        this.context = context.getApplicationContext();
    }

    // 创建安全的OkHttpClient
    public OkHttpClient createSecureHttpClient() {
        try {
            // 创建SSLContext
            SSLContext sslContext = SSLContext.getInstance("TLS");

            // 创建信任管理器
            X509TrustManager trustManager = createTrustManager();

            // 初始化SSLContext
            sslContext.init(null, new TrustManager[]{trustManager}, null);

            // 创建SSLSocketFactory
            SSLSocketFactory sslSocketFactory = sslContext.getSocketFactory();

            return new OkHttpClient.Builder()
                .sslSocketFactory(sslSocketFactory, trustManager)
                .hostnameVerifier(createHostnameVerifier())
                .connectTimeout(10, TimeUnit.SECONDS)
                .readTimeout(30, TimeUnit.SECONDS)
                .writeTimeout(10, TimeUnit.SECONDS)
                .build();

        } catch (NoSuchAlgorithmException | KeyManagementException e) {
            Log.e(TAG, "创建安全HTTP客户端失败", e);
            return createDefaultHttpClient();
        }
    }

    // 创建信任管理器
    private X509TrustManager createTrustManager() {
        return new X509TrustManager() {
            @Override
            public void checkClientTrusted(X509Certificate[] chain, String authType)
                throws CertificateException {
                // 生产环境应该验证客户端证书
            }

            @Override
            public void checkServerTrusted(X509Certificate[] chain, String authType)
                throws CertificateException {
                // 生产环境应该验证服务器证书
                if (!isValidCertificateChain(chain)) {
                    throw new CertificateException("无效的证书链");
                }
            }

            @Override
            public X509Certificate[] getAcceptedIssuers() {
                return new X509Certificate[0];
            }
        };
    }

    // 验证证书链
    private boolean isValidCertificateChain(X509Certificate[] chain) {
        // 这里应该实现证书链验证逻辑
        // 生产环境需要检查证书的颁发者、有效期、域名等
        try {
            for (X509Certificate cert : chain) {
                cert.checkValidity(); // 检查证书有效期

                // 检查证书主题（域名）
                String cn = cert.getSubjectX500Principal().getName();
                if (!isValidDomain(cn)) {
                    return false;
                }
            }
            return true;
        } catch (Exception e) {
            Log.e(TAG, "证书验证失败", e);
            return false;
        }
    }

    // 验证域名
    private boolean isValidDomain(String cn) {
        // 简单的域名验证，生产环境需要更严格的验证
        return cn != null &&
               (cn.contains("example.com") || cn.contains("trusted-cdn.com"));
    }

    // 创建主机名验证器
    private HostnameVerifier createHostnameVerifier() {
        return new HostnameVerifier() {
            @Override
            public boolean verify(String hostname, SSLSession session) {
                // 生产环境应该验证主机名
                Log.d(TAG, "验证主机名: " + hostname);

                // 开发环境可以放宽限制
                if (BuildConfig.DEBUG) {
                    return true;
                }

                // 生产环境严格验证
                return isValidHostname(hostname);
            }
        };
    }

    // 验证主机名
    private boolean isValidHostname(String hostname) {
        // 允许的主机名列表
        String[] allowedHosts = {
            "api.example.com",
            "secure.example.com",
            "cdn.example.com"
        };

        return Arrays.asList(allowedHosts).contains(hostname.toLowerCase());
    }

    // 创建默认HTTP客户端
    private OkHttpClient createDefaultHttpClient() {
        return new OkHttpClient.Builder()
            .connectTimeout(10, TimeUnit.SECONDS)
            .readTimeout(30, TimeUnit.SECONDS)
            .writeTimeout(10, TimeUnit.SECONDS)
            .build();
    }

    // 检查是否为安全连接
    public static boolean isSecureConnection(String url) {
        return url != null && url.toLowerCase().startsWith("https://");
    }

    // 检查SSL/TLS版本
    public static boolean isSecureTlsVersion(String protocol) {
        return protocol != null &&
               (protocol.equals("TLSv1.2") || protocol.equals("TLSv1.3"));
    }
}
```

### 网络安全检查工具

```java
// NetworkSecurityChecker.java
import android.content.Context;
import android.content.pm.PackageManager;
import android.util.Log;

import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.util.Arrays;
import java.util.HashSet;
import java.util.Set;

public class NetworkSecurityChecker {
    private static final String TAG = "NetworkSecurityChecker";
    private static final Set<String> INSECURE_PROTOCOLS = new HashSet<>(Arrays.asList(
        "HTTP", "FTP", "TELNET", "SMTP"
    ));

    private static final Set<String> SECURE_PROTOCOLS = new HashSet<>(Arrays.asList(
        "HTTPS", "FTPS", "SFTP"
    ));

    // 检查URL安全性
    public static SecurityResult checkUrlSecurity(String url) {
        SecurityResult result = new SecurityResult(url);

        if (url == null || url.isEmpty()) {
            result.setSecure(false);
            result.setMessage("URL为空");
            return result;
        }

        try {
            String protocol = url.split(":")[0].toUpperCase();

            if (SECURE_PROTOCOLS.contains(protocol)) {
                result.setSecure(true);
                result.setMessage("使用安全协议: " + protocol);
            } else if (INSECURE_PROTOCOLS.contains(protocol)) {
                result.setSecure(false);
                result.setMessage("使用不安全协议: " + protocol);
            } else {
                result.setSecure(false);
                result.setMessage("未知协议: " + protocol);
            }

            // 检查URL中是否包含敏感信息
            if (containsSensitiveInfo(url)) {
                result.setSecure(false);
                result.setMessage("URL包含敏感信息: " + result.getMessage());
            }

        } catch (Exception e) {
            result.setSecure(false);
            result.setMessage("URL安全检查失败: " + e.getMessage());
        }

        return result;
    }

    // 检查是否包含敏感信息
    private static boolean containsSensitiveInfo(String url) {
        String[] sensitiveKeywords = {
            "password", "token", "secret", "key", "auth",
            "credit", "card", "ssn", "personal"
        };

        String lowerUrl = url.toLowerCase();
        for (String keyword : sensitiveKeywords) {
            if (lowerUrl.contains(keyword)) {
                Log.w(TAG, "URL包含敏感信息: " + keyword);
                return true;
            }
        }

        return false;
    }

    // 检查应用权限
    public static SecurityResult checkAppPermissions(Context context) {
        SecurityResult result = new SecurityResult("AppPermissions");
        PackageManager pm = context.getPackageManager();

        try {
            // 检查网络权限
            boolean hasInternetPermission = pm.checkPermission(
                "android.permission.INTERNET",
                context.getPackageName()
            ) == PackageManager.PERMISSION_GRANTED;

            boolean hasNetworkStatePermission = pm.checkPermission(
                "android.permission.ACCESS_NETWORK_STATE",
                context.getPackageName()
            ) == PackageManager.PERMISSION_GRANTED;

            result.setSecure(hasInternetPermission && hasNetworkStatePermission);

            StringBuilder message = new StringBuilder();
            message.append("网络权限: ").append(hasInternetPermission ? "已授予" : "未授予").append(", ");
            message.append("网络状态权限: ").append(hasNetworkStatePermission ? "已授予" : "未授予");

            result.setMessage(message.toString());

        } catch (Exception e) {
            result.setSecure(false);
            result.setMessage("权限检查失败: " + e.getMessage());
        }

        return result;
    }

    // 生成安全哈希
    public static String generateSecurityHash(String data) {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] hash = digest.digest(data.getBytes());

            StringBuilder hexString = new StringBuilder();
            for (byte b : hash) {
                String hex = Integer.toHexString(0xff & b);
                if (hex.length() == 1) {
                    hexString.append('0');
                }
                hexString.append(hex);
            }

            return hexString.toString();

        } catch (NoSuchAlgorithmException e) {
            Log.e(TAG, "生成安全哈希失败", e);
            return null;
        }
    }

    // 安全检查结果
    public static class SecurityResult {
        private final String checkType;
        private boolean isSecure;
        private String message;

        public SecurityResult(String checkType) {
            this.checkType = checkType;
            this.isSecure = true;
            this.message = "检查通过";
        }

        public String getCheckType() {
            return checkType;
        }

        public boolean isSecure() {
            return isSecure;
        }

        public void setSecure(boolean secure) {
            isSecure = secure;
        }

        public String getMessage() {
            return message;
        }

        public void setMessage(String message) {
            this.message = message;
        }

        @Override
        public String toString() {
            return String.format("%s: %s - %s", checkType, isSecure ? "安全" : "不安全", message);
        }
    }
}
```

### 网络请求优化策略

```java
// NetworkOptimizer.java
import android.content.Context;
import android.util.Log;

import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

public class NetworkOptimizer {
    private static final String TAG = "NetworkOptimizer";
    private static final int MAX_CONCURRENT_REQUESTS = 5;
    private static final long REQUEST_TIMEOUT = 30; // 秒

    private final ExecutorService networkExecutor;
    private final RequestQueue requestQueue;
    private final Context context;

    public NetworkOptimizer(Context context) {
        this.context = context.getApplicationContext();
        this.networkExecutor = Executors.newFixedThreadPool(MAX_CONCURRENT_REQUESTS);
        this.requestQueue = new RequestQueue();
    }

    // 请求队列管理
    private static class RequestQueue {
        private final PriorityQueue<NetworkRequest> queue;

        public RequestQueue() {
            this.queue = new PriorityQueue<>((r1, r2) -> {
                // 优先级：高 > 中 > 低
                return Integer.compare(r2.getPriority().getValue(), r1.getPriority().getValue());
            });
        }

        public void enqueue(NetworkRequest request) {
            queue.offer(request);
        }

        public NetworkRequest dequeue() {
            return queue.poll();
        }

        public boolean isEmpty() {
            return queue.isEmpty();
        }
    }

    // 网络请求优先级
    public enum RequestPriority {
        HIGH(3),
        MEDIUM(2),
        LOW(1);

        private final int value;

        RequestPriority(int value) {
            this.value = value;
        }

        public int getValue() {
            return value;
        }
    }

    // 网络请求封装
    public static class NetworkRequest {
        private final String url;
        private final RequestPriority priority;
        private final long timestamp;
        private final boolean requiresNetwork;

        public NetworkRequest(String url, RequestPriority priority, boolean requiresNetwork) {
            this.url = url;
            this.priority = priority;
            this.timestamp = System.currentTimeMillis();
            this.requiresNetwork = requiresNetwork;
        }

        public String getUrl() {
            return url;
        }

        public RequestPriority getPriority() {
            return priority;
        }

        public long getTimestamp() {
            return timestamp;
        }

        public boolean isRequiresNetwork() {
            return requiresNetwork;
        }
    }

    // 执行网络请求
    public void executeRequest(NetworkRequest request) {
        if (request.isRequiresNetwork()) {
            NetworkUtils networkUtils = new NetworkUtils(context);
            if (!networkUtils.isNetworkConnected()) {
                Log.w(TAG, "网络不可用，取消请求: " + request.getUrl());
                return;
            }
        }

        networkExecutor.execute(() -> {
            long startTime = System.currentTimeMillis();

            try {
                // 执行请求逻辑
                executeRequestInternal(request);

                long duration = System.currentTimeMillis() - startTime;
                Log.d(TAG, String.format("请求完成: %s, 耗时: %dms",
                    request.getUrl(), duration));

            } catch (Exception e) {
                Log.e(TAG, "请求失败: " + request.getUrl(), e);

                // 错误重试逻辑
                if (shouldRetry(request, e)) {
                    retryRequest(request);
                }
            }
        });
    }

    // 执行请求内部逻辑
    private void executeRequestInternal(NetworkRequest request) {
        // 这里实现具体的请求逻辑
        Log.d(TAG, "执行请求: " + request.getUrl());

        // 模拟请求延迟
        try {
            Thread.sleep(1000);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }

    // 判断是否需要重试
    private boolean shouldRetry(NetworkRequest request, Exception error) {
        // 根据错误类型判断是否重试
        if (error instanceof java.net.SocketTimeoutException) {
            return true;
        }

        if (error instanceof java.net.ConnectException) {
            return request.getPriority() == RequestPriority.HIGH;
        }

        return false;
    }

    // 重试请求
    private void retryRequest(NetworkRequest request) {
        Log.d(TAG, "重试请求: " + request.getUrl());

        // 延迟重试
        networkExecutor.schedule(() -> {
            if (request.isRequiresNetwork()) {
                NetworkUtils networkUtils = new NetworkUtils(context);
                if (networkUtils.isNetworkConnected()) {
                    executeRequest(request);
                }
            } else {
                executeRequest(request);
            }
        }, 2, TimeUnit.SECONDS);
    }

    // 关闭优化器
    public void shutdown() {
        networkExecutor.shutdown();
        try {
            if (!networkExecutor.awaitTermination(10, TimeUnit.SECONDS)) {
                networkExecutor.shutdownNow();
            }
        } catch (InterruptedException e) {
            networkExecutor.shutdownNow();
            Thread.currentThread().interrupt();
        }
    }
}
```

```java
// NetworkPerformanceMonitor.java
import android.util.Log;

import java.util.HashMap;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicLong;

public class NetworkPerformanceMonitor {
    private static final String TAG = "NetworkPerformanceMonitor";
    private static NetworkPerformanceMonitor instance;

    private final Map<String, RequestStats> requestStatsMap;
    private final AtomicLong totalRequests;
    private final AtomicLong failedRequests;
    private final AtomicLong totalResponseTime;

    private NetworkPerformanceMonitor() {
        this.requestStatsMap = new ConcurrentHashMap<>();
        this.totalRequests = new AtomicLong(0);
        this.failedRequests = new AtomicLong(0);
        this.totalResponseTime = new AtomicLong(0);
    }

    public static NetworkPerformanceMonitor getInstance() {
        if (instance == null) {
            instance = new NetworkPerformanceMonitor();
        }
        return instance;
    }

    // 记录请求开始
    public void recordRequestStart(String requestId, String url) {
        RequestStats stats = new RequestStats(requestId, url);
        stats.setStartTime(System.currentTimeMillis());
        requestStatsMap.put(requestId, stats);
        totalRequests.incrementAndGet();
    }

    // 记录请求成功
    public void recordRequestSuccess(String requestId, int responseCode, long responseSize) {
        RequestStats stats = requestStatsMap.get(requestId);
        if (stats != null) {
            stats.setEndTime(System.currentTimeMillis());
            stats.setResponseCode(responseCode);
            stats.setResponseSize(responseSize);
            stats.setSuccess(true);

            long responseTime = stats.getEndTime() - stats.getStartTime();
            totalResponseTime.addAndGet(responseTime);

            Log.d(TAG, String.format("请求成功: %s, 状态码: %d, 响应时间: %dms, 响应大小: %d bytes",
                requestId, responseCode, responseTime, responseSize));
        }
    }

    // 记录请求失败
    public void recordRequestFailure(String requestId, String error) {
        RequestStats stats = requestStatsMap.get(requestId);
        if (stats != null) {
            stats.setEndTime(System.currentTimeMillis());
            stats.setError(error);
            stats.setSuccess(false);

            failedRequests.incrementAndGet();

            Log.e(TAG, String.format("请求失败: %s, 错误: %s", requestId, error));
        }
    }

    // 获取性能统计
    public PerformanceStats getPerformanceStats() {
        PerformanceStats stats = new PerformanceStats();

        stats.setTotalRequests(totalRequests.get());
        stats.setFailedRequests(failedRequests.get());
        stats.setTotalResponseTime(totalResponseTime.get());

        if (stats.getTotalRequests() > 0) {
            stats.setSuccessRate((stats.getTotalRequests() - stats.getFailedRequests()) * 100.0 / stats.getTotalRequests());
            stats.setAverageResponseTime(stats.getTotalResponseTime() / stats.getTotalRequests());
        }

        // 按URL统计
        Map<String, UrlStats> urlStatsMap = new HashMap<>();
        for (RequestStats requestStats : requestStatsMap.values()) {
            String url = requestStats.getUrl();
            UrlStats urlStats = urlStatsMap.computeIfAbsent(url, k -> new UrlStats(url));
            urlStats.addRequestStats(requestStats);
        }

        stats.setUrlStatsMap(urlStatsMap);

        return stats;
    }

    // 清理旧的统计数据
    public void cleanupOldStats(long cutoffTime) {
        requestStatsMap.entrySet().removeIf(entry ->
            entry.getValue().getStartTime() < cutoffTime);
    }

    // 请求统计数据
    public static class RequestStats {
        private final String requestId;
        private final String url;
        private long startTime;
        private long endTime;
        private int responseCode;
        private long responseSize;
        private boolean success;
        private String error;

        public RequestStats(String requestId, String url) {
            this.requestId = requestId;
            this.url = url;
            this.startTime = System.currentTimeMillis();
        }

        // Getters and Setters
        public long getResponseTime() {
            return endTime - startTime;
        }

        public String getRequestId() { return requestId; }
        public String getUrl() { return url; }
        public long getStartTime() { return startTime; }
        public long getEndTime() { return endTime; }
        public int getResponseCode() { return responseCode; }
        public long getResponseSize() { return responseSize; }
        public boolean isSuccess() { return success; }
        public String getError() { return error; }

        public void setStartTime(long startTime) { this.startTime = startTime; }
        public void setEndTime(long endTime) { this.endTime = endTime; }
        public void setResponseCode(int responseCode) { this.responseCode = responseCode; }
        public void setResponseSize(long responseSize) { this.responseSize = responseSize; }
        public void setSuccess(boolean success) { this.success = success; }
        public void setError(String error) { this.error = error; }
    }

    // URL统计
    public static class UrlStats {
        private final String url;
        private long totalRequests;
        private long successfulRequests;
        private long totalResponseTime;
        private long totalResponseSize;

        public UrlStats(String url) {
            this.url = url;
        }

        public void addRequestStats(RequestStats stats) {
            totalRequests++;
            if (stats.isSuccess()) {
                successfulRequests++;
                totalResponseTime += stats.getResponseTime();
                totalResponseSize += stats.getResponseSize();
            }
        }

        // Getters
        public String getUrl() { return url; }
        public long getTotalRequests() { return totalRequests; }
        public long getSuccessfulRequests() { return successfulRequests; }
        public double getSuccessRate() {
            return totalRequests > 0 ? (successfulRequests * 100.0 / totalRequests) : 0;
        }
        public long getAverageResponseTime() {
            return successfulRequests > 0 ? (totalResponseTime / successfulRequests) : 0;
        }
        public long getAverageResponseSize() {
            return successfulRequests > 0 ? (totalResponseSize / successfulRequests) : 0;
        }
    }

    // 性能统计
    public static class PerformanceStats {
        private long totalRequests;
        private long failedRequests;
        private long totalResponseTime;
        private double successRate;
        private long averageResponseTime;

        public void setFailedRequests(long failedRequests) { this.failedRequests = failedRequests; }

    }
```
