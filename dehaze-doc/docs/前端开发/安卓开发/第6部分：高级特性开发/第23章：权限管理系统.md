## 📱 运行时权限申请

### 权限请求的最佳实践

运行时权限是Android 6.0引入的重要特性，需要开发者仔细处理权限请求流程，提供良好的用户体验。

```java
// RuntimePermissionManager.java
import android.Manifest;
import android.app.Activity;
import android.content.Context;
import android.content.pm.PackageManager;
import android.os.Build;
import android.util.Log;

import androidx.activity.result.ActivityResultLauncher;
import androidx.activity.result.contract.ActivityResultContracts;
import androidx.appcompat.app.AlertDialog;
import androidx.core.app.ActivityCompat;
import androidx.core.content.ContextCompat;

import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class RuntimePermissionManager {
    private static final String TAG = "RuntimePermissionManager";

    // 权限请求结果回调
    public interface PermissionCallback {
        void onPermissionsGranted(List<String> grantedPermissions);
        void onPermissionsDenied(List<String> deniedPermissions);
        void onPermissionsPermanentlyDenied(List<String> permanentlyDeniedPermissions);
        void onPermissionError(String error);
    }

    // 权限请求配置
    public static class PermissionRequest {
        private final String[] permissions;
        private final String rationale;
        private final int requestCode;
        private final PermissionCallback callback;

        public PermissionRequest(String[] permissions, String rationale,
                              int requestCode, PermissionCallback callback) {
            this.permissions = permissions;
            this.rationale = rationale;
            this.requestCode = requestCode;
            this.callback = callback;
        }

        public String[] getPermissions() { return permissions; }
        public String getRationale() { return rationale; }
        public int getRequestCode() { return requestCode; }
        public PermissionCallback getCallback() { return callback; }
    }

    // 权限管理器
    public static class PermissionManager {
        private final Activity activity;
        private final ActivityResultLauncher<String[]> permissionLauncher;
        private PermissionRequest currentRequest;

        public PermissionManager(Activity activity) {
            this.activity = activity;
            this.permissionLauncher = activity.registerForActivityResult(
                new ActivityResultContracts.RequestMultiplePermissions(),
                this::handlePermissionResult
            );
        }

        // 请求权限
        public void requestPermissions(String[] permissions, String rationale,
                                     int requestCode, PermissionCallback callback) {
            if (Build.VERSION.SDK_INT < Build.VERSION_CODES.M) {
                // 低于Android 6.0，直接授予权限
                callback.onPermissionsGranted(Arrays.asList(permissions));
                return;
            }

            // 检查哪些权限还未授予
            List<String> deniedPermissions = new ArrayList<>();
            for (String permission : permissions) {
                if (ContextCompat.checkSelfPermission(activity, permission)
                    != PackageManager.PERMISSION_GRANTED) {
                    deniedPermissions.add(permission);
                }
            }

            if (deniedPermissions.isEmpty()) {
                // 所有权限都已授予
                callback.onPermissionsGranted(Arrays.asList(permissions));
                return;
            }

            // 保存当前请求
            currentRequest = new PermissionRequest(
                permissions, rationale, requestCode, callback);

            // 检查是否需要显示权限说明
            boolean shouldShowRationale = false;
            for (String permission : deniedPermissions) {
                if (ActivityCompat.shouldShowRequestPermissionRationale(activity, permission)) {
                    shouldShowRationale = true;
                    break;
                }
            }

            if (shouldShowRationale) {
                // 显示权限说明对话框
                showPermissionRationaleDialog(
                    deniedPermissions.toArray(new String[0]), rationale);
            } else {
                // 直接请求权限
                permissionLauncher.launch(deniedPermissions.toArray(new String[0]));
            }
        }

        // 显示权限说明对话框
        private void showPermissionRationaleDialog(String[] permissions, String rationale) {
            new AlertDialog.Builder(activity)
                .setTitle("权限请求")
                .setMessage(rationale)
                .setPositiveButton("授权", (dialog, which) -> {
                    permissionLauncher.launch(permissions);
                })
                .setNegativeButton("取消", (dialog, which) -> {
                    if (currentRequest != null && currentRequest.getCallback() != null) {
                        currentRequest.getCallback()
                            .onPermissionsDenied(Arrays.asList(permissions));
                    }
                })
                .setCancelable(false)
                .show();
        }

        // 处理权限请求结果
        private void handlePermissionResult(java.util.Map<String, Boolean> results) {
            if (currentRequest == null || currentRequest.getCallback() == null) {
                return;
            }

            List<String> grantedPermissions = new ArrayList<>();
            List<String> deniedPermissions = new ArrayList<>();
            List<String> permanentlyDeniedPermissions = new ArrayList<>();

            for (java.util.Map.Entry<String, Boolean> entry : results.entrySet()) {
                String permission = entry.getKey();
                boolean granted = entry.getValue();

                if (granted) {
                    grantedPermissions.add(permission);
                } else {
                    deniedPermissions.add(permission);
                    // 检查是否为永久拒绝
                    if (!ActivityCompat.shouldShowRequestPermissionRationale(activity, permission)) {
                        permanentlyDeniedPermissions.add(permission);
                    }
                }
            }

            PermissionCallback callback = currentRequest.getCallback();

            // 调用相应的回调
            if (!grantedPermissions.isEmpty()) {
                callback.onPermissionsGranted(grantedPermissions);
            }

            if (!deniedPermissions.isEmpty()) {
                callback.onPermissionsDenied(deniedPermissions);
            }

            if (!permanentlyDeniedPermissions.isEmpty()) {
                callback.onPermissionsPermanentlyDenied(permanentlyDeniedPermissions);
            }

            currentRequest = null;
        }
    }

    // 批量权限请求器
    public static class BatchPermissionRequester {
        private final Activity activity;
        private final PermissionManager permissionManager;

        public BatchPermissionRequester(Activity activity) {
            this.activity = activity;
            this.permissionManager = new PermissionManager(activity);
        }

        // 请求相机权限
        public void requestCameraPermission(PermissionCallback callback) {
            String[] permissions = PermissionCompatibility.getCameraPermissions();
            String rationale = "应用需要相机权限来拍照和录制视频。请在设置中允许相机访问权限。";

            permissionManager.requestPermissions(permissions, rationale, 1001, callback);
        }

        // 请求存储权限
        public void requestStoragePermission(PermissionCallback callback) {
            String[] permissions = PermissionCompatibility.getStoragePermissions();
            String rationale = "应用需要存储权限来保存和读取文件。请在设置中允许存储访问权限。";

            if (permissions.length > 0) {
                permissionManager.requestPermissions(permissions, rationale, 1002, callback);
            } else {
                callback.onPermissionsGranted(Arrays.asList(permissions));
            }
        }

        // 请求位置权限
        public void requestLocationPermission(PermissionCallback callback) {
            String[] permissions = PermissionCompatibility.getLocationPermissions();
            String rationale = "应用需要位置权限来提供基于位置的服务。请在设置中允许位置访问权限。";

            permissionManager.requestPermissions(permissions, rationale, 1003, callback);
        }

        // 请求录音权限
        public void requestRecordAudioPermission(PermissionCallback callback) {
            String[] permissions = {Manifest.permission.RECORD_AUDIO};
            String rationale = "应用需要录音权限来录制音频。请在设置中允许麦克风访问权限。";

            permissionManager.requestPermissions(permissions, rationale, 1004, callback);
        }

        // 请求联系人权限
        public void requestContactsPermission(PermissionCallback callback) {
            String[] permissions = {
                Manifest.permission.READ_CONTACTS,
                Manifest.permission.WRITE_CONTACTS
            };
            String rationale = "应用需要联系人权限来读取和管理联系人信息。请在设置中允许联系人访问权限。";

            permissionManager.requestPermissions(permissions, rationale, 1005, callback);
        }

        // 请求电话权限
        public void requestPhonePermission(PermissionCallback callback) {
            String[] permissions = {
                Manifest.permission.READ_PHONE_STATE,
                Manifest.permission.CALL_PHONE
            };
            String rationale = "应用需要电话权限来读取通话状态和拨打电话。请在设置中允许电话访问权限。";

            permissionManager.requestPermissions(permissions, rationale, 1006, callback);
        }

        // 请求短信权限
        public void requestSmsPermission(PermissionCallback callback) {
            String[] permissions = {
                Manifest.permission.READ_SMS,
                Manifest.permission.SEND_SMS,
                Manifest.permission.RECEIVE_SMS
            };
            String rationale = "应用需要短信权限来读取和发送短信。请在设置中允许短信访问权限。";

            permissionManager.requestPermissions(permissions, rationale, 1007, callback);
        }

        // 请求日历权限
        public void requestCalendarPermission(PermissionCallback callback) {
            String[] permissions = {
                Manifest.permission.READ_CALENDAR,
                Manifest.permission.WRITE_CALENDAR
            };
            String rationale = "应用需要日历权限来读取和管理日历事件。请在设置中允许日历访问权限。";

            permissionManager.requestPermissions(permissions, rationale, 1008, callback);
        }

        // 自定义权限请求
        public void requestCustomPermissions(String[] permissions, String rationale,
                                          PermissionCallback callback) {
            permissionManager.requestPermissions(permissions, rationale, 1009, callback);
        }
    }

    // 权限状态检查器
    public static class PermissionStatusChecker {
        private final Context context;

        public PermissionStatusChecker(Context context) {
            this.context = context.getApplicationContext();
        }

        // 检查单个权限状态
        public boolean isPermissionGranted(String permission) {
            return ContextCompat.checkSelfPermission(context, permission)
                == PackageManager.PERMISSION_GRANTED;
        }

        // 检查多个权限状态
        public PermissionStatus checkPermissions(String... permissions) {
            List<String> granted = new ArrayList<>();
            List<String> denied = new ArrayList<>();

            for (String permission : permissions) {
                if (isPermissionGranted(permission)) {
                    granted.add(permission);
                } else {
                    denied.add(permission);
                }
            }

            return new PermissionStatus(granted, denied);
        }

        // 检查权限组状态
        public PermissionStatus checkPermissionGroup(String groupName) {
            String[] groupPermissions = getGroupPermissions(groupName);
            return checkPermissions(groupPermissions);
        }

        // 获取权限组对应的权限列表
        private String[] getGroupPermissions(String groupName) {
            switch (groupName) {
                case "CAMERA":
                    return new String[]{Manifest.permission.CAMERA};
                case "CONTACTS":
                    return new String[]{
                        Manifest.permission.READ_CONTACTS,
                        Manifest.permission.WRITE_CONTACTS
                    };
                case "LOCATION":
                    return new String[]{
                        Manifest.permission.ACCESS_FINE_LOCATION,
                        Manifest.permission.ACCESS_COARSE_LOCATION
                    };
                case "MICROPHONE":
                    return new String[]{Manifest.permission.RECORD_AUDIO};
                case "PHONE":
                    return new String[]{
                        Manifest.permission.READ_PHONE_STATE,
                        Manifest.permission.CALL_PHONE
                    };
                case "SMS":
                    return new String[]{
                        Manifest.permission.READ_SMS,
                        Manifest.permission.SEND_SMS,
                        Manifest.permission.RECEIVE_SMS
                    };
                case "STORAGE":
                    return PermissionCompatibility.getStoragePermissions();
                default:
                    return new String[]{};
            }
        }

        // 权限状态结果类
        public static class PermissionStatus {
            private final List<String> grantedPermissions;
            private final List<String> deniedPermissions;

            public PermissionStatus(List<String> granted, List<String> denied) {
                this.grantedPermissions = granted;
                this.deniedPermissions = denied;
            }

            public boolean allGranted() {
                return deniedPermissions.isEmpty();
            }

            public boolean hasDenied() {
                return !deniedPermissions.isEmpty();
            }

            public List<String> getGrantedPermissions() { return grantedPermissions; }
            public List<String> getDeniedPermissions() { return deniedPermissions; }

            @Override
            public String toString() {
                return String.format("PermissionStatus{granted=%d, denied=%d}",
                    grantedPermissions.size(), deniedPermissions.size());
            }
        }
    }

    // 使用示例
    public static class PermissionUsageExample {
        private Activity activity;
        private BatchPermissionRequester batchRequester;
        private PermissionStatusChecker statusChecker;

        public PermissionUsageExample(Activity activity) {
            this.activity = activity;
            this.batchRequester = new BatchPermissionRequester(activity);
            this.statusChecker = new PermissionStatusChecker(activity);
        }

        // 检查并请求相机权限
        public void checkAndRequestCameraPermission() {
            PermissionStatus status = statusChecker.checkPermissions(
                PermissionCompatibility.getCameraPermissions());

            if (status.allGranted()) {
                // 权限已授予，执行相机相关操作
                Log.d(TAG, "相机权限已授予");
                openCamera();
            } else {
                // 请求相机权限
                batchRequester.requestCameraPermission(new PermissionCallback() {
                    @Override
                    public void onPermissionsGranted(List<String> grantedPermissions) {
                        Log.d(TAG, "相机权限授予成功");
                        openCamera();
                    }

                    @Override
                    public void onPermissionsDenied(List<String> deniedPermissions) {
                        Log.w(TAG, "相机权限被拒绝");

                    }

                    @Override
                    public void onPermissionsPermanentlyDenied(List<String> permanentlyDeniedPermissions) {
                        Log.w(TAG, "相机权限被永久拒绝");
                        showSettingsDialog();
                    }

                    @Override
                    public void onPermissionError(String error) {
                        Log.e(TAG, "相机权限请求错误: " + error);
                    }
                });
            }
        }

        private void openCamera() {
            // 实现打开相机的逻辑
            Log.d(TAG, "打开相机");
        }

        private void showPermissionDeniedMessage() {
            // 显示权限被拒绝的消息
            Log.d(TAG, "显示权限被拒绝的消息");
        }

        private void showSettingsDialog() {
            // 显示引导用户到设置页面的对话框
            Log.d(TAG, "显示设置对话框");
        }
    }
```

## 🛡️ 权限最佳实践

### 权限请求的用户体验优化

良好的权限请求体验是应用成功的关键因素之一。本节介绍如何优化权限请求流程，提供用户友好的体验。

```java
// PermissionBestPractices.java
import android.Manifest;
import android.app.Activity;
import android.content.Context;
import android.content.DialogInterface;
import android.content.Intent;
import android.content.pm.PackageManager;
import android.net.Uri;
import android.os.Build;
import android.provider.Settings;
import android.util.Log;

import androidx.appcompat.app.AlertDialog;
import androidx.core.app.ActivityCompat;
import androidx.core.content.ContextCompat;

import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class PermissionBestPractices {
    private static final String TAG = "PermissionBestPractices";

    // 权限请求策略
    public static class PermissionRequestStrategy {
        public enum RequestTiming {
            JUST_IN_TIME,     // 恰当时间请求
            ON_STARTUP,        // 应用启动时请求
            BEFORE_FEATURE,   // 功能使用前请求
            BATCH_REQUEST      // 批量请求
        }

        public enum RequestPriority {
            CRITICAL,         // 关键权限
            IMPORTANT,        // 重要权限
            OPTIONAL          // 可选权限
        }

        public enum RequestFrequency {
            ONCE,             // 只请求一次

        // 判断是否应该请求权限
        public boolean shouldRequestPermission(String permission, String feature) {
            // 检查权限是否已授予
            if (ContextCompat.checkSelfPermission(context, permission)
                == PackageManager.PERMISSION_GRANTED) {
                return false;
            }

            // 检查权限重要性
            PermissionMetadata metadata = getPermissionMetadata(permission);
            if (metadata == null) {
                return true; // 未知权限，保守处理
            }

            // 关键权限必须请求
            if (metadata.getPriority() == PermissionRequestStrategy.RequestPriority.CRITICAL) {
                return true;
            }

            // 重要权限在相关功能使用时请求
            if (metadata.getPriority() == PermissionRequestStrategy.RequestPriority.IMPORTANT &&
                feature != null && metadata.getRelatedFeatures().contains(feature)) {
                return true;
            }

            return false;
        }

        // 权限请求统计
        public static class PermissionRequestStats {
            private final String permission;
            private int requestCount;
            private int grantCount;
            private int denyCount;
            private int permanentDenyCount;
            private long lastRequestTime;

            public PermissionRequestStats(String permission) {
                this.permission = permission;
                this.requestCount = 0;
                this.grantCount = 0;
                this.denyCount = 0;
                this.permanentDenyCount = 0;
                this.lastRequestTime = 0;
            }

            // 记录权限请求
            public void recordRequest(boolean granted, boolean permanentlyDenied) {
                requestCount++;
                lastRequestTime = System.currentTimeMillis();

                if (granted) {
                    grantCount++;
                } else {
                    denyCount++;
                    if (permanentlyDenied) {
                        permanentDenyCount++;
                    }
                }
            }

            // 计算授权率
            public double getGrantRate() {
                return requestCount > 0 ? (double) grantCount / requestCount * 100 : 0;
            }

            // 计算永久拒绝率
            public double getPermanentDenyRate() {
                return requestCount > 0 ? (double) permanentDenyCount / requestCount * 100 : 0;
            }

            // Getters
            public String getPermission() { return permission; }
            public int getRequestCount() { return requestCount; }
            public int getGrantCount() { return grantCount; }
            public int getDenyCount() { return denyCount; }
            public int getPermanentDenyCount() { return permanentDenyCount; }
            public long getLastRequestTime() { return lastRequestTime; }

            @Override
            public String toString() {
                return String.format(
                    "PermissionStats{permission=%s, requests=%d, grants=%d, denies=%d, permanent_denies=%d, grant_rate=%.1f%%}",
                    permission, requestCount, grantCount, denyCount, permanentDenyCount, getGrantRate()
                );
            }
        }

        // 生成权限请求建议
        public static String generateOptimizationAdvice(Map<String, PermissionRequestStats> statsMap) {
            StringBuilder advice = new StringBuilder();
            advice.append("权限请求优化建议：\n");

            for (Map.Entry<String, PermissionRequestStats> entry : statsMap.entrySet()) {
                String permission = entry.getKey();
                PermissionRequestStats stats = entry.getValue();

                if (stats.getGrantRate() < 50) {
                    advice.append(String.format("• %s: 授权率较低(%.1f%%)，建议改进权限说明和请求时机\n",
                        permission, stats.getGrantRate()));
                }

                if (stats.getPermanentDenyRate() > 30) {
                    advice.append(String.format("• %s: 永久拒绝率较高(%.1f%%)，建议提前教育用户或优化功能设计\n",
                        permission, stats.getPermanentDenyRate()));
                }
            }

            return advice.toString();
        }
```

```java
            boolean hasCoarseLocation = ContextCompat.checkSelfPermission(context,

            } else if (hasFineLocation) {

            } else if (hasCoarseLocation) {
                return LocationPermissionStatus.COARSE_ONLY;

                return LocationPermissionStatus.NONE_GRANTED;
            }
```
