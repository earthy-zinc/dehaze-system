# 第24章：多媒体应用开发

## 📷 Camera2 API使用

### Camera2架构概述

Camera2是Android 5.0引入的新一代相机API，相比过时的Camera API提供了更强大和灵活的相机控制能力。

```java
// Camera2Manager.java
import android.annotation.SuppressLint;
import android.content.Context;
import android.content.pm.PackageManager;
import android.graphics.ImageFormat;
import android.graphics.SurfaceTexture;
import android.hardware.camera2.CameraAccessException;
import android.hardware.camera2.CameraCharacteristics;
import android.hardware.camera2.CameraDevice;
import android.hardware.camera2.CameraManager;
import android.hardware.camera2.CaptureRequest;
import android.hardware.camera2.CaptureResult;
import android.hardware.camera2.TotalCaptureResult;
import android.hardware.camera2.params.StreamConfigurationMap;
import android.media.Image;
import android.media.ImageReader;
import android.os.Handler;
import android.os.HandlerThread;
import android.util.Log;
import android.util.Size;
import android.view.Surface;
import android.view.TextureView;

import androidx.annotation.NonNull;
import androidx.core.app.ActivityCompat;

import java.util.Arrays;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;

public class Camera2Manager {
    private static final String TAG = "Camera2Manager";

    // Camera2状态枚举
    public enum CameraState {
        CLOSED,          // 相机已关闭
        OPENED,          // 相机已打开
        PREPARING,       // 正在准备
        READY,           // 相机就绪
        CAPTURING,       // 正在拍照
        FOCUSING,        // 正在对焦
        ERROR           // 错误状态
    }

    // 相机回调接口
    public interface CameraCallback {
        void onCameraOpened();
        void onCameraClosed();
        void onCameraError(String error);
        void onCaptureCompleted(Image image);
        void onCaptureFailed(String error);
        void onFocusComplete(boolean success);
        void onPreviewStarted();
    }

    // 成员变量
    private final Context context;
    private final CameraManager cameraManager;
    private final CameraCallback callback;
    private final Handler backgroundHandler;
    private final HandlerThread backgroundThread;

    private CameraDevice cameraDevice;
    private CameraCharacteristics cameraCharacteristics;
    private CaptureRequest.Builder previewRequestBuilder;
    private CaptureRequest captureRequest;
    private CameraCaptureSession cameraCaptureSession;
    private ImageReader imageReader;
    private String cameraId;
    private CameraState currentState;

    public Camera2Manager(Context context, CameraCallback callback) {
        this.context = context;
        this.callback = callback;
        this.cameraManager = (CameraManager) context.getSystemService(Context.CAMERA_SERVICE);
        this.backgroundThread = new HandlerThread("Camera2Background");
        this.backgroundThread.start();
        this.backgroundHandler = new Handler(backgroundThread.getLooper());
        this.currentState = CameraState.CLOSED;
    }

    // 打开相机
    @SuppressLint("MissingPermission")
    public void openCamera() {
        if (currentState != CameraState.CLOSED) {
            Log.w(TAG, "相机已打开或正在打开");
            return;
        }

        if (!hasCameraPermission()) {
            callback.onCameraError("缺少相机权限");
            return;
        }

        try {
            setState(CameraState.PREPARING);

            // 获取后置相机ID
            cameraId = getCameraId(false);
            if (cameraId == null) {
                // 如果没有后置相机，获取前置相机
                cameraId = getCameraId(true);
                if (cameraId == null) {
                    callback.onCameraError("没有可用的相机");
                    return;
                }
            }

            // 获取相机特性
            cameraCharacteristics = cameraManager.getCameraCharacteristics(cameraId);

            // 打开相机设备
            cameraManager.openCamera(cameraId, cameraStateCallback, backgroundHandler);

        } catch (CameraAccessException e) {
            Log.e(TAG, "打开相机失败", e);
            callback.onCameraError("打开相机失败: " + e.getMessage());
        }
    }

    // 相机状态回调
    private final CameraDevice.StateCallback cameraStateCallback = new CameraDevice.StateCallback() {
        @Override
        public void onOpened(@NonNull CameraDevice camera) {
            Log.d(TAG, "相机设备已打开");
            cameraDevice = camera;
            setState(CameraState.OPENED);
            callback.onCameraOpened();

            // 开始预览
            startPreview();
        }

        @Override
        public void onDisconnected(@NonNull CameraDevice camera) {
            Log.w(TAG, "相机设备已断开连接");
            camera.close();
            cameraDevice = null;
            setState(CameraState.CLOSED);
            callback.onCameraClosed();
        }

        @Override
        public void onError(@NonNull CameraDevice camera, int error) {
            Log.e(TAG, "相机设备错误: " + error);
            camera.close();
            cameraDevice = null;
            setState(CameraState.ERROR);
            callback.onCameraError("相机错误: " + getErrorMessage(error));
        }

        @Override
        public void onClosed(@NonNull CameraDevice camera) {
            super.onClosed(camera);
            Log.d(TAG, "相机设备已关闭");
            setState(CameraState.CLOSED);
            callback.onCameraClosed();
        }
    };

    // 开始预览
    public void startPreview(TextureView textureView) {
        if (cameraDevice == null || currentState == CameraState.CLOSED) {
            Log.w(TAG, "相机设备未打开");
            return;
        }

        try {
            // 设置纹理视图的监听器
            SurfaceTexture surfaceTexture = textureView.getSurfaceTexture();
            if (surfaceTexture == null) {
                Log.w(TAG, "纹理视图未准备好");
                return;
            }

            // 设置预览尺寸
            Size previewSize = getOptimalPreviewSize(textureView.getWidth(), textureView.getHeight());
            surfaceTexture.setDefaultBufferSize(previewSize.getWidth(), previewSize.getHeight());

            Surface previewSurface = new Surface(surfaceTexture);

            // 设置拍照用的ImageReader
            setupImageReader();

            // 创建预览请求
            previewRequestBuilder = cameraDevice.createCaptureRequest(CameraDevice.TEMPLATE_PREVIEW);
            previewRequestBuilder.addTarget(previewSurface);

            // 设置自动对焦
            setupAutoFocus(previewRequestBuilder);

            // 创建相机会话
            cameraDevice.createCaptureSession(
                Arrays.asList(previewSurface, imageReader.getSurface()),
                sessionStateCallback,
                backgroundHandler
            );

        } catch (CameraAccessException e) {
            Log.e(TAG, "创建预览会话失败", e);
            callback.onCameraError("创建预览失败: " + e.getMessage());
        }
    }

    // 相机会话状态回调
    private final CameraCaptureSession.StateCallback sessionStateCallback =
        new CameraCaptureSession.StateCallback() {

        @Override
        public void onConfigured(@NonNull CameraCaptureSession session) {
            Log.d(TAG, "相机会话已配置");
            cameraCaptureSession = session;
            setState(CameraState.READY);

            try {
                // 开始重复预览
                previewRequest = previewRequestBuilder.build();
                cameraCaptureSession.setRepeatingRequest(previewRequest,
                    captureCallback, backgroundHandler);
                callback.onPreviewStarted();

            } catch (CameraAccessException e) {
                Log.e(TAG, "开始预览失败", e);
                callback.onCameraError("开始预览失败: " + e.getMessage());
            }
        }

        @Override
        public void onConfigureFailed(@NonNull CameraCaptureSession session) {
            Log.e(TAG, "相机会话配置失败");
            callback.onCameraError("相机会话配置失败");
        }

        @Override
        public void onReady(@NonNull CameraCaptureSession session) {
            super.onReady(session);
            Log.d(TAG, "相机会话已就绪");
        }

        @Override
        public void onActive(@NonNull CameraCaptureSession session) {
            super.onActive(session);
            Log.d(TAG, "相机会话已激活");
        }

        @Override
        public void onClosed(@NonNull CameraCaptureSession session) {
            super.onClosed(session);
            Log.d(TAG, "相机会话已关闭");
        }
    };

    // 捕获回调
    private final CameraCaptureSession.CaptureCallback captureCallback =
        new CameraCaptureSession.CaptureCallback() {

        @Override
        public void onCaptureProgressed(@NonNull CameraCaptureSession session,
                                       @NonNull CaptureRequest request,
                                       @NonNull CaptureResult partialResult) {
            // 捕获进度更新
            checkFocusResult(partialResult);
        }

        @Override
        public void onCaptureCompleted(@NonNull CameraCaptureSession session,
                                     @NonNull CaptureRequest request,
                                     @NonNull TotalCaptureResult result) {
            // 捕获完成
            checkFocusResult(result);
        }

        @Override
        public void onCaptureFailed(@NonNull CameraCaptureSession session,
                                   @NonNull CaptureRequest request,
                                   @NonNull CaptureFailure failure) {
            Log.e(TAG, "捕获失败: " + failure.getReason());
            callback.onCaptureError("捕获失败: " + failure.getReason());
        }

        private void checkFocusResult(CaptureResult result) {
            Integer afState = result.get(CaptureResult.CONTROL_AF_STATE);
            if (afState != null) {
                boolean focused = afState == CaptureResult.CONTROL_AF_STATE_FOCUSED_LOCKED ||
                               afState == CaptureResult.CONTROL_AF_STATE_PASSIVE_FOCUSED;
                callback.onFocusComplete(focused);
            }
        }
    };

    // 拍照
    public void takePicture() {
        if (cameraDevice == null || cameraCaptureSession == null) {
            callback.onCameraError("相机未准备好");
            return;
        }

        if (currentState == CameraState.CAPTURING) {
            Log.w(TAG, "正在拍照中，请稍后");
            return;
        }

        try {
            setState(CameraState.CAPTURING);

            // 创建拍照请求
            final CaptureRequest.Builder captureBuilder =
                cameraDevice.createCaptureRequest(CameraDevice.TEMPLATE_STILL_CAPTURE);

            captureBuilder.addTarget(imageReader.getSurface());

            // 设置自动对焦
            captureBuilder.set(CaptureRequest.CONTROL_AF_MODE,
                CaptureRequest.CONTROL_AF_MODE_AUTO_PICTURE);

            // 设置曝光模式
            captureBuilder.set(CaptureRequest.CONTROL_AE_MODE,
                CaptureRequest.CONTROL_AE_MODE_ON_AUTO_FLASH);

            // 停止预览
            cameraCaptureSession.stopRepeating();

            // 执行拍照
            cameraCaptureSession.capture(captureBuilder.build(),
                new CameraCaptureSession.CaptureCallback() {

                    @Override
                    public void onCaptureCompleted(@NonNull CameraCaptureSession session,
                                                 @NonNull CaptureRequest request,
                                                 @NonNull TotalCaptureResult result) {
                        Log.d(TAG, "拍照完成");
                        // 恢复预览
                        restartPreview();
                    }

                    @Override
                    public void onCaptureFailed(@NonNull CameraCaptureSession session,
                                               @NonNull CaptureRequest request,
                                               @NonNull CaptureFailure failure) {
                        Log.e(TAG, "拍照失败: " + failure.getReason());
                        callback.onCaptureFailed("拍照失败: " + failure.getReason());
                        setState(CameraState.READY);
                        // 恢复预览
                        restartPreview();
                    }
                }, backgroundHandler);

        } catch (CameraAccessException e) {
            Log.e(TAG, "拍照失败", e);
            callback.onCaptureFailed("拍照失败: " + e.getMessage());
            setState(CameraState.READY);
        }
    }

    // 设置ImageReader
    private void setupImageReader() {
        Size largestImageSize = getLargestImageSize();
        imageReader = ImageReader.newInstance(largestImageSize.getWidth(),
                                           largestImageSize.getHeight(),
                                           ImageFormat.JPEG, 1);

        imageReader.setOnImageAvailableListener(reader -> {
            Image image = reader.acquireLatestImage();
            if (image != null) {
                callback.onCaptureCompleted(image);
                image.close(); // 记得关闭Image
            }
        }, backgroundHandler);
    }

    // 设置自动对焦
    private void setupAutoFocus(CaptureRequest.Builder requestBuilder) {
        // 检查相机是否支持自动对焦
        Integer[] afModes = cameraCharacteristics.get(
            CameraCharacteristics.CONTROL_AF_AVAILABLE_MODES);

        if (afModes != null && afModes.length > 0) {
            for (int mode : afModes) {
                if (mode == CaptureRequest.CONTROL_AF_MODE_AUTO ||
                    mode == CaptureRequest.CONTROL_AF_MODE_CONTINUOUS_PICTURE) {
                    requestBuilder.set(CaptureRequest.CONTROL_AF_MODE, mode);
                    break;
                }
            }
        }
    }

    // 恢复预览
    private void restartPreview() {
        if (cameraCaptureSession == null || previewRequestBuilder == null) {
            return;
        }

        try {
            cameraCaptureSession.setRepeatingRequest(previewRequest,
                captureCallback, backgroundHandler);
            setState(CameraState.READY);
        } catch (CameraAccessException e) {
            Log.e(TAG, "恢复预览失败", e);
            callback.onCameraError("恢复预览失败: " + e.getMessage());
        }
    }

    // 手动对焦
    public void focusAtPoint(float x, float y) {
        if (cameraCaptureSession == null) {
            return;
        }

        try {
            // 创建对焦区域
            android.graphics.Rect focusArea = new android.graphics.Rect(
                (int) (x - 1000), (int) (y - 1000),
                (int) (x + 1000), (int) (y + 1000));

            // 创建对焦请求
            CaptureRequest.Builder focusRequestBuilder =
                cameraDevice.createCaptureRequest(CameraDevice.TEMPLATE_PREVIEW);

            focusRequestBuilder.set(CaptureRequest.CONTROL_AF_REGIONS,
                new android.hardware.camera2.params.MeteringRectangle[]{
                    new android.hardware.camera2.params.MeteringRectangle(focusArea, 1)
                });

            focusRequestBuilder.set(CaptureRequest.CONTROL_AF_MODE,
                CaptureRequest.CONTROL_AF_MODE_AUTO);

            focusRequestBuilder.addTarget(previewRequestBuilder.getTargets().iterator().next());

            // 执行对焦
            cameraCaptureSession.capture(focusRequestBuilder.build(),
                new CameraCaptureSession.CaptureCallback() {

                    @Override
                    public void onCaptureCompleted(@NonNull CameraCaptureSession session,
                                                 @NonNull CaptureRequest request,
                                                 @NonNull TotalCaptureResult result) {
                        // 对焦完成
                    }
                }, backgroundHandler);

        } catch (CameraAccessException e) {
            Log.e(TAG, "手动对焦失败", e);
        }
    }

    // 切换前后摄像头
    public void switchCamera() {
        if (cameraDevice != null) {
            closeCamera();
            // 获取相反的相机ID
            cameraId = cameraId.contains("front") ? getCameraId(false) : getCameraId(true);
            if (cameraId != null) {
                openCamera();
            }
        }
    }

    // 获取最优预览尺寸
    private Size getOptimalPreviewSize(int textureViewWidth, int textureViewHeight) {
        try {
            StreamConfigurationMap map = cameraCharacteristics.get(
                CameraCharacteristics.SCALER_STREAM_CONFIGURATION_MAP);

            if (map == null) {
                return new Size(1920, 1080); // 默认尺寸
            }

            Size[] outputSizes = map.getOutputSizes(SurfaceTexture.class);

            // 根据视图尺寸计算合适的预览尺寸
            int targetWidth = textureViewWidth;
            int targetHeight = textureViewHeight;

            // 寻找最接近目标尺寸的预览尺寸
            Size optimalSize = null;
            double minDiff = Double.MAX_VALUE;

            for (Size size : outputSizes) {
                double ratio = (double) size.getWidth() / size.getHeight();
                double targetRatio = (double) targetWidth / targetHeight;
                double diff = Math.abs(ratio - targetRatio);

                if (diff < minDiff) {
                    minDiff = diff;
                    optimalSize = size;
                }
            }

            return optimalSize != null ? optimalSize : outputSizes[0];

        } catch (Exception e) {
            Log.e(TAG, "获取预览尺寸失败", e);
            return new Size(1920, 1080);
        }
    }

    // 获取最大图片尺寸
    private Size getLargestImageSize() {
        try {
            StreamConfigurationMap map = cameraCharacteristics.get(
                CameraCharacteristics.SCALER_STREAM_CONFIGURATION_MAP);

            if (map == null) {
                return new Size(1920, 1080);
            }

            Size[] outputSizes = map.getOutputSizes(ImageFormat.JPEG);

            // 返回最大尺寸
            return outputSizes[outputSizes.length - 1];

        } catch (Exception e) {
            Log.e(TAG, "获取图片尺寸失败", e);
            return new Size(1920, 1080);
        }
    }

    // 获取相机ID
    private String getCameraId(boolean frontFacing) {
        try {
            String[] cameraIds = cameraManager.getCameraIdList();

            for (String id : cameraIds) {
                CameraCharacteristics characteristics =
                    cameraManager.getCameraCharacteristics(id);

                Integer facing = characteristics.get(CameraCharacteristics.LENS_FACING);

                if (facing != null) {
                    if (frontFacing && facing == CameraCharacteristics.LENS_FACING_FRONT) {
                        return id;
                    } else if (!frontFacing && facing == CameraCharacteristics.LENS_FACING_BACK) {
                        return id;
                    }
                }
            }
        } catch (CameraAccessException e) {
            Log.e(TAG, "获取相机ID失败", e);
        }

        return null;
    }

    // 检查相机权限
    private boolean hasCameraPermission() {
        return ActivityCompat.checkSelfPermission(context,
            android.Manifest.permission.CAMERA) == PackageManager.PERMISSION_GRANTED;
    }

    // 获取错误消息
    private String getErrorMessage(int error) {
        switch (error) {
            case CameraDevice.StateCallback.ERROR_CAMERA_IN_USE:
                return "相机正在被其他应用使用";
            case CameraDevice.StateCallback.ERROR_MAX_CAMERAS_IN_USE:
                return "已达到最大相机使用数量";
            case CameraDevice.StateCallback.ERROR_CAMERA_DISABLED:
                return "相机被禁用";
            case CameraDevice.StateCallback.ERROR_CAMERA_DEVICE:
                return "相机设备错误";
            case CameraDevice.StateCallback.ERROR_CAMERA_SERVICE:
                return "相机服务错误";
            default:
                return "未知相机错误: " + error;
        }
    }

    // 设置状态
    private void setState(CameraState state) {
        CameraState oldState = currentState;
        currentState = state;
        Log.d(TAG, "相机状态变化: " + oldState + " -> " + state);
    }

    // 获取当前状态
    public CameraState getCurrentState() {
        return currentState;
    }

    // 获取相机信息
    public CameraInfo getCameraInfo() {
        if (cameraCharacteristics == null) {
            return null;
        }

        try {
            CameraInfo info = new CameraInfo();
            info.cameraId = cameraId;

            Integer facing = cameraCharacteristics.get(CameraCharacteristics.LENS_FACING);
            info.isFrontFacing = facing != null && facing == CameraCharacteristics.LENS_FACING_FRONT;

            Float[] focalLengths = cameraCharacteristics.get(CameraCharacteristics.LENS_INFO_AVAILABLE_FOCAL_LENGTHS);
            if (focalLengths != null && focalLengths.length > 0) {
                info.focalLength = focalLengths[0];
            }

            Size[] sensorSizes = cameraCharacteristics.get(CameraCharacteristics.SENSOR_INFO_PHYSICAL_SIZE);
            if (sensorSizes != null && sensorSizes.length > 0) {
                info.sensorSize = sensorSizes[0];
            }

            return info;
        } catch (Exception e) {
            Log.e(TAG, "获取相机信息失败", e);
            return null;
        }
    }

    // 相机信息类
    public static class CameraInfo {
        public String cameraId;
        public boolean isFrontFacing;
        public float focalLength;
        public Size sensorSize;

        @Override
        public String toString() {
            return String.format("CameraInfo{id=%s, front=%s, focal=%.2f, sensor=%dx%d}",
                cameraId, isFrontFacing, focalLength,
                sensorSize != null ? sensorSize.getWidth() : 0,
                sensorSize != null ? sensorSize.getHeight() : 0);
        }
    }

    // 关闭相机
    public void closeCamera() {
        if (cameraCaptureSession != null) {
            cameraCaptureSession.close();
            cameraCaptureSession = null;
        }

        if (cameraDevice != null) {
            cameraDevice.close();
            cameraDevice = null;
        }

        if (imageReader != null) {
            imageReader.close();
            imageReader = null;
        }

        setState(CameraState.CLOSED);
    }

    // 释放资源
    public void release() {
        closeCamera();

        if (backgroundThread != null) {
            backgroundThread.quitSafely();
        }
    }
}
```

```java
// AudioRecorderManager.java
import android.content.Context;
import android.media.AudioFormat;
import android.media.AudioManager;
import android.media.AudioRecord;
import android.media.MediaRecorder;
import android.media.MediaScannerConnection;
import android.os.Environment;
import android.util.Log;

import java.io.File;
import java.io.FileOutputStream;
import java.io.IOException;

public class AudioRecorderManager {
    private static final String TAG = "AudioRecorderManager";

    // 音频格式配置
    public static class AudioConfig {
        public static final int SAMPLE_RATE = 44100; // 采样率
        public static final int CHANNEL_CONFIG = AudioFormat.CHANNEL_IN_MONO; // 单声道
        public static final int AUDIO_FORMAT = AudioFormat.ENCODING_PCM_16BIT; // 16位PCM
        public static final int BITRATE = 128000; // 比特率

        // 获取音频缓冲区大小
        public static int getBufferSize() {
            return AudioRecord.getMinBufferSize(SAMPLE_RATE, CHANNEL_CONFIG, AUDIO_FORMAT);
        }

        // 获取录制格式
        public static int getOutputFormat() {
            return MediaRecorder.OutputFormat.MPEG_4;
        }

        // 获取音频编码器
        public static int getAudioEncoder() {
            return MediaRecorder.AudioEncoder.AAC;
        }
    }

    // 录音状态
    public enum RecordingState {
        IDLE,       // 空闲
        PREPARING,  // 准备中
        RECORDING,  // 录制中
        PAUSED,     // 暂停
        STOPPED,    // 已停止
        ERROR       // 错误
    }

    // 录音回调接口
    public interface RecordingCallback {
        void onRecordingStarted();
        void onRecordingStopped();
        void onRecordingPaused();
        void onRecordingResumed();
        void onRecordingError(String error);
        void onRecordingProgress(long durationMs, double amplitude);
        void onFileSaved(String filePath);
    }

    // 成员变量
    private final Context context;
    private final RecordingCallback callback;
    private MediaRecorder mediaRecorder;
    private AudioRecord audioRecord;
    private Thread recordingThread;
    private volatile boolean isRecording;
    private volatile boolean isPaused;
    private String currentFilePath;
    private RecordingState currentState;
    private long recordingStartTime;
    private long pausedDuration;

    public AudioRecorderManager(Context context, RecordingCallback callback) {
        this.context = context;
        this.callback = callback;
        this.currentState = RecordingState.IDLE;
    }

    // 准备录音
    public boolean prepareRecording() {
        if (currentState == RecordingState.PREPARING || currentState == RecordingState.RECORDING) {
            Log.w(TAG, "录音已在进行中");
            return false;
        }

        try {
            setState(RecordingState.PREPARING);

            // 创建MediaRecorder
            mediaRecorder = new MediaRecorder();

            // 设置音频源
            mediaRecorder.setAudioSource(MediaRecorder.AudioSource.MIC);

            // 设置输出格式
            mediaRecorder.setOutputFormat(AudioConfig.getOutputFormat());

            // 设置音频编码器
            mediaRecorder.setAudioEncoder(AudioConfig.getAudioEncoder());

            // 设置输出文件路径
            currentFilePath = generateAudioFilePath();
            mediaRecorder.setOutputFile(currentFilePath);

            // 设置比特率
            mediaRecorder.setAudioEncodingBitRate(AudioConfig.BITRATE);

            // 设置采样率
            mediaRecorder.setAudioSamplingRate(AudioConfig.SAMPLE_RATE);

            // 准备录音
            mediaRecorder.prepare();

            setState(RecordingState.IDLE);
            Log.d(TAG, "录音准备完成");
            return true;

        } catch (IOException e) {
            Log.e(TAG, "录音准备失败", e);
            setState(RecordingState.ERROR);
            callback.onRecordingError("录音准备失败: " + e.getMessage());
            return false;
        }
    }

    // 开始录音
    public boolean startRecording() {
        if (currentState != RecordingState.IDLE) {
            Log.w(TAG, "录音状态不正确: " + currentState);
            return false;
        }

        if (mediaRecorder == null) {
            callback.onRecordingError("录音器未准备");
            return false;
        }

        try {
            setState(RecordingState.RECORDING);
            recordingStartTime = System.currentTimeMillis();
            pausedDuration = 0;

            mediaRecorder.start();
            isRecording = true;
            isPaused = false;

            // 启动录音监控线程
            startRecordingMonitor();

            callback.onRecordingStarted();
            Log.d(TAG, "开始录音");
            return true;

        } catch (Exception e) {
            Log.e(TAG, "开始录音失败", e);
            setState(RecordingState.ERROR);
            callback.onRecordingError("开始录音失败: " + e.getMessage());
            return false;
        }
    }

    // 停止录音
    public boolean stopRecording() {
        if (currentState != RecordingState.RECORDING && currentState != RecordingState.PAUSED) {
            Log.w(TAG, "录音未在进行中");
            return false;
        }

        try {
            setState(RecordingState.STOPPED);
            isRecording = false;
            isPaused = false;

            if (mediaRecorder != null) {
                mediaRecorder.stop();
                mediaRecorder.reset();
            }

            // 停止监控线程
            stopRecordingMonitor();

            // 扫描文件
            scanAudioFile(currentFilePath);

            callback.onRecordingStopped();
            callback.onFileSaved(currentFilePath);

            Log.d(TAG, "录音已停止，文件: " + currentFilePath);
            return true;

        } catch (Exception e) {
            Log.e(TAG, "停止录音失败", e);
            setState(RecordingState.ERROR);
            callback.onRecordingError("停止录音失败: " + e.getMessage());
            return false;
        }
    }

    // 暂停录音
    public boolean pauseRecording() {
        if (currentState != RecordingState.RECORDING) {
            Log.w(TAG, "录音未在进行中");
            return false;
        }

        try {
            setState(RecordingState.PAUSED);
            isPaused = true;
            pausedDuration += System.currentTimeMillis() - recordingStartTime;

            if (mediaRecorder != null) {
                mediaRecorder.pause();
            }

            callback.onRecordingPaused();
            Log.d(TAG, "录音已暂停");
            return true;

        } catch (Exception e) {
            Log.e(TAG, "暂停录音失败", e);
            callback.onRecordingError("暂停录音失败: " + e.getMessage());
            return false;
        }
    }

    // 恢复录音
    public boolean resumeRecording() {
        if (currentState != RecordingState.PAUSED) {
            Log.w(TAG, "录音未暂停");
            return false;
        }

        try {
            setState(RecordingState.RECORDING);
            isPaused = false;
            recordingStartTime = System.currentTimeMillis();

            if (mediaRecorder != null) {
                mediaRecorder.resume();
            }

            callback.onRecordingResumed();
            Log.d(TAG, "录音已恢复");
            return true;

        } catch (Exception e) {
            Log.e(TAG, "恢复录音失败", e);
            callback.onRecordingError("恢复录音失败: " + e.getMessage());
            return false;
        }
    }

    // 启动录音监控线程
    private void startRecordingMonitor() {
        recordingThread = new Thread(() -> {
            while (isRecording) {
                if (!isPaused) {
                    try {
                        // 计算录音时长
                        long currentDuration = System.currentTimeMillis() - recordingStartTime + pausedDuration;

                        // 获取音量振幅
                        double amplitude = getCurrentAmplitude();

                        // 通知进度
                        callback.onRecordingProgress(currentDuration, amplitude);

                        Thread.sleep(100); // 每100ms更新一次

                    } catch (InterruptedException e) {
                        Thread.currentThread().interrupt();
                        break;
                    }
                } else {
                    try {
                        Thread.sleep(200);
                    } catch (InterruptedException e) {
                        Thread.currentThread().interrupt();
                        break;
                    }
                }
            }
        });

        recordingThread.start();
    }

    // 停止录音监控线程
    private void stopRecordingMonitor() {
        if (recordingThread != null) {
            isRecording = false;
            recordingThread.interrupt();
            try {
                recordingThread.join(1000);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
            recordingThread = null;
        }
    }

    // 获取当前音量振幅
    private double getCurrentAmplitude() {
        if (audioRecord == null) {
            return 0;
        }

        short[] buffer = new short[AudioConfig.getBufferSize()];
        int readSize = audioRecord.read(buffer, 0, buffer.length);

        if (readSize > 0) {
            double sum = 0;
            for (int i = 0; i < readSize; i++) {
                sum += Math.abs(buffer[i]);
            }
            return sum / readSize;
        }

        return 0;
    }

    // 生成音频文件路径
    private String generateAudioFilePath() {
        File audioDir = new File(context.getExternalFilesDir(Environment.DIRECTORY_MUSIC), "recordings");
        if (!audioDir.exists()) {
            audioDir.mkdirs();
        }

        String fileName = "audio_" + System.currentTimeMillis() + ".mp4";
        return new File(audioDir, fileName).getAbsolutePath();
    }

    // 扫描音频文件
    private void scanAudioFile(String filePath) {
        MediaScannerConnection.scanFile(
            context.getApplicationContext(),
            new String[]{filePath},
            new String[]{"audio/mp4"},
            (path, uri) -> {
                Log.d(TAG, "音频文件扫描完成: " + path);
            }
        );
    }

    // 获取录音时长
    public long getRecordingDuration() {
        if (currentState == RecordingState.RECORDING) {
            return System.currentTimeMillis() - recordingStartTime + pausedDuration;
        } else if (currentState == RecordingState.PAUSED) {
            return pausedDuration;
        }
        return 0;
    }

    // 获取当前录音文件路径
    public String getCurrentFilePath() {
        return currentFilePath;
    }

    // 获取录音状态
    public RecordingState getState() {
        return currentState;
    }

    // 设置状态
    private void setState(RecordingState state) {
        RecordingState oldState = currentState;
        currentState = state;
        Log.d(TAG, "录音状态变化: " + oldState + " -> " + state);
    }

    // 检查是否正在录音
    public boolean isRecording() {
        return currentState == RecordingState.RECORDING;
    }

    // 检查是否暂停
    public boolean isPaused() {
        return currentState == RecordingState.PAUSED;
    }

    // 释放资源
    public void release() {
        stopRecording();

        if (mediaRecorder != null) {
            mediaRecorder.release();
            mediaRecorder = null;
        }

        if (audioRecord != null) {
            audioRecord.release();
            audioRecord = null;
        }

        setState(RecordingState.IDLE);
    }

// AudioPlayerManager.java
import android.content.Context;
import android.media.AudioAttributes;
import android.media.AudioFocusRequest;
import android.media.AudioManager;
import android.media.MediaPlayer;
import android.media.PlaybackParams;
import android.net.Uri;
import android.os.Build;
import android.os.Handler;
import android.os.Looper;
import android.util.Log;

import java.io.IOException;

public class AudioPlayerManager {
    private static final String TAG = "AudioPlayerManager";

    // 播放状态
    public enum PlaybackState {
        IDLE,       // 空闲
        PREPARING,  // 准备中
        PREPARED,   // 已准备
        PLAYING,    // 播放中
        PAUSED,     // 暂停
        STOPPED,    // 已停止
        COMPLETED,  // 播放完成
        ERROR       // 错误
    }

    // 播放回调接口
    public interface PlaybackCallback {
        void onPlaybackStarted();
        void onPlaybackPaused();
        void onPlaybackStopped();
        void onPlaybackCompleted();
        void onPlaybackError(String error);
        void onPlaybackProgress(int position, int duration);
    }

    // 成员变量
    private final Context context;
    private final PlaybackCallback callback;
    private MediaPlayer mediaPlayer;
    private final AudioManager audioManager;
    private AudioFocusRequest audioFocusRequest;
    private boolean hasAudioFocus;
    private final Handler progressHandler;
    private Runnable progressRunnable;
    private PlaybackState currentState;
    private String currentFilePath;

    public AudioPlayerManager(Context context, PlaybackCallback callback) {
        this.context = context;
        this.callback = callback;
        this.audioManager = (AudioManager) context.getSystemService(Context.AUDIO_SERVICE);
        this.progressHandler = new Handler(Looper.getMainLooper());
        this.currentState = PlaybackState.IDLE;
    }

    // 准备播放
    public boolean preparePlayback(String filePath) {
        if (currentState == PlaybackState.PREPARING || currentState == PlaybackState.PLAYING) {
            Log.w(TAG, "播放器已在使用中");
            return false;
        }

        try {
            setState(PlaybackState.PREPARING);
            currentFilePath = filePath;

            // 释放之前的播放器
            if (mediaPlayer != null) {
                mediaPlayer.release();
            }

            // 创建新的播放器
            mediaPlayer = new MediaPlayer();
            setupMediaPlayerListeners();

            // 设置音频源
            mediaPlayer.setDataSource(filePath);

            // 准备播放器
            mediaPlayer.prepareAsync();

            return true;

        } catch (IOException e) {
            Log.e(TAG, "准备播放失败", e);
            setState(PlaybackState.ERROR);
            callback.onPlaybackError("准备播放失败: " + e.getMessage());
            return false;
        }
    }

    // 设置播放器监听器
    private void setupMediaPlayerListeners() {
        mediaPlayer.setOnPreparedListener(mp -> {
            setState(PlaybackState.PREPARED);
            Log.d(TAG, "播放器准备完成");
        });

        mediaPlayer.setOnStartListener(() -> {
            setState(PlaybackState.PLAYING);
            callback.onPlaybackStarted();
            startProgressUpdate();
            Log.d(TAG, "开始播放");
        });

        mediaPlayer.setOnPauseListener(() -> {
            setState(PlaybackState.PAUSED);
            callback.onPlaybackPaused();
            stopProgressUpdate();
            Log.d(TAG, "暂停播放");
        });

        mediaPlayer.setOnCompletionListener(mp -> {
            setState(PlaybackState.COMPLETED);
            callback.onPlaybackCompleted();
            stopProgressUpdate();
            abandonAudioFocus();
            Log.d(TAG, "播放完成");
        });

        mediaPlayer.setOnErrorListener((mp, what, extra) -> {
            Log.e(TAG, "播放错误: " + what + ", extra: " + extra);
            setState(PlaybackState.ERROR);
            callback.onPlaybackError("播放错误: " + getErrorMessage(what));
            stopProgressUpdate();
            abandonAudioFocus();
            return true;
        });
    }

    // 开始播放
    public boolean startPlayback() {
        if (currentState != PlaybackState.PREPARED && currentState != PlaybackState.PAUSED &&
            currentState != PlaybackState.STOPPED) {
            Log.w(TAG, "播放器状态不正确: " + currentState);
            return false;
        }

        if (mediaPlayer == null) {
            callback.onPlaybackError("播放器未准备");
            return false;
        }

        // 请求音频焦点
        if (!requestAudioFocus()) {
            callback.onPlaybackError("无法获得音频焦点");
            return false;
        }

        try {
            mediaPlayer.start();
            return true;
        } catch (Exception e) {
            Log.e(TAG, "开始播放失败", e);
            callback.onPlaybackError("开始播放失败: " + e.getMessage());
            return false;
        }
    }

    // 暂停播放
    public boolean pausePlayback() {
        if (currentState != PlaybackState.PLAYING) {
            Log.w(TAG, "播放器未在播放");
            return false;
        }

        try {
            mediaPlayer.pause();
            return true;
        } catch (Exception e) {
            Log.e(TAG, "暂停播放失败", e);
            callback.onPlaybackError("暂停播放失败: " + e.getMessage());
            return false;
        }
    }

    // 停止播放
    public boolean stopPlayback() {
        if (currentState == PlaybackState.IDLE || currentState == PlaybackState.STOPPED) {
            return true;
        }

        try {
            mediaPlayer.stop();
            setState(PlaybackState.STOPPED);
            stopProgressUpdate();
            abandonAudioFocus();
            callback.onPlaybackStopped();
            return true;
        } catch (Exception e) {
            Log.e(TAG, "停止播放失败", e);
            callback.onPlaybackError("停止播放失败: " + e.getMessage());
            return false;
        }
    }

    // 跳转到指定位置
    public boolean seekTo(int position) {
        if (mediaPlayer == null) {
            return false;
        }

        try {
            mediaPlayer.seekTo(position);
            return true;
        } catch (Exception e) {
            Log.e(TAG, "跳转失败", e);
            return false;
        }
    }

    // 设置音量
    public void setVolume(float leftVolume, float rightVolume) {
        if (mediaPlayer != null) {
            mediaPlayer.setVolume(leftVolume, rightVolume);
        }
    }

    // 设置循环播放
    public void setLooping(boolean looping) {
        if (mediaPlayer != null) {
            mediaPlayer.setLooping(looping);
        }
    }

    // 设置播放速度
    public void setPlaybackSpeed(float speed) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M && mediaPlayer != null) {
            PlaybackParams params = new PlaybackParams();
            params.setSpeed(speed);
            mediaPlayer.setPlaybackParams(params);
        }
    }

    // 请求音频焦点
    private boolean requestAudioFocus() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            // Android 8.0+ 使用新的音频焦点API
            AudioAttributes playbackAttributes = new AudioAttributes.Builder()
                .setUsage(AudioAttributes.USAGE_MEDIA)
                .setContentType(AudioAttributes.CONTENT_TYPE_MUSIC)
                .build();

            audioFocusRequest = new AudioFocusRequest.Builder(
                AudioManager.AUDIOFOCUS_GAIN)
                .setAudioAttributes(playbackAttributes)
                .setOnAudioFocusChangeListener(focusChangeListener)
                .build();

            int result = audioManager.requestAudioFocus(audioFocusRequest);
            hasAudioFocus = result == AudioManager.AUDIOFOCUS_REQUEST_GRANTED;
        } else {
            // 兼容旧版本
            int result = audioManager.requestAudioFocus(
                focusChangeListener,
                AudioManager.STREAM_MUSIC,
                AudioManager.AUDIOFOCUS_GAIN);
            hasAudioFocus = result == AudioManager.AUDIOFOCUS_REQUEST_GRANTED;
        }

        return hasAudioFocus;
    }

    // 放弃音频焦点
    private void abandonAudioFocus() {
        if (!hasAudioFocus) {
            return;
        }

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O && audioFocusRequest != null) {
            audioManager.abandonAudioFocusRequest(audioFocusRequest);
        } else {
            audioManager.abandonAudioFocus(focusChangeListener);
        }

        hasAudioFocus = false;
    }

    // 音频焦点监听器
    private final AudioManager.OnAudioFocusChangeListener focusChangeListener =
        new AudioManager.OnAudioFocusChangeListener() {

            @Override
            public void onAudioFocusChange(int focusChange) {
                switch (focusChange) {
                    case AudioManager.AUDIOFOCUS_GAIN:
                        // 获得音频焦点
                        Log.d(TAG, "获得音频焦点");
                        if (mediaPlayer != null) {
                            mediaPlayer.setVolume(1.0f, 1.0f);
                            if (currentState == PlaybackState.PAUSED) {
                                mediaPlayer.start();
                            }
                        }
                        break;

                    case AudioManager.AUDIOFOCUS_LOSS:
                        // 永久失去音频焦点
                        Log.d(TAG, "永久失去音频焦点");
                        if (mediaPlayer != null) {
                            mediaPlayer.stop();
                        }
                        break;

                    case AudioManager.AUDIOFOCUS_LOSS_TRANSIENT:
                        // 暂时失去音频焦点
                        Log.d(TAG, "暂时失去音频焦点");
                        if (mediaPlayer != null && currentState == PlaybackState.PLAYING) {
                            mediaPlayer.pause();
                        }
                        break;

                    case AudioManager.AUDIOFOCUS_LOSS_TRANSIENT_CAN_DUCK:
                        // 可以降低音量
                        Log.d(TAG, "降低音量处理");
                        if (mediaPlayer != null) {
                            mediaPlayer.setVolume(0.3f, 0.3f);
                        }
                        break;
                }
            }
        };

    // 开始进度更新
    private void startProgressUpdate() {
        stopProgressUpdate();

        progressRunnable = new Runnable() {
            @Override
            public void run() {
                if (mediaPlayer != null && currentState == PlaybackState.PLAYING) {
                    int position = mediaPlayer.getCurrentPosition();
                    int duration = mediaPlayer.getDuration();
                    callback.onPlaybackProgress(position, duration);

                    // 继续更新
                    progressHandler.postDelayed(this, 100);
                }
            }
        };

        progressHandler.post(progressRunnable);
    }

    // 停止进度更新
    private void stopProgressUpdate() {
        if (progressRunnable != null) {
            progressHandler.removeCallbacks(progressRunnable);
            progressRunnable = null;
        }
    }

    // 获取当前播放位置
    public int getCurrentPosition() {
        return mediaPlayer != null ? mediaPlayer.getCurrentPosition() : 0;
    }

    // 获取播放时长
    public int getDuration() {
        return mediaPlayer != null ? mediaPlayer.getDuration() : 0;
    }

    // 获取播放状态
    public PlaybackState getState() {
        return currentState;
    }

    // 检查是否正在播放
    public boolean isPlaying() {
        return mediaPlayer != null && mediaPlayer.isPlaying();
    }

    // 获取当前文件路径
    public String getCurrentFilePath() {
        return currentFilePath;
    }

    // 设置状态
    private void setState(PlaybackState state) {
        PlaybackState oldState = currentState;
        currentState = state;
        Log.d(TAG, "播放状态变化: " + oldState + " -> " + state);
    }

    // 获取错误消息
    private String getErrorMessage(int what) {
        switch (what) {
            case MediaPlayer.MEDIA_ERROR_UNKNOWN:
                return "未知媒体错误";
            case MediaPlayer.MEDIA_ERROR_SERVER_DIED:
                return "媒体服务器死亡";
            case MediaPlayer.MEDIA_ERROR_IO:
                return "IO错误";
            case MediaPlayer.MEDIA_ERROR_MALFORMED:
                return "媒体格式错误";
            case MediaPlayer.MEDIA_ERROR_UNSUPPORTED:
                return "不支持的媒体";
            case MediaPlayer.MEDIA_ERROR_TIMED_OUT:
                return "媒体超时";
            default:
                return "播放错误: " + what;
        }
    }

    // 释放资源
    public void release() {
        stopProgressUpdate();
        abandonAudioFocus();

        if (mediaPlayer != null) {
            mediaPlayer.release();
            mediaPlayer = null;
        }

        setState(PlaybackState.IDLE);
    }
}
```

```java
            mediaRecorder.setMaxDuration(60 * 60 * 1000); // 60分钟

            // 设置信息监听器
            setupInfoListener();

            // 设置错误监听器
            setupErrorListener();

            // 准备录制
            mediaRecorder.prepare();

            setState(RecordingState.IDLE);
            Log.d(TAG, "视频录制准备完成");
            return true;

        } catch (IOException e) {
            Log.e(TAG, "准备视频录制失败", e);

            return false;

            return false;

            recordingStartTime = System.currentTimeMillis();

            return false;

                    callback.onRecordingProgress(duration);
                    Thread.sleep(1000); // 每秒更新一次
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    break;
                }
            }
        });

        recordingMonitorThread.start();

        if (recordingMonitorThread != null) {
            isRecording = false;
            recordingMonitorThread.interrupt();
            try {
                recordingMonitorThread.join(1000);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
            recordingMonitorThread = null;
        }

            setState(RecordingState.ERROR);

            }

            return System.currentTimeMillis() - recordingStartTime;
        }
        return 0;

            default:
                return "录制错误: " + what;
        }
    }

        stopRecording();

        if (mediaRecorder != null) {
            mediaRecorder.release();
            mediaRecorder = null;
        }

        setState(RecordingState.IDLE);

public class VideoPlayerManager {
    private static final String TAG = "VideoPlayerManager";

    // 播放状态
    public enum PlaybackState {
        IDLE,       // 空闲
        PREPARING,  // 准备中
        PREPARED,   // 已准备
        PLAYING,    // 播放中
        PAUSED,     // 暂停
        STOPPED,    // 已停止
        COMPLETED,  // 播放完成
        ERROR       // 错误

    // 播放回调接口
    public interface PlaybackCallback {
        void onVideoPrepared();
        void onPlaybackStarted();
        void onPlaybackPaused();
        void onPlaybackStopped();
        void onPlaybackCompleted();
        void onPlaybackError(String error);
        void onPlaybackProgress(int position, int duration);
    }

    // 成员变量
    private final Context context;
    private final PlaybackCallback callback;
    private MediaPlayer mediaPlayer;
    private final Handler progressHandler;
    private Runnable progressRunnable;
    private PlaybackState currentState;
    private String currentVideoPath;
    private SurfaceHolder surfaceHolder;

    public VideoPlayerManager(Context context, PlaybackCallback callback) {
        this.context = context;
        this.callback = callback;
        this.progressHandler = new Handler(Looper.getMainLooper());
        this.currentState = PlaybackState.IDLE;

    // 设置SurfaceView
    public void setSurfaceView(SurfaceView surfaceView) {
        surfaceView.getHolder().addCallback(new SurfaceHolder.Callback() {
            @Override
            public void surfaceCreated(SurfaceHolder holder) {
                surfaceHolder = holder;

                    setSurface(surfaceHolder.getSurface());
                }
            }

            @Override
            public void surfaceChanged(SurfaceHolder holder, int format, int width, int height) {
                Log.d(TAG, "Surface尺寸变化: " + width + "x" + height);
            }

            @Override
            public void surfaceDestroyed(SurfaceHolder holder) {
                surfaceHolder = null;
                Log.d(TAG, "Surface已销毁");
                if (mediaPlayer != null) {
                    mediaPlayer.setSurface(null);
                }
            }
        });
    }

    // 设置Surface

        if (mediaPlayer != null) {
            mediaPlayer.setSurface(surface);
        }
    }

    // 准备播放
    public boolean preparePlayback(String videoPath) {
        if (currentState == PlaybackState.PREPARING || currentState == PlaybackState.PLAYING) {
            Log.w(TAG, "播放器已在使用中");
            return false;
        }

        try {
            setState(PlaybackState.PREPARING);
            currentVideoPath = videoPath;

            // 释放之前的播放器
            if (mediaPlayer != null) {
                mediaPlayer.release();
            }

            // 创建新的播放器
            mediaPlayer = new MediaPlayer();
            setupMediaPlayerListeners();

            // 设置视频源
            mediaPlayer.setDataSource(videoPath);

            // 设置Surface
            if (surfaceHolder != null) {
                mediaPlayer.setSurface(surfaceHolder.getSurface());
            }

            // 准备播放器
            mediaPlayer.prepareAsync();

            return true;

        } catch (IOException e) {
            Log.e(TAG, "准备视频播放失败", e);
            setState(PlaybackState.ERROR);
            callback.onPlaybackError("准备播放失败: " + e.getMessage());
            return false;
        }
    }

    // 设置播放器监听器
    private void setupMediaPlayerListeners() {
        mediaPlayer.setOnPreparedListener(mp -> {
            setState(PlaybackState.PREPARED);
            callback.onVideoPrepared();
            Log.d(TAG, "视频准备完成");

            setState(PlaybackState.ERROR);

            stopProgressUpdate();
            return true;

        mediaPlayer.setOnSeekCompleteListener(mp -> {
            Log.d(TAG, "跳转完成");

            Log.d(TAG, "视频尺寸变化: " + width + "x" + height);

        if (currentState != PlaybackState.PREPARED && currentState != PlaybackState.PAUSED &&
            currentState != PlaybackState.STOPPED) {

            return false;

            mediaPlayer.start();
```

```java
                    byte[] encodedData = encodeData(inputData);
                    if (encodedData != null) {
                        // 解码
                        byte[] decodedData = decodeData(encodedData);
                        if (decodedData != null) {
                            outputQueue.offer(decodedData);
                        }
                    }
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    break;
                }
            }
        }

                    return outputQueue.take();
                } catch (InterruptedException e) {

                decoder.stop();
```

```java
    // 获取媒体文件缩略图
    public Bitmap getMediaThumbnail(String filePath, int kind, int width, int height) {
        try {
            File file = new File(filePath);
            if (!file.exists()) {
                return null;
            }

            // 根据文件类型获取不同的缩略图
            if (filePath.toLowerCase().endsWith(".jpg") ||
                filePath.toLowerCase().endsWith(".jpeg") ||
                filePath.toLowerCase().endsWith(".png")) {
                // 图片缩略图
                return getBitmapFromFile(file, width, height);
            } else if (filePath.toLowerCase().endsWith(".mp4") ||
                       filePath.toLowerCase().endsWith(".3gp") ||
                       filePath.toLowerCase().endsWith(".avi")) {
                // 视频缩略图
                return getVideoThumbnail(file, width, height);
            } else if (filePath.toLowerCase().endsWith(".mp3") ||
                       filePath.toLowerCase().endsWith(".wav") ||
                       filePath.toLowerCase().endsWith(".m4a")) {
                // 音频缩略图
                return getAudioThumbnail(file, width, height);
            }

        } catch (Exception e) {
            Log.e(TAG, "获取缩略图失败", e);
        }

        return null;
    }

    // 从文件获取Bitmap
    private Bitmap getBitmapFromFile(File file, int width, int height) {
        try {
            BitmapFactory.Options options = new BitmapFactory.Options();
            options.inJustDecodeBounds = true;
            BitmapFactory.decodeFile(file, options);

            options.inSampleSize = calculateInSampleSize(options, width, height);
            options.inJustDecodeBounds = false;

            return BitmapFactory.decodeFile(file, options);
        } catch (Exception e) {
            Log.e(TAG, "获取Bitmap失败", e);
            return null;
        }
    }

    // 获取视频缩略图
    private Bitmap getVideoThumbnail(File file, int width, int height) {

            return null;
        }
    }

    // 获取音频缩略图
    private Bitmap getAudioThumbnail(File file, int width, int height) {
        try {
            MediaMetadataRetriever retriever = new MediaMetadataRetriever();
            retriever.setDataSource(MediaMetadataRetriever.DataSource.FILE_URI,
                Uri.fromFile(file));

            byte[] art = retriever.getEmbeddedPicture();
            if (art != null) {
                return BitmapFactory.decodeByteArray(art, 0, art.length);
            }

            // 如果没有封面图，返回默认的音频图标
            return null;
        } catch (Exception e) {
            Log.e(TAG, "获取音频缩略图失败", e);
            return null;
        }
    }

    // 计算采样率
    private int calculateInSampleSize(BitmapFactory.Options options, int reqWidth, int reqHeight) {
        final int height = options.outHeight;
        final int width = options.outWidth;
        int inSampleSize = 1;

        if (height > reqHeight || width > reqWidth) {
            final int halfHeight = height / 2;
            final int halfWidth = width / 2;

            while ((halfHeight / inSampleSize) >= reqHeight &&
                   (halfWidth / inSampleSize) >= reqWidth) {
                inSampleSize *= 2;
            }
        }

        return inSampleSize;
    }

    // 删除媒体文件
    public boolean deleteMediaFile(MediaFileInfo fileInfo) {
        try {
            if (fileInfo.getUri() != null) {
                // 使用MediaStore删除
                int deleted = contentResolver.delete(fileInfo.getUri(), null, null);
                if (deleted > 0) {
                    Log.d(TAG, "媒体文件已删除: " + fileInfo.getDisplayName());
                    return true;
                }
            }

            // 备用方案：直接删除文件
            if (fileInfo.getPath() != null) {
                File file = new File(fileInfo.getPath());
                if (file.exists() && file.delete()) {
                    Log.d(TAG, "文件已删除: " + file.getPath());
                    return true;
                }
            }

        } catch (Exception e) {
            Log.e(TAG, "删除媒体文件失败", e);
        }

        return false;
    }
```
