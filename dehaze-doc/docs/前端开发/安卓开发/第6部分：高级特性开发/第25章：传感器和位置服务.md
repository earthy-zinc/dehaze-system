# 第25章：传感器和位置服务

传感器和位置服务是移动设备的重要特性，为开发者提供了丰富的环境感知和位置定位能力。本章将详细介绍Android传感器框架、各类传感器的使用方法、数据处理算法以及位置服务的开发技术。

## 25.1 Android传感器框架概述

Android提供了完整的传感器框架，支持访问设备的各种传感器硬件。传感器框架采用事件驱动机制，通过监听器模式提供传感器数据。

### 25.1.1 传感器类型和分类

Android平台支持的传感器主要分为三类：

1. **运动传感器**：检测设备的运动和加速度变化
2. **位置传感器**：检测设备的位置和方向
3. **环境传感器**：检测周围环境条件

```java
// 传感器类型常量
```

```java
public class SensorManagerHelper {

        info.append("厂商: ").append(sensor.getVendor()).append("\n");
        info.append("版本: ").append(sensor.getVersion()).append("\n");
        info.append("功耗: ").append(sensor.getPower()).append(" mA\n");
        info.append("分辨率: ").append(sensor.getResolution()).append("\n");
        info.append("最大范围: ").append(sensor.getMaximumRange()).append("\n");
        info.append("最小延迟: ").append(sensor.getMinDelay()).append(" μs\n");
        info.append("是否是唤醒传感器: ").append(sensor.isWakeUpSensor()).append("\n");
        return info.toString();
    }

    /**
     * 获取特定类型的默认传感器
     */
    public Sensor getDefaultSensor(int sensorType) {
        return sensorManager.getDefaultSensor(sensorType);
    }

    /**
     * 获取延迟级别对应的微秒数
     */
    public int getDelayInMicros(int delay) {
        switch (delay) {
            case SensorManager.SENSOR_DELAY_FASTEST:
                return 0; // 最快延迟，应用不应使用
            case SensorManager.SENSOR_DELAY_GAME:
                return 20000; // 20ms，适合游戏
            case SensorManager.SENSOR_DELAY_UI:
                return 60000; // 60ms，适合UI更新
            case SensorManager.SENSOR_DELAY_NORMAL:
                return 200000; // 200ms，默认延迟
            default:
                return delay;
        }
    }

    /**
     * 检查传感器权限
     */
    public boolean hasSensorPermission(String permission) {
        return ContextCompat.checkSelfPermission(context, permission)
               == PackageManager.PERMISSION_GRANTED;
    }
}
```

## 25.2 运动传感器

运动传感器检测设备的运动状态，包括加速度、旋转和重力等信息。

### 25.2.1 加速度计

加速度计测量施加在设备上的加速度力，包括重力加速度：

```java
public class AccelerometerManager {
    private static final String TAG = "AccelerometerManager";
    private SensorManager sensorManager;
    private Sensor accelerometer;
    private AccelerometerListener listener;

    public interface AccelerometerListener {
        void onAccelerationChanged(float x, float y, float z, long timestamp);
        void onShakeDetected();
        void onTiltChanged(float tiltAngle);
    }

    private static final float SHAKE_THRESHOLD = 12.0f; // 摇晃阈值
    private static final int SHAKE_COOLDOWN = 1000; // 摇晃冷却时间（毫秒）

    private long lastShakeTime = 0;
    private float[] gravity = new float[3];
    private float[] linearAcceleration = new float[3];

    // 低通滤波器系数
    private static final float ALPHA = 0.8f;

    public AccelerometerManager(Context context, AccelerometerListener listener) {
        this.sensorManager = (SensorManager) context.getSystemService(Context.SENSOR_SERVICE);
        this.accelerometer = sensorManager.getDefaultSensor(Sensor.TYPE_ACCELEROMETER);
        this.listener = listener;
    }

    /**
     * 注册加速度计监听器
     */
    public boolean register() {
        if (accelerometer != null) {
            return sensorManager.registerListener(
                sensorEventListener,
                accelerometer,
                SensorManager.SENSOR_DELAY_NORMAL
            );
        }
        return false;
    }

    /**
     * 注销加速度计监听器
     */
    public void unregister() {
        sensorManager.unregisterListener(sensorEventListener);
    }

    private final SensorEventListener sensorEventListener = new SensorEventListener() {
        @Override
        public void onSensorChanged(SensorEvent event) {
            if (event.sensor.getType() == Sensor.TYPE_ACCELEROMETER) {
                float x = event.values[0];
                float y = event.values[1];
                float z = event.values[2];
                long timestamp = event.timestamp;

                // 应用低通滤波器分离重力
                gravity[0] = ALPHA * gravity[0] + (1 - ALPHA) * x;
                gravity[1] = ALPHA * gravity[1] + (1 - ALPHA) * y;
                gravity[2] = ALPHA * gravity[2] + (1 - ALPHA) * z;

                // 计算线性加速度（移除重力影响）
                linearAcceleration[0] = x - gravity[0];
                linearAcceleration[1] = y - gravity[1];
                linearAcceleration[2] = z - gravity[2];

                // 检测摇晃
                detectShake(linearAcceleration);

                // 计算倾斜角度
                float tiltAngle = calculateTiltAngle(x, y, z);

                // 通知监听器
                if (listener != null) {
                    listener.onAccelerationChanged(x, y, z, timestamp);
                    listener.onTiltChanged(tiltAngle);
                }

                Log.d(TAG, String.format("加速度: X=%.2f, Y=%.2f, Z=%.2f, 倾斜角=%.1f°",
                    x, y, z, tiltAngle));
            }
        }

        @Override
        public void onAccuracyChanged(Sensor sensor, int accuracy) {
            Log.d(TAG, "加速度计精度变化: " + accuracy);
        }
    };

    /**
     * 检测设备摇晃
     */
    private void detectShake(float[] acceleration) {
        float magnitude = (float) Math.sqrt(
            acceleration[0] * acceleration[0] +
            acceleration[1] * acceleration[1] +
            acceleration[2] * acceleration[2]
        );

        long currentTime = System.currentTimeMillis();

        if (magnitude > SHAKE_THRESHOLD && (currentTime - lastShakeTime) > SHAKE_COOLDOWN) {
            lastShakeTime = currentTime;
            if (listener != null) {
                listener.onShakeDetected();
            }
            Log.d(TAG, "检测到设备摇晃! 幅度: " + magnitude);
        }
    }

    /**
     * 计算设备倾斜角度
     */
    private float calculateTiltAngle(float x, float y, float z) {
        // 计算设备相对于垂直方向的倾斜角度
        double magnitude = Math.sqrt(x * x + y * y + z * z);
        double tilt = Math.toDegrees(Math.acos(z / magnitude));

        // 限制角度范围在0-90度
        return (float) Math.min(90, Math.max(0, tilt));
    }

    /**
     * 获取当前线性加速度
     */
    public float[] getLinearAcceleration() {
        return linearAcceleration.clone();
    }

    /**
     * 获取当前重力向量
     */
    public float[] getGravity() {
        return gravity.clone();
    }
}
```

### 25.2.2 陀螺仪

陀螺仪测量设备的旋转角速度：

```java
public class GyroscopeManager {
    private static final String TAG = "GyroscopeManager";
    private SensorManager sensorManager;
    private Sensor gyroscope;
    private GyroscopeListener listener;

    public interface GyroscopeListener {
        void onRotationChanged(float x, float y, float z, long timestamp);
        void onOrientationChanged(float pitch, float roll, float yaw);
        void onRotationDetected(float rotationSpeed);
    }

    // 旋转积分
    private float[] orientation = new float[3]; // [pitch, roll, yaw]
    private long lastTimestamp = 0;

    // 旋转矩阵和四元数
    private float[] rotationMatrix = new float[9];
    private float[] quaternion = new float[4];

    public GyroscopeManager(Context context, GyroscopeListener listener) {
        this.sensorManager = (SensorManager) context.getSystemService(Context.SENSOR_SERVICE);
        this.gyroscope = sensorManager.getDefaultSensor(Sensor.TYPE_GYROSCOPE);
        this.listener = listener;
    }

    /**
     * 注册陀螺仪监听器
     */
    public boolean register() {
        if (gyroscope != null) {
            return sensorManager.registerListener(
                sensorEventListener,
                gyroscope,
                SensorManager.SENSOR_DELAY_GAME
            );
        }
        return false;
    }

    /**
     * 注销陀螺仪监听器
     */
    public void unregister() {
        sensorManager.unregisterListener(sensorEventListener);
    }

    private final SensorEventListener sensorEventListener = new SensorEventListener() {
        @Override
        public void onSensorChanged(SensorEvent event) {
            if (event.sensor.getType() == Sensor.TYPE_GYROSCOPE) {
                float x = event.values[0]; // 绕X轴旋转速度（弧度/秒）
                float y = event.values[1]; // 绕Y轴旋转速度（弧度/秒）
                float z = event.values[2]; // 绕Z轴旋转速度（弧度/秒）
                long timestamp = event.timestamp;

                // 计算总旋转速度
                float rotationSpeed = (float) Math.sqrt(x * x + y * y + z * z);

                // 积分计算方向角
                if (lastTimestamp != 0) {
                    float dt = (timestamp - lastTimestamp) * 1e-9f; // 转换为秒

                    orientation[0] += x * dt; // pitch
                    orientation[1] += y * dt; // roll
                    orientation[2] += z * dt; // yaw

                    // 标准化角度到 -180° 到 180°
                    for (int i = 0; i < 3; i++) {
                        orientation[i] = normalizeAngle(orientation[i]);
                    }
                }
                lastTimestamp = timestamp;

                // 转换为角度
                float pitchDegrees = (float) Math.toDegrees(orientation[0]);
                float rollDegrees = (float) Math.toDegrees(orientation[1]);
                float yawDegrees = (float) Math.toDegrees(orientation[2]);

                // 通知监听器
                if (listener != null) {
                    listener.onRotationChanged(x, y, z, timestamp);
                    listener.onOrientationChanged(pitchDegrees, rollDegrees, yawDegrees);
                    listener.onRotationDetected(rotationSpeed);
                }

                Log.d(TAG, String.format("陀螺仪: X=%.4f, Y=%.4f, Z=%.4f rad/s | 方向: P=%.1f°, R=%.1f°, Y=%.1f°",
                    x, y, z, pitchDegrees, rollDegrees, yawDegrees));
            }
        }

        @Override
        public void onAccuracyChanged(Sensor sensor, int accuracy) {
            Log.d(TAG, "陀螺仪精度变化: " + accuracy);
        }
    };

    /**
     * 标准化角度到 -180° 到 180°
     */
    private float normalizeAngle(float angle) {
        while (angle > Math.PI) {
            angle -= 2 * Math.PI;
        }
        while (angle < -Math.PI) {
            angle += 2 * Math.PI;
        }
        return angle;
    }

    /**
     * 重置方向角度
     */
    public void resetOrientation() {
        Arrays.fill(orientation, 0);
        lastTimestamp = 0;
    }

    /**
     * 获取当前方向角度（弧度）
     */
    public float[] getOrientation() {
        return orientation.clone();
    }

    /**
     * 获取当前方向角度（度）
     */
    public float[] getOrientationDegrees() {
        float[] degrees = new float[3];
        for (int i = 0; i < 3; i++) {
            degrees[i] = (float) Math.toDegrees(orientation[i]);
        }
        return degrees;
    }
}
```

### 25.2.3 磁力计

磁力计测量地磁场强度，用于确定设备的方向：

```java
public class MagnetometerManager {
    private static final String TAG = "MagnetometerManager";
    private SensorManager sensorManager;
    private Sensor magnetometer;
    private Sensor accelerometer;
    private MagnetometerListener listener;

    public interface MagnetometerListener {
        void onMagneticFieldChanged(float x, float y, float z, long timestamp);
        void onCompassChanged(float azimuth, float pitch, float roll);
        void onNorthDirectionChanged(float direction);
    }

    private float[] magneticField = new float[3];
    private float[] gravity = new float[3];
    private float[] rotationMatrix = new float[9];
    private float[] orientation = new float[3];

    private boolean hasGravity = false;
    private boolean hasMagneticField = false;

    public MagnetometerManager(Context context, MagnetometerListener listener) {
        this.sensorManager = (SensorManager) context.getSystemService(Context.SENSOR_SERVICE);
        this.magnetometer = sensorManager.getDefaultSensor(Sensor.TYPE_MAGNETIC_FIELD);
        this.accelerometer = sensorManager.getDefaultSensor(Sensor.TYPE_ACCELEROMETER);
        this.listener = listener;
    }

    /**
     * 注册传感器监听器
     */
    public boolean register() {
        boolean magnetRegistered = false;
        boolean accelRegistered = false;

        if (magnetometer != null) {
            magnetRegistered = sensorManager.registerListener(
                magnetometerListener,
                magnetometer,
                SensorManager.SENSOR_DELAY_NORMAL
            );
        }

        if (accelerometer != null) {
            accelRegistered = sensorManager.registerListener(
                accelerometerListener,
                accelerometer,
                SensorManager.SENSOR_DELAY_NORMAL
            );
        }

        return magnetRegistered && accelRegistered;
    }

    /**
     * 注销传感器监听器
     */
    public void unregister() {
        sensorManager.unregisterListener(magnetometerListener);
        sensorManager.unregisterListener(accelerometerListener);
    }

    private final SensorEventListener magnetometerListener = new SensorEventListener() {
        @Override
        public void onSensorChanged(SensorEvent event) {
            if (event.sensor.getType() == Sensor.TYPE_MAGNETIC_FIELD) {
                System.arraycopy(event.values, 0, magneticField, 0, 3);
                hasMagneticField = true;

                float x = event.values[0];
                float y = event.values[1];
                float z = event.values[2];
                long timestamp = event.timestamp;

                // 计算磁场强度
                float magnitude = (float) Math.sqrt(x * x + y * y + z * z);

                if (listener != null) {
                    listener.onMagneticFieldChanged(x, y, z, timestamp);
                }

                Log.d(TAG, String.format("磁场: X=%.2f, Y=%.2f, Z=%.2f μT, 强度=%.2f μT",
                    x, y, z, magnitude));

                // 尝试计算方向
                calculateOrientation();
            }
        }

        @Override
        public void onAccuracyChanged(Sensor sensor, int accuracy) {
            Log.d(TAG, "磁力计精度变化: " + accuracy);
        }
    };

    private final SensorEventListener accelerometerListener = new SensorEventListener() {
        @Override
        public void onSensorChanged(SensorEvent event) {
            if (event.sensor.getType() == Sensor.TYPE_ACCELEROMETER) {
                // 低通滤波获取重力
                final float alpha = 0.8f;
                gravity[0] = alpha * gravity[0] + (1 - alpha) * event.values[0];
                gravity[1] = alpha * gravity[1] + (1 - alpha) * event.values[1];
                gravity[2] = alpha * gravity[2] + (1 - alpha) * event.values[2];
                hasGravity = true;

                // 尝试计算方向
                calculateOrientation();
            }
        }

        @Override
        public void onAccuracyChanged(Sensor sensor, int accuracy) {
            // 不处理
        }
    };

    /**
     * 计算设备方向
     */
    private void calculateOrientation() {
        if (hasGravity && hasMagneticField) {
            boolean success = SensorManager.getRotationMatrix(
                rotationMatrix,
                null,
                gravity,
                magneticField
            );

            if (success) {
                // 获取方向角（ azimuth, pitch, roll ）
                SensorManager.getOrientation(rotationMatrix, orientation);

                // 转换为角度
                float azimuth = (float) Math.toDegrees(orientation[0]); // 方位角
                float pitch = (float) Math.toDegrees(orientation[1]);    // 俯仰角
                float roll = (float) Math.toDegrees(orientation[2]);    // 横滚角

                // 标准化方位角到 0-360°
                if (azimuth < 0) {
                    azimuth += 360;
                }

                if (listener != null) {
                    listener.onCompassChanged(azimuth, pitch, roll);
                    listener.onNorthDirectionChanged(azimuth);
                }

                Log.d(TAG, String.format("罗盘: 方位=%.1f°, 俯仰=%.1f°, 横滚=%.1f°",
                    azimuth, pitch, roll));
            }
        }
    }

    /**
     * 获取当前磁场强度
     */
    public float[] getMagneticField() {
        return magneticField.clone();
    }

    /**
     * 获取当前方向角度（度）
     */
    public float[] getOrientation() {
        if (hasGravity && hasMagneticField) {
            float[] degrees = new float[3];
            degrees[0] = (float) Math.toDegrees(orientation[0]);
            degrees[1] = (float) Math.toDegrees(orientation[1]);
            degrees[2] = (float) Math.toDegrees(orientation[2]);

            // 标准化方位角
            if (degrees[0] < 0) {
                degrees[0] += 360;
            }

            return degrees;
        }
        return new float[3];
    }
}
```

## 25.3 环境传感器

环境传感器检测周围环境的物理条件，如光线、温度、气压等。

### 25.3.1 光线传感器

光线传感器检测环境光照强度：

```java
public class LightSensorManager {
    private static final String TAG = "LightSensorManager";
    private SensorManager sensorManager;
    private Sensor lightSensor;
    private LightSensorListener listener;

    public interface LightSensorListener {
        void onLightChanged(float lux, long timestamp);
        void onBrightnessLevelChanged(int level);
        void onEnvironmentChanged(String environment);
    }

    // 光照强度阈值（勒克斯）
    private static final float DARK_THRESHOLD = 1.0f;      // 黑暗
    private static final float DIM_THRESHOLD = 10.0f;      // 昏暗
    private static final float NORMAL_THRESHOLD = 500.0f;  // 正常
    private static final float BRIGHT_THRESHOLD = 10000.0f; // 明亮
    private static final float SUNLIGHT_THRESHOLD = 50000.0f; // 阳光直射

    private int lastBrightnessLevel = -1;

    public LightSensorManager(Context context, LightSensorListener listener) {
        this.sensorManager = (SensorManager) context.getSystemService(Context.SENSOR_SERVICE);
        this.lightSensor = sensorManager.getDefaultSensor(Sensor.TYPE_LIGHT);
        this.listener = listener;
    }

    /**
     * 注册光线传感器监听器
     */
    public boolean register() {
        if (lightSensor != null) {
            return sensorManager.registerListener(
                sensorEventListener,
                lightSensor,
                SensorManager.SENSOR_DELAY_NORMAL
            );
        }
        return false;
    }

    /**
     * 注销光线传感器监听器
     */
    public void unregister() {
        sensorManager.unregisterListener(sensorEventListener);
    }

    private final SensorEventListener sensorEventListener = new SensorEventListener() {
        @Override
        public void onSensorChanged(SensorEvent event) {
            if (event.sensor.getType() == Sensor.TYPE_LIGHT) {
                float lux = event.values[0];
                long timestamp = event.timestamp;

                // 计算亮度等级（0-255）
                int brightnessLevel = calculateBrightnessLevel(lux);

                // 判断环境类型
                String environment = getEnvironmentType(lux);

                // 通知监听器
                if (listener != null) {
                    listener.onLightChanged(lux, timestamp);
                    if (brightnessLevel != lastBrightnessLevel) {
                        listener.onBrightnessLevelChanged(brightnessLevel);
                        lastBrightnessLevel = brightnessLevel;
                    }
                    listener.onEnvironmentChanged(environment);
                }

                Log.d(TAG, String.format("光照强度: %.2f lux, 亮度等级: %d, 环境: %s",
                    lux, brightnessLevel, environment));
            }
        }

        @Override
        public void onAccuracyChanged(Sensor sensor, int accuracy) {
            Log.d(TAG, "光线传感器精度变化: " + accuracy);
        }
    };

    /**
     * 根据光照强度计算亮度等级
     */
    private int calculateBrightnessLevel(float lux) {
        // 将光照强度映射到0-255范围
        if (lux <= DARK_THRESHOLD) {
            return 1;   // 最暗
        } else if (lux <= DIM_THRESHOLD) {
            return (int) (1 + (lux - DARK_THRESHOLD) / (DIM_THRESHOLD - DARK_THRESHOLD) * 30);
        } else if (lux <= NORMAL_THRESHOLD) {
            return (int) (31 + (lux - DIM_THRESHOLD) / (NORMAL_THRESHOLD - DIM_THRESHOLD) * 100);
        } else if (lux <= BRIGHT_THRESHOLD) {
            return (int) (131 + (lux - NORMAL_THRESHOLD) / (BRIGHT_THRESHOLD - NORMAL_THRESHOLD) * 100);
        } else {
            return 231 + Math.min(24, (int) ((lux - BRIGHT_THRESHOLD) / (SUNLIGHT_THRESHOLD - BRIGHT_THRESHOLD) * 24));
        }
    }

    /**
     * 获取环境类型描述
     */
    private String getEnvironmentType(float lux) {
        if (lux <= DARK_THRESHOLD) {
            return "非常黑暗";
        } else if (lux <= DIM_THRESHOLD) {
            return "昏暗";
        } else if (lux <= 50) {
            return "室内照明";
        } else if (lux <= 200) {
            return "办公室照明";
        } else if (lux <= NORMAL_THRESHOLD) {
            return "正常室内照明";
        } else if (lux <= BRIGHT_THRESHOLD) {
            return "明亮环境";
        } else if (lux <= SUNLIGHT_THRESHOLD) {
            return "非常明亮";
        } else {
            return "阳光直射";
        }
    }

    /**
     * 获取推荐的屏幕亮度

    public int getRecommendedBrightness(float lux) {

            return 30;   // 昏暗环境使用低亮度
        } else if (lux <= NORMAL_THRESHOLD) {
            return 120;  // 正常环境使用中等亮度
        } else if (lux <= BRIGHT_THRESHOLD) {
            return 200;  // 明亮环境使用较高亮度
        } else {

        }
    }
```

### 25.3.2 气压计

气压计测量大气压力，可以用于天气预测和海拔高度估算：

```java
public class PressureSensorManager {
    private static final String TAG = "PressureSensorManager";
    private SensorManager sensorManager;
    private Sensor pressureSensor;
    private PressureSensorListener listener;

    public interface PressureSensorListener {
        void onPressureChanged(float pressure, long timestamp);
        void onAltitudeChanged(float altitude);
        void onWeatherChanged(String weatherCondition);
        void onPressureTrendChanged(String trend);
    }

    // 气压相关常量
    private static final float SEA_LEVEL_PRESSURE = 1013.25f; // 海平面标准气压（hPa）
    private static final float PRESSURE_CHANGE_THRESHOLD = 0.5f; // 气压变化阈值（hPa）

    private float[] pressureHistory = new float[10]; // 保存最近10个气压读数
    private int historyIndex = 0;
    private boolean historyFilled = false;
    private float lastSignificantPressure = 0;

    public PressureSensorManager(Context context, PressureSensorListener listener) {
        this.sensorManager = (SensorManager) context.getSystemService(Context.SENSOR_SERVICE);
        this.pressureSensor = sensorManager.getDefaultSensor(Sensor.TYPE_PRESSURE);
        this.listener = listener;
    }

    /**
     * 注册气压传感器监听器
     */
    public boolean register() {
        if (pressureSensor != null) {
            return sensorManager.registerListener(
                sensorEventListener,
                pressureSensor,
                SensorManager.SENSOR_DELAY_NORMAL
            );
        }
        return false;
    }

    /**
     * 注销气压传感器监听器
     */
    public void unregister() {
        sensorManager.unregisterListener(sensorEventListener);
    }

    private final SensorEventListener sensorEventListener = new SensorEventListener() {
        @Override
        public void onSensorChanged(SensorEvent event) {
            if (event.sensor.getType() == Sensor.TYPE_PRESSURE) {
                float pressure = event.values[0]; // 气压（hPa）
                long timestamp = event.timestamp;

                // 估算海拔高度
                float altitude = calculateAltitude(pressure);

                // 分析气压趋势
                String trend = analyzePressureTrend(pressure);

                // 预测天气条件
                String weatherCondition = predictWeather(pressure, trend);

                // 更新历史记录
                updatePressureHistory(pressure);

                if (listener != null) {
                    listener.onPressureChanged(pressure, timestamp);
                    listener.onAltitudeChanged(altitude);
                    listener.onWeatherChanged(weatherCondition);
                    listener.onPressureTrendChanged(trend);
                }

                Log.d(TAG, String.format("气压: %.2f hPa, 海拔: %.1f m, 趋势: %s, 天气: %s",
                    pressure, altitude, trend, weatherCondition));
            }
        }

        @Override
        public void onAccuracyChanged(Sensor sensor, int accuracy) {
            Log.d(TAG, "气压传感器精度变化: " + accuracy);
        }
    };

    /**
     * 根据气压计算海拔高度
     */
    private float calculateAltitude(float pressure) {
        // 使用国际标准大气模型计算海拔高度
        // h = 44330 * (1 - (P/P0)^(1/5.255))
        return 44330f * (1f - (float) Math.pow(pressure / SEA_LEVEL_PRESSURE, 1f / 5.255f));
    }

    /**
     * 分析气压变化趋势
     */
    private String analyzePressureTrend(float currentPressure) {
        // 更新历史记录
        pressureHistory[historyIndex] = currentPressure;
        historyIndex = (historyIndex + 1) % pressureHistory.length;

        if (!historyFilled && historyIndex == 0) {
            historyFilled = true;
        }

        // 分析趋势
        if (historyFilled || historyIndex >= 3) {
            float sum = 0;
            int count = historyFilled ? pressureHistory.length : historyIndex;

            // 计算最近几个读数的平均值
            for (int i = 0; i < count; i++) {
                sum += pressureHistory[i];
            }
            float average = sum / count;

            float change = currentPressure - average;

            if (Math.abs(change) < PRESSURE_CHANGE_THRESHOLD) {
                return "稳定";
            } else if (change > 0) {
                return "上升";
            } else {
                return "下降";
            }
        }

        return "数据收集中";
    }

    /**
     * 根据气压和趋势预测天气
     */
    private String predictWeather(float pressure, String trend) {
        // 根据气压值和变化趋势预测天气
        if (pressure < 1000) {
            if ("下降".equals(trend)) {
                return "暴风雨即将来临";
            } else {
                return "阴雨天气";
            }
        } else if (pressure < 1010) {
            if ("下降".equals(trend)) {
                return "可能下雨";
            } else if ("上升".equals(trend)) {
                return "天气转好";
            } else {
                return "阴天";
            }
        } else if (pressure < 1020) {
            if ("上升".equals(trend)) {
                return "天气晴朗";
            } else {
                return "多云天气";
            }
        } else {
            if ("上升".equals(trend)) {
                return "晴朗稳定";
            } else if ("下降".equals(trend)) {
                return "天气变化";
            } else {
                return "晴朗";
            }
        }
    }

    /**
     * 更新气压历史记录
     */
    private void updatePressureHistory(float pressure) {
        if (lastSignificantPressure == 0) {
            lastSignificantPressure = pressure;
        }

        // 检测是否有显著变化
        if (Math.abs(pressure - lastSignificantPressure) > PRESSURE_CHANGE_THRESHOLD) {
            lastSignificantPressure = pressure;
        }
    }

    /**
     * 计算两个海拔高度之间的气压差
     */
    public static float calculatePressureDifference(float altitude1, float altitude2) {
        // P2 = P1 * (1 - (h2-h1)/44330)^5.255
        float heightDiff = altitude2 - altitude1;
        return SEA_LEVEL_PRESSURE * (float) Math.pow(1 - heightDiff / 44330f, 5.255f);
    }

    /**
     * 校准海平面气压
     */
    public void calibrateSeaLevelPressure(float knownAltitude, float measuredPressure) {
        // 根据已知海拔和测量气压反推海平面气压
        // P0 = P * (1 - h/44330)^(-5.255)
        float calibratedSeaLevelPressure = measuredPressure *
            (float) Math.pow(1 - knownAltitude / 44330f, -5.255f);

        Log.d(TAG, "校准海平面气压: " + calibratedSeaLevelPressure + " hPa");
    }
}
```

### 25.4.2 Fused Location Provider

Google Play Services的Fused Location Provider提供了更智能的位置服务：

```java
public class FusedLocationManager {

     * 检查位置权限

            if (callback != null) {
                callback.onLocationFailure("缺少位置权限");

                    if (callback != null) {
                        callback.onLocationFailure("获取位置失败: " + e.getMessage());
                    }
                    Log.e(TAG, "获取当前位置失败", e);
                });
        } catch (SecurityException e) {
            if (callback != null) {
                callback.onLocationFailure("安全异常: " + e.getMessage());
            }
            Log.e(TAG, "获取当前位置安全异常", e);
        }
    }

    /**
     * 获取最后已知位置

        if (!isGooglePlayServicesAvailable()) {
            if (callback != null) {
                callback.onLocationFailure("Google Play Services不可用");
            }
            return;
        }

        if (!hasLocationPermission()) {
            if (callback != null) {

            if (callback != null) {
                callback.onLocationFailure("Google Play Services不可用");

            return;
        }

            if (callback != null) {
                callback.onLocationFailure("缺少位置权限");

            return;

                .setMaxUpdateDelayMillis(UPDATE_INTERVAL * 2)

                    if (callback != null) {
                        callback.onLocationFailure("启动位置更新失败: " + e.getMessage());
                    }
                    Log.e(TAG, "启动位置更新失败", e);
                });
        } catch (SecurityException e) {
            if (callback != null) {
                callback.onLocationFailure("安全异常: " + e.getMessage());
            }
            Log.e(TAG, "启动位置更新安全异常", e);
        }
    }

    /**
     * 停止位置更新
     */
    public void stopLocationUpdates() {
        fusedLocationClient.removeLocationUpdates(locationCallback)
            .addOnSuccessListener(aVoid -> {
                Log.d(TAG, "位置更新已停止");
            })
            .addOnFailureListener(e -> {
                Log.e(TAG, "停止位置更新失败", e);
            });
```

```java
public class SmartActivityTracker {
    private static final String TAG = "SmartActivityTracker";

    private Context context;
    private SensorManager sensorManager;
    private LocationServiceManager locationServiceManager;

    // 传感器管理器
    private AccelerometerManager accelerometerManager;
    private StepDetectorManager stepDetectorManager;
    private LightSensorManager lightSensorManager;

    // 数据收集
    private List<ActivityData> activityDataList;
    private ActivityData currentActivity;

    // 监听器
    private ActivityTrackingListener listener;

    public interface ActivityTrackingListener {
        void onStepDetected(int stepCount, float distance);
        void onActivityChanged(String activityType, float confidence);
        void onLocationUpdated(Location location);
        void onEnvironmentChanged(String environment);
        void onTrackingDataCollected(ActivityData data);
    }

    // 活动类型
    public static final String ACTIVITY_STILL = "静止";
    public static final String ACTIVITY_WALKING = "步行";
    public static final String ACTIVITY_RUNNING = "跑步";
    public static final String ACTIVITY_CYCLING = "骑行";
    public static final String ACTIVITY_DRIVING = "驾驶";

    public SmartActivityTracker(Context context, ActivityTrackingListener listener) {
        this.context = context;
        this.listener = listener;
        this.sensorManager = (SensorManager) context.getSystemService(Context.SENSOR_SERVICE);
        this.locationServiceManager = new LocationServiceManager(context, new LocationCallback());
        this.activityDataList = new ArrayList<>();

        initializeSensors();
    }

    /**
     * 初始化传感器
     */
    private void initializeSensors() {
        // 步数检测
        stepDetectorManager = new StepDetectorManager(context, new StepCallback());

        // 加速度检测（用于活动识别）
        accelerometerManager = new AccelerometerManager(context, new AccelerometerCallback());

        // 环境光线检测
        lightSensorManager = new LightSensorManager(context, new LightCallback());
    }

    /**
     * 开始运动追踪
     */
    public void startTracking() {
        Log.d(TAG, "开始运动追踪");

        // 启动传感器
        stepDetectorManager.register();
        accelerometerManager.register();
        lightSensorManager.register();

        // 启动位置更新
        locationServiceManager.startGPSLocationUpdates();

        // 创建新的活动数据
        currentActivity = new ActivityData();
        currentActivity.setStartTime(System.currentTimeMillis());
        currentActivity.setActivityType(ACTIVITY_STILL);
    }

    /**
     * 停止运动追踪
     */
    public void stopTracking() {
        Log.d(TAG, "停止运动追踪");

        // 停止传感器
        stepDetectorManager.unregister();
        accelerometerManager.unregister();
        lightSensorManager.unregister();

        // 停止位置更新
        locationServiceManager.stopLocationUpdates();

        // 完成当前活动
        if (currentActivity != null) {
            currentActivity.setEndTime(System.currentTimeMillis());
            activityDataList.add(currentActivity);

            if (listener != null) {
                listener.onTrackingDataCollected(currentActivity);
            }
        }
    }

    /**
     * 获取今日运动数据
     */
    public DailyActivitySummary getTodaySummary() {
        DailyActivitySummary summary = new DailyActivitySummary();
        Calendar today = Calendar.getInstance();
        today.set(Calendar.HOUR_OF_DAY, 0);
        today.set(Calendar.MINUTE, 0);
        today.set(Calendar.SECOND, 0);
        today.set(Calendar.MILLISECOND, 0);

        for (ActivityData data : activityDataList) {
            if (data.getStartTime() >= today.getTimeInMillis()) {
                summary.addData(data);
            }
        }

        return summary;

    /**
     * 位置更新回调

    private class LocationCallback implements LocationServiceManager.LocationServiceCallback {
        @Override
        public void onLocationChanged(Location location) {
            if (currentActivity != null) {
                currentActivity.addLocation(location);

                // 根据速度判断活动类型
                float speed = location.getSpeed();
                String activityType = detectActivityBySpeed(speed);
                updateActivityType(activityType, 0.8f); // 高置信度
            }

            if (listener != null) {
                listener.onLocationUpdated(location);
            }
        }

        @Override
        public void onLocationError(String error) {
            Log.e(TAG, "位置更新错误: " + error);
        }

        @Override
        public void onProviderEnabled(String provider) {}

        @Override
        public void onProviderDisabled(String provider) {}

        @Override
        public void onStatusChanged(String provider, int status, Bundle extras) {}
    }

    /**
     * 步数检测回调
     */
    private class StepCallback implements StepDetectorManager.StepDetectorListener {
        @Override
        public void onStepDetected(int stepCount, float stepLength) {
            if (currentActivity != null) {
                currentActivity.incrementStepCount();
                currentActivity.addDistance(stepLength);
            }

            if (listener != null) {
                listener.onStepDetected(currentActivity != null ? currentActivity.getStepCount() : stepCount,
                                     currentActivity != null ? currentActivity.getDistance() : stepCount * stepLength);
            }

            // 根据步数频率判断活动类型
            updateActivityType(detectActivityBySteps(), 0.6f);
        }

        @Override
        public void onStepFrequencyChanged(float frequency) {
            // 根据步频进一步判断活动类型
            if (frequency > 2.5f) {
                updateActivityType(ACTIVITY_RUNNING, 0.7f);
            } else if (frequency > 1.5f) {
                updateActivityType(ACTIVITY_WALKING, 0.7f);
            }
        }
    }

    /**
     * 加速度检测回调
     */
    private class AccelerometerCallback implements AccelerometerManager.AccelerometerListener {
        @Override
        public void onAccelerationChanged(float x, float y, float z, long timestamp) {
            // 计算加速度幅度
            float magnitude = (float) Math.sqrt(x * x + y * y + z * z);

            if (currentActivity != null) {
                currentActivity.addAccelerationMagnitude(magnitude);
            }
        }

        @Override
        public void onShakeDetected() {
            // 检测到摇晃，可能是在跑步或其他剧烈活动
            updateActivityType(ACTIVITY_RUNNING, 0.5f);
        }

        @Override
        public void onTiltChanged(float tiltAngle) {
            // 根据设备倾斜角度判断用户状态
            if (currentActivity != null) {
                currentActivity.setTiltAngle(tiltAngle);
            }
        }
    }

     * 光线传感器回调
     */
    private class LightCallback implements LightSensorManager.LightSensorListener {
        @Override
        public void onLightChanged(float lux, long timestamp) {
            // 可以根据光线判断是室内还是室外
            if (currentActivity != null) {
                currentActivity.setLightLevel(lux);
            }
        }

        @Override
        public void onBrightnessLevelChanged(int level) {
            // 不处理
        }

        @Override
        public void onEnvironmentChanged(String environment) {

            }
        }

    /**
     * 根据速度检测活动类型
     */
    private String detectActivityBySpeed(float speed) {
        if (speed < 0.5f) {
            return ACTIVITY_STILL;
        } else if (speed < 2.0f) {
            return ACTIVITY_WALKING;
        } else if (speed < 5.0f) {
            return ACTIVITY_RUNNING;
        } else if (speed < 15.0f) {
            return ACTIVITY_CYCLING;
        } else {
            return ACTIVITY_DRIVING;
        }
    }

    /**
     * 根据步数模式检测活动类型
     */
    private String detectActivityBySteps() {
        if (currentActivity == null) {
            return ACTIVITY_STILL;
        }

        long duration = System.currentTimeMillis() - currentActivity.getStartTime();
        if (duration == 0) return ACTIVITY_STILL;

        float stepsPerMinute = (currentActivity.getStepCount() * 60000f) / duration;

        if (stepsPerMinute < 10) {
            return ACTIVITY_STILL;
        } else if (stepsPerMinute < 140) {
            return ACTIVITY_WALKING;
        } else {
            return ACTIVITY_RUNNING;
        }
    }

    /**
     * 更新活动类型
     */
    private void updateActivityType(String newType, float confidence) {
        if (currentActivity == null) return;

        String oldType = currentActivity.getActivityType();
        if (!oldType.equals(newType)) {
            // 活动类型改变，完成旧活动，开始新活动
            currentActivity.setEndTime(System.currentTimeMillis());
            activityDataList.add(currentActivity);

            if (listener != null) {
                listener.onTrackingDataCollected(currentActivity);
                listener.onActivityChanged(newType, confidence);
            }

            // 开始新活动
            currentActivity = new ActivityData();
            currentActivity.setStartTime(System.currentTimeMillis());
            currentActivity.setActivityType(newType);
        }
    }

    public static class ActivityData {
        private long startTime;
        private long endTime;
        private String activityType;
        private int stepCount;
        private float distance;
        private List<Location> locations;
        private List<Float> accelerationMagnitudes;
        private float lightLevel;
        private float tiltAngle;

        public ActivityData() {
            this.locations = new ArrayList<>();
            this.accelerationMagnitudes = new ArrayList<>();
        }

        // Getters and Setters
        public long getStartTime() { return startTime; }
        public void setStartTime(long startTime) { this.startTime = startTime; }

        public long getEndTime() { return endTime; }
        public void setEndTime(long endTime) { this.endTime = endTime; }

        public String getActivityType() { return activityType; }
        public void setActivityType(String activityType) { this.activityType = activityType; }

        public int getStepCount() { return stepCount; }
        public void incrementStepCount() { this.stepCount++; }

        public float getDistance() { return distance; }
        public void addDistance(float additionalDistance) { this.distance += additionalDistance; }

        public List<Location> getLocations() { return locations; }
        public void addLocation(Location location) { this.locations.add(location); }

        public List<Float> getAccelerationMagnitudes() { return accelerationMagnitudes; }
        public void addAccelerationMagnitude(float magnitude) { this.accelerationMagnitudes.add(magnitude); }

        public float getLightLevel() { return lightLevel; }
        public void setLightLevel(float lightLevel) { this.lightLevel = lightLevel; }

        public float getTiltAngle() { return tiltAngle; }
        public void setTiltAngle(float tiltAngle) { this.tiltAngle = tiltAngle; }

        public long getDuration() {
            return endTime > 0 ? endTime - startTime : System.currentTimeMillis() - startTime;
        }

        public float getAverageSpeed() {
            long duration = getDuration();
            return duration > 0 ? (distance * 1000f) / duration : 0; // m/s
        }

        public float getAverageAcceleration() {
            if (accelerationMagnitudes.isEmpty()) return 0;
            float sum = 0;
            for (float mag : accelerationMagnitudes) {
                sum += mag;
            }
            return sum / accelerationMagnitudes.size();
        }
    }

     * 每日活动总结
     */
    public static class DailyActivitySummary {

        private long activeTime;

        // Getters
        public int getTotalSteps() { return totalSteps; }
        public float getTotalDistance() { return totalDistance; }
        public long getActiveTime() { return activeTime; }
        public Map<String, Long> getActivityDurations() { return activityDurations; }

        public String getSummaryText() {
```

```java
            if (sensor == null) return;

            // 根据应用状态和优先级选择延迟
            int delay = calculateOptimalDelay(sensorType, isInBackground, priority);

                    backgroundHandler.post(() -> {

                    backgroundHandler.post(() -> {
                        listener.onAccuracyChanged(sensor, accuracy);
                    });
                }
            };

            boolean registered = sensorManager.registerListener(
                optimizedListener, sensor, delay);

            if (registered) {
                activeListeners.put(sensorType, optimizedListener);
            }
        }

        /**
         * 计算最优延迟
         */
        private int calculateOptimalDelay(int sensorType, boolean isInBackground, int priority) {
            // 后台模式下使用低频率
            if (isInBackground) {
                return SensorManager.SENSOR_DELAY_NORMAL;
            }

            // 根据传感器类型和优先级调整
            switch (sensorType) {
                case Sensor.TYPE_ACCELEROMETER:
                case Sensor.TYPE_GYROSCOPE:
                    return priority == 1 ? SensorManager.SENSOR_DELAY_FASTEST :
                           priority == 2 ? SensorManager.SENSOR_DELAY_GAME :
                           SensorManager.SENSOR_DELAY_UI;

                case Sensor.TYPE_LIGHT:
                case Sensor.TYPE_PRESSURE:
                case Sensor.TYPE_MAGNETIC_FIELD:
                    return SensorManager.SENSOR_DELAY_NORMAL;

                default:
                    return SensorManager.SENSOR_DELAY_UI;
            }

    public static class SensorDataFilter {
        private static final float LOW_PASS_ALPHA = 0.8f;
        private static final float HIGH_PASS_ALPHA = 0.2f;

        /**
         * 低通滤波器（去高频噪声）
         */
        public static float[] lowPassFilter(float[] input, float[] output) {
            if (output == null) {
                return input.clone();
            }

            for (int i = 0; i < input.length; i++) {
                output[i] = LOW_PASS_ALPHA * output[i] + (1 - LOW_PASS_ALPHA) * input[i];
            }

            return output;
        }

        /**
         * 高通滤波器（去低频漂移）
         */
        public static float[] highPassFilter(float[] input, float[] output, float[] prevInput) {
            if (output == null || prevInput == null) {
                return new float[input.length];
            }

            for (int i = 0; i < input.length; i++) {
                output[i] = HIGH_PASS_ALPHA * (output[i] + input[i] - prevInput[i]);
                prevInput[i] = input[i];
            }

            return output;
        }

        /**
         * 中值滤波器（去脉冲噪声）
         */
        public static float medianFilter(float[] data) {
            Arrays.sort(data.clone());
            return data[data.length / 2];
        }

        /**
         * 移动平均滤波器
         */
        public static class MovingAverageFilter {
            private final float[] buffer;
            private int index;
            private int count;
            private float sum;

            public MovingAverageFilter(int windowSize) {
                this.buffer = new float[windowSize];
                this.index = 0;
                this.count = 0;
                this.sum = 0;
            }

            public float filter(float value) {
                sum -= buffer[index];
                buffer[index] = value;
                sum += buffer[index];

                index = (index + 1) % buffer.length;
                if (count < buffer.length) {
                    count++;
                }

                return sum / count;
            }

            public void reset() {
                Arrays.fill(buffer, 0);
                index = 0;
                count = 0;
                sum = 0;
            }
        }
    }

    /**
     * 传感器数据验证器
     */
    public static class SensorDataValidator {
        private static final float ACCELEROMETER_MAX = 78.4f; // 8g
        private static final float GYROSCOPE_MAX = 35.0f;     // 2000度/秒转弧度
        private static final float MAGNETIC_FIELD_MAX = 2000f; // 2000μT

        /**
         * 验证加速度数据
         */
        public static boolean validateAccelerometer(float x, float y, float z) {
            float magnitude = (float) Math.sqrt(x * x + y * y + z * z);
            return magnitude <= ACCELEROMETER_MAX && !Float.isNaN(magnitude) && !Float.isInfinite(magnitude);
        }

        /**
         * 验证陀螺仪数据
         */
        public static boolean validateGyroscope(float x, float y, float z) {
            return Math.abs(x) <= GYROSCOPE_MAX &&
                   Math.abs(y) <= GYROSCOPE_MAX &&
                   Math.abs(z) <= GYROSCOPE_MAX &&
                   !Float.isNaN(x) && !Float.isInfinite(x);
        }

        /**
         * 验证磁场数据
         */
        public static boolean validateMagneticField(float x, float y, float z) {
            float magnitude = (float) Math.sqrt(x * x + y * y + z * z);
            return magnitude <= MAGNETIC_FIELD_MAX && magnitude >= 20 && // 地磁场强度通常在20-100μT之间
                   !Float.isNaN(magnitude) && !Float.isInfinite(magnitude);
        }

        /**
         * 检测异常值
         */
        public static boolean isOutlier(float value, float[] history, float threshold) {
            if (history.length == 0) return false;

            float mean = calculateMean(history);
            float stdDev = calculateStandardDeviation(history, mean);

            return Math.abs(value - mean) > threshold * stdDev;
        }

        private static float calculateMean(float[] data) {
            float sum = 0;
            for (float value : data) {
                sum += value;
            }
            return sum / data.length;
        }

        private static float calculateStandardDeviation(float[] data, float mean) {
            float sum = 0;
            for (float value : data) {
                sum += (value - mean) * (value - mean);
            }
            return (float) Math.sqrt(sum / data.length);
        }
    }
```
