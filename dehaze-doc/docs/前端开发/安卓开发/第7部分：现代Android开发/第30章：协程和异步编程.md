# 第30章：协程和异步编程

Kotlin协程是现代Android开发中最重要的异步编程技术之一，它提供了一种简洁、高效的方式来处理异步操作和并发编程。协程让异步代码看起来像同步代码一样简单，同时具备强大的错误处理和资源管理能力。本章将深入探讨Kotlin协程的使用、原理和最佳实践。

## 30.1 协程基础概念

协程是一种并发编程的方式，它允许在单个线程中实现暂停和恢复执行的函数。

### 30.1.1 协程的基本概念

```java
public class CoroutineBasics {
    private static final String TAG = "CoroutineBasics";

    /**
     * 协程基本概念演示
     */
    public static class CoroutineConcepts {
        /**
         * 协程创建示例
         */
        public static void demonstrateBasicCoroutines() {
            // 创建协程作用域
            CoroutineScope scope = new CoroutineScope(SupervisorJob() + Dispatchers.Main);

            // 启动协程
            scope.launch(Dispatchers.IO) {
                Log.d(TAG, "协程开始执行");
                delay(1000); // 挂起协程1秒
                Log.d(TAG, "协程执行完成");
            };

            // 异步执行并返回结果
            Deferred<String> deferred = scope.async(Dispatchers.IO) {
                Log.d(TAG, "异步协程开始");
                delay(2000);
                return "异步结果";
            };

            // 等待异步结果
            scope.launch(Dispatchers.Main) {
                try {
                    String result = deferred.await();
                    Log.d(TAG, "异步结果: " + result);
                } catch (Exception e) {
                    Log.e(TAG, "等待异步结果失败", e);
                }
            };
        }

        /**
         * 协程构建器对比
         */
        public static void coroutineBuildersComparison() {
            CoroutineScope scope = new CoroutineScope(SupervisorJob() + Dispatchers.Main);

            // launch - 启动新协程，不返回结果
            scope.launch(Dispatchers.IO) {
                performBackgroundTask("launch");
            };

            // async - 启动新协程，返回Deferred<T>
            Deferred<Integer> result = scope.async(Dispatchers.IO) -> {
                return performCalculation("async");
            };

            // runBlocking - 阻塞当前线程直到协程完成
            runBlocking(Dispatchers.IO) {
                performBackgroundTask("runBlocking");
            };

            // withContext - 切换上下文并返回结果
            scope.launch(Dispatchers.Main) -> {
                String result = withContext(Dispatchers.IO) {
                    performBackgroundTask("withContext");
                    return "任务完成";
                };
                Log.d(TAG, result);
            };
        }

        private static void performBackgroundTask(String name) {
            Log.d(TAG, name + " 开始执行");
            try {
                Thread.sleep(1000);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
            Log.d(TAG, name + " 执行完成");
        }

        private static int performCalculation(String name) {
            Log.d(TAG, name + " 开始计算");
            try {
                Thread.sleep(2000);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
            Log.d(TAG, name + " 计算完成");
            return 42;
        }
    }

    /**
     * 协程作用域管理
     */
    public static class CoroutineScopeManager {
        private final CoroutineScope scope;

        public CoroutineScopeManager(Dispatcher dispatcher) {
            this.scope = new CoroutineScope(SupervisorJob() + dispatcher);
        }

        /**
         * 启动协程
         */
        public Job launch(Dispatcher dispatcher, suspend () -> Unit block) {
            return scope.launch(dispatcher) {
                try {
                    block.invoke();
                } catch (Exception e) {
                    Log.e(TAG, "协程执行异常", e);
                    handleException(e);
                }
            };
        }

        /**
         * 异步执行
         */
        public <T> Deferred<T> async(Dispatcher dispatcher, suspend () -> T block) {
            return scope.async(dispatcher) {
                try {
                    return block.invoke();
                } catch (Exception e) {
                    Log.e(TAG, "异步执行异常", e);
                    handleException(e);
                    throw e;
                }
            };
        }

        /**
         * 取消所有协程
         */
        public void cancelAll() {
            scope.cancel("取消所有协程");
        }

        /**
         * 检查作用域是否活跃
         */
        public boolean isActive() {
            return scope.isActive();
        }

        private void handleException(Exception e) {
            // 异常处理逻辑
            Log.e(TAG, "协程异常处理", e);
        }
    }

    /**
     * 协程调度器类型
     */
    public static class CoroutineDispatchers {
        public static final Dispatcher MAIN = Dispatchers.Main;
        public static final Dispatcher IO = Dispatchers.IO;
        public static final Dispatcher DEFAULT = Dispatchers.Default;
        public static final Dispatcher UNCONFINED = Dispatchers.Unconfined;

        /**
         * 获取调度器描述
         */
        public static String getDispatcherDescription(Dispatcher dispatcher) {
            if (dispatcher == MAIN) {
                return "主线程调度器 - 用于UI操作";
            } else if (dispatcher == IO) {
                return "IO调度器 - 用于网络和文件操作";
            } else if (dispatcher == DEFAULT) {
                return "默认调度器 - 用于CPU密集型操作";
            } else if (dispatcher == UNCONFINED) {
                return "无限制调度器 - 在当前线程执行";
            } else {
                return "未知调度器";
            }
        }

        /**
         * 根据操作类型选择调度器
         */
        public static Dispatcher selectDispatcher(String operationType) {
            switch (operationType.toLowerCase()) {
                case "ui":
                case "main":
                    return MAIN;
                case "network":
                case "file":
                case "io":
                    return IO;
                case "calculation":
                case "cpu":
                case "default":
                    return DEFAULT;
                default:
                    return IO;
            }
        }
    }
}
```

### 30.1.2 协程生命周期管理

```java
public class CoroutineLifecycleManagement {
    private static final String TAG = "CoroutineLifecycle";

    /**
     * 协程生命周期管理器
     */
    public static class CoroutineLifecycleManager implements LifecycleObserver {
        private final Map<String, Job> coroutineJobs = new HashMap<>();
        private final LifecycleOwner lifecycleOwner;

        public CoroutineLifecycleManager(LifecycleOwner lifecycleOwner) {
            this.lifecycleOwner = lifecycleOwner;
            lifecycleOwner.getLifecycle().addObserver(this);
        }

        /**
         * 启动生命周期感知的协程
         */
        public Job launchLifecycleAware(String key, CoroutineScope scope, Dispatcher dispatcher,
                                         suspend () -> Unit block) {
            // 取消已存在的协程
            cancelCoroutine(key);

            Job job = scope.launch(dispatcher) {
                // 检查生命周期状态
                if (!isLifecycleActive()) {
                    return;
                }

                try {
                    block.invoke();
                } catch (Exception e) {
                    Log.e(TAG, "协程执行异常: " + key, e);
                }
            };

            coroutineJobs.put(key, job);
            return job;
        }

        /**
         * 取消指定协程
         */
        public void cancelCoroutine(String key) {
            Job job = coroutineJobs.remove(key);
            if (job != null && job.isActive()) {
                job.cancel();
            }
        }

        /**
         * 检查生命周期是否活跃
         */
        public boolean isLifecycleActive() {
            Lifecycle.State currentState = lifecycleOwner.getLifecycle().getCurrentState();
            return currentState.isAtLeast(Lifecycle.State.STARTED);
        }

        /**
         * 生命周期状态变化回调
         */
        @OnLifecycleEvent(Lifecycle.Event.ON_STOP)
        public void onStop() {
            // 停止时取消所有协程
            cancelAllCoroutines();
        }

        @OnLifecycleEvent(Lifecycle.Event.ON_DESTROY)
        public void onDestroy() {
            // 销毁时清理资源
            cancelAllCoroutines();
            coroutineJobs.clear();
        }

        private void cancelAllCoroutines() {
            for (Map.Entry<String, Job> entry : coroutineJobs.entrySet()) {
                Job job = entry.getValue();
                if (job.isActive()) {
                    job.cancel();
                }
            }
            coroutineJobs.clear();
        }
    }

    /**
     * ViewModel协程管理器
     */
    public static class ViewModelCoroutineManager {
        private final Map<String, Job> viewModelJobs = new HashMap<>();
        private final CoroutineScope viewModelScope;

        public ViewModelCoroutineManager(CoroutineScope viewModelScope) {
            this.viewModelScope = viewModelScope;
        }

        /**
         * 在ViewModel中启动协程
         */
        public Job launchInViewModel(String key, Dispatcher dispatcher, suspend () -> Unit block) {
            // 取消已存在的协程
            cancelViewModelJob(key);

            Job job = viewModelScope.launch(dispatcher) {
                try {
                    block.invoke();
                } catch (Exception e) {
                    Log.e(TAG, "ViewModel协程执行异常: " + key, e);
                }
            };

            viewModelJobs.put(key, job);
            return job;
        }

        /**
         * 在ViewModel中异步执行
         */
        public <T> Deferred<T> asyncInViewModel(String key, Dispatcher dispatcher, suspend () -> T block) {
            cancelViewModelJob(key);

            Deferred<T> deferred = viewModelScope.async(dispatcher) {
                try {
                    return block.invoke();
                } catch (Exception e) {
                    Log.e(TAG, "ViewModel异步执行异常: " + key, e);
                    throw e;
                }
            };

            // 将Deferred作为Job保存
            viewModelJobs.put(key, deferred);
            return deferred;
        }

        /**
         * 取消ViewModel协程
         */
        public void cancelViewModelJob(String key) {
            Job job = viewModelJobs.remove(key);
            if (job != null && job.isActive()) {
                job.cancel();
            }
        }

        /**
         * 取消所有ViewModel协程
         */
        public void cancelAllViewModelJobs() {
            for (Map.Entry<String, Job> entry : viewModelJobs.entrySet()) {
                Job job = entry.getValue();
                if (job.isActive()) {
                    job.cancel();
                }
            }
            viewModelJobs.clear();
        }

        /**
         * 清理资源
         */
        public void cleanup() {
            cancelAllViewModelJobs();
        }
    }

    /**
     * 应用级别协程管理器
     */
    public static class ApplicationCoroutineManager {
        private static final String TAG = "AppCoroutineManager";
        private static ApplicationCoroutineManager instance;
        private final CoroutineScope applicationScope;
        private final Map<String, Job> applicationJobs = new ConcurrentHashMap<>();

        private ApplicationCoroutineManager() {
            this.applicationScope = new CoroutineScope(
                SupervisorJob() + Dispatchers.Default
            );
        }

        public static synchronized ApplicationCoroutineManager getInstance() {
            if (instance == null) {
                instance = new ApplicationCoroutineManager();
            }
            return instance;
        }

        /**
         * 在应用级别启动协程
         */
        public Job launchInApplication(String key, Dispatcher dispatcher, suspend () -> Unit block) {
            // 取消已存在的协程
            cancelApplicationJob(key);

            Job job = applicationScope.launch(dispatcher) {
                try {
                    block.invoke();
                } catch (Exception e) {
                    Log.e(TAG, "应用协程执行异常: " + key, e);
                }
            };

            applicationJobs.put(key, job);
            return job;
        }

        /**
         * 启动长期运行的后台任务
         */
        public Job launchBackgroundTask(String taskName, suspend () -> Unit block) {
            return launchInApplication(
                taskName,
                Dispatchers.IO,
                () -> {
                    Log.d(TAG, "启动后台任务: " + taskName);
                    try {
                        block.invoke();
                    } catch (Exception e) {
                        Log.e(TAG, "后台任务异常: " + taskName, e);
                    } finally {
                        Log.d(TAG, "后台任务结束: " + taskName);
                    }
                }
            );
        }

        /**
         * 取消应用协程
         */
        public void cancelApplicationJob(String key) {
            Job job = applicationJobs.remove(key);
            if (job != null && job.isActive()) {
                job.cancel();
            }
        }

        /**
         * 取消所有应用协程
         */
        public void cancelAllApplicationJobs() {
            for (Map.Entry<String, Job> entry : applicationJobs.entrySet()) {
                Job job = entry.getValue();
                if (job.isActive()) {
                    job.cancel();
                }
            }
            applicationJobs.clear();
        }

        /**
         * 获取应用协程作用域
         */
        public CoroutineScope getApplicationScope() {
            return applicationScope;
        }

        /**
         * 清理资源
         */
        public void cleanup() {
            cancelAllApplicationJobs();
            applicationScope.cancel();
        }
    }
}
```

## 30.2 协程并发和同步

协程提供了丰富的并发和同步原语，让我们能够轻松处理复杂的并发场景。

### 30.2.1 并发协程管理

```java
public class CoroutineConcurrency {
    private static final String TAG = "CoroutineConcurrency";

    /**
     * 并发协程管理器
     */
    public static class ConcurrentCoroutineManager {
        private final CoroutineScope scope;
        private final ExecutorService executor;

        public ConcurrentCoroutineManager() {
            this.scope = new CoroutineScope(SupervisorJob() + Dispatchers.Default);
            this.executor = Executors.newFixedThreadPool(4);
        }

        /**
         * 并发执行多个任务
         */
        public <T> List<T> executeConcurrent(List<suspend () -> T> tasks) {
            return scope.async(Dispatchers.IO) {
                val deferredList = tasks.map { task ->
                    async { task() }
                }
                deferredList.awaitAll()
            }.awaitBlocking();
        }

        /**
         * 并发执行并收集结果
         */
        public <T> Flow<T> executeConcurrentFlow(List<suspend () -> T> tasks) {
            return flow {
                tasks.forEach { task ->
                    launch {
                        val result = task()
                        emit(result)
                    }
                }
            }.flowOn(Dispatchers.IO);
        }

        /**
         * 并发执行带超时
         */
        public <T> List<T> executeConcurrentWithTimeout(
                List<suspend () -> T> tasks,
                long timeoutMillis
        ) {
            return scope.async {
                withTimeout(timeoutMillis) {
                    val deferredList = tasks.map { task ->
                        async { task() }
                    }
                    deferredList.awaitAll()
                }
            }.awaitBlocking();
        }

        /**
         * 批量处理任务
         */
        public <T, R> List<R> processBatch(
                List<T> items,
                int batchSize,
                suspend (List<T>) -> R processor
        ) {
            return scope.async(Dispatchers.IO) {
                val batches = items.chunked(batchSize)
                val deferredList = batches.map { batch ->
                    async { processor(batch) }
                }
                deferredList.awaitAll()
            }.awaitBlocking();
        }

        /**
         * 限流并发执行
         */
        public <T> Flow<T> executeWithRateLimit(
                List<suspend () -> T> tasks,
                int maxConcurrency,
                long delayMillis
        ) {
            return flow {
                val semaphore = Semaphore(maxConcurrency)
                tasks.forEach { task ->
                    launch {
                        semaphore.acquire()
                        try {
                            val result = task()
                            emit(result)
                        } finally {
                            semaphore.release()
                            delay(delayMillis)
                        }
                    }
                }
            }.flowOn(Dispatchers.IO);
        }

        /**
         * 管道模式处理
         */
        public <T, R> Flow<R> pipelineProcess(
                Flow<T> inputFlow,
                int bufferSize,
                suspend (T) -> R processor
        ) {
            return inputFlow
                .map { input ->
                    scope.async(Dispatchers.IO) {
                        processor(input)
                    }
                }
                .buffer(bufferSize)
                .map { deferred ->
                    deferred.await()
                }
                .flowOn(Dispatchers.IO);
        }

        /**
         * 工作者池模式
         */
        public <T> Flow<T> workerPoolProcess(
                List<suspend () -> T> tasks,
                int workerCount
        ) {
            return channelFlow {
                val inputChannel = Channel<suspend () -> T>(Channel.UNLIMITED)

                // 启动工作者
                repeat(workerCount) { workerId ->
                    launch(Dispatchers.IO) {
                        for (task in inputChannel) {
                            try {
                                val result = task()
                                send(result)
                            } catch (e: Exception) {
                                Log.e(TAG, "工作者 $workerId 执行任务失败", e)
                            }
                        }
                    }
                }

                // 发送任务
                tasks.forEach { task ->
                    inputChannel.send(task)
                }

                // 关闭通道
                inputChannel.close()
                close()
            }.flowOn(Dispatchers.IO);
        }

        /**
         * 清理资源
         */
        public void cleanup() {
            scope.cancel();
            executor.shutdown();
        }
    }

    /**
     * 协程同步原语
     */
    public static class CoroutineSynchronization {
        /**
         * 互斥锁示例
         */
        public static class MutexExample {
            private final Mutex mutex = new Mutex();
            private int counter = 0;

            public suspend fun incrementCounter(): Int {
                mutex.withLock {
                    counter++
                    delay(100) // 模拟耗时操作
                    return counter
                }
            }

            public suspend fun getCounter(): Int {
                return mutex.withLock {
                    counter
                }
            }
        }

        /**
         * 信号量示例
         */
        public static class SemaphoreExample {
            private final Semaphore semaphore = new Semaphore(3); // 最多3个并发访问

            public suspend fun performLimitedOperation(operation: suspend () -> String): String {
                semaphore.acquire()
                try {
                    return operation()
                } finally {
                    semaphore.release()
                }
            }
        }

        /**
         * 条件变量示例
         */
        public static class ConditionVariableExample {
            private val condition = ConditionVariable(Mutex())
            private var dataReady = false
            private var data: String? = null

            public suspend fun waitForData(): String {
                condition.withLock {
                    while (!dataReady) {
                        condition.wait()
                    }
                    return data!!
                }
            }

            public suspend fun provideData(newData: String) {
                condition.withLock {
                    data = newData
                    dataReady = true
                    condition.signalAll()
                }
            }
        }

        /**
         * Actor示例
         */
        public static class ActorExample {
            private val actor = scope.actor<String>(capacity = Channel.UNLIMITED) { message ->
                // 处理消息
                processMessage(message)
            }

            public suspend fun sendMessage(message: String) {
                actor.send(message)
            }

            private suspend fun processMessage(message: String) {
                Log.d(TAG, "处理消息: $message")
                delay(100) // 模拟处理时间
            }
        }

        /**
         * 生产者-消费者模式
         */
        public static class ProducerConsumerPattern {
            private val channel = Channel<String>(capacity = Channel.UNLIMITED)
            private val producers = mutableListOf<Job>()
            private val consumers = mutableListOf<Job>()

            public suspend fun startProducers(count: Int) {
                repeat(count) { producerId ->
                    val producer = launch(Dispatchers.IO) {
                        repeat(10) { itemId ->
                            val item = "Producer-$producerId-Item-$itemId"
                            channel.send(item)
                            Log.d(TAG, "生产: $item")
                            delay(100)
                        }
                    }
                    producers.add(producer)
                }
            }

            public suspend fun startConsumers(count: Int) {
                repeat(count) { consumerId ->
                    val consumer = launch(Dispatchers.IO) {
                        for (item in channel) {
                            Log.d(TAG, "消费者-$consumerId 消费: $item")
                            delay(150)
                        }
                    }
                    consumers.add(consumer)
                }
            }

            public suspend fun stop() {
                producers.forEach { it.cancel() }
                consumers.forEach { it.cancel() }
                channel.close()
            }
        }
    }
}
```

### 30.2.2 协程异常处理

```java
public class CoroutineExceptionHandling {
    private static final String TAG = "CoroutineException";

    /**
     * 协程异常处理器
     */
    public static class CoroutineExceptionHandler {
        /**
         * 全局异常处理器
         */
        public static final CoroutineExceptionHandler GLOBAL_HANDLER =
            new CoroutineExceptionHandler() { (context, throwable) ->
                Log.e(TAG, "全局协程异常: ${context[CoroutineName]}", throwable);
                // 发送错误报告
                // 记录崩溃日志
            };

        /**
         * 创建带异常处理的协程作用域
         */
        public static CoroutineScope createSafeScope(Dispatcher dispatcher) {
            return new CoroutineScope(
                SupervisorJob() + dispatcher + GLOBAL_HANDLER
            );
        }

        /**
         * 安全启动协程
         */
        public static Job launchSafely(
                CoroutineScope scope,
                Dispatcher dispatcher,
                suspend () -> Unit block
        ) {
            scope.launch(dispatcher + GLOBAL_HANDLER) {
                try {
                    block.invoke();
                } catch (Exception e) {
                    Log.e(TAG, "协程执行异常", e);
                    // 处理异常
                    handleException(e);
                }
            };
        }

        /**
         * 安全异步执行
         */
        public static <T> Deferred<T> asyncSafely(
                CoroutineScope scope,
                Dispatcher dispatcher,
                suspend () -> T block
        ) {
            return scope.async(dispatcher + GLOBAL_HANDLER) {
                try {
                    return block.invoke();
                } catch (Exception e) {
                    Log.e(TAG, "异步执行异常", e);
                    handleException(e);
                    throw e;
                }
            };
        }

        /**
         * 异常处理逻辑
         */
        private static void handleException(Exception e) {
            // 根据异常类型进行不同处理
            if (e instanceof CancellationException) {
                Log.d(TAG, "协程被取消");
            } else if (e instanceof TimeoutCancellationException) {
                Log.w(TAG, "协程执行超时");
            } else if (e instanceof IOException) {
                Log.e(TAG, "IO异常", e);
            } else {
                Log.e(TAG, "未知异常", e);
            }
        }
    }

    /**
     * 协程错误恢复机制
     */
    public static class CoroutineErrorRecovery {
        /**
         * 重试机制
         */
        public static suspend fun <T> retry(
                maxAttempts: Int,
                delayMillis: Long = 1000,
                block: suspend () -> T
        ): T {
            repeat(maxAttempts - 1) { attempt ->
                try {
                    return block()
                } catch (e: Exception) {
                    Log.w(TAG, "重试第 ${attempt + 1} 次", e);
                    delay(delayMillis * (attempt + 1)); // 指数退避
                }
            }
            return block() // 最后一次不捕获异常
        }

        /**
         * 条件重试
         */
        public static suspend fun <T> retryWhen(
                maxAttempts: Int,
                condition: suspend (Exception, Int) -> Boolean,
                block: suspend () -> T
        ): T {
            repeat(maxAttempts - 1) { attempt ->
                try {
                    return block()
                } catch (e: Exception) {
                    if (condition(e, attempt + 1)) {
                        Log.w(TAG, "条件重试第 ${attempt + 1} 次", e);
                        delay(1000L * (attempt + 1));
                    } else {
                        throw e
                    }
                }
            }
            return block()
        }

        /**
         * 超时重试
         */
        public static suspend fun <T> retryWithTimeout(
                maxAttempts: Int,
                timeoutMillis: Long,
                block: suspend () -> T
        ): T {
            repeat(maxAttempts - 1) { attempt ->
                try {
                    return withTimeout(timeoutMillis) {
                        block()
                    }
                } catch (e: TimeoutCancellationException) {
                    Log.w(TAG, "超时重试第 ${attempt + 1} 次", e);
                    delay(1000L * (attempt + 1));
                }
            }
            return withTimeout(timeoutMillis) {
                block()
            }
        }

        /**
         * 熔断器模式
         */
        public static class CircuitBreaker<T> {
            private var state: State = State.CLOSED
            private var failureCount = 0
            private var lastFailureTime = 0L
            private val maxFailures = 5
            private val timeoutMillis = 60000L // 1分钟

            enum class State {
                CLOSED, OPEN, HALF_OPEN
            }

            public suspend fun execute(block: suspend () -> T): T {
                when (state) {
                    State.OPEN -> {
                        if (System.currentTimeMillis() - lastFailureTime > timeoutMillis) {
                            state = State.HALF_OPEN
                        } else {
                            throw CircuitBreakerOpenException("熔断器开启")
                        }
                    }
                    State.HALF_OPEN -> {
                        try {
                            val result = block()
                            reset()
                            return result
                        } catch (e: Exception) {
                            trip()
                            throw e
                        }
                    }
                    State.CLOSED -> {
                        try {
                            val result = block()
                            reset()
                            return result
                        } catch (e: Exception) {
                            trip()
                            throw e
                        }
                    }
                }
            }

            private fun trip() {
                failureCount++
                lastFailureTime = System.currentTimeMillis()
                if (failureCount >= maxFailures) {
                    state = State.OPEN
                    Log.w(TAG, "熔断器开启，失败次数: $failureCount")
                }
            }

            private fun reset() {
                failureCount = 0
                state = State.CLOSED
            }

            private static class CircuitBreakerOpenException extends RuntimeException {
                public CircuitBreakerOpenException(String message) {
                    super(message);
                }
            }
        }

        /**
         * 降级机制
         */
        public static class FallbackMechanism {
            public static suspend fun <T> withFallback(
                    primaryOperation: suspend () -> T,
                    fallbackOperation: suspend () -> T
            ): T {
                try {
                    return primaryOperation()
                } catch (e: Exception) {
                    Log.w(TAG, "主操作失败，使用降级方案", e)
                    return fallbackOperation()
                }
            }

            public static suspend fun <T> withTimeoutFallback(
                    operation: suspend () -> T,
                    timeoutMillis: Long,
                    fallbackOperation: suspend () -> T
            ): T {
                try {
                    return withTimeout(timeoutMillis) {
                        operation()
                    }
                } catch (e: TimeoutCancellationException) {
                    Log.w(TAG, "操作超时，使用降级方案", e)
                    return fallbackOperation()
                }
            }
        }
    }
}
```

```java
                    val localUsers = database.userDao().getAllUsers().first()

                                }
                                ConflictResolutionStrategy.MERGE -> {

                                    val updatedRows = database.userDao().updateUser(mergedUser)
                                    syncResult.addUpdated(updatedRows)
                                }

                        }
```

## 30.4 协程性能优化

协程虽然提供了便利的异步编程方式，但在实际使用中需要注意性能优化。

### 30.4.1 协程性能监控

```java
public class CoroutinePerformanceOptimization {
    private static final String TAG = "CoroutinePerf";

    /**
     * 协程性能监控器
     */
    public static class CoroutinePerformanceMonitor {
        private final Map<String, PerformanceMetrics> metricsMap = new ConcurrentHashMap<>();

        /**
         * 监控协程执行
         */
        public <T> Deferred<T> monitorCoroutine(
                String name,
                CoroutineScope scope,
                Dispatcher dispatcher,
                suspend () -> T block
        ) {
            return scope.async(dispatcher) {
                val startTime = System.nanoTime();
                val startMemory = getUsedMemory();

                try {
                    val result = block();

                    val endTime = System.nanoTime();
                    val endMemory = getUsedMemory();

                    recordMetrics(name, startTime, endTime, startMemory, endMemory, null);

                    return result;

                } catch (Exception e) {
                    val endTime = System.nanoTime();
                    val endMemory = getUsedMemory();

                    recordMetrics(name, startTime, endTime, startMemory, endMemory, e);
                    throw e;
                }
            };
        }

        /**
         * 记录性能指标
         */
        private void recordMetrics(
                String name,
                long startTime,
                long endTime,
                long startMemory,
                long endMemory,
                Exception exception
        ) {
            val duration = (endTime - startTime) / 1_000_000; // 转换为毫秒
            val memoryDelta = endMemory - startMemory;

            PerformanceMetrics metrics = metricsMap.computeIfAbsent(name, k -> new PerformanceMetrics());

            metrics.addExecution(duration, memoryDelta, exception);

            if (metrics.getTotalExecutions() % 10 == 0) {
                Log.d(TAG, String.format("协程 %s 性能统计: %s", name, metrics.getSummary()));
            }
        }

        private long getUsedMemory() {
            Runtime runtime = Runtime.getRuntime();
            return runtime.totalMemory() - runtime.freeMemory();
        }

        /**
         * 获取性能指标
         */
        public PerformanceMetrics getMetrics(String name) {
            return metricsMap.get(name);
        }

        /**
         * 获取所有性能指标
         */
        public Map<String, PerformanceMetrics> getAllMetrics() {
            return new HashMap<>(metricsMap);
        }

        /**
         * 清理指标
         */
        public void clearMetrics() {
            metricsMap.clear();
        }
    }

    /**
     * 性能指标类
     */
    public static class PerformanceMetrics {
        private final List<Long> executionTimes = new ArrayList<>();
        private final List<Long> memoryDeltas = new ArrayList<>();
        private final List<Exception> exceptions = new ArrayList<>();

        public synchronized void addExecution(long duration, long memoryDelta, Exception exception) {
            executionTimes.add(duration);
            memoryDeltas.add(memoryDelta);
            if (exception != null) {
                exceptions.add(exception);
            }
        }

        public synchronized long getAverageExecutionTime() {
            if (executionTimes.isEmpty()) return 0;
            return executionTimes.stream().mapToLong(Long::longValue).sum() / executionTimes.size();
        }

        public synchronized long getMaxExecutionTime() {
            return executionTimes.stream().mapToLong(Long::longValue).max().orElse(0);
        }

        public synchronized long getMinExecutionTime() {
            return executionTimes.stream().mapToLong(Long::longValue).min().orElse(0);
        }

        public synchronized long getAverageMemoryDelta() {
            if (memoryDeltas.isEmpty()) return 0;
            return memoryDeltas.stream().mapToLong(Long::longValue).sum() / memoryDeltas.size();
        }

        public synchronized double getErrorRate() {
            int totalExecutions = executionTimes.size();
            if (totalExecutions == 0) return 0.0;
            return (double) exceptions.size() / totalExecutions * 100;
        }

        public synchronized int getTotalExecutions() {
            return executionTimes.size();
        }

        public synchronized String getSummary() {
            return String.format(
                "执行次数: %d, 平均耗时: %dms, 最大耗时: %dms, 内存变化: %dKB, 错误率: %.2f%%",
                getTotalExecutions(),
                getAverageExecutionTime(),
                getMaxExecutionTime(),
                getAverageMemoryDelta() / 1024,
                getErrorRate()
            );
        }
    }

    /**
     * 协程池优化
     */
    public static class CoroutinePoolOptimizer {
        private static final int CPU_COUNT = Runtime.getRuntime().availableProcessors();
        private static final int DEFAULT_POOL_SIZE = CPU_COUNT + 1;

        /**
         * 创建IO密集型协程调度器
         */
        public static Dispatcher createIODispatcher() {
            return Dispatchers.IO.limitedParallelism(64);
        }

        /**
         * 创建CPU密集型协程调度器
         */
        public static Dispatcher createCPUDispatcher() {
            return Dispatchers.Default.limitedParallelism(CPU_COUNT);
        }

        /**
         * 创建自定义调度器
         */
        public static Dispatcher createCustomDispatcher(
                String name,
                int threadCount,
                ThreadPriority priority
        ) {
            val threadFactory = ThreadFactory { runnable ->
                Thread(runnable, name).apply {
                    setPriority(priority.priority)
                }
            }

            val executor = Executors.newFixedThreadPool(threadCount, threadFactory)
            return executor.asCoroutineDispatcher()
        }

        /**
         * 动态调整线程池大小
         */
        public static class DynamicThreadPool {
            private val minThreads = 2
            private val maxThreads = CPU_COUNT * 2
            private var currentThreads = minThreads
            private val executor = Executors.newFixedThreadPool(currentThreads) as ThreadPoolExecutor
            private val dispatcher = executor.asCoroutineDispatcher()

            public synchronized Dispatcher getDispatcher() {
                return dispatcher
            }

            public synchronized void adjustPoolSize(int newThreadCount) {
                val boundedSize = maxOf(minThreads, minOf(newThreadCount, maxThreads))
                if (boundedSize != currentThreads) {
                    executor.setCorePoolSize(boundedSize)
                    executor.setMaximumPoolSize(boundedSize)
                    currentThreads = boundedSize
                    Log.d(TAG, "线程池大小调整为: $boundedSize")
                }
            }

            public synchronized void increasePoolSize() {
                adjustPoolSize(currentThreads + 1)
            }

            public synchronized void decreasePoolSize() {
                adjustPoolSize(currentThreads - 1)
            }
        }
    }

    /**
     * 协程内存优化
     */
    public static class CoroutineMemoryOptimizer {
        private static final long MEMORY_THRESHOLD = 50 * 1024 * 1024; // 50MB

        /**
         * 内存敏感的协程作用域
         */
        public static class MemoryAwareScope {
            private final CoroutineScope scope;
            private final MemoryMonitor memoryMonitor;

            public MemoryAwareScope(CoroutineScope scope) {
                this.scope = scope;
                this.memoryMonitor = new MemoryMonitor();
            }

            public <T> Deferred<T> launchMemoryAware(
                    Dispatcher dispatcher,
                    suspend () -> T block
            ) {
                return scope.async(dispatcher) {
                    memoryMonitor.checkMemory()
                    try {
                        return block()
                    } finally {
                        memoryMonitor.checkMemory()
                    }
                };
            }
        }

        /**
         * 内存监控器
         */
        private static class MemoryMonitor {
            public void checkMemory() {
                val usedMemory = getUsedMemory()
```
