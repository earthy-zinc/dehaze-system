```kotlin
// 型变示例
// Producer - 协变（只读）
interface Producer<out T> {
    fun produce(): T
}

class StringProducer : Producer<String> {
    override fun produce(): String = "Hello"
}

// Consumer - 逆变（只写）
interface Consumer<in T> {
    fun consume(item: T)
}

class StringConsumer : Consumer<String> {
    override fun consume(item: String) {
        println("Consuming: $item")
    }
}

// 型变的实际应用
fun demonstrateVariance() {
    // 协变示例
    val stringProducer: Producer<String> = StringProducer()
    val anyProducer: Producer<Any> = stringProducer // 协变：Producer<String> 是 Producer<Any> 的子类型

    val anyValue = anyProducer.produce()
    println("Produced: $anyValue")

    // 逆变示例
    val anyConsumer: Consumer<Any> = object : Consumer<Any> {
        override fun consume(item: Any) {
            println("Consuming Any: $item")
        }
    }
    val stringConsumer: Consumer<String> = anyConsumer // 逆变：Consumer<Any> 是 Consumer<String> 的子类型

    stringConsumer.consume("Hello")

    // 星号投影
    val starProducer: Producer<*> = stringProducer
    val producedValue = starProducer.produce() // 类型为Any?
    println("Star produced: $producedValue")
}

// 型变点的使用
class Box<out T>(private val value: T) {
    // 只能读取，不能写入
    fun getValue(): T = value

    // 这样编译错误：
    // fun setValue(value: T) { this.value = value }
}

class MutableBox<T>(private var value: T) {
    fun getValue(): T = value
    fun setValue(value: T) {
        this.value = value
    }

    // 使用型变点
    fun copy(): MutableBox<@UnsafeVariance T> {
        return MutableBox(value)
    }
}
```

### 9.2.2 注解处理器

```kotlin
// 注解处理器接口
interface AnnotationProcessor {
    fun processAnnotations(elements: List<AnnotatedElement>): ProcessingResult
}

data class ProcessingResult(
    val success: Boolean,
    val messages: List<String> = emptyList(),
    val generatedFiles: List<GeneratedFile> = emptyList()
)

data class GeneratedFile(
    val fileName: String,
    val content: String,
    val packageName: String = ""
)

// 实际的注解处理器示例
class ValidationProcessor : AnnotationProcessor {
    override fun processAnnotations(elements: List<AnnotatedElement>): ProcessingResult {
        val generatedFiles = mutableListOf<GeneratedFile>()
        val messages = mutableListOf<String>()

        elements.forEach { element ->
            val requiredAnnotations = element.annotations.filterIsInstance<Required>()

            if (requiredAnnotations.isNotEmpty()) {
                val validatorClass = generateValidator(element, requiredAnnotations)
                generatedFiles.add(validatorClass)
                messages.add("Generated validator for ${element.simpleName}")
            }
        }

        return ProcessingResult(
            success = true,
            messages = messages,
            generatedFiles = generatedFiles
        )
    }

    private fun generateValidator(element: AnnotatedElement, annotations: List<Required>): GeneratedFile {
        val className = "${element.simpleName}Validator"
        val packageName = element.packageName
        val fields = annotations.joinToString("\n    ") { annotation ->
            "private fun validate${annotation.propertyName.capitalize()}(value: String?): Boolean { ${generateValidationCode(annotation)} }"
        }

        val content = """
            package $packageName

            import kotlin.reflect.KProperty1

            class $className {
                fun validate(obj: ${element.simpleName}): ValidationResult {
                    val errors = mutableListOf<String>()

                    $fields

                    return if (errors.isEmpty()) {
                        ValidationResult.Success
                    } else {
                        ValidationResult.Failure(errors)
                    }
                }
            }
        """.trimIndent()

        return GeneratedFile(
            fileName = "$className.kt",
            content = content,
            packageName = packageName
        )
    }

    private fun generateValidationCode(annotation: Required): String {
        return """
            if (value.isNullOrBlank()) {
                errors.add("${annotation.message}")
                return false
            }
            true
        """.trimIndent()
    }
}
```

```kotlin
// 动态代理示例
class DynamicProxy(private val target: Any) : InvocationHandler {
    private val interceptors = mutableListOf<MethodInterceptor>()

    fun addInterceptor(interceptor: MethodInterceptor) {
        interceptors.add(interceptor)
    }

    override fun invoke(proxy: Any, method: Method, args: Array<out Any?>): Any? {
        // 前置拦截器
        interceptors.forEach { it.before(target, method, args) }

        try {
            val result = method.invoke(target, *args)

            // 后置拦截器
            interceptors.forEach { it.after(target, method, args, result) }

            return result
        } catch (e: Exception) {
            // 异常拦截器
            interceptors.forEach { it.onError(target, method, args, e) }
            throw e
        }
    }
}

interface MethodInterceptor {
    fun before(target: Any, method: Method, args: Array<out Any?>) {}
    fun after(target: Any, method: Method, args: Array<out Any?>, result: Any?) {}
    fun onError(target: Any, method: Method, args: Array<out Any?>, error: Exception) {}
}
```

```kotlin
// HTML DSL 示例
class Tag(val name: String) {
    val attributes = mutableMapOf<String, String>()
    val children = mutableListOf<Tag>()

    fun attribute(name: String, value: String) {
        attributes[name] = value
    }

    fun child(tag: Tag) {
        children.add(tag)
    }

    fun addText(text: String) {
        children.add(TextNode(text))
    }

    override fun toString(): String {
        val attributesStr = if (attributes.isNotEmpty()) {
            " " + attributes.entries.joinToString(" ") { "${it.key}=\"${it.value}\"" }
        } else ""

        val childrenStr = children.joinToString("") { it.toString() }

        return if (children.isNotEmpty()) {
            "<$name$attributesStr>$childrenStr</$name>"
        } else {
            "<$name$attributesStr />"
        }
    }
}

class TextNode(val text: String) {
    override fun toString(): String = text
}

// HTML构建DSL
fun html(block: Tag.() -> Unit): Tag {
    val root = Tag("html")
    root.block()
    return root
}

fun Tag.head(block: Tag.() -> Unit) {
    val head = Tag("head")
    head.block()
    child(head)
}

fun Tag.body(block: Tag.() -> Unit) {
    val body = Tag("body")
    body.block()
    child(body)
}

fun Tag.title(text: String) {
    val title = Tag("title")
    title.addText(text)
    child(title)
}

fun Tag.h1(text: String) {
    val h1 = Tag("h1")

}
```

```kotlin
// SQL构建DSL
class SqlBuilder {
    private var selectClause = "*"
    private var fromClause = ""
    private var whereClause = ""
    private val joinClauses = mutableListOf<String>()
    private val groupByClauses = mutableListOf<String>()
    private val orderByClauses = mutableListOf<String>()
    private var limitClause = ""

    fun select(vararg columns: String): SqlBuilder {
        selectClause = if (columns.isEmpty()) "*" else columns.joinToString(", ")
        return this
    }

    fun from(table: String): SqlBuilder {
        fromClause = table
        return this
    }

    fun where(condition: String): SqlBuilder {
        whereClause = if (whereClause.isEmpty()) {
            "WHERE $condition"
        } else {
            "$whereClause AND $condition"
        }
        return this
    }

    fun innerJoin(table: String, condition: String): SqlBuilder {
        joinClauses.add("INNER JOIN $table ON $condition")
        return this
    }

    fun leftJoin(table: String, condition: String): SqlBuilder {
        joinClauses.add("LEFT JOIN $table ON $condition")
        return this
    }

    fun groupBy(vararg columns: String): SqlBuilder {
        groupByClauses.addAll(columns)
        return this
    }

    fun orderBy(column: String, direction: String = "ASC"): SqlBuilder {
        orderByClauses.add("$column $direction")
        return this
    }

    fun limit(count: Int): SqlBuilder {
        limitClause = "LIMIT $count"
        return this
    }

    fun build(): String {
        val parts = mutableListOf<String>()

        parts.add("SELECT $selectClause")
        parts.add("FROM $fromClause")

        joinClauses.forEach { parts.add(it) }

        if (whereClause.isNotEmpty()) {
            parts.add(whereClause)
        }

        if (groupByClauses.isNotEmpty()) {
            parts.add("GROUP BY ${groupByClauses.joinToString(", ")}")
        }

        if (orderByClauses.isNotEmpty()) {
            parts.add("ORDER BY ${orderByClauses.joinToString(", ")}")
        }

        if (limitClause.isNotEmpty()) {
            parts.add(limitClause)
        }

        return parts.joinToString(" ")
    }
}
```

```kotlin
object TypeChecker {

    // 安全的类型转换
    inline fun <reified T> safeCast(value: Any?): T? {
        return if (value is T) value else null
    }

    // 类型检查
    inline fun <reified T> isType(value: Any?): Boolean {
        return value is T
    }

    // 获取类型信息
    inline fun <reified T> getTypeInfo(): TypeInfo {
        return TypeInfo(
            simpleName = T::class.simpleName ?: "Unknown",
            qualifiedName = T::class.qualifiedName ?: "Unknown",
            isSealed = T::class.isSealed,
            isData = T::class.isData,
            isCompanionObject = T::class.isCompanion
        )
    }

    // 检查是否为集合类型
    fun isCollection(value: Any?): Boolean {
        return value is Collection<*>
    }

    // 获取集合元素类型
    fun getCollectionElementType(value: Collection<*>: Class<*>? {
        if (value.isEmpty()) return null
        return value.firstOrNull()?.let { it?.javaClass }
    }
}

object ObjectCreator {

    // 创建对象实例
    fun <T> createInstance(
        clazz: Class<T>,
        arguments: Map<String, Any?> = emptyMap()
    ): T {
        val constructor = findBestConstructor(clazz, arguments)
        val args = prepareArguments(constructor, arguments)
        return constructor.call(*args)
    }

    private fun <T> findBestConstructor(
        clazz: Class<T>,
        arguments: Map<String, Any?>
    ): KFunction<T> {
        val constructors = clazz.kotlin.constructors
        return constructors.maxByOrNull { constructor ->
            val paramNames = constructor.parameters.map { it.name ?: "" }
            val matchCount = paramNames.count { it in arguments.keys }
            matchCount
        } ?: throw IllegalArgumentException("No suitable constructor found")
    }

    private fun <T> prepareArguments(
        constructor: KFunction<T>,
        arguments: Map<String, Any?>
    ): Array<Any?> {
        return constructor.parameters.map { param ->
            val paramName = param.name ?: ""
            arguments[paramName]
        }.toTypedArray()
    }
}

    fun hasMethod(obj: Any, methodName: String, vararg parameterTypes: KClass<*>): Boolean {
        return obj::class.functions.any { method ->
            method.name == methodName && method.parameters.drop(1).size == parameterTypes.size &&
            method.parameters.drop(1).zip(parameterTypes).all { (param, type) ->
                param.type.classifier == type
            }
        }
    }
```

```kotlin
// 内存优化工具类
object MemoryOptimizer {

    // 对象池
    private val objectPools = mutableMapOf<Class<*>, MutableList<Any>>()

    fun <T> getFromPool(clazz: Class<T>, factory: () -> T): T {
        val pool = objectPools.getOrPut(clazz) { mutableListOf() }
        return synchronized(pool) {
            if (pool.isNotEmpty()) {
                @Suppress("UNCHECKED_CAST")
                pool.removeAt(pool.size - 1) as T
            } else {
                factory()
            }
        }
    }

    fun <T> returnToPool(obj: T) {
        val pool = objectPools.getOrPut(obj!!::class) { mutableListOf() }
        synchronized(pool) {
            pool.add(obj)
        }
    }

    // 内存使用监控
    fun getMemoryUsage(): MemoryInfo {
        val runtime = Runtime.getRuntime()

            totalMemory = totalMemory,
            usedMemory = usedMemory,
            freeMemory = freeMemory,
            maxMemory = maxMemory,
            usagePercentage = (usedMemory.toDouble() / maxMemory * 100)
        )
    }

    // 执行内存清理
    fun gc() {
        System.gc()
        Thread.sleep(100) // 给GC一些时间
    }

    // 内存泄漏检测
    fun detectMemoryLeaks(): List<MemoryLeak> {
        val leaks = mutableListOf<MemoryLeak>()

        val staticFields = MemoryLeakDetector.getStaticFields()
```

```kotlin
class WeakReferenceManager<T> {
    private val references = mutableMapOf<String, WeakReference<T>>()

    fun put(key: String, value: T) {
        references[key] = WeakReference(value)
    }

    }

    fun remove(key: String): T? {
        val ref = references.remove(key)
        return ref?.get()

    fun cleanup() {

// 内存感知缓存
class MemoryAwareCache<K, V>(
    private val maxSize: Int,
    private val maxMemoryMB: Double = 100.0
) {
    private val cache = mutableMapOf<K, V>()
    private val accessOrder = mutableListOf<K>()
    private val memoryCalculator = MemoryCalculator()

    private var currentMemoryUsage = 0L

    fun put(key: K, value: V) {
        // 计算新值的内存占用
        val valueSize = memoryCalculator.calculateSize(value)

        // 检查是否超过最大内存限制
        if (currentMemoryUsage + valueSize > maxMemoryMB * 1024 * 1024) {
            cleanup()
        }

        // 如果缓存已满，移除最少使用的项
        if (cache.size >= maxSize) {
            removeLeastRecentlyUsed()
        }

        // 移除现有的键（如果存在）
        cache[key]?.let { oldValue ->
            currentMemoryUsage -= memoryCalculator.calculateSize(oldValue)
            accessOrder.remove(key)
        }

        // 添加新值
        cache[key] = value
        accessOrder.add(key)
        currentMemoryUsage += valueSize
    }

    fun get(key: K): V? {
        return cache[key]?.also {
            // 更新访问顺序
            accessOrder.remove(key)
            accessOrder.add(key)
        }
    }

    private fun removeLeastRecentlyUsed() {
        if (accessOrder.isNotEmpty()) {
            val lruKey = accessOrder.removeAt(0)
            cache.remove(lruKey)?.let { oldValue ->
                currentMemoryUsage -= memoryCalculator.calculateSize(oldValue)
            }
        }
    }

    private fun cleanup() {
        // 移除最旧的项目直到内存使用在限制内
        while (currentMemoryUsage > maxMemoryMB * 1024 * 1024 && accessOrder.isNotEmpty()) {
            removeLeastRecentlyUsed()
        }
    }

    fun clear() {
        cache.clear()
        accessOrder.clear()
        currentMemoryUsage = 0L
    }

    fun size(): Int = cache.size

    fun memoryUsage(): Long = currentMemoryUsage
}

// 内存计算器
class MemoryCalculator {
    private val primitiveSizes = mapOf(
        Byte::class to 1,
        Short::class to 2,
        Int::class to 4,
        Long::class to 8,
        Float::class to 4,
        Double::class to 8,
        Boolean::class to 1,
        Char::class to 2
    )

    fun calculateSize(obj: Any?): Long {
        if (obj == null) return 0

        val clazz = obj!!::class

        // 基本类型
        primitiveSizes[clazz]?.let { return it.toLong() }

        return when {
            obj is String -> obj.toByteArray().size + 24 // String对象开销
            obj is Collection<*> -> {
                var size = 24 // 集合对象开销
                obj.forEach { size += calculateSize(it) }
                size
            }
            obj is Map<*, *> -> {
                var size = 48 // Map对象开销
                obj.entries.forEach { entry ->
                    size += calculateSize(entry.key)
                    size += calculateSize(entry.value)
                }
                size
            }
            obj is Array<*> -> {
                var size = 24 // 数组对象开销
                obj.forEach { size += calculateSize(it) }
                size
            }
            else -> {
                // 简化的对象大小估算
                64 + obj::class.declaredMembers.size * 8
            }
        }
    }
}
```

```kotlin
object PerformanceProfiler {

    private val methodTimes = mutableMapOf<String, Long>()
    private val methodCounts = mutableMapOf<String, Int>()
    private val memoryUsage = mutableMapOf<String, Long>()

    inline fun <T> profileMethod(
        methodName: String,
        trackMemory: Boolean = false,
        block: () -> T
    ): T {
        val startTime = System.nanoTime()
        val startMemory = if (trackMemory) {
            Runtime.getRuntime().totalMemory() - Runtime.getRuntime().freeMemory()
        } else 0

        // 增加调用次数
        methodCounts[methodName] = methodCounts.getOrDefault(methodName, 0) + 1

        try {
            val result = block()
            val endTime = System.nanoTime()
            val duration = endTime - startTime

            // 更新执行时间
            methodTimes[methodName] = methodTimes.getOrDefault(methodName, 0L) + duration

            if (trackMemory) {
                val endMemory = Runtime.getRuntime().totalMemory() - Runtime.getRuntime().freeMemory()
                val memoryUsed = endMemory - startMemory
                memoryUsage[methodName] = memoryUsage.getOrDefault(methodName, 0L) + memoryUsed
            }

            return result
        } catch (e: Exception) {
            val endTime = System.nanoTime()
            val duration = endTime - startTime
            methodTimes[methodName] = methodTimes.getOrDefault(methodName, 0L) + duration

            throw e
        }

        return PerformanceReport(
            methodTimes = methodTimes.mapValues { it.value / 1_000_000 }, // 转换为毫秒
            methodCounts = methodCounts,
            memoryUsage = memoryUsage,
            totalMethods = methodCounts.size
        )
    }

    fun clear() {
        methodTimes.clear()
        methodCounts.clear()
        memoryUsage.clear()
    }
```

```kotlin
        return benchmarkResult
    }

    private fun calculateStandardDeviation(measurements: List<Long>): Double {
        val mean = measurements.average()
        val variance = measurements.map { (it - mean).pow(2) }.average()
        return Math.sqrt(variance)
    }

    fun compareResults(name1: String, name2: String): ComparisonResult {
        val result1 = results.find { it.name == name1 }
        val result2 = results.find { it.name == name2 }

        return if (result1 != null && result2 != null) {
            val speedup = result2.averageTime.toDouble() / result1.averageTime.toDouble()
            ComparisonResult(
                name1 = name1,
                name2 = name2,
                speedup = speedup,
                winner = if (speedup > 1.0) name1 else name2,
                improvement = "${String.format("%.2f", (if (speedup > 1.0) (speedup - 1.0) else (1.0 / speedup - 1.0) * 100)}%"
            )
        } else {
            throw IllegalArgumentException("Benchmark results not found")
        }
    }

    fun generateReport(): BenchmarkReport {
        return BenchmarkReport(
            results = results.toList(),
            generatedAt = System.currentTimeMillis(),
            summary = generateSummary()
        )
    }

    private fun generateSummary(): String {
        return buildString {
            appendLine("Benchmark Summary:")
            appendLine("Total benchmarks: ${results.size}")

            if (results.isNotEmpty()) {
                val fastest = results.minByOrNull { it.averageTime }
                val slowest = results.maxByOrNull { it.averageTime }

                fastest?.let {
                    appendLine("Fastest: ${it.name} (${it.averageTime / 1_000_000}ms avg)")
                }
                slowest?.let {
                    appendLine("Slowest: ${it.name} (${it.averageTime / 1_000_000}ms avg)")
                }
            }
        }
    }
}

data class BenchmarkResult<T>(
    val name: String,
    val averageTime: Double,
    val medianTime: Long,
    val minTime: Long,
    val maxTime: Long,
    val standardDeviation: Double,
    val iterations: Int,
    val timestamp: Long
) {
    fun formatTime(nanos: Long): String = "${nanos / 1_000_000}ms"
    fun formatTime(nanos: Double): String = "${String.format("%.2f", nanos / 1_000_000)}ms"
}

data class ComparisonResult(
    val name1: String,
    val name2: String,
    val speedup: Double,
    val winner: String,
    val improvement: String
)

data class BenchmarkReport(
    val results: List<BenchmarkResult<*>>,
    val generatedAt: Long,
    val summary: String
)

// 性能基准测试示例
fun performanceBenchmarkDemo() {
    val benchmark = PerformanceBenchmark()

    // 字符串连接基准测试
    val stringConcatResult = benchmark("String Concatenation") {
        val builder = StringBuilder()
        repeat(1000) { i ->
            builder.append("Line $i\n")
        }
        builder.length
    }

    // 列表操作基准测试
    val listOperationResult = benchmark("List Operations") {
        val list = (1..10000).toList()
        list.filter { it % 2 == 0 }.map { it * it }.sum()
    }

    // 集合操作基准测试
    val setOperationResult = benchmark("Set Operations") {
        val set = (1..5000).toSet()
        set.filter { it % 3 == 0 }.map { it.toString() }.joinToString(", ")
        set.size
    }

    // 比较结果
    val comparison = benchmark.compareResults("String Concatenation", "List Operations")

    println("Benchmark Results:")
    println("String Concatenation: ${stringConcatResult.formatTime(stringConcatResult.averageTime)} avg")
    println("List Operations: ${listOperationResult.formatTime(listOperationResult.averageTime)} avg")
    println("Set Operations: ${setOperationResult.formatTime(setOperationResult.averageTime)} avg")
    println()
    println("Comparison:")
    println("Speedup: ${comparison.speedup}x")
    println("Winner: ${comparison.winner}")
    println("Improvement: ${comparison.improvement}")

    // 生成报告
    val report = benchmark.generateReport()
    println("\n${report.summary}")
}
```
