### 5.4.3 惰性求值

```groovy
// 使用迭代器实现惰性求值
class LazyList implements Iterator {
    private Iterator source
    private Closure predicate
    private Object nextValue
    private boolean hasNextValue = false

    LazyList(Iterable source, Closure predicate) {
        this.source = source.iterator()
        this.predicate = predicate
    }

    boolean hasNext() {
        if (hasNextValue) {
            return true
        }

        while (source.hasNext()) {
            def candidate = source.next()
            if (predicate(candidate)) {
                nextValue = candidate
                hasNextValue = true
                return true
            }
        }
        return false
    }

    Object next() {
        if (!hasNext()) {
            throw new NoSuchElementException()
        }
        hasNextValue = false
        return nextValue
    }
}

// 使用惰性列表
def numbers = 1..1000000
def lazyEvens = new LazyList(numbers) { it % 2 == 0 }

// 只在需要时计算
def first10Evens = []
lazyEvens.take(10).each { first10Evens << it }
println "First 10 evens: ${first10Evens}"
```

// CSV数据处理
class CSVProcessor {
    static List<Map<String, Object>> processCSV(String csvContent) {
        def lines = csvContent.split('\n')*.trim()
        def headers = lines[0].split(',')*.trim()

        lines[1..-1].collect { line ->
            def values = line.split(',')*.trim()
            headers.collectEntries { header ->
                [header, values[headers.indexOf(header)]]
            }
        }
    }

    static String formatAsTable(List<Map<String, Object>> data) {
        if (!data) return ""

        def headers = data[0].keySet()
        def columnWidths = headers.collect { header ->
            Math.max(header.length(), data.max { row -> row[header].toString().length() })
        }

        def headerRow = headers.collectWithIndex { header, index ->
            header.padRight(columnWidths[index])
        }.join(' | ')

        def separator = columnWidths.collect { width -> '-' * width }.join('-+-')

        def dataRows = data.collect { row ->
            headers.collectWithIndex { header, index ->
                row[header].toString().padRight(columnWidths[index])
            }.join(' | ')
        }

        return [headerRow, separator] + dataRows
    }
}

        def setTime = System.currentTimeMillis() - start

        start = System.currentTimeMillis()
        def foundInList = target in searchList
        def listTime = System.currentTimeMillis() - start

        println "Search in Set: ${setTime}ms"
        println "Search in List: ${listTime}ms"
        println "Set is ${listTime / setTime}x faster"
    }

    // 避免创建不必要的中间集合
    static def avoidIntermediateCollections() {
        def data = (1..10000).toList()

        // 不好的做法：创建多个中间集合
        def badWay = data
            .findAll { it % 2 == 0 }        // 集合1
            .collect { it * 2 }            // 集合2
            .findAll { it > 100 }          // 集合3

        // 好的做法：使用迭代器或流
        def goodWay = data
            .iterator()
            .findAll { it % 2 == 0 }
            .collect { it * 2 }
            .findAll { it > 100 }
    }

    // 使用对象池复用对象
    static def useObjectPool() {
        def pool = [] as Stack
        def maxSize = 100

        def getObject = {
            pool.isEmpty() ? new StringBuilder() : pool.pop()
        }

        def returnObject = { obj ->
            if (pool.size() < maxSize) {
                obj.setLength(0)  // 清空
                pool.push(obj)
            }
        }

        // 使用对象池
        def builder = getObject()
        builder.append("Hello World")
        println builder.toString()
        returnObject(builder)
    }
}

// 集合工具类
class CollectionUtils {
    // 安全的集合操作
    static def safeGet(List list, int index, defaultValue = null) {
        index >= 0 && index < list.size() ? list[index] : defaultValue
    }

    static def safeGet(Map map, key, defaultValue = null) {
        map.containsKey(key) ? map[key] : defaultValue
    }

    // 集合分页
    static def paginate(List list, int page, int size) {
        def start = (page - 1) * size
        def end = Math.min(start + size, list.size())
        start < list.size() ? list[start..<end] : []
    }

    // 集合去重（保留顺序）
    static def uniquePreserveOrder(List list) {
        def seen = [] as Set
        def result = []
        list.each { item ->
            if (!seen.contains(item)) {
                seen << item
                result << item
            }
        }
        result
    }

    // 深度比较两个Map
    static def deepEquals(Map map1, Map map2) {
        if (map1.size() != map2.size()) {
            return false
        }

        return map1.every { key, value ->
            if (map2.containsKey(key)) {
                def otherValue = map2[key]
                if (value instanceof Map && otherValue instanceof Map) {
                    deepEquals(value, otherValue)
                } else {
                    value == otherValue
                }
            } else {
                false
            }
        }
    }

    // 随机抽样
    static def randomSample(List list, int count) {
        if (count >= list.size()) {
            return new ArrayList(list)
        }

        def shuffled = new ArrayList(list)
        Collections.shuffle(shuffled)
        shuffled[0..<count]
    }

    // 分组并统计
    static def groupAndCount(List list, Closure keyExtractor) {
        list.groupBy(keyExtractor)
            .collectEntries { key, group ->
                [key, group.size()]
            }
    }

    // 检查集合是否有序
    static def isSorted(List list, Closure comparator = { a, b -> a <=> b }) {
        for (int i = 1; i < list.size(); i++) {
            if (comparator(list[i-1], list[i]) > 0) {
                return false
            }
        }
        true
    }

    // 扁平化嵌套集合
    static def flatten(List collection, int maxDepth = Integer.MAX_VALUE) {
        if (maxDepth <= 0) {
            return collection
        }

        def result = []
        collection.each { item ->
            if (item instanceof Collection) {
                result.addAll(flatten(item, maxDepth - 1))
            } else {
                result << item
            }
        }
        result
    }

    // 集合分割
    static def partition(List list, Closure predicate) {
        def matching = []
        def nonMatching = []
        list.each { item ->
            if (predicate(item)) {
                matching << item
            } else {
                nonMatching << item
            }
        }
        [matching, nonMatching]
    }
}
