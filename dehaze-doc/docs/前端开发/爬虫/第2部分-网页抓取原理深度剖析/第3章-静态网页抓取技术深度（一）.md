## 3.1 正则表达式与文本处理高级技巧

### 3.1.1 正则表达式性能优化

正则表达式在文本处理中扮演着重要角色，但不当的使用会导致严重的性能问题：

```python
import re
import time
from typing import Pattern, List, Dict, Optional
import concurrent.futures
from functools import lru_cache

class RegexOptimizer:
    """正则表达式优化器"""

    def __init__(self):
        self.compiled_patterns = {}
        self.performance_stats = {}

    def compile_with_cache(self, pattern: str, flags: int = 0) -> Pattern:
        """缓存编译后的正则表达式"""
        cache_key = (pattern, flags)

        if cache_key not in self.compiled_patterns:
            self.compiled_patterns[cache_key] = re.compile(pattern, flags)

        return self.compiled_patterns[cache_key]

    def optimize_pattern(self, pattern: str) -> str:
        """优化正则表达式模式"""
        # 移除不必要的捕获组
        optimized = re.sub(r'\((?!\?:)(?=[^?*+])', r'(?:', pattern)

        # 使用原子组防止回溯
        optimized = re.sub(r'(\(\?\:)([^)]+)\)', r'(?>\2)', optimized)

        # 优化字符类
        optimized = re.sub(r'\[0-9\]', r'\d', optimized)
        optimized = re.sub(r'\[A-Za-z0-9_]', r'\w', optimized)

        # 预编译量词
        optimized = re.sub(r'\{(\d+),(\d+)\}', lambda m:
                          f'{{{m.group(1)},{m.group(2)}}}'
                          if int(m.group(1)) < int(m.group(2))
                          else f'{{{m.group(1)}}}', optimized)

        return optimized

    def benchmark_pattern(self, pattern: str, text: str, iterations: int = 1000) -> Dict:
        """基准测试正则表达式性能"""
        # 原始模式
        start_time = time.time()
        for _ in range(iterations):
            re.findall(pattern, text)
        original_time = time.time() - start_time

        # 优化后模式
        optimized_pattern = self.optimize_pattern(pattern)
        start_time = time.time()
        for _ in range(iterations):
            re.findall(optimized_pattern, text)
        optimized_time = time.time() - start_time

        # 编译模式
        compiled = self.compile_with_cache(optimized_pattern)
        start_time = time.time()
        for _ in range(iterations):
            compiled.findall(text)
        compiled_time = time.time() - start_time

        return {
            'original_time': original_time,
            'optimized_time': optimized_time,
            'compiled_time': compiled_time,
            'improvement_ratio': original_time / compiled_time if compiled_time > 0 else 0,
            'original_pattern': pattern,
            'optimized_pattern': optimized_pattern
        }

    @lru_cache(maxsize=128)
    def extract_urls(self, text: str) -> List[str]:
        """高效提取URL"""
        url_pattern = r'https?://[^\s<>"{}|\\^`\[\]]+'
        compiled_pattern = self.compile_with_cache(url_pattern)
        return compiled_pattern.findall(text)

    @lru_cache(maxsize=128)
    def extract_emails(self, text: str) -> List[str]:
        """高效提取邮箱地址"""
        email_pattern = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b'
        compiled_pattern = self.compile_with_cache(email_pattern, re.IGNORECASE)
        return compiled_pattern.findall(text)

    @lru_cache(maxsize=128)
    def extract_phone_numbers(self, text: str) -> List[str]:
        """高效提取电话号码"""
        phone_pattern = r'''
            (?:
                # 国际格式
                \+?\d{1,3}[-.\s]?
                # 区号
                \(?\d{2,4}\)?[-.\s]?
                # 号码
                \d{3,4}[-.\s]?\d{3,4}[-.\s]?\d{3,4}
            )
            |
            (?:
                # 中国手机号
                1[3-9]\d{9}
            )
        '''
        compiled_pattern = self.compile_with_cache(phone_pattern, re.VERBOSE)
        return compiled_pattern.findall(text)

class AdvancedTextProcessor:
    """高级文本处理器"""

    def __init__(self):
        self.regex_optimizer = RegexOptimizer()
        self.text_normalizers = {
            'whitespace': self._normalize_whitespace,
            'unicode': self._normalize_unicode,
            'case': self._normalize_case,
            'punctuation': self._normalize_punctuation
        }

    def extract_structured_data(self, text: str, schema: Dict) -> Dict:
        """根据模式提取结构化数据"""
        results = {}

        for field_name, field_config in schema.items():
            pattern = field_config['pattern']
            extraction_method = field_config.get('method', 'findall')
            multiple = field_config.get('multiple', True)
            normalizer = field_config.get('normalizer')

            compiled_pattern = self.regex_optimizer.compile_with_cache(
                pattern,
                field_config.get('flags', 0)
            )

            if extraction_method == 'findall':
                matches = compiled_pattern.findall(text)
            elif extraction_method == 'search':
                match = compiled_pattern.search(text)
                matches = [match.group()] if match else []
            elif extraction_method == 'finditer':
                matches = [m.groupdict() for m in compiled_pattern.finditer(text)]
            else:
                matches = []

            # 应用规范化器
            if normalizer and normalizer in self.text_normalizers:
                matches = [self.text_normalizers[normalizer](match) for match in matches]

            # 处理单值情况
            if not multiple and matches:
                results[field_name] = matches[0]
            else:
                results[field_name] = matches

        return results

    def _normalize_whitespace(self, text: str) -> str:
        """规范化空白字符"""
        return re.sub(r'\s+', ' ', text.strip())

    def _normalize_unicode(self, text: str) -> str:
        """规范化Unicode字符"""
        import unicodedata
        return unicodedata.normalize('NFKC', text)

    def _normalize_case(self, text: str) -> str:
        """规范化大小写"""
        return text.lower().strip()

    def _normalize_punctuation(self, text: str) -> str:
        """规范化标点符号"""
        # 移除多余的标点
        text = re.sub(r'[^\w\s.-]', '', text)
        # 规范化空格
        return re.sub(r'\s+', ' ', text.strip())

    def clean_html_text(self, html_text: str) -> str:
        """清理HTML文本"""
        # 移除HTML标签
        text = re.sub(r'<[^>]+>', ' ', html_text)
        # 移除HTML实体
        text = re.sub(r'&[a-zA-Z]+;', '', text)
        # 移除多余的空白
        text = re.sub(r'\s+', ' ', text)
        return text.strip()

    def extract_tables(self, text: str) -> List[List[str]]:
        """从文本中提取表格数据"""
        table_patterns = [
            # Markdown表格
            r'\|(.+)\|\s*\|[-\s|]+\|\s*((?:\|.+\|\s*)+)',
            # 简单的制表符分隔表格
            r'([^\n]+\t[^\n]+(?:\n[^\n]+\t[^\n]+)*)',
            # 逗号分隔表格
            r'("[^"]*"|[^,\n]+)(?:,("[^"]*"|[^,\n]+))*\s*(?:\n("[^"]*"|[^,\n]+)(?:,("[^"]*"|[^,\n]+))*)*'
        ]

        tables = []
        for pattern in table_patterns:
            matches = re.findall(pattern, text, re.MULTILINE)
            for match in matches:
                if isinstance(match, tuple):
                    row_data = [cell.strip() for cell in match if cell.strip()]
                else:
                    rows = match.split('\n')
                    row_data = []
                    for row in rows:
                        cells = re.split(r'\t+', row.strip())
                        row_data.append([cell.strip().strip('"') for cell in cells])

                if row_data:
                    tables.append(row_data)

        return tables

# 并行文本处理
class ParallelTextProcessor:
    """并行文本处理器"""

    def __init__(self, max_workers: int = 4):
        self.max_workers = max_workers
        self.text_processor = AdvancedTextProcessor()

    def process_texts_parallel(self, texts: List[str], schema: Dict) -> List[Dict]:
        """并行处理多个文本"""
        with concurrent.futures.ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            futures = [
                executor.submit(self.text_processor.extract_structured_data, text, schema)
                for text in texts
            ]

            results = []
            for future in concurrent.futures.as_completed(futures):
                try:
                    result = future.result()
                    results.append(result)
                except Exception as e:
                    print(f"Error processing text: {e}")
                    results.append({})

        return results

    def batch_extract_patterns(self, texts: List[str], patterns: Dict[str, str]) -> Dict[str, List]:
        """批量提取模式"""
        results = {pattern_name: [] for pattern_name in patterns.keys()}

        with concurrent.futures.ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            future_to_pattern = {}

            for pattern_name, pattern in patterns.items():
                for text in texts:
                    future = executor.submit(
                        self.text_processor.regex_optimizer.compile_with_cache(pattern).findall,
                        text
                    )
                    future_to_pattern[future] = pattern_name

            for future in concurrent.futures.as_completed(future_to_pattern):
                pattern_name = future_to_pattern[future]
                try:
                    matches = future.result()
                    results[pattern_name].extend(matches)
                except Exception as e:
                    print(f"Error extracting pattern {pattern_name}: {e}")

        return results
```

### 3.1.2 智能文本提取与清洗

```python
import jieba
import nltk
from collections import Counter
from typing import List, Dict, Tuple
import math

class IntelligentTextExtractor:
    """智能文本提取器"""

    def __init__(self):
        self.stop_words = self._load_stop_words()
        self.keyword_extractor = KeywordExtractor()

    def _load_stop_words(self) -> set:
        """加载停用词"""
        # 中文停用词
        chinese_stop_words = {
            '的', '了', '在', '是', '我', '有', '和', '就', '不', '人', '都', '一', '一个',
            '上', '也', '很', '到', '说', '要', '去', '你', '会', '着', '没有', '看', '好',
            '自己', '这', '那', '他', '她', '它', '们', '这个', '那个', '什么', '怎么',
            '可以', '但是', '因为', '所以', '如果', '虽然', '然而', '因此', '而且'
        }

        # 英文停用词
        english_stop_words = {
            'the', 'a', 'an', 'and', 'or', 'but', 'in', 'on', 'at', 'to', 'for', 'of',
            'with', 'by', 'is', 'are', 'was', 'were', 'be', 'been', 'being', 'have',
            'has', 'had', 'do', 'does', 'did', 'will', 'would', 'could', 'should',
            'may', 'might', 'must', 'can', 'this', 'that', 'these', 'those', 'i',
            'you', 'he', 'she', 'it', 'we', 'they', 'me', 'him', 'her', 'us', 'them'
        }

        return chinese_stop_words.union(english_stop_words)

    def extract_main_content(self, html_content: str) -> Dict:
        """提取网页主要内容"""
        from bs4 import BeautifulSoup

        soup = BeautifulSoup(html_content, 'html.parser')

        # 移除无关标签
        for tag in soup(['script', 'style', 'nav', 'footer', 'header', 'aside']):
            tag.decompose()

        # 计算文本密度
        text_density = self._calculate_text_density(soup)

        # 提取主要内容区域
        main_content = self._extract_main_region(soup, text_density)

        # 清理和规范化文本
        cleaned_text = self._clean_text(main_content.get_text())

        # 提取关键信息
        keywords = self.keyword_extractor.extract_keywords(cleaned_text)
        summary = self._generate_summary(cleaned_text)

        return {
            'title': self._extract_title(soup),
            'main_content': cleaned_text,
            'keywords': keywords,
            'summary': summary,
            'text_density': text_density,
            'word_count': len(cleaned_text.split()),
            'language': self._detect_language(cleaned_text)
        }

    def _calculate_text_density(self, soup) -> Dict:
        """计算文本密度"""
        density_map = {}

        for element in soup.find_all():
            if element.name in ['p', 'div', 'section', 'article']:
                text_length = len(element.get_text().strip())
                html_length = len(str(element))

                if html_length > 0:
                    density = text_length / html_length
                    density_map[element] = {
                        'density': density,
                        'text_length': text_length,
                        'html_length': html_length
                    }

        return density_map

    def _extract_main_region(self, soup, density_map: Dict):
        """提取主要内容区域"""
        if not density_map:
            return soup

        # 找到密度最高的元素
        max_density_element = max(density_map.items(), key=lambda x: x[1]['density'])[0]

        # 向上扩展到合适的容器
        current = max_density_element
        while current.parent:
            parent_text = current.parent.get_text().strip()
            current_text = current.get_text().strip()

            # 如果父元素的文本长度是当前元素的2倍以上，则使用父元素
            if len(parent_text) > len(current_text) * 2:
                current = current.parent
            else:
                break

        return current

    def _clean_text(self, text: str) -> str:
        """清理文本"""
        # 移除多余的空白
        text = re.sub(r'\s+', ' ', text)

        # 移除特殊字符
        text = re.sub(r'[^\w\s\u4e00-\u9fff.,!?;:]', '', text)

        # 移除多余的标点
        text = re.sub(r'[.,!?;:]{2,}', '.', text)

        return text.strip()

    def _extract_title(self, soup) -> str:
        """提取标题"""
        # 尝试多种方式提取标题
        title_selectors = [
            'h1',
            'title',
            '.title',
            '.headline',
            '[class*="title"]',
            '[class*="headline"]'
        ]

        for selector in title_selectors:
            element = soup.select_one(selector)
            if element:
                title = element.get_text().strip()
                if len(title) > 5:  # 标题长度至少5个字符
                    return title

        return ''

    def _detect_language(self, text: str) -> str:
        """检测语言"""
        chinese_chars = len(re.findall(r'[\u4e00-\u9fff]', text))
        english_chars = len(re.findall(r'[a-zA-Z]', text))
        total_chars = chinese_chars + english_chars

        if total_chars == 0:
            return 'unknown'

        chinese_ratio = chinese_chars / total_chars

        if chinese_ratio > 0.3:
            return 'chinese'
        elif chinese_ratio < 0.1:
            return 'english'
        else:
            return 'mixed'

    def _generate_summary(self, text: str, max_sentences: int = 3) -> str:
        """生成摘要"""
        sentences = re.split(r'[.!?。！？]', text)
        sentences = [s.strip() for s in sentences if len(s.strip()) > 10]

        if len(sentences) <= max_sentences:
            return '. '.join(sentences)

        # 简单的重要性评分：选择较长的句子
        sentences_with_scores = [
            (sentence, len(sentence.split()))
            for sentence in sentences
        ]

        sentences_with_scores.sort(key=lambda x: x[1], reverse=True)
        selected_sentences = [s[0] for s in sentences_with_scores[:max_sentences]]

        # 按原文顺序排列
        ordered_sentences = []
        for sentence in sentences:
            if sentence in selected_sentences:
                ordered_sentences.append(sentence)

        return '. '.join(ordered_sentences)

class KeywordExtractor:
    """关键词提取器"""

    def __init__(self):
        self.word_freq_cache = {}

    def extract_keywords(self, text: str, top_k: int = 10) -> List[Tuple[str, float]]:
        """提取关键词"""
        # 分词
        words = self._tokenize(text)

        # 过滤停用词
        filtered_words = [word for word in words if word not in self.stop_words and len(word) > 1]

        # 计算词频
        word_freq = Counter(filtered_words)

        # 计算TF-IDF权重
        keywords = self._calculate_tfidf(word_freq, len(filtered_words))

        # 返回top-k关键词
        return sorted(keywords.items(), key=lambda x: x[1], reverse=True)[:top_k]

    def _tokenize(self, text: str) -> List[str]:
        """分词"""
        words = []

        # 检测是否包含中文
        if re.search(r'[\u4e00-\u9fff]', text):
            # 中文分词
            import jieba
            words.extend(jieba.lcut(text))

        # 英文分词
        english_words = re.findall(r'\b[a-zA-Z]+\b', text)
        words.extend([word.lower() for word in english_words])

        return words

    def _calculate_tfidf(self, word_freq: Counter, total_words: int) -> Dict[str, float]:
        """计算TF-IDF权重"""
        tfidf_scores = {}

        for word, freq in word_freq.items():
            tf = freq / total_words
            # 简化的IDF计算（实际应用中需要语料库）
            idf = math.log(1000 / (freq + 1))
            tfidf_scores[word] = tf * idf

        return tfidf_scores
```

## 3.2 HTML解析与数据结构化

### 3.2.1 高级HTML解析技术

```python
from bs4 import BeautifulSoup, Tag, NavigableString
from lxml import etree, html
import json
import xml.etree.ElementTree as ET
from dataclasses import dataclass, asdict
from typing import List, Dict, Any, Optional, Union
import hashlib

@dataclass
class ParsedElement:
    """解析后的元素数据结构"""
    tag: str
    text: str
    attributes: Dict[str, str]
    children: List['ParsedElement']
    xpath: str
    css_selector: str
    content_hash: str

class AdvancedHTMLParser:
    """高级HTML解析器"""

    def __init__(self, parser_type: str = 'lxml'):
        self.parser_type = parser_type
        self.element_cache = {}
        self.xpath_cache = {}

    def parse_html_structured(self, html_content: str) -> Dict[str, Any]:
        """结构化解析HTML"""
        if self.parser_type == 'beautifulsoup':
            soup = BeautifulSoup(html_content, 'lxml')
            return self._parse_with_beautifulsoup(soup)
        elif self.parser_type == 'lxml':
            tree = html.fromstring(html_content)
            return self._parse_with_lxml(tree)
        else:
            raise ValueError(f"Unsupported parser type: {self.parser_type}")

    def _parse_with_beautifulsoup(self, soup: BeautifulSoup) -> Dict[str, Any]:
        """使用BeautifulSoup解析"""
        result = {
            'title': self._extract_title(soup),
            'meta': self._extract_meta(soup),
            'structure': self._build_structure(soup),
            'forms': self._extract_forms(soup),
            'tables': self._extract_tables(soup),
            'links': self._extract_links(soup),
            'images': self._extract_images(soup),
            'lists': self._extract_lists(soup)
        }

        return result

    def _build_structure(self, soup: BeautifulSoup) -> ParsedElement:
        """构建DOM结构"""
        def build_element(element, xpath='') -> ParsedElement:
            if isinstance(element, NavigableString):
                return None

            # 生成XPath
            if xpath:
                element_xpath = f"{xpath}/{element.name}"
            else:
                element_xpath = f"/{element.name}"

            # 生成CSS选择器
            css_selector = self._generate_css_selector(element)

            # 提取文本内容
            text = element.get_text(strip=True)

            # 计算内容哈希
            content_hash = hashlib.md5(text.encode()).hexdigest()

            # 递归处理子元素
            children = []
            for child in element.children:
                child_element = build_element(child, element_xpath)
                if child_element:
                    children.append(child_element)

            return ParsedElement(
                tag=element.name,
                text=text,
                attributes=dict(element.attrs),
                children=children,
                xpath=element_xpath,
                css_selector=css_selector,
                content_hash=content_hash
            )

        return build_element(soup)

    def _generate_css_selector(self, element: Tag) -> str:
        """生成CSS选择器"""
        selectors = []
        current = element

        while current and current.name:
            selector_part = current.name

            # 添加ID选择器
            if current.get('id'):
                selector_part += f"#{current['id']}"
            # 添加类选择器
            elif current.get('class'):
                classes = ' '.join(current['class'])
                selector_part += f".{classes.replace(' ', '.')}"

            selectors.insert(0, selector_part)
            current = current.parent

        return ' > '.join(selectors)

    def _extract_title(self, soup: BeautifulSoup) -> str:
        """提取标题"""
        title_tag = soup.find('title')
        return title_tag.get_text().strip() if title_tag else ''

    def _extract_meta(self, soup: BeautifulSoup) -> Dict[str, str]:
        """提取meta标签"""
        meta_data = {}

        for meta in soup.find_all('meta'):
            name = meta.get('name') or meta.get('property')
            content = meta.get('content', '')

            if name:
                meta_data[name] = content

        return meta_data

    def _extract_forms(self, soup: BeautifulSoup) -> List[Dict]:
        """提取表单"""
        forms = []

        for form in soup.find_all('form'):
            form_data = {
                'action': form.get('action', ''),
                'method': form.get('method', 'GET'),
                'fields': []
            }

            # 提取表单字段
            for input_tag in form.find_all(['input', 'select', 'textarea']):
                field_data = {
                    'name': input_tag.get('name', ''),
                    'type': input_tag.get('type', 'text'),
                    'value': input_tag.get('value', ''),
                    'placeholder': input_tag.get('placeholder', ''),
                    'required': input_tag.has_attr('required')
                }

                if input_tag.name == 'select':
                    field_data['options'] = [
                        {'value': opt.get('value', ''), 'text': opt.get_text().strip()}
                        for opt in input_tag.find_all('option')
                    ]

                form_data['fields'].append(field_data)

            forms.append(form_data)

        return forms

    def _extract_tables(self, soup: BeautifulSoup) -> List[Dict]]:
        """提取表格"""
        tables = []

        for table in soup.find_all('table'):
            table_data = {
                'headers': [],
                'rows': [],
                'caption': ''
            }

            # 提取标题
            caption = table.find('caption')
            if caption:
                table_data['caption'] = caption.get_text().strip()

            # 提取表头
            thead = table.find('thead')
            if thead:
                header_row = thead.find('tr')
                if header_row:
                    table_data['headers'] = [
                        th.get_text().strip() for th in header_row.find_all(['th', 'td'])
                    ]

            # 提取数据行
            tbody = table.find('tbody') or table
            for row in tbody.find_all('tr'):
                cells = [cell.get_text().strip() for cell in row.find_all(['td', 'th'])]
                if cells:
                    table_data['rows'].append(cells)

            tables.append(table_data)

        return tables

    def _extract_links(self, soup: BeautifulSoup) -> List[Dict]:
        """提取链接"""
        links = []

        for link in soup.find_all('a', href=True):
            link_data = {
                'url': link['href'],
                'text': link.get_text().strip(),
                'title': link.get('title', ''),
                'target': link.get('target', ''),
                'rel': link.get('rel', [])
            }
            links.append(link_data)

        return links

    def _extract_images(self, soup: BeautifulSoup) -> List[Dict]:
        """提取图片"""
        images = []

        for img in soup.find_all('img', src=True):
            img_data = {
                'src': img['src'],
                'alt': img.get('alt', ''),
                'title': img.get('title', ''),
                'width': img.get('width', ''),
                'height': img.get('height', '')
            }
            images.append(img_data)

        return images

    def _extract_lists(self, soup: BeautifulSoup) -> List[Dict]:
        """提取列表"""
        lists = []

        for list_tag in soup.find_all(['ul', 'ol', 'dl']):
            list_data = {
                'type': list_tag.name,
                'items': []
            }

            if list_tag.name in ['ul', 'ol']:
                for li in list_tag.find_all('li', recursive=False):
                    list_data['items'].append({
                        'text': li.get_text().strip(),
                        'type': 'list_item'
                    })

            elif list_tag.name == 'dl':
                for dt, dd in zip(list_tag.find_all('dt'), list_tag.find_all('dd')):
                    list_data['items'].append({
                        'term': dt.get_text().strip(),
                        'description': dd.get_text().strip(),
                        'type': 'definition'
                    })

            lists.append(list_data)

        return lists

    def _parse_with_lxml(self, tree: etree.Element) -> Dict[str, Any]:
        """使用lxml解析"""
        result = {
            'title': self._extract_title_lxml(tree),
            'meta': self._extract_meta_lxml(tree),
            'text_content': tree.xpath('//body//text()'),
            'links': self._extract_links_lxml(tree),
            'images': self._extract_images_lxml(tree)
        }

        return result

    def _extract_title_lxml(self, tree: etree.Element) -> str:
        """使用lxml提取标题"""
        titles = tree.xpath('//title/text()')
        return titles[0].strip() if titles else ''

    def _extract_meta_lxml(self, tree: etree.Element) -> Dict[str, str]:
        """使用lxml提取meta标签"""
        meta_data = {}

        for meta in tree.xpath('//meta[@name or @property]'):
            name = meta.get('name') or meta.get('property')
            content = meta.get('content', '')

            if name:
                meta_data[name] = content

        return meta_data

    def _extract_links_lxml(self, tree: etree.Element) -> List[Dict]:
        """使用lxml提取链接"""
        links = []

        for link in tree.xpath('//a[@href]'):
            link_text = link.xpath('string()').strip()
            link_data = {
                'url': link.get('href'),
                'text': link_text,
                'title': link.get('title', ''),
                'target': link.get('target', '')
            }
            links.append(link_data)

        return links

    def _extract_images_lxml(self, tree: etree.Element) -> List[Dict]:
        """使用lxml提取图片"""
        images = []

        for img in tree.xpath('//img[@src]'):
            img_data = {
                'src': img.get('src'),
                'alt': img.get('alt', ''),
                'title': img.get('title', ''),
                'width': img.get('width', ''),
                'height': img.get('height', '')
            }
            images.append(img_data)

        return images

class DataStructureNormalizer:
    """数据结构规范化器"""

    def __init__(self):
        self.schemas = {}

    def normalize_to_json(self, parsed_data: Dict[str, Any], schema_name: str = 'default') -> str:
        """规范化为JSON格式"""
        if schema_name not in self.schemas:
            self.schemas[schema_name] = self._create_default_schema()

        schema = self.schemas[schema_name]
        normalized_data = self._apply_schema(parsed_data, schema)

        return json.dumps(normalized_data, ensure_ascii=False, indent=2)

    def normalize_to_xml(self, parsed_data: Dict[str, Any], root_tag: str = 'data') -> str:
        """规范化为XML格式"""
        root = ET.Element(root_tag)
        self._dict_to_xml(parsed_data, root)

        return ET.tostring(root, encoding='unicode')

    def normalize_to_csv(self, parsed_data: Dict[str, Any], table_name: str = 'data') -> str:
        """规范化为CSV格式"""
        import csv
        import io

        output = io.StringIO()

        # 处理表格数据
        if 'tables' in parsed_data and parsed_data['tables']:
            writer = csv.writer(output)

            for table in parsed_data['tables']:
                if table['headers']:
                    writer.writerow(table['headers'])

                for row in table['rows']:
                    writer.writerow(row)

                writer.writerow([])  # 空行分隔表格

        return output.getvalue()

    def _create_default_schema(self) -> Dict:
        """创建默认模式"""
        return {
            'required_fields': ['title', 'text_content'],
            'field_mappings': {
                'title': 'title',
                'content': 'text_content',
                'meta': 'meta',
                'links': 'links'
            },
            'data_types': {
                'title': 'string',
                'content': 'string',
                'meta': 'object',
                'links': 'array'
            }
        }

    def _apply_schema(self, data: Dict[str, Any], schema: Dict) -> Dict[str, Any]:
        """应用数据模式"""
        normalized = {}

        for field_name, field_mapping in schema['field_mappings'].items():
            if field_mapping in data:
                normalized[field_name] = data[field_mapping]

        return normalized

    def _dict_to_xml(self, data: Any, parent: ET.Element):
        """将字典转换为XML元素"""
        if isinstance(data, dict):
            for key, value in data.items():
                child = ET.SubElement(parent, key)
                self._dict_to_xml(value, child)
        elif isinstance(data, list):
            for item in data:
                child = ET.SubElement(parent, 'item')
                self._dict_to_xml(item, child)
        else:
            parent.text = str(data)
```
