## 3.3 分页技术与URL规律发现

### 3.3.1 智能分页识别与抓取

分页是网站组织大量内容的常见方式，智能识别和抓取分页内容是爬虫技术的重要组成部分：

```python
import re
import urllib.parse
from typing import List, Dict, Optional, Tuple, Union
from dataclasses import dataclass
from abc import ABC, abstractmethod
import requests
from bs4 import BeautifulSoup

@dataclass
class PaginationInfo:
    """分页信息数据结构"""
    current_page: int
    total_pages: Optional[int]
    next_page_url: Optional[str]
    prev_page_url: Optional[str]
    page_param: str
    page_pattern: str
    pagination_type: str  # 'offset', 'page', 'cursor', 'infinite'

class PaginationDetector:
    """分页检测器"""

    def __init__(self):
        self.pagination_patterns = {
            'page_param': [
                r'page=(\d+)',
                r'p=(\d+)',
                r'pg=(\d+)',
                r'pageNumber=(\d+)',
                r'pageNum=(\d+)'
            ],
            'offset_param': [
                r'offset=(\d+)',
                r'start=(\d+)',
                r'begin=(\d+)',
                r'skip=(\d+)'
            ],
            'cursor_param': [
                r'cursor=([^&]+)',
                r'after=([^&]+)',
                r'next=([^&]+)',
                r'token=([^&]+)'
            ],
            'page_path': [
                r'/page/(\d+)',
                r'/p/(\d+)',
                r'/(\d+)/?',
                r'/page-\d+/'
            ]
        }

        self.pagination_selectors = {
            'next_button': [
                'a[href*="next"]',
                'a:contains("下一页")',
                'a:contains("Next")',
                '.next a',
                '.pagination-next a',
                '[rel="next"]'
            ],
            'prev_button': [
                'a[href*="prev"]',
                'a:contains("上一页")',
                'a:contains("Previous")',
                '.prev a',
                '.pagination-prev a',
                '[rel="prev"]'
            ],
            'page_numbers': [
                '.pagination a',
                '.pager a',
                '.page-numbers a',
                '[class*="page"] a',
                '.pagination li a'
            ],
            'total_info': [
                '.total-pages',
                '.page-info',
                '.pagination-info',
                '[class*="total"]',
                '[class*="count"]'
            ]
        }

    def detect_pagination_type(self, url: str, html_content: str) -> PaginationInfo:
        """检测分页类型和提取分页信息"""
        soup = BeautifulSoup(html_content, 'html.parser')

        # 检测URL中的分页参数
        pagination_type, page_param, page_value = self._detect_url_pagination(url)

        # 检测页面中的分页元素
        page_elements = self._detect_pagination_elements(soup)

        # 计算总页数
        total_pages = self._calculate_total_pages(soup, page_elements)

        # 获取下一页和上一页链接
        next_url, prev_url = self._extract_nav_links(soup)

        # 构建分页模式
        page_pattern = self._build_page_pattern(url, page_param, pagination_type)

        return PaginationInfo(
            current_page=int(page_value) if page_value else 1,
            total_pages=total_pages,
            next_page_url=next_url,
            prev_page_url=prev_url,
            page_param=page_param,
            page_pattern=page_pattern,
            pagination_type=pagination_type
        )

    def _detect_url_pagination(self, url: str) -> Tuple[str, str, Optional[str]]:
        """检测URL中的分页参数"""
        parsed_url = urllib.parse.urlparse(url)
        query_params = urllib.parse.parse_qs(parsed_url.query)

        # 检测各种分页参数
        for param_type, patterns in self.pagination_patterns.items():
            for pattern in patterns:
                if param_type in ['page_param', 'offset_param', 'cursor_param']:
                    for param in query_params:
                        if re.search(pattern, f"{param}={query_params[param][0]}"):
                            match = re.search(pattern, f"{param}={query_params[param][0]}")
                            if match:
                                return param_type, param, match.group(1)
                elif param_type == 'page_path':
                    match = re.search(pattern, parsed_url.path)
                    if match:
                        return param_type, 'path', match.group(1)

        return 'unknown', '', None

    def _detect_pagination_elements(self, soup: BeautifulSoup) -> Dict[str, List]:
        """检测页面中的分页元素"""
        elements = {}

        for element_type, selectors in self.pagination_selectors.items():
            elements[element_type] = []
            for selector in selectors:
                found = soup.select(selector)
                if found:
                    elements[element_type].extend(found)

        return elements

    def _calculate_total_pages(self, soup: BeautifulSoup, page_elements: Dict) -> Optional[int]:
        """计算总页数"""
        # 从页面元素中提取总页数信息
        total_info_selectors = [
            '.total-pages',
            '.page-info',
            '.pagination-info',
            '[class*="total"]'
        ]

        for selector in total_info_selectors:
            element = soup.select_one(selector)
            if element:
                text = element.get_text()
                # 尝试从文本中提取总页数
                total_match = re.search(r'(\d+)', text)
                if total_match:
                    return int(total_match.group(1))

        # 从页码导航中推断总页数
        if 'page_numbers' in page_elements and page_elements['page_numbers']:
            page_numbers = []
            for link in page_elements['page_numbers']:
                text = link.get_text().strip()
                if text.isdigit():
                    page_numbers.append(int(text))

            if page_numbers:
                return max(page_numbers)

        return None

    def _extract_nav_links(self, soup: BeautifulSoup) -> Tuple[Optional[str], Optional[str]]:
        """提取导航链接"""
        next_url = None
        prev_url = None

        # 提取下一页链接
        for selector in self.pagination_selectors['next_button']:
            link = soup.select_one(selector)
            if link and link.get('href'):
                next_url = link['href']
                break

        # 提取上一页链接
        for selector in self.pagination_selectors['prev_button']:
            link = soup.select_one(selector)
            if link and link.get('href'):
                prev_url = link['href']
                break

        return next_url, prev_url

    def _build_page_pattern(self, url: str, page_param: str, pagination_type: str) -> str:
        """构建分页URL模式"""
        parsed_url = urllib.parse.urlparse(url)
        query_params = urllib.parse.parse_qs(parsed_url.query)

        if pagination_type in ['page_param', 'offset_param']:
            # 构建参数型分页模式
            if page_param in query_params:
                # 替换参数值为占位符
                query_params[page_param] = ['{page}']
            else:
                # 添加分页参数
                query_params[page_param] = ['{page}']

            new_query = urllib.parse.urlencode(query_params, doseq=True)
            return f"{parsed_url.scheme}://{parsed_url.netloc}{parsed_url.path}?{new_query}"

        elif pagination_type == 'page_path':
            # 构建路径型分页模式
            path_pattern = re.sub(r'/\d+', '/{page}', parsed_url.path)
            return f"{parsed_url.scheme}://{parsed_url.netloc}{path_pattern}"

        else:
            # 未知类型，返回原URL
            return url

class PaginationCrawler:
    """分页爬虫"""

    def __init__(self, session: requests.Session):
        self.session = session
        self.detector = PaginationDetector()
        self.visited_urls = set()
        self.crawled_data = []

    def crawl_paginated_content(
        self,
        start_url: str,
        max_pages: int = 100,
        delay: float = 1.0
    ) -> List[Dict]:
        """爬取分页内容"""
        current_url = start_url
        page_count = 0

        while current_url and page_count < max_pages:
            if current_url in self.visited_urls:
                break

            print(f"正在爬取第 {page_count + 1} 页: {current_url}")

            try:
                # 获取页面内容
                response = self.session.get(current_url, timeout=10)
                response.raise_for_status()

                self.visited_urls.add(current_url)

                # 检测分页信息
                pagination_info = self.detector.detect_pagination_type(
                    current_url,
                    response.text
                )

                # 提取页面数据
                page_data = self._extract_page_data(response.text, pagination_info)
                self.crawled_data.append(page_data)

                # 获取下一页URL
                current_url = self._get_next_page_url(pagination_info, current_url)

                page_count += 1

                # 延迟
                if delay > 0:
                    time.sleep(delay)

            except Exception as e:
                print(f"爬取页面失败: {e}")
                break

        return self.crawled_data

    def _extract_page_data(self, html_content: str, pagination_info: PaginationInfo) -> Dict:
        """提取页面数据"""
        soup = BeautifulSoup(html_content, 'html.parser')

        # 提取主要内容区域
        main_content = self._extract_main_content(soup)

        return {
            'page_number': pagination_info.current_page,
            'url': pagination_info.page_pattern.format(page=pagination_info.current_page),
            'content': main_content,
            'pagination_info': {
                'current_page': pagination_info.current_page,
                'total_pages': pagination_info.total_pages,
                'pagination_type': pagination_info.pagination_type
            }
        }

    def _extract_main_content(self, soup: BeautifulSoup) -> Dict:
        """提取主要内容"""
        # 移除导航、页脚等无关元素
        for tag in soup(['nav', 'footer', 'header', 'aside', '.pagination']):
            tag.decompose()

        # 提取文章内容
        content_selectors = [
            '.content',
            '.main-content',
            '.article-content',
            '.post-content',
            'article',
            '.entry-content'
        ]

        main_content = ""
        for selector in content_selectors:
            element = soup.select_one(selector)
            if element:
                main_content = element.get_text(strip=True)
                break

        if not main_content:
            # 如果没有找到内容区域，提取body文本
            body = soup.find('body')
            if body:
                main_content = body.get_text(strip=True)

        return {
            'text': main_content,
            'word_count': len(main_content.split()),
            'char_count': len(main_content)
        }

    def _get_next_page_url(self, pagination_info: PaginationInfo, current_url: str) -> Optional[str]:
        """获取下一页URL"""
        # 优先使用检测到的下一页链接
        if pagination_info.next_page_url:
            return self._resolve_url(pagination_info.next_page_url, current_url)

        # 根据分页模式生成下一页URL
        if pagination_info.pagination_type in ['page_param', 'offset_param', 'page_path']:
            next_page = pagination_info.current_page + 1

            # 检查是否超过总页数
            if pagination_info.total_pages and next_page > pagination_info.total_pages:
                return None

            return pagination_info.page_pattern.format(page=next_page)

        return None

    def _resolve_url(self, url: str, base_url: str) -> str:
        """解析相对URL"""
        return urllib.parse.urljoin(base_url, url)

class InfiniteScrollCrawler:
    """无限滚动爬虫"""

    def __init__(self, session: requests.Session):
        self.session = session
        self.scroll_patterns = {
            'load_more_button': [
                '.load-more',
                '.load-more-button',
                '[class*="load-more"]',
                'button:contains("加载更多")',
                'a:contains("Load More")'
            ],
            'infinite_scroll': [
                '.infinite-scroll',
                '[data-infinite-scroll]',
                '.lazy-load'
            ]
        }

    def crawl_infinite_scroll(
        self,
        start_url: str,
        max_scrolls: int = 50,
        delay: float = 2.0
    ) -> List[Dict]:
        """爬取无限滚动内容"""
        from selenium import webdriver
        from selenium.webdriver.common.by import By
        from selenium.webdriver.support.ui import WebDriverWait
        from selenium.webdriver.support import expected_conditions as EC
        import time

        # 使用Selenium处理JavaScript渲染
        options = webdriver.ChromeOptions()
        options.add_argument('--headless')
        options.add_argument('--no-sandbox')
        options.add_argument('--disable-dev-shm-usage')

        driver = webdriver.Chrome(options=options)
        scraped_data = []

        try:
            driver.get(start_url)
            WebDriverWait(driver, 10).until(
                EC.presence_of_element_located((By.TAG_NAME, 'body'))
            )

            scroll_count = 0
            last_height = driver.execute_script("return document.body.scrollHeight")

            while scroll_count < max_scrolls:
                # 滚动到底部
                driver.execute_script("window.scrollTo(0, document.body.scrollHeight);")

                # 等待新内容加载
                time.sleep(delay)

                # 检查是否有新内容
                new_height = driver.execute_script("return document.body.scrollHeight")
                if new_height == last_height:
                    # 检查是否有"加载更多"按钮
                    load_more_clicked = self._click_load_more_button(driver)
                    if not load_more_clicked:
                        break
                else:
                    last_height = new_height

                # 提取当前页面内容
                page_data = self._extract_current_content(driver)
                scraped_data.append({
                    'scroll_number': scroll_count + 1,
                    'content': page_data,
                    'page_height': new_height
                })

                scroll_count += 1

        finally:
            driver.quit()

        return scraped_data

    def _click_load_more_button(self, driver) -> bool:
        """点击加载更多按钮"""
        for selector in self.scroll_patterns['load_more_button']:
            try:
                button = driver.find_element(By.CSS_SELECTOR, selector)
                if button.is_displayed() and button.is_enabled():
                    driver.execute_script("arguments[0].click();", button)
                    return True
            except:
                continue
        return False

    def _extract_current_content(self, driver) -> Dict:
        """提取当前页面内容"""
        from bs4 import BeautifulSoup

        html = driver.page_source
        soup = BeautifulSoup(html, 'html.parser')

        # 提取文章或内容项
        content_items = []

        # 常见的内容选择器
        item_selectors = [
            '.item',
            '.post',
            '.article',
            '.product',
            '.card',
            '[class*="item"]',
            '[class*="post"]'
        ]

        for selector in item_selectors:
            items = soup.select(selector)
            if items:
                for item in items:
                    content_items.append({
                        'title': item.get_text(strip=True),
                        'html': str(item)
                    })
                break

        return {
            'items': content_items,
            'total_items': len(content_items),
            'page_title': driver.title
        }
```

### 3.3.2 URL规律发现与生成

```python
import re
import urllib.parse
from typing import List, Dict, Tuple, Optional, Set
from collections import defaultdict, Counter
import difflib
import math

class URLPatternAnalyzer:
    """URL模式分析器"""

    def __init__(self):
        self.url_patterns = {}
        self.parameter_patterns = defaultdict(list)

    def analyze_urls(self, urls: List[str]) -> Dict[str, any]:
        """分析URL列表，发现规律"""
        if not urls:
            return {}

        # 解析URL结构
        parsed_urls = [urllib.parse.urlparse(url) for url in urls]

        # 分析路径模式
        path_patterns = self._analyze_path_patterns([p.path for p in parsed_urls])

        # 分析查询参数
        query_patterns = self._analyze_query_patterns([p.query for p in parsed_urls])

        # 检测分页模式
        pagination_patterns = self._detect_pagination_patterns(urls)

        # 检测序列化模式
        sequence_patterns = self._detect_sequence_patterns(urls)

        return {
            'path_patterns': path_patterns,
            'query_patterns': query_patterns,
            'pagination_patterns': pagination_patterns,
            'sequence_patterns': sequence_patterns,
            'total_urls': len(urls),
            'unique_domains': len(set(p.netloc for p in parsed_urls))
        }

    def _analyze_path_patterns(self, paths: List[str]) -> List[Dict]:
        """分析路径模式"""
        # 分割路径为段
        path_segments = []
        for path in paths:
            segments = [seg for seg in path.split('/') if seg]
            path_segments.append(segments)

        if not path_segments:
            return []

        # 找出每个位置的段类型
        segment_types = []
        max_length = max(len(segments) for segments in path_segments)

        for i in range(max_length):
            segments_at_position = []
            for segments in path_segments:
                if i < len(segments):
                    segments_at_position.append(segments[i])

            segment_type = self._classify_segment_type(segments_at_position)
            segment_types.append(segment_type)

        # 生成路径模式
        patterns = []
        for i, segment_type in enumerate(segment_types):
            pattern = {
                'position': i,
                'type': segment_type['type'],
                'pattern': segment_type['pattern'],
                'examples': segment_type['examples'][:5]  # 只保留前5个例子
            }
            patterns.append(pattern)

        return patterns

    def _classify_segment_type(self, segments: List[str]) -> Dict:
        """分类段类型"""
        if not segments:
            return {'type': 'empty', 'pattern': '', 'examples': []}

        # 检查是否为数字
        numeric_segments = [seg for seg in segments if seg.isdigit()]
        if len(numeric_segments) / len(segments) > 0.7:
            return {
                'type': 'numeric',
                'pattern': '\\d+',
                'examples': numeric_segments[:5]
            }

        # 检查是否为日期
        date_patterns = [
            r'\d{4}-\d{2}-\d{2}',  # YYYY-MM-DD
            r'\d{4}/\d{2}/\d{2}',  # YYYY/MM/DD
            r'\d{2}-\d{2}-\d{4}',  # MM-DD-YYYY
            r'\d{8}',             # YYYYMMDD
        ]

        for date_pattern in date_patterns:
            date_segments = [seg for seg in segments if re.match(date_pattern, seg)]
            if len(date_segments) / len(segments) > 0.5:
                return {
                    'type': 'date',
                    'pattern': date_pattern,
                    'examples': date_segments[:5]
                }

        # 检查是否为UUID
        uuid_pattern = r'[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}'
        uuid_segments = [seg for seg in segments if re.match(uuid_pattern, seg, re.IGNORECASE)]
        if len(uuid_segments) / len(segments) > 0.5:
            return {
                'type': 'uuid',
                'pattern': uuid_pattern,
                'examples': uuid_segments[:5]
            }

        # 检查是否为ID类型（数字+字母组合）
        id_pattern = r'[a-zA-Z0-9]{8,}'
        id_segments = [seg for seg in segments if re.match(id_pattern, seg)]
        if len(id_segments) / len(segments) > 0.5:
            return {
                'type': 'id',
                'pattern': id_pattern,
                'examples': id_segments[:5]
            }

        # 检查是否有共同前缀
        common_prefix = self._find_common_prefix(segments)
        if common_prefix:
            return {
                'type': 'prefixed',
                'pattern': f'{common_prefix}.*',
                'examples': segments[:5]
            }

        # 默认为文本类型
        return {
            'type': 'text',
            'pattern': '[^/]+',
            'examples': segments[:5]
        }

    def _find_common_prefix(self, strings: List[str]) -> str:
        """找出字符串的共同前缀"""
        if not strings:
            return ''

        common_prefix = strings[0]
        for string in strings[1:]:
            for i, (c1, c2) in enumerate(zip(common_prefix, string)):
                if c1 != c2:
                    common_prefix = common_prefix[:i]
                    break
            else:
                common_prefix = common_prefix[:len(string)]

        return common_prefix

    def _analyze_query_patterns(self, queries: List[str]) -> Dict[str, any]:
        """分析查询参数模式"""
        param_stats = defaultdict(lambda: {'count': 0, 'values': set(), 'patterns': []})

        for query in queries:
            if not query:
                continue

            params = urllib.parse.parse_qs(query)
            for param, values in params.items():
                param_stats[param]['count'] += 1
                param_stats[param]['values'].update(values)

        # 分析每个参数的值模式
        for param, stats in param_stats.items():
            values = list(stats['values'])
            if len(values) > 1:
                pattern = self._analyze_value_patterns(values)
                stats['patterns'] = pattern

        return dict(param_stats)

    def _analyze_value_patterns(self, values: List[str]) -> List[Dict]:
        """分析值的模式"""
        patterns = []

        # 数字模式
        numeric_values = [v for v in values if v.isdigit()]
        if numeric_values:
            patterns.append({
                'type': 'numeric',
                'count': len(numeric_values),
                'examples': numeric_values[:5]
            })

        # 日期模式
        date_patterns = [r'\d{4}-\d{2}-\d{2}', r'\d{8}']
        for date_pattern in date_patterns:
            date_values = [v for v in values if re.match(date_pattern, v)]
            if date_values:
                patterns.append({
                    'type': 'date',
                    'pattern': date_pattern,
                    'count': len(date_values),
                    'examples': date_values[:5]
                })

        # 文本模式
        text_values = [v for v in values if not v.isdigit() and not re.match(date_patterns[0], v)]
        if text_values:
            patterns.append({
                'type': 'text',
                'count': len(text_values),
                'examples': text_values[:5]
            })

        return patterns

    def _detect_pagination_patterns(self, urls: List[str]) -> List[Dict]:
        """检测分页模式"""
        patterns = []

        # 检测基于页码的分页
        page_params = ['page', 'p', 'pg', 'pageNumber', 'pageNum']
        for param in page_params:
            param_pattern = f'{param}=(\\d+)'
            matching_urls = [url for url in urls if re.search(param_pattern, url)]

            if len(matching_urls) >= 2:  # 至少需要2个URL才能确定是分页
                page_numbers = []
                for url in matching_urls:
                    match = re.search(param_pattern, url)
                    if match:
                        page_numbers.append(int(match.group(1)))

                if page_numbers:
                    patterns.append({
                        'type': 'page_param',
                        'parameter': param,
                        'pattern': param_pattern,
                        'count': len(matching_urls),
                        'page_range': [min(page_numbers), max(page_numbers)],
                        'example_urls': matching_urls[:3]
                    })

        # 检测基于偏移量的分页
        offset_params = ['offset', 'start', 'begin', 'skip']
        for param in offset_params:
            param_pattern = f'{param}=(\\d+)'
            matching_urls = [url for url in urls if re.search(param_pattern, url)]

            if len(matching_urls) >= 2:
                offsets = []
                for url in matching_urls:

class URLGenerator:
    """URL生成器"""

    def __init__(self):
        self.pattern_analyzer = URLPatternAnalyzer()

    def generate_urls_from_pattern(
        self,
        base_url: str,
        pattern_info: Dict,
        count: int = 100
    ) -> List[str]:
        """根据模式生成URL"""
        generated_urls = []

        pagination_patterns = pattern_info.get('pagination_patterns', [])

        for pattern in pagination_patterns:
            if pattern['type'] == 'page_param':
                urls = self._generate_page_param_urls(base_url, pattern, count)
                generated_urls.extend(urls)
            elif pattern['type'] == 'offset_param':
                urls = self._generate_offset_param_urls(base_url, pattern, count)
                generated_urls.extend(urls)
            elif pattern['type'] == 'page_path':
                urls = self._generate_page_path_urls(base_url, pattern, count)
                generated_urls.extend(urls)

        return generated_urls

    def _generate_page_param_urls(self, base_url: str, pattern: Dict, count: int) -> List[str]:
        """生成基于页码参数的URL"""
        urls = []
        param = pattern['parameter']

        parsed_url = urllib.parse.urlparse(base_url)
        query_params = urllib.parse.parse_qs(parsed_url.query)

        for page in range(1, count + 1):
            query_params[param] = [str(page)]
            new_query = urllib.parse.urlencode(query_params, doseq=True)
            new_url = urllib.parse.urlunparse((
                parsed_url.scheme,
                parsed_url.netloc,
                parsed_url.path,
                parsed_url.params,
                new_query,
                parsed_url.fragment
            ))
            urls.append(new_url)

        return urls

    def _generate_offset_param_urls(self, base_url: str, pattern: Dict, count: int) -> List[str]:
        """生成基于偏移量参数的URL"""
        urls = []
        param = pattern['parameter']
        offset_step = 20  # 默认步长

        parsed_url = urllib.parse.urlparse(base_url)
        query_params = urllib.parse.parse_qs(parsed_url.query)

        for i in range(count):
            offset = i * offset_step
            query_params[param] = [str(offset)]
            new_query = urllib.parse.urlencode(query_params, doseq=True)
            new_url = urllib.parse.urlunparse((
                parsed_url.scheme,
                parsed_url.netloc,
                parsed_url.path,
                parsed_url.params,
                new_query,
                parsed_url.fragment
            ))
            urls.append(new_url)

        return urls

    def _generate_page_path_urls(self, base_url: str, pattern: Dict, count: int) -> List[str]:
        """生成基于路径的URL"""
        urls = []
        path_pattern = pattern['pattern']

        parsed_url = urllib.parse.urlparse(base_url)

        for page in range(1, count + 1):
            new_path = re.sub(r'/\d+', f'/{page}', parsed_url.path)
            new_url = urllib.parse.urlunparse((
                parsed_url.scheme,
                parsed_url.netloc,
                new_path,
                parsed_url.params,
                parsed_url.query,
                parsed_url.fragment
            ))
            urls.append(new_url)

        return urls
```

```python
class NetworkMonitor:
    """网络请求监控器"""

    def __init__(self):
        self.requests = []

    def start_monitoring(self, driver):
        """开始监控网络请求"""
        self.monitoring = True

        # 注入JavaScript来拦截网络请求
        monitor_script = """
        (function() {
            const originalFetch = window.fetch;
            const originalXHR = window.XMLHttpRequest;
            const requests = [];

            // 拦截fetch请求
            window.fetch = function(...args) {
                const url = args[0];
                const options = args[1] || {};
                const requestId = Date.now() + Math.random();

                const requestData = {
                    id: requestId,
                    type: 'fetch',
                    url: url,
                    method: options.method || 'GET',
                    headers: options.headers || {},
                    body: options.body || '',
                    timestamp: Date.now(),
                    status: 'pending'
                };

                requests.push(requestData);

                return originalFetch.apply(this, args)
                    .then(response => {
                        const responseClone = response.clone();
                        responseClone.text().then(body => {
                            requestData.status = 'completed';
                            requestData.responseStatus = response.status;
                            requestData.responseHeaders = {};
                            response.headers.forEach((value, key) => {
                                requestData.responseHeaders[key] = value;
                            });
                            requestData.responseBody = body.substring(0, 1000); // 限制长度
                        });
                        return response;
                    })
                    .catch(error => {
                        requestData.status = 'error';
                        requestData.error = error.message;
                        throw error;
                    });
            };

            // 拦截XHR请求
            const XHROpen = originalXHR.prototype.open;
            const XHRSend = originalXHR.prototype.send;

            originalXHR.prototype.open = function(method, url) {
                this._requestId = Date.now() + Math.random();
                this._method = method;
                this._url = url;

                const requestData = {
                    id: this._requestId,
                    type: 'xhr',
                    method: method,
                    url: url,
                    headers: {},
                    body: '',
                    timestamp: Date.now(),
                    status: 'pending'
                };

                requests.push(requestData);

                return XHROpen.apply(this, arguments);
            };

            originalXHR.prototype.send = function(data) {
                const xhr = this;
                const requestId = xhr._requestId;

                const requestData = requests.find(r => r.id === requestId);
                if (requestData) {
                    requestData.body = data || '';

                    xhr.addEventListener('load', function() {
                        requestData.status = 'completed';
                        requestData.responseStatus = xhr.status;
                        requestData.responseHeaders = {};

                        try {
                            const responseHeaders = xhr.getAllResponseHeaders();
                            const headerLines = responseHeaders.split('\\r\\n');
                            headerLines.forEach(line => {
                                const parts = line.split(': ');

                    });
                }

                return XHRSend.apply(this, arguments);
            };

            // 将请求数据暴露给外部
            window.networkRequests = requests;
        })();
        """

        driver.execute_script(monitor_script)

    def _monitor_requests(self, driver):

        while self.monitoring:
            try:
                # 获取网络请求数据
                requests_data = driver.execute_script("""

class APIAnalyzer:
    """API分析器"""

    def __init__(self):
        self.api_patterns = {}
        self.request_analyzer = RequestAnalyzer()

    def analyze_api_requests(self, requests: List[APIRequest]) -> Dict[str, Any]:
        """分析API请求"""
        if not requests:
            return {}

        # 分组请求
        grouped_requests = self._group_similar_requests(requests)

        # 分析每个组的模式
        patterns = {}
        for group_key, group_requests in grouped_requests.items():
            pattern = self._analyze_request_group(group_requests)
            patterns[group_key] = pattern

        # 分析参数变化模式
        parameter_patterns = self._analyze_parameter_patterns(requests)

        # 检测认证机制
        auth_mechanisms = self._detect_authentication_mechanisms(requests)

        # 分析响应格式
        response_formats = self._analyze_response_formats(requests)

        return {
            'patterns': patterns,
            'parameter_patterns': parameter_patterns,
            'authentication': auth_mechanisms,
            'response_formats': response_formats,
            'total_requests': len(requests),
            'unique_endpoints': len(grouped_requests)
        }

    def _group_similar_requests(self, requests: List[APIRequest]) -> Dict[str, List[APIRequest]]:
        """将相似的请求分组"""
        groups = {}

        for request in requests:
            # 提取URL模式（去掉参数）
            url_pattern = self._extract_url_pattern(request.url)
            group_key = f"{request.method}:{url_pattern}"

            if group_key not in groups:
                groups[group_key] = []
            groups[group_key].append(request)

        return groups

    def _extract_url_pattern(self, url: str) -> str:
        """提取URL模式"""
        parsed = urlparse(url)
        base_url = f"{parsed.scheme}://{parsed.netloc}{parsed.path}"

        # 替换数字ID为占位符
        pattern = re.sub(r'/\d+', '/{id}', base_url)
        pattern = re.sub(r'/[a-f0-9-]{36}', '/{uuid}', pattern)
        pattern = re.sub(r'/[a-zA-Z0-9]{8,}', '/{identifier}', pattern)

        return pattern

    def _analyze_request_group(self, requests: List[APIRequest]) -> APIPattern:
        """分析请求组"""
        if not requests:
            return None

        # 提取共同特征
        first_request = requests[0]
        url_pattern = self._extract_url_pattern(first_request.url)
        method = first_request.method

        # 分析参数
        all_params = set()
        for request in requests:
            all_params.update(request.params.keys())

        # 分析请求头
        common_headers = self._find_common_headers(requests)

        # 分析请求体
        body_schema = self._analyze_body_schema(requests)

        # 分析响应
        response_schema = self._analyze_response_schema(requests)

        return APIPattern(
            url_pattern=url_pattern,
            method=method,
            parameters=list(all_params),
            headers=common_headers,
            body_schema=body_schema,
            response_schema=response_schema,
            examples=requests[:5]  # 保留前5个例子
        )

    def _find_common_headers(self, requests: List[APIRequest]) -> Dict[str, str]:
        """找出共同的请求头"""
        if not requests:
            return {}

        header_counts = {}
        for request in requests:
            for header, value in request.headers.items():
                if header not in header_counts:
                    header_counts[header] = {}
                if value not in header_counts[header]:
                    header_counts[header][value] = 0
                header_counts[header][value] += 1

        # 找出出现频率最高的值
        common_headers = {}
        for header, values in header_counts.items():
            if len(values) == 1:  # 所有请求中该头部的值都相同
                common_headers[header] = list(values.keys())[0]
            else:
                # 选择出现次数最多的值
                most_common_value = max(values.items(), key=lambda x: x[1])[0]
                if values[most_common_value] / len(requests) > 0.8:  # 出现频率超过80%
                    common_headers[header] = most_common_value

        return common_headers

    def _analyze_body_schema(self, requests: List[APIRequest]) -> Optional[Dict]:
        """分析请求体模式"""
        bodies_with_content = [req.body for req in requests if req.body and req.body.strip()]

        if not bodies_with_content:
            return None

        # 尝试解析JSON
        json_bodies = []
        for body in bodies_with_content:
            try:
                json_body = json.loads(body)
                json_bodies.append(json_body)
            except:
                continue

        if json_bodies:
            return self._analyze_json_schema(json_bodies)
        else:
            # 分析表单数据
            return self._analyze_form_schema(bodies_with_content)

    def _analyze_json_schema(self, json_objects: List[Dict]) -> Dict:
        """分析JSON模式"""
        if not json_objects:
            return {}

        # 收集所有字段
        all_fields = set()
        field_types = {}
        field_examples = {}

        for obj in json_objects:
            for key, value in obj.items():
                all_fields.add(key)
                field_types[key] = type(value).__name__
                if key not in field_examples:
                    field_examples[key] = value

        return {
            'type': 'json',
            'fields': list(all_fields),
            'field_types': field_types,
            'examples': field_examples
        }

    def _analyze_form_schema(self, bodies: List[str]) -> Dict:
        """分析表单数据模式"""
        all_params = set()
        param_examples = {}

        for body in bodies:
            try:
                params = parse_qs(body)
                for param, values in params.items():
                    all_params.add(param)
                    if param not in param_examples:
                        param_examples[param] = values[0] if values else ''
            except:
                continue

        return {
            'type': 'form',
            'parameters': list(all_params),
            'examples': param_examples
        }

    def _analyze_response_schema(self, requests: List[APIRequest]) -> Optional[Dict]:
        """分析响应模式"""
        responses_with_content = [
            req for req in requests
            if req.response_body and req.response_status and 200 <= req.response_status < 300
        ]

        if not responses_with_content:
            return None

        # 分析内容类型
        content_types = set()
        for req in responses_with_content:
            content_type = req.response_headers.get('content-type', '').split(';')[0]
            content_types.add(content_type)

        # 分析JSON响应
        json_responses = []
        for req in responses_with_content:
            content_type = req.response_headers.get('content-type', '').lower()
            if 'json' in content_type:
                try:
                    json_response = json.loads(req.response_body)
                    json_responses.append(json_response)
                except:
                    continue

        schema = {
            'content_types': list(content_types),
            'status_codes': list(set(req.response_status for req in responses_with_content))
        }

        if json_responses:
            schema['json_schema'] = self._analyze_json_schema(json_responses)

        return schema

    def _analyze_parameter_patterns(self, requests: List[APIRequest]) -> Dict[str, Any]:
        """分析参数变化模式"""
        parameter_analysis = {}

        # 按URL模式分组
        grouped_requests = self._group_similar_requests(requests)

        for group_key, group_requests in grouped_requests.items():
            params_by_name = defaultdict(list)

            for request in group_requests:
                for param, value in request.params.items():
                    params_by_name[param].append(value)

            group_analysis = {}
            for param, values in params_by_name.items():
                group_analysis[param] = self._analyze_parameter_values(values)

            parameter_analysis[group_key] = group_analysis

        return parameter_analysis

    def _analyze_parameter_values(self, values: List[str]) -> Dict[str, Any]:
        """分析参数值的模式"""
        if not values:
            return {}

        unique_values = list(set(values))
        numeric_values = []
        date_values = []

        for value in values:
            if value.isdigit():
                numeric_values.append(int(value))
            elif re.match(r'\d{4}-\d{2}-\d{2}', value):
                date_values.append(value)

        analysis = {
            'unique_count': len(unique_values),
            'total_count': len(values),
            'most_common': max(set(values), key=values.count) if values else None,
            'examples': unique_values[:5]
        }

        if numeric_values:
            analysis['numeric_info'] = {
                'type': 'numeric',
                'min': min(numeric_values),
                'max': max(numeric_values),
                'range': max(numeric_values) - min(numeric_values)
            }

        if date_values:
            analysis['date_info'] = {
                'type': 'date',
                'examples': date_values[:5]
            }

        return analysis

    def _detect_authentication_mechanisms(self, requests: List[APIRequest]) -> List[Dict]:
        """检测认证机制"""
        auth_mechanisms = []

        # 检查各种认证头
        auth_headers = ['Authorization', 'X-API-Key', 'X-Auth-Token', 'Cookie']

        for header in auth_headers:
            header_values = []
            for request in requests:
                if header in request.headers:
                    header_values.append(request.headers[header])

            if header_values:
                # 分析认证类型
                auth_type = self._identify_auth_type(header, header_values)
                auth_mechanisms.append({
                    'type': auth_type,
                    'header': header,
                    'present_in_requests': len(header_values),
                    'examples': header_values[:3]
                })

        return auth_mechanisms

    def _identify_auth_type(self, header: str, values: List[str]) -> str:
        """识别认证类型"""
        if header == 'Authorization':
            for value in values:
                if value.startswith('Bearer '):
                    return 'Bearer Token'
                elif value.startswith('Basic '):
                    return 'Basic Auth'
                elif value.startswith('Digest '):
                    return 'Digest Auth'
                elif value.startswith('OAuth '):
                    return 'OAuth'
            return 'Unknown Auth'

        elif header == 'X-API-Key':
            return 'API Key'

        elif header == 'Cookie':
            # 检查是否包含会话相关的cookie
            for value in values:
                if 'session' in value.lower() or 'token' in value.lower():
                    return 'Session Cookie'
            return 'Cookie'

        return 'Custom Auth'

    def _analyze_response_formats(self, requests: List[APIRequest]) -> Dict[str, Any]:
        """分析响应格式"""
        format_analysis = {}

        # 按内容类型分组
        content_type_groups = defaultdict(list)
        for request in requests:
            if request.response_headers:
                content_type = request.response_headers.get('content-type', '').split(';')[0]
                content_type_groups[content_type].append(request)

        for content_type, requests_list in content_type_groups.items():
            format_analysis[content_type] = {
                'count': len(requests_list),
                'avg_size': sum(len(req.response_body or '') for req in requests_list) / len(requests_list)
            }

            if 'json' in content_type:
                # 分析JSON结构
                json_responses = []
                for req in requests_list:
                    try:
                        json_resp = json.loads(req.response_body or '{}')
                        json_responses.append(json_resp)
                    except:
                        continue

                if json_responses:
                    format_analysis[content_type]['structure'] = self._analyze_json_structure(json_responses)

        return format_analysis

    def _analyze_json_structure(self, json_objects: List[Dict]) -> Dict:
        """分析JSON结构"""
        if not json_objects:
            return {}

        # 分析根结构
        root_types = set()
        for obj in json_objects:
            if isinstance(obj, dict):
                root_types.add('object')
            elif isinstance(obj, list):
                root_types.add('array')
            else:
                root_types.add('primitive')

        structure = {
            'root_type': list(root_types),
            'field_analysis': {}
        }

        # 如果是对象数组，分析数组元素的结构
        array_objects = []
        for obj in json_objects:
            if isinstance(obj, list) and obj and isinstance(obj[0], dict):
                array_objects.extend([item for item in obj if isinstance(item, dict)])

        if array_objects:
            field_analysis = self._analyze_json_schema(array_objects)
            structure['field_analysis'] = field_analysis

        return structure

class RequestAnalyzer:
    """请求分析器"""

    def __init__(self):
        self.session = requests.Session()

    def test_api_endpoint(self, url: str, method: str = 'GET', **kwargs) -> Dict[str, Any]:
        """测试API端点"""
        try:
            if method.upper() == 'GET':
                response = self.session.get(url, **kwargs)
            elif method.upper() == 'POST':
                response = self.session.post(url, **kwargs)
            elif method.upper() == 'PUT':
                response = self.session.put(url, **kwargs)
            elif method.upper() == 'DELETE':
                response = self.session.delete(url, **kwargs)
            else:
                return {'error': f'Unsupported method: {method}'}

            return {
                'status_code': response.status_code,
                'headers': dict(response.headers),
                'content': response.text[:1000],  # 限制长度
                'content_type': response.headers.get('content-type', ''),
                'success': response.status_code < 400
            }

        except Exception as e:
            return {'error': str(e), 'success': False}
```
