### 14.1.2 传感器管理器实现

```typescript
// SensorManager.ets
import sensor from '@ohos.sensor'
import hilog from '@ohos.hilog'

export interface SensorData {
  type: SensorType
  timestamp: number
  values: number[]
  accuracy?: number
}

export interface SensorListener {
  onSensorDataChanged(data: SensorData): void
  onSensorAccuracyChanged?(accuracy: number): void
}

export class SensorManager {
  private static instance: SensorManager
  private listeners: Map<number, Set<SensorListener>> = new Map()
  private activeSensors: Set<number> = new Set()
  private readonly TAG = 'SensorManager'

  static getInstance(): SensorManager {
    if (!SensorManager.instance) {
      SensorManager.instance = new SensorManager()
    }
    return SensorManager.instance
  }

  /**
   * 注册传感器监听器
   * @param sensorType 传感器类型
   * @param listener 监听器
   * @param samplingInterval 采样间隔
   */
  async registerSensor(
    sensorType: SensorType,
    listener: SensorListener,
    samplingInterval: number = 100000000 // 100ms
  ): Promise<boolean> {
    try {
      // 检查传感器是否支持
      const isSupported = await this.isSensorSupported(sensorType)
      if (!isSupported) {
        hilog.error(0x0000, this.TAG, `Sensor type ${sensorType} not supported`)
        return false
      }

      // 添加监听器
      if (!this.listeners.has(sensorType)) {
        this.listeners.set(sensorType, new Set())
      }
      this.listeners.get(sensorType)?.add(listener)

      // 如果传感器未激活，则激活它
      if (!this.activeSensors.has(sensorType)) {
        await this.activateSensor(sensorType, samplingInterval)
        this.activeSensors.add(sensorType)
      }

      hilog.info(0x0000, this.TAG, `Sensor ${sensorType} registered successfully`)
      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to register sensor: ${error}`)
      return false
    }
  }

  /**
   * 注销传感器监听器
   * @param sensorType 传感器类型
   * @param listener 监听器
   */
  async unregisterSensor(sensorType: SensorType, listener: SensorListener): Promise<void> {
    const listeners = this.listeners.get(sensorType)
    if (listeners) {
      listeners.delete(listener)

      // 如果没有监听器了，停用传感器
      if (listeners.size === 0) {
        await this.deactivateSensor(sensorType)
        this.activeSensors.delete(sensorType)
        this.listeners.delete(sensorType)
      }
    }
  }

  /**
   * 激活传感器
   * @param sensorType 传感器类型
   * @param samplingInterval 采样间隔
   */
  private async activateSensor(sensorType: SensorType, samplingInterval: number): Promise<void> {
    try {
      const sensorDataListener = new sensor.SensorDataListener((data: sensor.SensorData) => {
        this.handleSensorData(sensorType, data)
      })

      // 根据传感器类型激活
      switch (sensorType) {
        case SensorType.ACCELEROMETER:
          sensor.on(sensor.SensorId.ACCELEROMETER, sensorDataListener, { interval: samplingInterval })
          break
        case SensorType.GYROSCOPE:
          sensor.on(sensor.SensorId.GYROSCOPE, sensorDataListener, { interval: samplingInterval })
          break
        case SensorType.MAGNETIC_FIELD:
          sensor.on(sensor.SensorId.MAGNETIC_FIELD, sensorDataListener, { interval: samplingInterval })
          break
        case SensorType.ORIENTATION:
          sensor.on(sensor.SensorId.ORIENTATION, sensorDataListener, { interval: samplingInterval })
          break
        case SensorType.PROXIMITY:
          sensor.on(sensor.SensorId.PROXIMITY, sensorDataListener, { interval: samplingInterval })
          break
        case SensorType.AMBIENT_LIGHT:
          sensor.on(sensor.SensorId.AMBIENT_LIGHT, sensorDataListener, { interval: samplingInterval })
          break
        case SensorType.PRESSURE:
          sensor.on(sensor.SensorId.PRESSURE, sensorDataListener, { interval: samplingInterval })
          break
        case SensorType.STEP_COUNTER:
          sensor.on(sensor.SensorId.STEP_COUNTER, sensorDataListener, { interval: samplingInterval })
          break
        default:
          throw new Error(`Unsupported sensor type: ${sensorType}`)
      }
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to activate sensor ${sensorType}: ${error}`)
      throw error
    }
  }

  /**
   * 停用传感器
   * @param sensorType 传感器类型
   */
  private async deactivateSensor(sensorType: SensorType): Promise<void> {
    try {
      switch (sensorType) {
        case SensorType.ACCELEROMETER:
          sensor.off(sensor.SensorId.ACCELEROMETER)
          break
        case SensorType.GYROSCOPE:
          sensor.off(sensor.SensorId.GYROSCOPE)
          break
        case SensorType.MAGNETIC_FIELD:
          sensor.off(sensor.SensorId.MAGNETIC_FIELD)
          break
        case SensorType.ORIENTATION:
          sensor.off(sensor.SensorId.ORIENTATION)
          break
        case SensorType.PROXIMITY:
          sensor.off(sensor.SensorId.PROXIMITY)
          break
        case SensorType.AMBIENT_LIGHT:
          sensor.off(sensor.SensorId.AMBIENT_LIGHT)
          break
        case SensorType.PRESSURE:
          sensor.off(sensor.SensorId.PRESSURE)
          break
        case SensorType.STEP_COUNTER:
          sensor.off(sensor.SensorId.STEP_COUNTER)
          break
      }
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to deactivate sensor ${sensorType}: ${error}`)
    }
  }

  /**
   * 处理传感器数据
   * @param sensorType 传感器类型
   * @param data 传感器数据
   */
  private handleSensorData(sensorType: SensorType, data: sensor.SensorData): void {
    const listeners = this.listeners.get(sensorType)
    if (!listeners || listeners.size === 0) {
      return
    }

    const sensorData: SensorData = {
      type: sensorType,
      timestamp: data.timestamp,
      values: this.extractValues(sensorType, data),
      accuracy: data.accuracy
    }

    // 通知所有监听器
    listeners.forEach(listener => {
      try {
        listener.onSensorDataChanged(sensorData)
      } catch (error) {
        hilog.error(0x0000, this.TAG, `Error in sensor listener: ${error}`)
      }
    })
  }

  /**
   * 提取传感器值
   * @param sensorType 传感器类型
   * @param data 原始数据
   */
  private extractValues(sensorType: SensorType, data: sensor.SensorData): number[] {
    switch (sensorType) {
      case SensorType.ACCELEROMETER:
        return [(data as sensor.AccelerometerResponse).x,
                (data as sensor.AccelerometerResponse).y,
                (data as sensor.AccelerometerResponse).z]
      case SensorType.GYROSCOPE:
        return [(data as sensor.GyroscopeResponse).x,
                (data as sensor.GyroscopeResponse).y,
                (data as sensor.GyroscopeResponse).z]
      case SensorType.MAGNETIC_FIELD:
        return [(data as sensor.MagneticFieldResponse).x,
                (data as sensor.MagneticFieldResponse).y,
                (data as sensor.MagneticFieldResponse).z]
      case SensorType.ORIENTATION:
        return [(data as sensor.OrientationResponse).alpha,
                (data as sensor.OrientationResponse).beta,
                (data as sensor.OrientationResponse).gamma]
      case SensorType.PROXIMITY:
        return [(data as sensor.ProximityResponse).distance]
      case SensorType.AMBIENT_LIGHT:
        return [(data as sensor.LightResponse).intensity]
      case SensorType.PRESSURE:
        return [(data as sensor.PressureResponse).pressure]
      case SensorType.STEP_COUNTER:
        return [(data as sensor.StepCounterResponse).steps]
      default:
        return []
    }
  }

  /**
   * 检查传感器是否支持
   * @param sensorType 传感器类型
   */
  private async isSensorSupported(sensorType: SensorType): Promise<boolean> {
    try {
      return await sensor.getSensorList().then(list => {
        return list.some(sensor => sensor.sensorId === this.getSensorId(sensorType))
      })
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to check sensor support: ${error}`)
      return false
    }
  }

  /**
   * 获取传感器ID
   * @param sensorType 传感器类型
   */
  private getSensorId(sensorType: SensorType): number {
    switch (sensorType) {
      case SensorType.ACCELEROMETER:
        return sensor.SensorId.ACCELEROMETER
      case SensorType.GYROSCOPE:
        return sensor.SensorId.GYROSCOPE
      case SensorType.MAGNETIC_FIELD:
        return sensor.SensorId.MAGNETIC_FIELD
      case SensorType.ORIENTATION:
        return sensor.SensorId.ORIENTATION
      case SensorType.PROXIMITY:
        return sensor.SensorId.PROXIMITY
      case SensorType.AMBIENT_LIGHT:
        return sensor.SensorId.AMBIENT_LIGHT
      case SensorType.PRESSURE:
        return sensor.SensorId.PRESSURE
      case SensorType.STEP_COUNTER:
        return sensor.SensorId.STEP_COUNTER
      default:
        return -1
    }
  }

  /**
   * 释放所有传感器资源
   */
  async release(): Promise<void> {
    for (const sensorType of this.activeSensors) {
      await this.deactivateSensor(sensorType)
    }
    this.activeSensors.clear()
    this.listeners.clear()
    hilog.info(0x0000, this.TAG, 'All sensors released')
  }
}
```

```typescript
// MotionDetector.ets
import { SensorManager, SensorData, SensorListener, SensorType } from './SensorManager'

@Entry
@Component
struct MotionDetector {
  @State accelerationData: string = '等待数据...'
  @State gyroData: string = '等待数据...'
  @State motionStatus: string = '静止'
  @State stepCount: number = 0
  @State isRecording: boolean = false

  private sensorManager = SensorManager.getInstance()
  private motionThreshold: number = 2.0 // 运动阈值

  async aboutToAppear() {
    await this.initSensors()
  }

  async aboutToDisappear() {
    await this.cleanupSensors()
  }

  private async initSensors() {
    // 注册加速度传感器监听器
    await this.sensorManager.registerSensor(
      SensorType.ACCELEROMETER,
      {
        onSensorDataChanged: (data: SensorData) => {
          this.handleAccelerationData(data)
        }
      },
      50000000 // 50ms
    )

    // 注册陀螺仪传感器监听器
    await this.sensorManager.registerSensor(
      SensorType.GYROSCOPE,
      {
        onSensorDataChanged: (data: SensorData) => {
          this.handleGyroData(data)
        }
      },
      50000000 // 50ms
    )

    // 注册计步传感器监听器
    await this.sensorManager.registerSensor(
      SensorType.STEP_COUNTER,
      {
        onSensorDataChanged: (data: SensorData) => {
          this.handleStepData(data)
        }
      },
      1000000000 // 1s
    )
  }

  private async cleanupSensors() {
    // 释放传感器资源
    await this.sensorManager.release()
  }

  private handleAccelerationData(data: SensorData) {
    if (data.values.length >= 3) {
      const x = data.values[0].toFixed(2)
      const y = data.values[1].toFixed(2)
      const z = data.values[2].toFixed(2)

      this.accelerationData = `X: ${x}, Y: ${y}, Z: ${z}`

      // 检测运动状态
      const magnitude = Math.sqrt(data.values[0] ** 2 + data.values[1] ** 2 + data.values[2] ** 2)
      if (magnitude > this.motionThreshold) {
        this.motionStatus = '运动中'
      } else {
        this.motionStatus = '静止'
      }
    }
  }

  private handleGyroData(data: SensorData) {
    if (data.values.length >= 3) {
      const x = data.values[0].toFixed(2)
      const y = data.values[1].toFixed(2)
      const z = data.values[2].toFixed(2)

      this.gyroData = `X: ${x}, Y: ${y}, Z: ${z}`
    }
  }

  private handleStepData(data: SensorData) {
    if (data.values.length >= 1) {
      this.stepCount = Math.floor(data.values[0])
    }
  }
```

### 14.2.1 位置服务管理

```typescript
// LocationManager.ets
import geolocation from '@ohos.geolocation'
import hilog from '@ohos.hilog'

export interface LocationInfo {
  latitude: number
  longitude: number
  altitude?: number
  accuracy?: number
  speed?: number
  timestamp: number
  address?: string
}

export interface LocationListener {
  onLocationChanged(location: LocationInfo): void
  onLocationError?(error: string): void
}

export class LocationManager {
  private static instance: LocationManager
  private listeners: Set<LocationListener> = new Set()
  private isLocationEnabled: boolean = false
  private locationRequest: geolocation.LocationRequest = {
    priority: geolocation.LocationRequestPriority.ACCURACY_HIGH,
    timeInterval: 1000,
    distanceInterval: 1,
    maxAccuracy: 100
  }
  private readonly TAG = 'LocationManager'

  static getInstance(): LocationManager {
    if (!LocationManager.instance) {
      LocationManager.instance = new LocationManager()
    }
    return LocationManager.instance
  }

  /**
   * 检查位置服务是否可用
   */
  async isLocationServiceAvailable(): Promise<boolean> {
    try {
      this.isLocationEnabled = await geolocation.isLocationEnabled()
      return this.isLocationEnabled
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to check location service: ${error}`)
      return false
    }
  }

  /**
   * 请求位置权限
   */
  async requestLocationPermission(): Promise<boolean> {
    try {
      // 这里需要根据实际权限管理API进行调整
      // const result = await requestPermission('ohos.permission.APPROXIMATELY_LOCATION')
      // return result === 0
      return true // 简化处理
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to request location permission: ${error}`)
      return false
    }
  }

  /**
   * 获取当前位置（单次定位）
   */
  async getCurrentLocation(): Promise<LocationInfo | null> {
    try {
      const location = await geolocation.getCurrentLocation(this.locationRequest)

      const locationInfo: LocationInfo = {
        latitude: location.latitude,
        longitude: location.longitude,
        altitude: location.altitude,
        accuracy: location.accuracy,
        speed: location.speed,
        timestamp: location.timeStamp
      }

      // 尝试获取地址信息
      const addresses = await this.reverseGeocode(locationInfo.latitude, locationInfo.longitude)
      if (addresses && addresses.length > 0) {
        locationInfo.address = addresses[0]
      }

      return locationInfo
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to get current location: ${error}`)
      return null
    }
  }

  /**
   * 开始连续定位
   */
  async startLocationUpdates(): Promise<boolean> {
    try {
      if (!this.isLocationEnabled) {
        const enabled = await this.isLocationServiceAvailable()
        if (!enabled) {
          hilog.error(0x0000, this.TAG, 'Location service is not enabled')
          return false
        }
      }

      const locationChangeListener = new geolocation.LocationChangeListener((location: geolocation.Location) => {
        this.handleLocationUpdate(location)
      })

      await geolocation.startLocationUpdate(this.locationRequest, locationChangeListener)
      hilog.info(0x0000, this.TAG, 'Location updates started')
      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to start location updates: ${error}`)
      return false
    }
  }

  /**
   * 停止连续定位
   */
  async stopLocationUpdates(): Promise<void> {
    try {
      await geolocation.stopLocationUpdate()
      hilog.info(0x0000, this.TAG, 'Location updates stopped')
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to stop location updates: ${error}`)
    }
  }

  /**
   * 添加位置监听器
   */
  addLocationListener(listener: LocationListener): void {
    this.listeners.add(listener)
  }

  /**
   * 移除位置监听器
   */
  removeLocationListener(listener: LocationListener): void {
    this.listeners.delete(listener)
  }

  /**
   * 处理位置更新
   */
  private async handleLocationUpdate(location: geolocation.Location): Promise<void> {
    const locationInfo: LocationInfo = {
      latitude: location.latitude,
      longitude: location.longitude,
      altitude: location.altitude,
      accuracy: location.accuracy,
      speed: location.speed,
      timestamp: location.timeStamp
    }

    // 通知所有监听器
    this.listeners.forEach(listener => {
      try {
        listener.onLocationChanged(locationInfo)
      } catch (error) {
        hilog.error(0x0000, this.TAG, `Error in location listener: ${error}`)
      }
    })
  }

  /**
   * 反向地理编码（坐标转地址）
   */
  async reverseGeocode(latitude: number, longitude: number): Promise<string[] | null> {
    try {
      const reverseGeocodeRequest: geolocation.ReverseGeocodeRequest = {
        latitude: latitude,
        longitude: longitude,
        maxItems: 5
      }

      const addresses = await geolocation.getAddressesFromLocation(reverseGeocodeRequest)
      return addresses.map(addr => `${addr.administrativeArea} ${addr.subAdministrativeArea} ${addr.locality} ${addr.subLocality} ${addr.roadName} ${addr.subRoadName}`.trim())
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to reverse geocode: ${error}`)
      return null
    }
  }

  /**
   * 地理编码（地址转坐标）
   */
  async geocode(address: string): Promise<LocationInfo | null> {
    try {
      const geocodeRequest: geolocation.GeocodeRequest = {
        description: address,
        maxItems: 1
      }

      const locations = await geolocation.getLocationFromAddress(geocodeRequest)
      if (locations.length > 0) {
        const location = locations[0]
        return {
          latitude: location.latitude,
          longitude: location.longitude,
          timestamp: Date.now()
        }
      }
      return null
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to geocode address: ${error}`)
      return null
    }
  }

  /**
   * 计算两点间距离（米）
   */
  calculateDistance(lat1: number, lon1: number, lat2: number, lon2: number): number {
    const R = 6371e3 // 地球半径（米）
    const φ1 = lat1 * Math.PI / 180
    const φ2 = lat2 * Math.PI / 180
    const Δφ = (lat2 - lat1) * Math.PI / 180
    const Δλ = (lon2 - lon1) * Math.PI / 180

    const a = Math.sin(Δφ / 2) * Math.sin(Δφ / 2) +
              Math.cos(φ1) * Math.cos(φ2) *
              Math.sin(Δλ / 2) * Math.sin(Δλ / 2)
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))

    return R * c
  }

  /**
   * 设置定位请求参数
   */
  setLocationRequest(request: Partial<geolocation.LocationRequest>): void {
    this.locationRequest = { ...this.locationRequest, ...request }
  }
}
```

### 14.2.2 地图组件实现

```typescript
// MapView.ets
import { LocationManager, LocationInfo, LocationListener } from './LocationManager'

@Component
export struct MapView {
  @State currentLocation: LocationInfo | null = null
  @State mapCenter: LocationInfo = { latitude: 39.9042, longitude: 116.4074, timestamp: Date.now() }
  @State markers: MapMarker[] = []
  @State zoomLevel: number = 15
  @State isTracking: boolean = false
  @State addressInfo: string = '获取地址中...'

  private locationManager = LocationManager.getInstance()

  async aboutToAppear() {
    await this.initializeLocation()
    this.setupLocationListener()
  }

  async aboutToDisappear() {
    await this.locationManager.stopLocationUpdates()
  }

  private async initializeLocation() {
    // 检查位置服务可用性
    const isAvailable = await this.locationManager.isLocationServiceAvailable()
    if (!isAvailable) {
      // 提示用户开启位置服务
      return
    }

    // 获取当前位置
```

### 14.3.1 蓝牙管理器

```typescript
// BluetoothManager.ets
import bluetooth from '@ohos.bluetooth'
import hilog from '@ohos.hilog'

export interface BluetoothDevice {
  id: string
  name: string
  address: string
  rssi?: number
  deviceClass?: number
  bondState?: number
}

export interface BluetoothListener {
  onDeviceFound(device: BluetoothDevice): void
  onDeviceConnected(deviceId: string): void
  onDeviceDisconnected(deviceId: string): void
  onDataReceived(deviceId: string, data: ArrayBuffer): void
  onBluetoothStateChanged(state: BluetoothState): void
  onError(error: string): void
}

export enum BluetoothState {
  OFF = 'OFF',
  TURNING_ON = 'TURNING_ON',
  ON = 'ON',
  TURNING_OFF = 'TURNING_OFF',
  ERROR = 'ERROR'
}

export class BluetoothManager {
  private static instance: BluetoothManager
  private listeners: Set<BluetoothListener> = new Set()
  private isScanning: boolean = false
  private connectedDevices: Map<string, bluetooth.BluetoothDevice> = new Map()
  private readonly TAG = 'BluetoothManager'

  static getInstance(): BluetoothManager {
    if (!BluetoothManager.instance) {
      BluetoothManager.instance = new BluetoothManager()
    }
    return BluetoothManager.instance
  }

  /**
   * 检查蓝牙是否可用
   */
  async isBluetoothAvailable(): Promise<boolean> {
    try {
      return await bluetooth.getState() === bluetooth.BluetoothState.STATE_ON
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to check bluetooth availability: ${error}`)
      return false
    }
  }

  /**
   * 开启蓝牙
   */
  async enableBluetooth(): Promise<boolean> {
    try {
      await bluetooth.enableBluetooth()
      hilog.info(0x0000, this.TAG, 'Bluetooth enabled')
      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to enable bluetooth: ${error}`)
      return false
    }
  }

  /**
   * 关闭蓝牙
   */
  async disableBluetooth(): Promise<void> {
    try {
      await bluetooth.disableBluetooth()
      hilog.info(0x0000, this.TAG, 'Bluetooth disabled')
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to disable bluetooth: ${error}`)
    }
  }

  /**
   * 开始设备扫描
   */
  async startDeviceDiscovery(): Promise<boolean> {
    try {
      if (this.isScanning) {
        hilog.warn(0x0000, this.TAG, 'Device discovery already in progress')
        return false
      }

      const isAvailable = await this.isBluetoothAvailable()
      if (!isAvailable) {
        hilog.error(0x0000, this.TAG, 'Bluetooth is not available')
        return false
      }

      // 设置设备发现监听器
      const deviceFoundListener = new bluetooth.DeviceFoundListener((device: bluetooth.BluetoothDevice) => {
        this.handleDeviceFound(device)
      })

      // 开始扫描
      await bluetooth.startBluetoothDiscovery(deviceFoundListener)
      this.isScanning = true

      // 设置扫描超时
      setTimeout(() => {
        if (this.isScanning) {
          this.stopDeviceDiscovery()
        }
      }, 30000) // 30秒超时

      hilog.info(0x0000, this.TAG, 'Device discovery started')
      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to start device discovery: ${error}`)
      return false
    }
  }

  /**
   * 停止设备扫描
   */
  async stopDeviceDiscovery(): Promise<void> {
    try {
      await bluetooth.stopBluetoothDiscovery()
      this.isScanning = false
      hilog.info(0x0000, this.TAG, 'Device discovery stopped')
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to stop device discovery: ${error}`)
    }
  }

  /**
   * 连接设备
   */
  async connectDevice(deviceId: string): Promise<boolean> {
    try {
      // 获取设备对象
      const device = await bluetooth.getRemoteDevice(deviceId)
      if (!device) {
        hilog.error(0x0000, this.TAG, `Device not found: ${deviceId}`)
        return false
      }

      // 建立GATT连接
      await bluetooth.createGattClientConnection(deviceId)
      this.connectedDevices.set(deviceId, device)

      // 设置连接状态监听器
      this.setupConnectionListeners(deviceId)

      hilog.info(0x0000, this.TAG, `Connected to device: ${deviceId}`)

      // 通知监听器
      this.listeners.forEach(listener => {
        listener.onDeviceConnected(deviceId)
      })

      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to connect to device ${deviceId}: ${error}`)
      return false
    }
  }

  /**
   * 断开设备连接
   */
  async disconnectDevice(deviceId: string): Promise<void> {
    try {
      await bluetooth.disconnectGattClient(deviceId)
      this.connectedDevices.delete(deviceId)

      hilog.info(0x0000, this.TAG, `Disconnected from device: ${deviceId}`)

      // 通知监听器
      this.listeners.forEach(listener => {
        listener.onDeviceDisconnected(deviceId)
      })
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to disconnect from device ${deviceId}: ${error}`)
    }
  }

  /**
   * 发送数据
   */
  async sendData(deviceId: string, serviceUuid: string, characteristicUuid: string, data: ArrayBuffer): Promise<boolean> {
    try {
      await bluetooth.writeCharacteristicValue(deviceId, serviceUuid, characteristicUuid, data)
      hilog.info(0x0000, this.TAG, `Data sent to device ${deviceId}`)
      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to send data to device ${deviceId}: ${error}`)
      return false
    }
  }

  /**
   * 读取数据
   */
  async readData(deviceId: string, serviceUuid: string, characteristicUuid: string): Promise<ArrayBuffer | null> {
    try {
      const data = await bluetooth.readCharacteristicValue(deviceId, serviceUuid, characteristicUuid)
      hilog.info(0x0000, this.TAG, `Data read from device ${deviceId}`)
      return data
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to read data from device ${deviceId}: ${error}`)
      return null
    }
  }

  /**
   * 订阅特征值通知
   */
  async subscribeToNotifications(deviceId: string, serviceUuid: string, characteristicUuid: string): Promise<boolean> {
    try {
      await bluetooth.setCharacteristicNotification(deviceId, serviceUuid, characteristicUuid, true)
      hilog.info(0x0000, this.TAG, `Subscribed to notifications from device ${deviceId}`)
      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to subscribe to notifications from device ${deviceId}: ${error}`)
      return false
    }
  }

  /**
   * 添加蓝牙监听器
   */
  addListener(listener: BluetoothListener): void {
    this.listeners.add(listener)
  }

  /**
   * 移除蓝牙监听器
   */
  removeListener(listener: BluetoothListener): void {
    this.listeners.delete(listener)
  }

  /**
   * 获取已连接设备列表
   */
  getConnectedDevices(): BluetoothDevice[] {
    const devices: BluetoothDevice[] = []
    this.connectedDevices.forEach((device, id) => {
      devices.push({
        id: id,
        name: device.name || 'Unknown Device',
        address: device.address,
        rssi: device.rssi,
        deviceClass: device.deviceClass,
        bondState: device.bondState
      })
    })
    return devices
  }

  /**
   * 处理设备发现
   */
  private handleDeviceFound(device: bluetooth.BluetoothDevice): void {
    const bluetoothDevice: BluetoothDevice = {
      id: device.address,
      name: device.name || 'Unknown Device',
      address: device.address,
      rssi: device.rssi,
      deviceClass: device.deviceClass,
      bondState: device.bondState
    }

    // 通知所有监听器
    this.listeners.forEach(listener => {
      try {
        listener.onDeviceFound(bluetoothDevice)
      } catch (error) {
        hilog.error(0x0000, this.TAG, `Error in device found listener: ${error}`)
      }
    })
  }

  /**
   * 设置连接监听器
   */
  private setupConnectionListeners(deviceId: string): void {
    // 设置特征值变化监听器
    const characteristicChangeListener = new bluetooth.CharacteristicChangeListener((deviceId: string, characteristic: bluetooth.Characteristic) => {
      this.listeners.forEach(listener => {
        if (characteristic.value) {
          listener.onDataReceived(deviceId, characteristic.value)
        }
      })
    })

    bluetooth.on('bleCharacteristicChange', characteristicChangeListener)
  }

  /**
   * 释放资源
   */
  async release(): Promise<void> {
    // 停止扫描
    if (this.isScanning) {
      await this.stopDeviceDiscovery()
    }

    // 断开所有连接
    for (const deviceId of this.connectedDevices.keys()) {
      await this.disconnectDevice(deviceId)
    }

    // 清除监听器
    this.listeners.clear()

    hilog.info(0x0000, this.TAG, 'Bluetooth manager released')
  }
}
```

```typescript
// WiFiManager.ets
import wifi from '@ohos.wifi'
import hilog from '@ohos.hilog'

export interface WiFiNetwork {
  ssid: string
  bssid: string
  rssi: number
  frequency: number
  capabilities: string
  isConnected: boolean
  securityType: WiFiSecurityType
}

export interface WiFiListener {
  onNetworkFound(network: WiFiNetwork): void
  onNetworkConnected(ssid: string): void
  onNetworkDisconnected(ssid: string): void
  onWiFiStateChanged(state: WiFiState): void
  onScanResultsAvailable(): void
  onError(error: string): void
}

export enum WiFiSecurityType {
  OPEN = 'OPEN',
  WEP = 'WEP',
  PSK = 'PSK',
  EAP = 'EAP'
}

export enum WiFiState {
  DISABLED = 'DISABLED',
  ENABLING = 'ENABLING',
  ENABLED = 'ENABLED',
  DISABLING = 'DISABLING',
  ERROR = 'ERROR'
}

export class WiFiManager {
  private static instance: WiFiManager
  private listeners: Set<WiFiListener> = new Set()
  private isScanning: boolean = false
  private connectedNetwork: WiFiNetwork | null = null
  private readonly TAG = 'WiFiManager'

  static getInstance(): WiFiManager {
    if (!WiFiManager.instance) {
      WiFiManager.instance = new WiFiManager()
    }
    return WiFiManager.instance
  }

  /**
   * 检查WiFi是否可用
   */
  async isWiFiAvailable(): Promise<boolean> {
    try {
      return await wifi.isWifiActive()
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to check WiFi availability: ${error}`)
      return false
    }
  }

  /**
   * 开启WiFi
   */
  async enableWiFi(): Promise<boolean> {
    try {
      await wifi.enableWifi()
      hilog.info(0x0000, this.TAG, 'WiFi enabled')
      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to enable WiFi: ${error}`)
      return false
    }
  }

  /**
   * 关闭WiFi
   */
  async disableWiFi(): Promise<void> {
    try {
      await wifi.disableWifi()
      hilog.info(0x0000, this.TAG, 'WiFi disabled')
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to disable WiFi: ${error}`)
    }
  }

  /**
   * 开始网络扫描
   */
  async startNetworkScan(): Promise<boolean> {
    try {
      if (this.isScanning) {
        hilog.warn(0x0000, this.TAG, 'Network scan already in progress')
        return false
      }

      const isAvailable = await this.isWiFiAvailable()
      if (!isAvailable) {
        hilog.error(0x0000, this.TAG, 'WiFi is not available')
        return false
      }

      // 开始扫描
      await wifi.scan()
      this.isScanning = true

      // 设置扫描超时
      setTimeout(() => {
        if (this.isScanning) {
          this.isScanning = false
          this.listeners.forEach(listener => {
            listener.onScanResultsAvailable()
          })
        }
      }, 10000) // 10秒超时

      hilog.info(0x0000, this.TAG, 'Network scan started')
      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to start network scan: ${error}`)
      return false
    }
  }

  /**
   * 获取扫描结果
   */
  async getScanResults(): Promise<WiFiNetwork[]> {
    try {
      const scanResults = await wifi.getScanResults()
      const networks: WiFiNetwork[] = []

      scanResults.forEach(result => {
        const network: WiFiNetwork = {
          ssid: result.ssid,
          bssid: result.bssid,
          rssi: result.rssi,
          frequency: result.frequency,
          capabilities: result.capabilities,
          isConnected: false, // 需要单独检查连接状态
          securityType: this.parseSecurityType(result.capabilities)
        }
        networks.push(network)
      })

      // 检查当前连接状态
      const connectedInfo = await this.getConnectedNetworkInfo()
      if (connectedInfo) {
        const connectedNetwork = networks.find(n => n.ssid === connectedInfo.ssid)
        if (connectedNetwork) {
          connectedNetwork.isConnected = true
          this.connectedNetwork = connectedNetwork
        }
      }

      return networks
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to get scan results: ${error}`)
      return []
    }
  }

  /**
   * 连接WiFi网络
   */
  async connectToNetwork(ssid: string, password: string, securityType: WiFiSecurityType): Promise<boolean> {
    try {
      // 创建网络配置
      const networkConfig: wifi.WifiDeviceConfig = {
        ssid: ssid,
        preSharedKey: password,
        isHiddenSsid: false,
        keyMgmt: this.getKeyMgmt(securityType)
      }

      // 添加网络配置
      const networkId = await wifi.addDeviceConfig(networkConfig)

      // 连接网络
      await wifi.connectToDevice(networkId)

      hilog.info(0x0000, this.TAG, `Connected to WiFi network: ${ssid}`)

      // 更新连接状态
      await this.updateConnectionStatus()

      // 通知监听器
      this.listeners.forEach(listener => {
        listener.onNetworkConnected(ssid)
      })

      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to connect to WiFi network ${ssid}: ${error}`)
      return false
    }
  }

  /**
   * 断开WiFi连接
   */
  async disconnectFromNetwork(): Promise<void> {
    try {
      await wifi.disconnect()
      this.connectedNetwork = null

      hilog.info(0x0000, this.TAG, 'Disconnected from WiFi network')

      // 通知监听器
      if (this.connectedNetwork) {
        this.listeners.forEach(listener => {
          listener.onNetworkDisconnected(this.connectedNetwork!.ssid)
        })
      }
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to disconnect from WiFi network: ${error}`)
    }
  }

  /**
   * 获取已连接网络信息
   */
  async getConnectedNetworkInfo(): Promise<WiFiNetwork | null> {
    try {
      const linkedInfo = await wifi.getLinkedInfo()
      if (linkedInfo && linkedInfo.ssid) {
        return {
          ssid: linkedInfo.ssid,
          bssid: linkedInfo.bssid || '',
          rssi: linkedInfo.rssi || 0,
          frequency: linkedInfo.frequency || 0,
          capabilities: linkedInfo.networkId?.toString() || '',
          isConnected: true,
          securityType: WiFiSecurityType.PSK // 默认值，实际需要根据情况判断
        }
      }
      return null
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to get connected network info: ${error}`)
      return null
    }
  }

  /**
   * 获取WiFi信号强度
   */
  async getSignalStrength(): Promise<number> {
    try {
      const linkedInfo = await wifi.getLinkedInfo()
      return linkedInfo?.rssi || 0
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to get signal strength: ${error}`)
      return 0
    }
  }

  /**
   * 添加WiFi监听器
   */
  addListener(listener: WiFiListener): void {
    this.listeners.add(listener)
  }

  /**
   * 移除WiFi监听器
   */
  removeListener(listener: WiFiListener): void {
    this.listeners.delete(listener)
  }

  /**
   * 更新连接状态
   */
  private async updateConnectionStatus(): Promise<void> {
    const connectedInfo = await this.getConnectedNetworkInfo()
    if (connectedInfo) {
      this.connectedNetwork = connectedInfo
    }
  }

  /**
   * 解析安全类型
   */
  private parseSecurityType(capabilities: string): WiFiSecurityType {
    if (capabilities.includes('WPA') || capabilities.includes('PSK')) {
      return WiFiSecurityType.PSK
    } else if (capabilities.includes('WEP')) {
      return WiFiSecurityType.WEP
    } else if (capabilities.includes('EAP')) {
      return WiFiSecurityType.EAP
    } else {
      return WiFiSecurityType.OPEN
    }
  }

  /**
   * 获取密钥管理类型
   */
  private getKeyMgmt(securityType: WiFiSecurityType): string {
    switch (securityType) {

      case WiFiSecurityType.WEP:
        return 'WEP'
      case WiFiSecurityType.PSK:
        return 'WPA-PSK'
      case WiFiSecurityType.EAP:
        return 'IEEE8021X'
      default:
        return 'NONE'
    }
  }

    this.listeners.clear()
```
