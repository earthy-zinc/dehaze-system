```typescript
// PermissionManager.ets
import abilityAccessCtrl from '@ohos.abilityAccessCtrl'
import bundleManager from '@ohos.bundle.bundleManager'
import hilog from '@ohos.hilog'

export interface PermissionRequest {
  permissions: string[]
  rationale?: string
  onGranted?: (grantedPermissions: string[]) => void
  onDenied?: (deniedPermissions: string[]) => void
  onPermanentDenied?: (permanentDeniedPermissions: string[]) => void
}

export interface PermissionStatus {
  granted: boolean
  canRequest: boolean
  shouldShowRationale: boolean
  isPermanentlyDenied: boolean
}

export class PermissionManager {
  private static instance: PermissionManager
  private accessTokenId: number = -1
  private atManager: abilityAccessCtrl.AtManager
  private readonly TAG = 'PermissionManager'

  static getInstance(): PermissionManager {
    if (!PermissionManager.instance) {
      PermissionManager.instance = new PermissionManager()
    }
    return PermissionManager.instance
  }

  constructor() {
    this.atManager = abilityAccessCtrl.createAtManager()
    this.initializeAccessTokenId()
  }

  /**
   * 初始化访问令牌ID
   */
  private async initializeAccessTokenId(): Promise<void> {
    try {
      const bundleInfo = await bundleManager.getBundleInfoForSelf(bundleManager.BundleFlag.GET_BUNDLE_INFO_WITH_APPLICATION)
      this.accessTokenId = bundleInfo.appInfo.accessTokenId
      hilog.info(0x0000, this.TAG, `Access token ID initialized: ${this.accessTokenId}`)
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to initialize access token ID: ${error}`)
    }
  }

  /**
   * 检查权限状态
   * @param permission 权限名称
   */
  async checkPermissionStatus(permission: string): Promise<PermissionStatus> {
    try {
      const grantStatus = await this.atManager.checkAccessToken(this.accessTokenId, permission)
      const granted = grantStatus === abilityAccessCtrl.GrantStatus.PERMISSION_GRANTED

      // 检查是否可以请求权限
      const canRequest = await this.canRequestPermission(permission)

      // 检查是否需要显示说明
      const shouldShowRationale = await this.shouldShowRequestPermissionRationale(permission)

      // 判断是否为永久拒绝
      const isPermanentlyDenied = !granted && !canRequest && !shouldShowRationale

      return {
        granted,
        canRequest,
        shouldShowRationale,
        isPermanentlyDenied
      }
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to check permission status for ${permission}: ${error}`)
      return {
        granted: false,
        canRequest: false,
        shouldShowRationale: false,
        isPermanentlyDenied: true
      }
    }
  }

  /**
   * 检查是否可以请求权限
   * @param permission 权限名称
   */
  private async canRequestPermission(permission: string): Promise<boolean> {
    try {
      // 这里简化处理，实际需要根据系统API实现
      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to check if can request permission ${permission}: ${error}`)
      return false
    }
  }

  /**
   * 检查是否应该显示权限说明
   * @param permission 权限名称
   */
  private async shouldShowRequestPermissionRationale(permission: string): Promise<boolean> {
    try {
      // 这里简化处理，实际需要根据系统API实现
      return false
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to check if should show rationale for ${permission}: ${error}`)
      return false
    }
  }

  /**
   * 请求权限
   * @param permissionRequest 权限请求配置
   */
  async requestPermissions(permissionRequest: PermissionRequest): Promise<void> {
    try {
      const { permissions, rationale, onGranted, onDenied, onPermanentDenied } = permissionRequest

      // 检查权限状态
      const permissionStatuses: Map<string, PermissionStatus> = new Map()
      const needRequestPermissions: string[] = []

      for (const permission of permissions) {
        const status = await this.checkPermissionStatus(permission)
        permissionStatuses.set(permission, status)

        if (!status.granted) {
          needRequestPermissions.push(permission)
        }
      }

      // 处理已授予的权限
      const grantedPermissions: string[] = []
      const deniedPermissions: string[] = []
      const permanentDeniedPermissions: string[] = []

      permissionStatuses.forEach((status, permission) => {
        if (status.granted) {
          grantedPermissions.push(permission)
        } else if (status.isPermanentlyDenied) {
          permanentDeniedPermissions.push(permission)
        } else {
          deniedPermissions.push(permission)
        }
      })

      // 如果没有需要请求的权限，直接返回结果
      if (needRequestPermissions.length === 0) {
        if (grantedPermissions.length > 0 && onGranted) {
          onGranted(grantedPermissions)
        }
        return
      }

      // 处理永久拒绝的权限
      if (permanentDeniedPermissions.length > 0 && onPermanentDenied) {
        onPermanentDenied(permanentDeniedPermissions)
      }

      // 请求权限
      if (needRequestPermissions.length > 0) {
        await this.performPermissionRequest(needRequestPermissions, (grantResults: boolean[]) => {
          needRequestPermissions.forEach((permission, index) => {
            if (grantResults[index]) {
              grantedPermissions.push(permission)
            } else {
              deniedPermissions.push(permission)
            }
          })

          // 回调结果
          if (grantedPermissions.length > 0 && onGranted) {
            onGranted(grantedPermissions)
          }
          if (deniedPermissions.length > 0 && onDenied) {
            onDenied(deniedPermissions)
          }
        })
      }
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to request permissions: ${error}`)
      if (permissionRequest.onDenied) {
        permissionRequest.onDenied(permissionRequest.permissions)
      }
    }
  }

  /**
   * 执行权限请求
   * @param permissions 权限列表
   * @param callback 结果回调
   */
  private async performPermissionRequest(
    permissions: string[],
    callback: (grantResults: boolean[]) => void
  ): Promise<void> {
    try {
      const grantResults = await this.atManager.requestPermissionsFromUser(
        this.accessTokenId,
        permissions
      )
      callback(grantResults.grantStatus.map(status => status === abilityAccessCtrl.GrantStatus.PERMISSION_GRANTED))
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to perform permission request: ${error}`)
      callback(permissions.map(() => false))
    }
  }

  /**
   * 批量检查权限
   * @param permissions 权限列表
   */
  async checkMultiplePermissions(permissions: string[]): Promise<Map<string, PermissionStatus>> {
    const results: Map<string, PermissionStatus> = new Map()

    for (const permission of permissions) {
      const status = await this.checkPermissionStatus(permission)
      results.set(permission, status)
    }

    return results
  }

  /**
   * 检查是否所有权限都已授予
   * @param permissions 权限列表
   */
  async areAllPermissionsGranted(permissions: string[]): Promise<boolean> {
    const statuses = await this.checkMultiplePermissions(permissions)
    for (const status of statuses.values()) {
      if (!status.granted) {
        return false
      }
    }
    return true
  }

  /**
   * 获取权限描述
   * @param permission 权限名称
   */
  getPermissionDescription(permission: string): string {
    const descriptions: Map<string, string> = new Map([
      ['ohos.permission.CAMERA', '访问摄像头'],
      ['ohos.permission.MICROPHONE', '访问麦克风'],
      ['ohos.permission.READ_CONTACTS', '读取联系人'],
      ['ohos.permission.WRITE_CONTACTS', '写入联系人'],
      ['ohos.permission.READ_CALENDAR', '读取日历'],
      ['ohos.permission.WRITE_CALENDAR', '写入日历'],
      ['ohos.permission.ACTIVITY_MOTION', '访问传感器'],
      ['ohos.permission.APPROXIMATELY_LOCATION', '获取大致位置'],
      ['ohos.permission.LOCATION', '获取精确位置'],
      ['ohos.permission.LOCATION_IN_BACKGROUND', '后台访问位置'],
      ['ohos.permission.READ_MEDIA', '读取媒体文件'],
      ['ohos.permission.WRITE_MEDIA', '写入媒体文件'],
      ['ohos.permission.READ_DOWNLOAD_DIRECTORY', '读取下载目录'],

    ])

    return descriptions.get(permission) || '未知权限'
  }

  /**
   * 获取权限申请理由
   * @param permission 权限名称
   */
  getPermissionRationale(permission: string): string {
```

### 15.2.1 加密工具类

```typescript
// CryptoUtils.ets
import cryptoFramework from '@ohos.security.cryptoFramework'
import hilog from '@ohos.hilog'

export interface EncryptionResult {
  ciphertext: Uint8Array
  iv: Uint8Array
  tag?: Uint8Array
}

export interface KeyPair {
  publicKey: Uint8Array
  privateKey: Uint8Array
}

export class CryptoUtils {
  private static instance: CryptoUtils
  private readonly TAG = 'CryptoUtils'

  static getInstance(): CryptoUtils {
    if (!CryptoUtils.instance) {
      CryptoUtils.instance = new CryptoUtils()
    }
    return CryptoUtils.instance
  }

  /**
   * AES-GCM 加密
   * @param plaintext 明文
   * @param key 加密密钥（32字节）
   * @param iv 初始化向量（12字节）
   */
  async aesGcmEncrypt(plaintext: string, key: Uint8Array, iv: Uint8Array): Promise<EncryptionResult> {
    try {
      // 创建AES GCM Cipher
      const cipher = cryptoFramework.createCipher('AES256|GCM|PKCS7')

      // 创建密钥
      const keyBlob: cryptoFramework.DataBlob = { data: key }
      const symKey = await cryptoFramework.createSymKey(keyBlob)

      // 初始化加密器
      const params: cryptoFramework.GcmParamsSpec = {
        iv: { data: iv },
        aad: null,
        authTag: null
      }

      await cipher.init(cryptoFramework.CryptoMode.ENCRYPT_MODE, symKey, params)

      // 加密数据
      const input: cryptoFramework.DataBlob = { data: new Uint8Array(Buffer.from(plaintext, 'utf8')) }
      const output = await cipher.doFinal(input)

      // 获取认证标签
      const tag = await cipher.getGcmAuthTag()

      return {
        ciphertext: output.data,
        iv: iv,
        tag: tag.data
      }
    } catch (error) {
      hilog.error(0x0000, this.TAG, `AES-GCM encryption failed: ${error}`)
      throw error
    }
  }

  /**
   * AES-GCM 解密
   * @param encryptionResult 加密结果
   * @param key 解密密钥（32字节）
   */
  async aesGcmDecrypt(encryptionResult: EncryptionResult, key: Uint8Array): Promise<string> {
    try {
      // 创建AES GCM Cipher
      const cipher = cryptoFramework.createCipher('AES256|GCM|PKCS7')

      // 创建密钥
      const keyBlob: cryptoFramework.DataBlob = { data: key }
      const symKey = await cryptoFramework.createSymKey(keyBlob)

      // 初始化解密器
      const params: cryptoFramework.GcmParamsSpec = {
        iv: { data: encryptionResult.iv },
        aad: null,
        authTag: encryptionResult.tag ? { data: encryptionResult.tag } : null
      }

      await cipher.init(cryptoFramework.CryptoMode.DECRYPT_MODE, symKey, params)

      // 解密数据
      const input: cryptoFramework.DataBlob = { data: encryptionResult.ciphertext }
      const output = await cipher.doFinal(input)

      // 转换为字符串
      return Buffer.from(output.data).toString('utf8')
    } catch (error) {
      hilog.error(0x0000, this.TAG, `AES-GCM decryption failed: ${error}`)
      throw error
    }
  }

  /**
   * 生成随机密钥
   * @param length 密钥长度（字节）
   */
  async generateRandomKey(length: number): Promise<Uint8Array> {
    try {
      const rand = cryptoFramework.createRandom()
      const result = await rand.generateRandom(length)
      return result.data
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to generate random key: ${error}`)
      throw error
    }
  }

  /**
   * 生成RSA密钥对
   * @param keySize 密钥长度（位）
   */
  async generateRSAKeyPair(keySize: number = 2048): Promise<KeyPair> {
    try {
      // 创建RSA密钥生成器
      const generator = cryptoFramework.createAsyKeyGenerator('RSA2048|PKCS1')

      // 生成密钥对
      const keyPair = await generator.generateKeyPair()

      // 获取公钥和私钥
      const pubKey = await keyPair.pubKey.getEncoded()
      const priKey = await keyPair.priKey.getEncoded()

      return {
        publicKey: pubKey.data,
        privateKey: priKey.data
      }
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to generate RSA key pair: ${error}`)
      throw error
    }
  }

  /**
   * RSA公钥加密
   * @param plaintext 明文
   * @param publicKey 公钥
   */
  async rsaPublicEncrypt(plaintext: string, publicKey: Uint8Array): Promise<Uint8Array> {
    try {
      // 创建RSA加密器
      const cipher = cryptoFramework.createCipher('RSA2048|PKCS1')

      // 创建公钥对象
      const keyBlob: cryptoFramework.DataBlob = { data: publicKey }
      const pubKey = await cryptoFramework.createAsyKeyGenerator('RSA2048|PKCS1').convertKey(keyBlob, null)

      // 初始化加密器
      await cipher.init(cryptoFramework.CryptoMode.ENCRYPT_MODE, pubKey, null)

      // 加密数据
      const input: cryptoFramework.DataBlob = { data: new Uint8Array(Buffer.from(plaintext, 'utf8')) }
      const output = await cipher.doFinal(input)

      return output.data
    } catch (error) {
      hilog.error(0x0000, this.TAG, `RSA public encryption failed: ${error}`)
      throw error
    }
  }

  /**
   * RSA私钥解密
   * @param ciphertext 密文
   * @param privateKey 私钥
   */
  async rsaPrivateDecrypt(ciphertext: Uint8Array, privateKey: Uint8Array): Promise<string> {
    try {
      // 创建RSA解密器
      const cipher = cryptoFramework.createCipher('RSA2048|PKCS1')

      // 创建私钥对象
      const keyBlob: cryptoFramework.DataBlob = { data: privateKey }
      const priKey = await cryptoFramework.createAsyKeyGenerator('RSA2048|PKCS1').convertKey(null, keyBlob)

      // 初始化解密器
      await cipher.init(cryptoFramework.CryptoMode.DECRYPT_MODE, priKey, null)

      // 解密数据
      const input: cryptoFramework.DataBlob = { data: ciphertext }
      const output = await cipher.doFinal(input)

      // 转换为字符串
      return Buffer.from(output.data).toString('utf8')
    } catch (error) {
      hilog.error(0x0000, this.TAG, `RSA private decryption failed: ${error}`)
      throw error
    }
  }

  /**
   * 计算哈希值
   * @param data 原始数据
   * @param algorithm 哈希算法
   */
  async calculateHash(data: string, algorithm: string = 'SHA256'): Promise<Uint8Array> {
    try {
      // 创建MD计算器
      const md = cryptoFramework.createMd(algorithm)

      // 计算哈希值
      const input: cryptoFramework.DataBlob = { data: new Uint8Array(Buffer.from(data, 'utf8')) }
      const output = await md.computeHash(input)

      return output.data
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Hash calculation failed: ${error}`)
      throw error
    }
  }

  /**
   * HMAC计算
   * @param data 原始数据
   * @param key 密钥
   * @param algorithm HMAC算法
   */
  async calculateHMAC(data: string, key: Uint8Array, algorithm: string = 'HMACSHA256'): Promise<Uint8Array> {
    try {
      // 创建HMAC计算器
      const hmac = cryptoFramework.createHmac(algorithm)

      // 创建密钥
      const keyBlob: cryptoFramework.DataBlob = { data: key }
      const symKey = await cryptoFramework.createSymKey(keyBlob)

      // 计算HMAC值
      const input: cryptoFramework.DataBlob = { data: new Uint8Array(Buffer.from(data, 'utf8')) }
      const output = await hmac.computeMac(input, symKey)

      return output.data
    } catch (error) {
      hilog.error(0x0000, this.TAG, `HMAC calculation failed: ${error}`)
      throw error
    }
  }

  /**
   * 字节数组转十六进制字符串
   * @param bytes 字节数组
   */
  bytesToHex(bytes: Uint8Array): string {
    return Array.from(bytes)
      .map(byte => byte.toString(16).padStart(2, '0'))
      .join('')
  }

  /**
   * 十六进制字符串转字节数组
   * @param hex 十六进制字符串
   */
  hexToBytes(hex: string): Uint8Array {
    const result = new Uint8Array(hex.length / 2)
    for (let i = 0; i < hex.length; i += 2) {
      result[i / 2] = parseInt(hex.substr(i, 2), 16)
    }
    return result
  }
```

```typescript
      hilog.error(0x0000, this.TAG, `Failed to initialize secure storage: ${error}`)
    }
  }

  /**
   * 初始化加密密钥
   */
  private async initializeEncryptionKey(): Promise<void> {
    try {
      // 尝试从存储中获取密钥
      const storedKey = this.storage?.getSync('encryption_key', '')

      if (storedKey) {
        // 使用存储的密钥
        this.encryptionKey = this.cryptoUtils.hexToBytes(storedKey)
      } else {
        // 生成新的密钥并存储
        this.encryptionKey = await this.cryptoUtils.generateRandomKey(32)
        const keyHex = this.cryptoUtils.bytesToHex(this.encryptionKey)
        this.storage?.putSync('encryption_key', keyHex)
        this.storage?.flushSync()
      }
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to initialize encryption key: ${error}`)
      // 使用默认密钥（实际应用中应该有更安全的处理方式）
      this.encryptionKey = await this.cryptoUtils.generateRandomKey(32)
    }
  }

  /**
   * 安全存储数据
   * @param key 数据键
   * @param value 数据值
   */
  async storeSecureData(key: string, value: string): Promise<boolean> {
    try {
      if (!this.storage) {
        throw new Error('Storage not initialized')
      }

      // 生成随机IV
      const iv = await this.cryptoUtils.generateRandomKey(12)

      // 加密数据
      const encryptionResult = await this.cryptoUtils.aesGcmEncrypt(value, this.encryptionKey, iv)

      // 创建存储项
      const dataItem: SecureDataItem = {
        key: key,
        value: this.cryptoUtils.bytesToHex(encryptionResult.ciphertext),
        timestamp: Date.now(),
        iv: this.cryptoUtils.bytesToHex(encryptionResult.iv),
        tag: encryptionResult.tag ? this.cryptoUtils.bytesToHex(encryptionResult.tag) : undefined
      }

      // 序列化并存储
      const serializedData = JSON.stringify(dataItem)
      this.storage.putSync(key, serializedData)
      this.storage.flushSync()

      hilog.info(0x0000, this.TAG, `Secure data stored for key: ${key}`)
      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to store secure data for key ${key}: ${error}`)
      return false
    }

   * 安全读取数据
   * @param key 数据键
   */
  async retrieveSecureData(key: string): Promise<string | null> {
    try {
      if (!this.storage) {
        throw new Error('Storage not initialized')
      }

      // 获取存储的数据
      const serializedData = this.storage.getSync(key, '')
      if (!serializedData) {
        return null

      // 反序列化数据
      const dataItem: SecureDataItem = JSON.parse(serializedData)

      // 验证数据完整性
      if (!this.validateDataItem(dataItem)) {
        hilog.error(0x0000, this.TAG, `Invalid data item for key: ${key}`)
        return null
      }

      // 重构加密结果
      const encryptionResult: EncryptionResult = {
        ciphertext: this.cryptoUtils.hexToBytes(dataItem.value),
        iv: this.cryptoUtils.hexToBytes(dataItem.iv || ''),
        tag: dataItem.tag ? this.cryptoUtils.hexToBytes(dataItem.tag) : undefined
      }

      // 解密数据
      const decryptedValue = await this.cryptoUtils.aesGcmDecrypt(encryptionResult, this.encryptionKey)

      hilog.info(0x0000, this.TAG, `Secure data retrieved for key: ${key}`)
      return decryptedValue
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to retrieve secure data for key ${key}: ${error}`)
      return null
    }
  }

  /**
   * 删除安全数据

      if (!this.storage) {
        throw new Error('Storage not initialized')
      }

      this.storage.deleteSync(key)
      this.storage.flushSync()

      hilog.info(0x0000, this.TAG, `Secure data deleted for key: ${key}`)
      return true
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to delete secure data for key ${key}: ${error}`)
      return false
    }

   * @param key 数据键
   */

    try {
      if (!this.storage) {
        return false
      }

      const serializedData = this.storage.getSync(key, '')
      return serializedData !== ''
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to check secure data existence for key ${key}: ${error}`)
      return false
    }

  /**
   * 获取所有存储的键
   */
  async getAllSecureKeys(): Promise<string[]> {

      if (!this.storage) {
        return []
      }

      const keys: string[] = []
      this.storage.forEachKey((key) => {
        keys.push(key)
      })

      // 排除系统键（如加密密钥）
      return keys.filter(key => key !== 'encryption_key')
    } catch (error) {

      return []
    }

        this.storage.deleteSync(key)

      dataItem.value &&

      }
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to get storage stats: ${error}`)
      return { itemCount: 0, totalSize: 0 }
    }
  }
```

```typescript
// SecureNetworkManager.ets
import http from '@ohos.net.http'
import { CryptoUtils } from './CryptoUtils'

export interface SecureRequestOptions {
  url: string
  method?: http.RequestMethod
  headers?: Record<string, string>
  data?: Object | string | ArrayBuffer
  timeout?: number
  enableSSLVerification?: boolean
  customCA?: string
}

export interface SecureResponse {
  statusCode: number
  headers: Record<string, string>
  data: Object | string | ArrayBuffer
  certificateInfo?: CertificateInfo
}

export interface CertificateInfo {
  issuer: string
  subject: string
  validFrom: Date
  validTo: Date
  fingerprint: string
}

export class SecureNetworkManager {
  private static instance: SecureNetworkManager
  private cryptoUtils: CryptoUtils
  private sslConfig: SSLConfiguration
  private readonly TAG = 'SecureNetworkManager'

  static getInstance(): SecureNetworkManager {
    if (!SecureNetworkManager.instance) {
      SecureNetworkManager.instance = new SecureNetworkManager()
    }
    return SecureNetworkManager.instance
  }

  constructor() {
    this.cryptoUtils = CryptoUtils.getInstance()
    this.sslConfig = this.createDefaultSSLConfig()
  }

  /**
   * 创建默认SSL配置
   */
  private createDefaultSSLConfig(): SSLConfiguration {
    return {
      enableSSLVerification: true,
      cipherSuites: [
        'TLS_AES_256_GCM_SHA384',
        'TLS_CHACHA20_POLY1305_SHA256',
        'TLS_AES_128_GCM_SHA256'
      ],
      protocols: ['TLSv1.3', 'TLSv1.2'],
      trustAnchors: [],
      certificateChain: []
    }
  }

  /**
   * 执行安全的HTTP请求
   * @param options 请求选项
   */
  async secureRequest(options: SecureRequestOptions): Promise<SecureResponse> {
    try {
      // 创建HTTP请求对象
      const httpRequest = http.createHttp()

      // 配置SSL/TLS
      await this.configureSSL(httpRequest, options)

      // 设置请求头
      const headers = this.buildSecureHeaders(options.headers || {})

      // 执行请求
      const response = await httpRequest.request(
        options.url,
        {
          method: options.method || http.RequestMethod.GET,
          header: headers,
          extraData: options.data,
          expectDataType: http.HttpDataType.STRING,
          usingCache: false,
          priority: 1,
          connectTimeout: options.timeout || 10000,
          readTimeout: options.timeout || 10000,
          usingProtocol: http.HttpProtocol.HTTP1_1
        }
      )

      // 解析响应
      const secureResponse: SecureResponse = {
        statusCode: response.responseCode,
        headers: response.header as Record<string, string>,
        data: response.result as Object | string | ArrayBuffer
      }

      // 如果启用了证书验证，提取证书信息
      if (options.enableSSLVerification !== false) {
        secureResponse.certificateInfo = await this.extractCertificateInfo(httpRequest)
      }

      // 验证响应完整性
      if (this.hasResponseIntegrityCheck(headers)) {
        const isValid = await this.verifyResponseIntegrity(secureResponse, headers)
        if (!isValid) {
          throw new Error('Response integrity verification failed')
        }
      }

      hilog.info(0x0000, this.TAG, `Secure request completed: ${options.url}`)
      return secureResponse
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Secure request failed for ${options.url}: ${error}`)
      throw error
    }
  }

  /**
   * 配置SSL/TLS
   * @param httpRequest HTTP请求对象
   * @param options 请求选项
   */
  private async configureSSL(httpRequest: http.HttpRequest, options: SecureRequestOptions): Promise<void> {
    try {
      // 创建SSL配置
      const sslConfig: http.SSLConfiguration = {
        caPath: options.customCA || '',
```

```typescript
import { CryptoUtils } from './CryptoUtils'

export interface SignatureData {
  timestamp: number
  nonce: string
  method: string
  path: string
  body: string
  headers: Record<string, string>
}

export interface SignatureResult {
  signature: string
  timestamp: number
  nonce: string
  publicKey?: string
}

export class RequestSignatureManager {
  private static instance: RequestSignatureManager
  private cryptoUtils: CryptoUtils
  private privateKey: Uint8Array
  private publicKey: Uint8Array
  private readonly TAG = 'RequestSignatureManager'

  static getInstance(): RequestSignatureManager {
    if (!RequestSignatureManager.instance) {
      RequestSignatureManager.instance = new RequestSignatureManager()
    }
    return RequestSignatureManager.instance
  }

  constructor() {
    this.cryptoUtils = CryptoUtils.getInstance()
    this.initializeKeyPair()
  }

  /**
   * 初始化密钥对
   */
  private async initializeKeyPair(): Promise<void> {
    try {
      const keyPair = await this.cryptoUtils.generateRSAKeyPair(2048)
      this.privateKey = keyPair.privateKey
      this.publicKey = keyPair.publicKey
    } catch (error) {
      console.error('Failed to initialize key pair:', error)
    }
  }

  /**
   * 生成请求签名
   * @param method HTTP方法
   * @param path 请求路径
   * @param body 请求体
   * @param headers 请求头
   * @param timestamp 时间戳
   */
  async generateRequestSignature(
    method: string,
    path: string,
    body: string,
    headers: Record<string, string>,
    timestamp?: number
  ): Promise<SignatureResult> {
    try {
      const ts = timestamp || Date.now()
      const nonce = this.generateNonce()

      // 构建签名数据
      const signatureData: SignatureData = {
        timestamp: ts,
        nonce: nonce,
        method: method.toUpperCase(),
        path: path,
        body: body || '',
        headers: this.normalizeHeaders(headers)
      }

      // 生成待签名字符串
      const stringToSign = this.buildStringToSign(signatureData)

      // 使用私钥签名
      const signature = await this.cryptoUtils.rsaPublicEncrypt(stringToSign, this.privateKey)

      // 编码签名
      const encodedSignature = this.cryptoUtils.bytesToHex(signature)

      return {
        signature: encodedSignature,
        timestamp: ts,
        nonce: nonce,
        publicKey: this.cryptoUtils.bytesToHex(this.publicKey)
      }
    } catch (error) {
      console.error('Failed to generate request signature:', error)
      throw error
    }
  }

  /**
   * 验证请求签名
   * @param signature 签名
   * @param method HTTP方法
   * @param path 请求路径
   * @param body 请求体
   * @param headers 请求头
   * @param timestamp 时间戳
   * @param nonce 随机数
   * @param publicKey 公钥
   */
  async verifyRequestSignature(
    signature: string,
    method: string,
    path: string,
    body: string,
    headers: Record<string, string>,
    timestamp: number,
    nonce: string,
    publicKey: string
  ): Promise<boolean> {
    try {
      // 验证时间戳（防止重放攻击）
      const now = Date.now()
      const timeWindow = 300000 // 5分钟时间窗口
      if (Math.abs(now - timestamp) > timeWindow) {
        console.error('Signature timestamp out of range')
        return false
      }

      // 检查nonce是否已使用（防止重放攻击）
      if (this.isNonceUsed(nonce)) {
        console.error('Nonce already used')
        return false
      }

      // 记录nonce使用
      this.recordNonceUsage(nonce, timestamp)

      // 构建签名数据
      const signatureData: SignatureData = {
        timestamp: timestamp,
        nonce: nonce,
        method: method.toUpperCase(),
        path: path,
        body: body || '',
        headers: this.normalizeHeaders(headers)
      }

      // 生成待签名字符串
      const stringToSign = this.buildStringToSign(signatureData)

      // 解码签名
      const decodedSignature = this.cryptoUtils.hexToBytes(signature)

      // 解码公钥
      const decodedPublicKey = this.cryptoUtils.hexToBytes(publicKey)

      // 使用公钥验证签名
      const decryptedData = await this.cryptoUtils.rsaPrivateDecrypt(decodedSignature, decodedPublicKey)

      // 比较解密后的数据与待签名字符串
      const isValid = decryptedData === stringToSign

      if (isValid) {
        hilog.info(0x0000, this.TAG, 'Request signature verification successful')
      } else {
        hilog.error(0x0000, this.TAG, 'Request signature verification failed')
      }

      return isValid
    } catch (error) {
      hilog.error(0x0000, this.TAG, `Failed to verify request signature: ${error}`)
      return false
    }
  }

  /**
   * 构建待签名字符串
   * @param data 签名数据
   */
  private buildStringToSign(data: SignatureData): string {
    // 按照特定顺序构建待签名字符串
    const parts = [
      data.timestamp.toString(),
      data.nonce,
      data.method,
      data.path,
      this.calculateBodyHash(data.body),
      this.buildHeaderString(data.headers)
    ]

    return parts.join('\n')
  }

  /**
   * 计算请求体哈希
   * @param body 请求体
   */
  private async calculateBodyHash(body: string): Promise<string> {
    if (!body) {
      return 'd41d8cd98f00b204e9800998ecf8427e' // MD5 of empty string
    }

    const hash = await this.cryptoUtils.calculateHash(body, 'SHA256')
    return this.cryptoUtils.bytesToHex(hash)
  }

  /**
   * 构建头部字符串
   * @param headers 请求头
   */
  private buildHeaderString(headers: Record<string, string>): string {
    // 只包含特定的头部字段，并按字母顺序排序
    const includedHeaders = [
      'content-type',
      'accept',
      'user-agent',
      'x-custom-header'
    ]

    const filteredHeaders: Record<string, string> = {}
    Object.keys(headers)
      .filter(key => includedHeaders.includes(key.toLowerCase()))
      .sort()
      .forEach(key => {
        filteredHeaders[key.toLowerCase()] = headers[key]
      })

    return Object.keys(filteredHeaders)
      .map(key => `${key}:${filteredHeaders[key]}`)
      .join('\n')
  }

  /**
   * 标准化请求头
   * @param headers 原始请求头
   */
  private normalizeHeaders(headers: Record<string, string>): Record<string, string> {
    const normalized: Record<string, string> = {}

    Object.keys(headers).forEach(key => {
      normalized[key.toLowerCase()] = headers[key]
    })

    return normalized
  }

  /**
   * 生成随机数
   */
  private generateNonce(): string {
    return Math.random().toString(36).substring(2) + Date.now().toString(36)
  }

  /**
   * 检查nonce是否已使用
   * @param nonce 随机数
   */
  private isNonceUsed(nonce: string): boolean {
    // 这里应该使用持久化存储来记录已使用的nonce
    // 简化处理，使用内存存储
    const usedNonces = this.getUsedNonces()
    return usedNonces.has(nonce)
  }

  /**
   * 记录nonce使用
   * @param nonce 随机数
   * @param timestamp 时间戳
   */
  private recordNonceUsage(nonce: string, timestamp: number): void {
    // 这里应该使用持久化存储来记录已使用的nonce
    // 简化处理，使用内存存储
    const usedNonces = this.getUsedNonces()
    usedNonces.set(nonce, timestamp)

    // 清理过期的nonce记录
    this.cleanupExpiredNonces()
  }

  /**
   * 获取已使用的nonce集合
   */
  private getUsedNonces(): Map<string, number> {
    // 这里应该使用持久化存储
    // 简化处理，使用静态变量
    if (!RequestSignatureManager.usedNonces) {
      RequestSignatureManager.usedNonces = new Map()
    }
    return RequestSignatureManager.usedNonces
  }

  /**
   * 清理过期的nonce记录
   */
  private cleanupExpiredNonces(): void {
    const usedNonces = this.getUsedNonces()
    const now = Date.now()
    const timeWindow = 300000 // 5分钟

    for (const [nonce, timestamp] of usedNonces.entries()) {
      if (now - timestamp > timeWindow) {
        usedNonces.delete(nonce)
      }
    }
  }

  /**
   * 设置自定义密钥对
   * @param privateKey 私钥
   * @param publicKey 公钥
   */
  setKeyPair(privateKey: Uint8Array, publicKey: Uint8Array): void {
    this.privateKey = privateKey
    this.publicKey = publicKey
  }

  /**
   * 获取公钥
   */
  getPublicKey(): string {
    return this.cryptoUtils.bytesToHex(this.publicKey)
  }

  private static usedNonces: Map<string, number> = new Map()
}
```

```typescript
  private async validatePassword(password: string): Promise<boolean> {
    // 这里应该实现实际的密码验证逻辑
    // 例如从安全存储中获取哈希值进行比对
    const storedPassword = await this.secureStorage.retrieveSecureData('user_password')
    if (!storedPassword) {
      return false
    }

    const inputHash = await this.cryptoUtils.calculateHash(password, 'SHA256')
    const storedHash = this.cryptoUtils.hexToBytes(storedPassword)

    return this.compareArrays(inputHash, storedHash)
  }

  private async startBiometricAuth() {
    try {
      // 检查生物识别权限
      const hasPermission = await this.checkBiometricPermission()
      if (!hasPermission) {
        this.authResult = {
          success: false,
          errorCode: 403,
          errorMessage: '需要生物识别权限'
        }
        return
      }

      // 生成挑战值
      const challenge = await this.cryptoUtils.generateRandomKey(32)

      // 执行生物识别认证
      const result = await this.biometricManager.authenticate(BiometricType.FINGERPRINT, challenge)

      if (result.success) {
        this.authResult = {
          success: true,
          userId: 'user123',
          token: await this.generateAuthToken(),
          factors: [AuthFactorType.PASSWORD, AuthFactorType.BIOMETRIC]
        }
      } else {
        this.authResult = {
          success: false,
          errorCode: result.errorCode || 401,
          errorMessage: result.errorMessage || '生物识别失败'
        }
      }
    } catch (error) {
      this.authResult = {
        success: false,
        errorCode: 500,
        errorMessage: '生物识别错误'
      }
    }
  }

  private compareArrays(arr1: Uint8Array, arr2: Uint8Array): boolean {
    if (arr1.length !== arr2.length) {
      return false
    }
    for (let i = 0; i < arr1.length; i++) {
      if (arr1[i] !== arr2[i]) {
        return false
      }
    }
    return true
  }
```
