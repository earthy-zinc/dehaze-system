# 第8章：数据管理与状态

在现代应用开发中，数据管理和状态控制是构建复杂应用的核心。鸿蒙提供了完整的状态管理机制，从简单的组件内状态到复杂的跨组件数据流，为开发者提供了灵活而强大的工具。

## 8.1 @State状态管理

### 8.1.1 @State基础概念

@State是鸿蒙中最基础的状态管理装饰器，用于管理组件的内部状态。当@State装饰的变量发生变化时，组件会自动重新渲染。

```typescript
@Entry
@Component
struct StateBasicsExample {
  @State count: number = 0
  @State message: string = 'Hello HarmonyOS'
  @State isVisible: boolean = true
  @State user: UserInfo = { name: '张三', age: 25 }

  build() {
    Column() {
      // 计数器示例
      Text('计数器：' + this.count)
        .fontSize(18)
        .margin(10)

      Row() {
        Button('增加')
          .onClick(() => {
            this.count++
          })
          .margin(5)

        Button('减少')
          .onClick(() => {
            this.count--
          })
          .margin(5)

        Button('重置')
          .onClick(() => {
            this.count = 0
          })
          .margin(5)
      }
      .margin(10)

      Divider()

      // 文本状态示例
      Text(this.message)
        .fontSize(16)
        .margin(10)

      TextInput({ placeholder: '输入消息' })
        .width('80%')
        .margin(10)
        .onChange((value: string) => {
          this.message = value
        })

      Divider()

      // 显示/隐藏示例
      if (this.isVisible) {
        Text('这是可切换的内容')
          .fontSize(16)
          .backgroundColor(Color.Blue)
          .fontColor(Color.White)
          .padding(10)
          .borderRadius(5)
          .margin(10)
      }

      Button(this.isVisible ? '隐藏' : '显示')
        .onClick(() => {
          this.isVisible = !this.isVisible
        })
        .margin(10)

      Divider()

      // 对象状态示例
      Text('用户信息：' + this.user.name + '，' + this.user.age + '岁')
        .fontSize(16)
        .margin(10)

      Row() {
        TextInput({ placeholder: '姓名' })
          .width(100)
          .margin(5)
          .onChange((value: string) => {
            this.user.name = value
          })

        TextInput({ placeholder: '年龄' })
          .width(80)
          .margin(5)
          .onChange((value: string) => {
            this.user.age = parseInt(value) || 0
          })
      }
      .margin(10)
    }
    .padding(20)
  }
}

// 用户信息接口定义
interface UserInfo {
  name: string
  age: number
}
```

### 8.1.2 @State响应式原理

@State的响应式机制基于数据劫持和依赖收集。当组件访问@State装饰的变量时，会建立依赖关系；当变量值发生变化时，会触发相关的UI更新。

```typescript
@Component
struct ReactiveExample {
  @State data: Array<string> = ['苹果', '香蕉', '橙子']
  @State selectedFruit: string = ''

  build() {
    Column() {
      Text('选择你喜欢的水果')
        .fontSize(18)
        .fontWeight(FontWeight.Bold)
        .margin(20)

      // 水果列表
      ForEach(this.data, (fruit: string, index: number) => {
        Row() {
          Text(fruit)
            .fontSize(16)
            .layoutWeight(1)

          Button('选择')
            .onClick(() => {
              this.selectedFruit = fruit
            })
        }
        .width('100%')
        .padding(10)
        .backgroundColor(this.selectedFruit === fruit ? Color.Green : Color.Gray)
        .margin(5)
        .borderRadius(5)
      })

      // 显示选择结果
      if (this.selectedFruit) {
        Text('你选择了：' + this.selectedFruit)
          .fontSize(16)
          .fontColor(Color.Blue)
          .margin(20)
      }

      // 添加新水果
      Row() {
        TextInput({ placeholder: '输入新水果' })
          .width(150)
          .margin(5)
          .id('fruitInput')

        Button('添加')
          .onClick(() => {
            const input = this.getUIContext().getRouter().getStackByName('fruitInput')[0]
            if (input && input.text) {
              this.data.push(input.text)
              // 清空输入框
              input.text = ''
            }
          })
          .margin(5)
      }
      .margin(20)
    }
    .padding(20)
  }
}
```

### 8.1.3 @State最佳实践

#### 1. 避免直接修改数组
```typescript
@Entry
@Component
struct ArrayStateExample {
  @State todos: Array<Todo> = [
    { id: 1, text: '学习鸿蒙开发', completed: false },
    { id: 2, text: '完成项目', completed: false }
  ]

  build() {
    Column() {
      // 添加新任务
      Row() {
        TextInput({ placeholder: '输入新任务' })
          .width(200)
          .margin(5)
          .id('todoInput')

        Button('添加')
          .onClick(() => {
            const input = this.getUIContext().getRouter().getStackByName('todoInput')[0]
            if (input && input.text.trim()) {
              // 正确做法：创建新数组
              this.todos = [...this.todos, {
                id: Date.now(),
                text: input.text.trim(),
                completed: false
              }]
              input.text = ''
            }
          })
      }
      .margin(10)

      // 任务列表
      ForEach(this.todos, (todo: Todo) => {
        Row() {
          Checkbox({ name: 'todo-' + todo.id, group: 'todoGroup' })
            .select(todo.completed)
            .onChange((checked: boolean) => {
              // 正确做法：创建新对象
              this.todos = this.todos.map(item =>
                item.id === todo.id ? { ...item, completed: checked } : item
              )
            })

          Text(todo.text)
            .fontSize(16)
            .decoration(todo.completed ? { type: TextDecorationType.LineThrough } : undefined)
            .layoutWeight(1)

          Button('删除')
            .onClick(() => {
              // 正确做法：使用filter创建新数组
              this.todos = this.todos.filter(item => item.id !== todo.id)
            })
        }
        .width('100%')
        .padding(10)
        .margin(5)
        .backgroundColor('#f0f0f0')
        .borderRadius(5)
      })
    }
    .padding(20)
  }
}

interface Todo {
  id: number
  text: string
  completed: boolean
}
```

#### 2. 状态初始化策略
```typescript
@Entry
@Component
struct StateInitializationExample {
  // 1. 直接初始化
  @State simpleValue: number = 100

  // 2. 函数初始化
  @State calculatedValue: number = this.calculateInitialValue()

  // 3. 异步数据初始化
  @State asyncData: Array<string> = []

  aboutToAppear() {
    // 组件出现前进行异步数据加载
    this.loadAsyncData()
  }

  private calculateInitialValue(): number {
    return Math.floor(Math.random() * 1000)
  }

  private async loadAsyncData() {
    try {
      // 模拟网络请求
      const response = await this.fetchData()
      this.asyncData = response.data
    } catch (error) {
      console.error('加载数据失败:', error)
    }
  }

  private async fetchData(): Promise<{ data: Array<string> }> {
    return new Promise(resolve => {
      setTimeout(() => {
        resolve({
          data: ['数据1', '数据2', '数据3', '数据4', '数据5']
        })
      }, 1000)
    })
  }

  build() {
    Column() {
      Text('简单值：' + this.simpleValue)
        .fontSize(16)
        .margin(10)

      Text('计算值：' + this.calculatedValue)
        .fontSize(16)
        .margin(10)

      Button('重新计算')
        .onClick(() => {
          this.calculatedValue = this.calculateInitialValue()
        })
        .margin(10)

      Divider()

      Text('异步数据：')
        .fontSize(16)
        .fontWeight(FontWeight.Bold)
        .margin(10)

      if (this.asyncData.length > 0) {
        ForEach(this.asyncData, (item: string) => {
          Text(item)
            .fontSize(14)
            .margin(5)
            .padding(10)
            .backgroundColor('#e0e0e0')
            .borderRadius(3)
        })
      } else {
        Text('正在加载数据...')
          .fontSize(14)
          .fontColor(Color.Gray)
          .margin(10)
      }

      Button('重新加载')
        .onClick(() => {
          this.asyncData = []
          this.loadAsyncData()
        })
        .margin(10)
    }
    .padding(20)
  }
}
```

## 8.2 @Prop与@Link属性传递

### 8.2.1 @Prop单向数据传递

@Prop用于父组件向子组件单向传递数据，子组件不能直接修改父组件传递的数据。

```typescript
// 子组件
@Component
struct ChildComponent {
  @Prop title: string = '默认标题'
  @Prop count: number = 0
  @Prop isEnabled: boolean = true

  build() {
    Column() {
      Text('子组件')
        .fontSize(18)
        .fontWeight(FontWeight.Bold)
        .margin(10)

      Text('标题：' + this.title)
        .fontSize(16)
        .margin(5)

      Text('计数：' + this.count)
        .fontSize(16)
        .margin(5)

      Text('状态：' + (this.isEnabled ? '启用' : '禁用'))
        .fontSize(16)
        .fontColor(this.isEnabled ? Color.Green : Color.Red)
        .margin(5)
    }
    .padding(15)
    .backgroundColor('#f5f5f5')
    .borderRadius(8)
    .margin(10)
  }
}

// 父组件
@Entry
@Component
struct PropExample {
  @State parentTitle: string = '父组件标题'
  @State parentCount: number = 10
  @State parentEnabled: boolean = true
```

## 8.3 @Provide与@Consume跨组件通信

### 8.3.1 @Provide与@Consume基础

@Provide和@Consume用于跨层级组件的数据共享，父组件使用@Provide提供数据，任意层级的子组件都可以使用@Consume消费数据。

```typescript
// 根组件
@Entry
@Component
struct ProvideConsumeExample {
  @Provide('appTheme') theme: string = 'light'
  @Provide('appLanguage') language: string = '中文'
  @Provide('userSettings') userSettings: UserSettings = {
    autoSave: true,
    fontSize: 16,
    showNotifications: true
  }

  build() {
    Column() {
      Text('应用设置提供者')
        .fontSize(20)
        .fontWeight(FontWeight.Bold)
        .margin(20)

      // 根控制面板
      this.buildRootControlPanel()

      // 第一层子组件
      FirstLevelChild()
    }
    .padding(20)
  }

  @Builder
  buildRootControlPanel() {
    Column() {
      Text('根组件控制面板')
        .fontSize(16)
        .fontWeight(FontWeight.Bold)
        .margin(10)

      Row() {
        Text('主题：')
          .fontSize(16)

        Select([
          { value: 'light' },
          { value: 'dark' },
          { value: 'auto' }
        ])
          .selected(0)
          .value(this.theme)
          .onSelect((index: number, value: string) => {
            this.theme = value
          })
      }
      .margin(10)

      Row() {
        Text('语言：')
          .fontSize(16)

        Select([
          { value: '中文' },
          { value: 'English' }
        ])
          .selected(0)
          .value(this.language)
          .onSelect((index: number, value: string) => {
            this.language = value
          })
      }
      .margin(10)
    }
    .padding(15)
    .backgroundColor('#f0f0f0')
    .borderRadius(8)
    .margin(10)
  }
}

// 第一层子组件
@Component
struct FirstLevelChild {
  build() {
    Column() {
      Text('第一层子组件')
        .fontSize(18)
        .fontWeight(FontWeight.Bold)

        .fontSize(14)

      SecondLevelChild()
```

```typescript
              Text(notification.message)
                .fontSize(14)
                .margin({ top: 5 })

              Text(this.formatTime(notification.timestamp))
                .fontSize(12)
                .fontColor(Color.Gray)
                .margin({ top: 5 })
            }
            .layoutWeight(1)
            .alignItems(HorizontalAlign.Start)

            // 根据类型显示不同的图标
            Text(this.getNotificationIcon(notification.type))
              .fontSize(20)
              .margin({ left: 10 })
          }
          .width('100%')
          .padding(12)
          .backgroundColor(this.getNotificationColor(notification.type))
          .borderRadius(6)
          .margin(3)
        })
      }
    }
    .padding(10)
  }

  private getNotificationIcon(type: string): string {
    switch (type) {
      case 'success': return '✅'
      case 'error': return '❌'
      case 'warning': return '⚠️'
      default: return 'ℹ️'
    }
  }

  private getNotificationColor(type: string): ResourceColor {
    switch (type) {
      case 'success': return '#d4edda'
      case 'error': return '#f8d7da'
      case 'warning': return '#fff3cd'
      default: return '#d1ecf1'
    }
  }

  private formatTime(timestamp: Date): string {
    const now = new Date()
    const diff = now.getTime() - timestamp.getTime()
    const minutes = Math.floor(diff / 60000)

    if (minutes < 1) return '刚刚'
    if (minutes < 60) return `${minutes}分钟前`

    const hours = Math.floor(minutes / 60)
    if (hours < 24) return `${hours}小时前`

    return timestamp.toLocaleDateString()
  }
}
```

### 8.5.1 Preferences首选项存储

Preferences是轻量级的键值对存储方案，适合存储应用配置和用户偏好。

```typescript
import data_preferences from '@ohos.data.preferences'
import BusinessError from '@ohos.base'

@Entry
@Component
struct PreferencesExample {
  @State preferencesData: Record<string, any> = {}
  @State isLoading: boolean = false
  @State message: string = ''

  private preferences: data_preferences.Preferences | null = null
  private readonly PREFERENCES_NAME = 'my_app_preferences'

  aboutToAppear() {
    this.initPreferences()
  }

  aboutToDisappear() {
    // 关闭首选项实例
    if (this.preferences) {
      this.preferences.flush()
    }
  }

  private async initPreferences() {
    try {
      this.isLoading = true
      this.message = '正在初始化首选项...'

      // 获取首选项实例
      this.preferences = await data_preferences.getPreferences(getContext(this), this.PREFERENCES_NAME)

      // 加载已存储的数据
      await this.loadStoredData()

      this.message = '首选项初始化成功'
    } catch (error) {
      const err = error as BusinessError.BusinessError
      this.message = `初始化失败: ${err.message}`
      console.error('Preferences初始化失败:', err)
    } finally {
      this.isLoading = false
    }
  }

  private async loadStoredData() {
    if (!this.preferences) return

    try {
      // 获取所有已存储的键
      const keys = await this.preferences.getAllKeys()
      const data: Record<string, any> = {}

      // 遍历所有键获取值
      for (const key of keys) {
        const value = await this.preferences.get(key, '')
        data[key] = value
      }

      this.preferencesData = data
    } catch (error) {
      const err = error as BusinessError.BusinessError
      console.error('加载数据失败:', err)
      this.message = `加载数据失败: ${err.message}`
    }
  }

  build() {
    Column() {
      Text('Preferences 首选项存储')
        .fontSize(24)
        .fontWeight(FontWeight.Bold)
        .margin(20)

      // 状态消息
      if (this.message) {
        Text(this.message)
          .fontSize(16)
          .fontColor(this.message.includes('失败') ? Color.Red : Color.Green)
          .margin(10)
      }

      if (this.isLoading) {
        LoadingProgress()
          .width(50)
          .height(50)
          .margin(20)
        return
      }

      // 数据输入区域
      this.buildDataInputArea()

      Divider()

      // 数据显示区域
      this.buildDataDisplayArea()

      Divider()

      // 操作按钮区域
      this.buildActionButtons()
    }
    .padding(20)
  }

  @Builder
  buildDataInputArea() {
    Column() {
      Text('数据存储')
        .fontSize(20)
        .fontWeight(FontWeight.Bold)
        .margin(15)

      // 快速存储按钮
      Row() {
        Button('存储用户信息')
          .onClick(() => this.storeUserInfo())
          .margin(10)

        Button('存储应用设置')
          .onClick(() => this.storeAppSettings())
          .margin(10)

        Button('存储主题配置')
          .onClick(() => this.storeThemeConfig())
          .margin(10)
      }

      // 自定义数据存储
      Column() {
        Text('自定义数据存储')
          .fontSize(16)
          .fontWeight(FontWeight.Bold)
          .margin(10)

        Row() {
          TextInput({ placeholder: '键名' })
            .width(100)
            .margin(5)

    .padding(15)
    .backgroundColor('#f5f5f5')
    .borderRadius(8)
    .margin(10)
  }

  @Builder

      Text('已存储数据')
        .fontSize(20)
        .fontWeight(FontWeight.Bold)
        .margin(15)

      if (Object.keys(this.preferencesData).length === 0) {
        Text('暂无存储数据')
          .fontSize(16)
          .fontColor(Color.Gray)
          .margin(20)
      } else {
        ForEach(Object.entries(this.preferencesData), ([key, value]) => {
          Row() {
            Text(key + ':')
              .fontSize(16)
              .fontWeight(FontWeight.Bold)
              .width(120)

            Text(JSON.stringify(value))
              .fontSize(14)
              .fontColor(Color.Blue)
              .layoutWeight(1)
              .maxLines(2)
              .textOverflow({ overflow: TextOverflow.Ellipsis })

    .backgroundColor('#fafafa')
    .borderRadius(8)
    .margin(10)

  private async storeUserInfo() {

  private async storeAppSettings() {

    try {
      const appSettings = {
        theme: 'dark',
        language: 'zh-CN',
        fontSize: 18,
        autoSave: true,
        notifications: false,
        lastModified: new Date().toISOString()
      }

      await this.preferences.put('appSettings', JSON.stringify(appSettings))
      await this.preferences.flush()

      this.message = '应用设置存储成功'
      await this.loadStoredData()
    } catch (error) {
      const err = error as BusinessError.BusinessError
      this.message = `存储失败: ${err.message}`
    }
  }

  // 存储主题配置
  private async storeThemeConfig() {
    if (!this.preferences) return

    try {
      const themeConfig = {
        primaryColor: '#007DFF',
        backgroundColor: '#FFFFFF',
        textColor: '#000000',
        accentColor: '#FF6B35',
        borderRadius: 8,
        shadow: true
      }

      await this.preferences.put('themeConfig', JSON.stringify(themeConfig))
      await this.preferences.flush()

      this.message = '主题配置存储成功'
      await this.loadStoredData()
    } catch (error) {
      const err = error as BusinessError.BusinessError
      this.message = `存储失败: ${err.message}`
    }
  }

  // 存储自定义数据
  private async storeCustomData() {
    if (!this.preferences) return

    try {
      // 获取输入框的值
      const keyInput = this.getUIContext().getRouter().getStackByName('keyInput')[0]
      const valueInput = this.getUIContext().getRouter().getStackByName('valueInput')[0]

      if (!keyInput?.text || !valueInput?.text) {
        this.message = '请输入键名和值'
        return
      }

      await this.preferences.put(keyInput.text, valueInput.text)
      await this.preferences.flush()

      // 清空输入框
      keyInput.text = ''
      valueInput.text = ''

      this.message = '数据存储成功'
      await this.loadStoredData()
    } catch (error) {
      const err = error as BusinessError.BusinessError
      this.message = `存储失败: ${err.message}`
    }
  }

  // 删除数据
  private async removeData(key: string) {
    if (!this.preferences) return

    try {
      await this.preferences.delete(key)
      await this.preferences.flush()

      this.message = `已删除: ${key}`
      await this.loadStoredData()
    } catch (error) {
      const err = error as BusinessError.BusinessError
      this.message = `删除失败: ${err.message}`
    }
  }

  // 清空所有数据
  private async clearAllData() {
    if (!this.preferences) return

    try {
      await this.preferences.clear()
      await this.preferences.flush()

      this.message = '所有数据已清空'
      await this.loadStoredData()
    } catch (error) {
      const err = error as BusinessError.BusinessError
      this.message = `清空失败: ${err.message}`
    }
  }

  // 导出数据
  private exportData() {
    try {
      const exportData = {
        exportTime: new Date().toISOString(),
        data: this.preferencesData
      }

      console.log('导出数据:', JSON.stringify(exportData, null, 2))
      this.message = '数据已导出到控制台'
    } catch (error) {
      this.message = '导出失败'
    }
  }
}
```

```typescript
    if (this.store) {
      this.store.close()

  private async initDatabase() {
    try {
      this.isLoading = true
      this.message = '正在初始化数据库...'

      // 创建数据库
      this.store = await relationalStore.getRdbStore(getContext(this), this.STORE_CONFIG)

      // 创建表
      await this.createTables()

      // 加载数据
      await this.loadData()

      this.message = '数据库初始化成功'
    } catch (error) {
      const err = error as BusinessError.BusinessError
      this.message = `数据库初始化失败: ${err.message}`
      console.error('数据库初始化失败:', err)
    } finally {
      this.isLoading = false
    }
  }

  private async createTables() {
    if (!this.store) return

    // 创建用户表
    const createUsersTable = `
      CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        email TEXT UNIQUE NOT NULL,
        age INTEGER NOT NULL,
        createTime INTEGER NOT NULL,
        updateTime INTEGER NOT NULL
      )
    `

    // 创建任务表
    const createTasksTable = `
      CREATE TABLE IF NOT EXISTS tasks (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        title TEXT NOT NULL,
        description TEXT,
        userId INTEGER NOT NULL,
        status TEXT NOT NULL DEFAULT 'pending',
        priority INTEGER NOT NULL DEFAULT 1,
        createTime INTEGER NOT NULL,
        updateTime INTEGER NOT NULL,
        FOREIGN KEY (userId) REFERENCES users(id)
      )
    `

    await this.store.executeSql(createUsersTable)
    await this.store.executeSql(createTasksTable)
  }

  private async loadData() {
    if (!this.store) return

    try {
      // 加载用户数据
      const usersResult = await this.store.query('SELECT * FROM users ORDER BY createTime DESC')
      const users: Array<User> = []

      while (usersResult.goToNextRow()) {
        users.push({
          id: usersResult.getLong(0),
          name: usersResult.getString(1),
          email: usersResult.getString(2),
          age: usersResult.getLong(3),
          createTime: usersResult.getLong(4),
          updateTime: usersResult.getLong(5)
        })
      }
      usersResult.close()

      this.users = users

      // 加载任务数据
      const tasksResult = await this.store.query(`
        SELECT t.*, u.name as userName
        FROM tasks t
        LEFT JOIN users u ON t.userId = u.id
        ORDER BY t.createTime DESC
      `)
      const tasks: Array<Task> = []

      while (tasksResult.goToNextRow()) {
        tasks.push({
          id: tasksResult.getLong(0),
          title: tasksResult.getString(1),
          description: tasksResult.getString(2),
          userId: tasksResult.getLong(3),
          status: tasksResult.getString(4) as Task['status'],
          priority: tasksResult.getLong(5),
          createTime: tasksResult.getLong(6),
          updateTime: tasksResult.getLong(7)
        })
      }
      tasksResult.close()

      this.tasks = tasks
    } catch (error) {
      const err = error as BusinessError.BusinessError
      this.message = `加载数据失败: ${err.message}`
    }
  }
```

- **维护性**：选择易于理解和维护的数据管理方案

- **类型匹配**：根据数据特点选择合适的存储方案
- **性能优化**：对大量数据使用分页和索引
- **安全考虑**：敏感数据需要进行加密存储
- **备份恢复**：实现数据备份和恢复机制

### 3. 错误处理和容错
- **异常捕获**：完善异步操作的错误处理

- **用户反馈**：及时向用户反馈操作结果
- **日志记录**：记录关键操作和错误信息

### 4. 性能优化
- **懒加载**：按需加载数据
- **缓存机制**：合理使用缓存减少访问次数
- **批量操作**：合并多个小操作为批量操作
- **内存管理**：及时释放不需要的数据引用
