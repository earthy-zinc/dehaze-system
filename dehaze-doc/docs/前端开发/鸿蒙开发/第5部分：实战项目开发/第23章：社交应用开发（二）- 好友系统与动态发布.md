```typescript
// friend.viewmodel.ets
export class FriendViewModel extends BaseViewModel {
  @State friends: Friendship[] = [];
  @State friendRequests: FriendRequest[] = [];
  @State sentRequests: FriendRequest[] = [];
  @State friendGroups: FriendGroup[] = [];
  @State friendSuggestions: FriendRecommendation[] = [];
  @State blockedUsers: User[] = [];
  @State searchResults: User[] = [];

  private friendService: FriendService;
  private userService: UserService;
  private notificationService: NotificationService;

  constructor(
    friendService: FriendService,
    userService: UserService,
    notificationService: NotificationService
  ) {
    super();
    this.friendService = friendService;
    this.userService = userService;
    this.notificationService = notificationService;
  }

  onInit(): void {
    this.loadFriendData();
  }

  onDestroy(): void {
    // 清理资源
  }

  async loadFriendData(): Promise<void> {
    await Promise.all([
      this.loadFriends(),
      this.loadFriendRequests(),
      this.loadFriendGroups(),
      this.loadFriendSuggestions(),
      this.loadBlockedUsers()
    ]);
  }

  async loadFriends(): Promise<void> {
    await this.executeAsync(
      async () => {
        const friends = await this.friendService.getFriends();
        this.friends = friends;
      },
      () => {
        console.info('Friends loaded successfully');
      }
    );
  }

  async loadFriendRequests(): Promise<void> {
    await this.executeAsync(
      async () => {
        const [receivedRequests, sentRequests] = await Promise.all([
          this.friendService.getReceivedFriendRequests(),
          this.friendService.getSentFriendRequests()
        ]);

        this.friendRequests = receivedRequests;
        this.sentRequests = sentRequests;
      },
      () => {
        console.info('Friend requests loaded successfully');
      }
    );
  }

  async loadFriendGroups(): Promise<void> {
    await this.executeAsync(
      async () => {
        const groups = await this.friendService.getFriendGroups();
        this.friendGroups = groups;
      },
      () => {
        console.info('Friend groups loaded successfully');
      }
    );
  }

  async loadFriendSuggestions(): Promise<void> {
    await this.executeAsync(
      async () => {
        const suggestions = await this.friendService.getFriendSuggestions();
        this.friendSuggestions = suggestions;
      },
      () => {
        console.info('Friend suggestions loaded successfully');
      }
    );
  }

  async loadBlockedUsers(): Promise<void> {
    await this.executeAsync(
      async () => {
        const blockedUsers = await this.friendService.getBlockedUsers();
        this.blockedUsers = blockedUsers;
      },
      () => {
        console.info('Blocked users loaded successfully');
      }
    );
  }

  async sendFriendRequest(
    targetUserId: string,
    message?: string,
    source?: FriendRequestSource
  ): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const friendRequest = await this.friendService.sendFriendRequest({
          targetId: targetUserId,
          message,
          source: source || FriendRequestSource.SEARCH
        });

        this.sentRequests.unshift(friendRequest);
        return true;
      },
      (success) => {
        if (success) {
          console.info('Friend request sent successfully');
          this.notificationService.showNotification({
            title: '好友请求已发送',
            body: '等待对方确认'
          });
        }
      }
    );

    return result || false;
  }

  async acceptFriendRequest(requestId: string): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const friendship = await this.friendService.acceptFriendRequest(requestId);

        // 从请求列表中移除
        this.friendRequests = this.friendRequests.filter(req => req.id !== requestId);

        // 添加到好友列表
        this.friends.unshift(friendship);

        // 发送通知
        const request = this.friendRequests.find(req => req.id === requestId);
        if (request) {
          this.notificationService.sendNotification({
            userId: request.requesterId,
            type: 'friend_request_accepted',
            title: '好友请求已接受',
            body: `${this.getCurrentUserName()}接受了你的好友请求`
          });
        }

        return true;
      },
      (success) => {
        if (success) {
          console.info('Friend request accepted successfully');
          this.notificationService.showNotification({
            title: '好友添加成功',
            body: '你们现在是好友了'
          });
        }
      }
    );

    return result || false;
  }

  async declineFriendRequest(requestId: string, reason?: string): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const success = await this.friendService.declineFriendRequest(requestId, reason);

        if (success) {
          this.friendRequests = this.friendRequests.filter(req => req.id !== requestId);
        }

        return success;
      },
      (success) => {
        if (success) {
          console.info('Friend request declined successfully');
        }
      }
    );

    return result || false;
  }

  async cancelFriendRequest(requestId: string): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const success = await this.friendService.cancelFriendRequest(requestId);

        if (success) {
          this.sentRequests = this.sentRequests.filter(req => req.id !== requestId);
        }

        return success;
      },
      (success) => {
        if (success) {
          console.info('Friend request cancelled successfully');
        }
      }
    );

    return result || false;
  }

  async removeFriend(friendId: string): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const success = await this.friendService.removeFriend(friendId);

        if (success) {
          this.friends = this.friends.filter(friend =>
            friend.user1Id === friendId || friend.user2Id === friendId
          );
        }

        return success;
      },
      (success) => {
        if (success) {
          console.info('Friend removed successfully');
          this.notificationService.showNotification({
            title: '好友已删除',
            body: '你们不再是好友关系'
          });
        }
      }
    );

    return result || false;
  }

  async blockUser(userId: string): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const user = await this.userService.getUserById(userId);
        if (!user) return false;

        const success = await this.friendService.blockUser(userId);

        if (success) {
          this.blockedUsers.push(user);

          // 如果是好友，同时从好友列表移除
          this.friends = this.friends.filter(friend =>
            friend.user1Id !== userId && friend.user2Id !== userId
          );
        }

        return success;
      },
      (success) => {
        if (success) {
          console.info('User blocked successfully');
          this.notificationService.showNotification({
            title: '用户已拉黑',
            body: '已将该用户加入黑名单'
          });
        }
      }
    );

    return result || false;
  }

  async unblockUser(userId: string): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const success = await this.friendService.unblockUser(userId);

        if (success) {
          this.blockedUsers = this.blockedUsers.filter(user => user.id !== userId);
        }

        return success;
      },
      (success) => {
        if (success) {
          console.info('User unblocked successfully');
        }
      }
    );

    return result || false;
  }

  async createFriendGroup(name: string, description?: string): Promise<FriendGroup | null> {
    const result = await this.executeAsync(
      async () => {
        const group = await this.friendService.createFriendGroup({
          name,
          description,
          memberIds: []
        });

        this.friendGroups.push(group);
        return group;
      },
      (group) => {
        if (group) {
          console.info('Friend group created successfully');
        }
      }
    );

    return result;
  }

  async updateFriendGroup(groupId: string, updates: Partial<FriendGroup>): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const updatedGroup = await this.friendService.updateFriendGroup(groupId, updates);

        if (updatedGroup) {
          const index = this.friendGroups.findIndex(g => g.id === groupId);
          if (index !== -1) {
            this.friendGroups[index] = updatedGroup;
          }
          return true;
        }

        return false;
      },
      (success) => {
        if (success) {
          console.info('Friend group updated successfully');
        }
      }
    );

    return result || false;
  }

  async deleteFriendGroup(groupId: string): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const success = await this.friendService.deleteFriendGroup(groupId);

        if (success) {
          this.friendGroups = this.friendGroups.filter(g => g.id !== groupId);
        }

        return success;
      },
      (success) => {
        if (success) {
          console.info('Friend group deleted successfully');
        }
      }
    );

    return result || false;
  }

  async addFriendsToGroup(groupId: string, userIds: string[]): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const updatedGroup = await this.friendService.addFriendsToGroup(groupId, userIds);

        if (updatedGroup) {
          const index = this.friendGroups.findIndex(g => g.id === groupId);
          if (index !== -1) {
            this.friendGroups[index] = updatedGroup;
          }
          return true;
        }

        return false;
      },
      (success) => {
        if (success) {
          console.info('Friends added to group successfully');
        }
      }
    );

    return result || false;
  }

  async removeFriendsFromGroup(groupId: string, userIds: string[]): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const updatedGroup = await this.friendService.removeFriendsFromGroup(groupId, userIds);

        if (updatedGroup) {
          const index = this.friendGroups.findIndex(g => g.id === groupId);
          if (index !== -1) {
            this.friendGroups[index] = updatedGroup;
          }
          return true;
        }

        return false;
      },
      (success) => {
        if (success) {
          console.info('Friends removed from group successfully');
        }
      }
    );

    return result || false;
  }

  async searchUsers(query: string, filters?: UserSearchFilters): Promise<void> {
    await this.executeAsync(
      async () => {
        const users = await this.userService.searchUsers(query, filters);
        this.searchResults = users;
      },
      () => {
        console.info('User search completed successfully');
      }
    );
  }

  async findNearbyUsers(radius: number = 10): Promise<User[]> {
    const result = await this.executeAsync(
      async () => {
        return await this.friendService.findNearbyUsers(radius);
      },
      (users) => {
        console.info('Nearby users found successfully');
      }
    );

    return result || [];
  }

  async syncContacts(contacts: ContactInfo[]): Promise<User[]> {
    const result = await this.executeAsync(
      async () => {
        return await this.friendService.syncContacts(contacts);
      },
      (users) => {
        console.info('Contacts synced successfully');
        if (users.length > 0) {
          this.notificationService.showNotification({
            title: '联系人同步完成',
            body: `发现${users.length}位好友在使用应用`
          });
        }
      }
    );

    return result || [];
  }

  async updateFriendSettings(friendId: string, settings: CustomFriendSettings): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        const success = await this.friendService.updateFriendSettings(friendId, settings);
        return success;
      },
      (success) => {
        if (success) {
          console.info('Friend settings updated successfully');
        }
      }
    );

    return result || false;
  }

  async updateFriendRecommendationFeedback(recommendationId: string, status: RecommendationStatus): Promise<void> {
    await this.executeAsync(
      async () => {
        await this.friendService.updateRecommendationFeedback(recommendationId, status);

        // 更新本地状态
        const index = this.friendSuggestions.findIndex(s => s.recommendedUserId === recommendationId);
        if (index !== -1) {
          this.friendSuggestions[index].status = status;
        }
      },
      () => {
        console.info('Recommendation feedback updated successfully');
      }
    );
  }

  // 辅助方法
  private getCurrentUserName(): string {
    // 获取当前用户名
    return '当前用户';
  }

  // Getters
  get friendsCount(): number {
    return this.friends.length;
  }

  get pendingRequestsCount(): number {
    return this.friendRequests.length;
  }

  get blockedUsersCount(): number {
    return this.blockedUsers.length;
  }

  getFriendById(userId: string): Friendship | null {
    return this.friends.find(friend =>
      friend.user1Id === userId || friend.user2Id === userId
    ) || null;
  }

  getFriendsInGroup(groupId: string): User[] {
    const group = this.friendGroups.find(g => g.id === groupId);
    if (!group) return [];

    return this.friends
      .filter(friend => group.memberIds.includes(
        friend.user1Id === this.getCurrentUserId() ? friend.user2Id : friend.user1Id
      ))
      .map(friend => {
        // 需要获取用户详细信息
        return {} as User;
      });
  }

  private getCurrentUserId(): string {
    // 获取当前用户ID
    return 'current-user-id';
  }
}

// 用户搜索过滤器

  hasAvatar?: boolean;
  verified?: boolean;
  interests?: string[];
}

  phone?: string;
  email?: string;
}
```

```typescript
// friends.list.view.ets
@Entry
@Component
struct FriendsListView {
  @ObjectLink friendViewModel: FriendViewModel;
  @State selectedTab: FriendTab = FriendTab.ALL_FRIENDS;
  @State searchQuery: string = '';
  @State showSearch: boolean = false;

  aboutToAppear(): void {
    this.friendViewModel.onInit();
  }

  aboutToDisappear(): void {
    this.friendViewModel.onDestroy();
  }

  build() {
    Column() {
      // 头部
      this.header()

      // 搜索栏
      if (this.showSearch) {
        this.searchBar()
      }

      // 标签页
      this.tabBar()

      // 内容区域
      this.contentArea()
    }
    .width('100%')
    .height('100%')
    .backgroundColor('#F5F5F5')
  }

  @Builder
  header() {

      Text('好友')
        .fontSize(18)
        .fontWeight(FontWeight.Bold)
        .fontColor('#333333')

      Blank()

      Row() {
        Button() {
          Image($r('app.media.search'))
            .width(20)
            .height(20)
            .fillColor('#333333')
        }
        .width(32)
        .height(32)
        .backgroundColor(Color.Transparent)
        .onClick(() => {
          this.showSearch = !this.showSearch;
        })

        Button() {
          Image($r('app.media.add_friend'))
            .width(20)
            .height(20)
            .fillColor('#333333')
        }
        .width(32)
        .height(32)
        .backgroundColor(Color.Transparent)
        .onClick(() => {
          this.navigateToFindFriends();
        })

        Button() {
          Image($r('app.media.more'))
            .width(20)
            .height(20)
            .fillColor('#333333')
        }
        .width(32)
        .height(32)
        .backgroundColor(Color.Transparent)
        .onClick(() => {
          this.showMoreOptions();
        })
      }
    }
    .width('100%')
    .padding({ left: 16, right: 16, top: 8, bottom: 8 })
    .backgroundColor(Color.White)
  }

  @Builder
  searchBar() {
    Row() {
      TextInput({ placeholder: '搜索好友...' })
        .width('100%')
        .height(36)
        .backgroundColor('#F8F9FA')
        .borderRadius(18)
        .margin({ right: 12 })
        .onChange((value: string) => {
          this.searchQuery = value;
          this.performSearch();
        })

      if (this.searchQuery) {
        Button() {
          Image($r('app.media.close'))
            .width(16)
            .height(16)
            .fillColor('#999999')
        }
        .width(32)
        .height(32)
        .backgroundColor(Color.Transparent)
        .onClick(() => {
          this.searchQuery = '';
          this.showSearch = false;
        })
      }
    }
    .width('100%')
    .padding({ left: 16, right: 16, top: 8, bottom: 8 })
    .backgroundColor(Color.White)
  }

  @Builder
  tabBar() {
    Row() {
      ForEach([
        { tab: FriendTab.ALL_FRIENDS, label: '全部好友', count: this.friendViewModel.friendsCount },
        { tab: FriendTab.GROUPS, label: '分组', count: this.friendViewModel.friendGroups.length },
        { tab: FriendTab.REQUESTS, label: '好友请求', count: this.friendViewModel.pendingRequestsCount },
        { tab: FriendTab.SUGGESTIONS, label: '推荐', count: this.friendViewModel.friendSuggestions.length }
      ], (item) => {
        Column() {
          Text(item.label)
            .fontSize(14)
            .fontColor(this.selectedTab === item.tab ? '#007DFF' : '#666666')

          if (item.count > 0) {
            Badge({
              value: item.count > 99 ? '99+' : item.count.toString(),
              position: BadgePosition.Right,
              style: { badgeSize: 16, badgeColor: '#FF4757' }
            })
          }
        }
        .layoutWeight(1)
        .justifyContent(FlexAlign.Center)
        .padding({ top: 12, bottom: 12 })
        .backgroundColor(this.selectedTab === item.tab ? Color.White : Color.Transparent)
        .border({
          width: { bottom: this.selectedTab === item.tab ? 2 : 0 },
          color: '#007DFF'
        })
        .onClick(() => {
          this.selectedTab = item.tab;
        })
      })
    }
    .width('100%')
    .backgroundColor('#F8F9FA')
  }

  @Builder
  contentArea() {
    if (this.selectedTab === FriendTab.ALL_FRIENDS) {
      this.allFriendsView()
    } else if (this.selectedTab === FriendTab.GROUPS) {
      this.groupsView()
    } else if (this.selectedTab === FriendTab.REQUESTS) {
      this.requestsView()
    } else if (this.selectedTab === FriendTab.SUGGESTIONS) {
      this.suggestionsView()
    }
  }

  @Builder
  allFriendsView() {
    if (this.searchQuery) {
      this.searchResultsView()
    } else {
      Column() {
        // 好友分组列表
        ForEach(this.getFriendsByGroups(), (group) => {
          Column() {
            // 分组标题
            Row() {
              Text(group.name)
                .fontSize(16)
                .fontWeight(FontWeight.Bold)
                .fontColor('#333333')

              Blank()

              Text(`${group.friends.length}人`)
                .fontSize(12)
                .fontColor('#999999')
            }
            .width('100%')
            .padding(16)
            .backgroundColor('#F8F9FA')

            // 好友列表
            ForEach(group.friends, (friendship: Friendship) => {
              FriendItemComponent({
                friendship: friendship,
                currentUserId: this.getCurrentUserId(),
                onClick: () => {
                  this.navigateToUserProfile(friendship);
                },
                onChat: () => {
                  this.navigateToChat(friendship);
                },
                onMore: () => {
                  this.showFriendOptions(friendship);
                }
              })
            })
          }
          .margin({ bottom: 8 })
        })
      }
      .layoutWeight(1)
    }
  }

  @Builder
  groupsView() {
    Column() {
      // 创建分组按钮
      Button('+ 创建分组')
        .fontSize(14)
        .fontColor('#007DFF')
        .backgroundColor(Color.Transparent)
        .border({ width: 1, color: '#007DFF' })
        .width('calc(100% - 32px)')
        .height(40)
        .margin({ top: 16, left: 16, right: 16 })
        .onClick(() => {
          this.showCreateGroupDialog();
        })

      // 分组列表
      ForEach(this.friendViewModel.friendGroups, (group: FriendGroup) => {
        FriendGroupComponent({
          group: group,
          friendViewModel: this.friendViewModel,
          onClick: () => {
            this.navigateToGroup(group);
          },
          onEdit: () => {
            this.showEditGroupDialog(group);
          },
          onDelete: () => {
            this.showDeleteGroupConfirm(group);
          }
        })
      })
    }
    .layoutWeight(1)
    .padding({ bottom: 16 })
  }

  @Builder
  requestsView() {
    Column() {
      // 收到的请求
      if (this.friendViewModel.friendRequests.length > 0) {
        Column() {
          Text('好友请求')
            .fontSize(16)
            .fontWeight(FontWeight.Bold)

            FriendRequestComponent({
              request: request,
              onAccept: () => {
                this.friendViewModel.acceptFriendRequest(request.id);
              },
              onDecline: () => {
                this.showDeclineReasonDialog(request);

        .width('100%')

        .width('100%')

            .margin({ bottom: 16 })

            .fontSize(16)
            .fontColor('#666666')

          Text('去发现更多好友吧')
            .fontSize(14)
            .fontColor('#999999')

          Button('发现好友')

            .fontColor('#007DFF')

            .height(36)
            .margin({ top: 16 })
            .onClick(() => {
              this.navigateToFindFriends();
            })
        }
        .width('100%')
        .height('100%')
        .justifyContent(FlexAlign.Center)

  }

  @Builder
  suggestionsView() {
    Column() {
      if (this.friendViewModel.friendSuggestions.length > 0) {
        ForEach(this.friendViewModel.friendSuggestions, (suggestion: FriendRecommendation) => {
          FriendSuggestionComponent({
            suggestion: suggestion,
            onAccept: () => {
              this.friendViewModel.sendFriendRequest(suggestion.recommendedUserId);
              this.friendViewModel.updateFriendRecommendationFeedback(suggestion.recommendedUserId, RecommendationStatus.ACCEPTED);
            },
            onReject: () => {
              this.friendViewModel.updateFriendRecommendationFeedback(suggestion.recommendedUserId, RecommendationStatus.DECLINED);
            },
            onHide: () => {
              this.friendViewModel.updateFriendRecommendationFeedback(suggestion.recommendedUserId, RecommendationStatus.HIDDEN);
            }
          })
        })
      } else {
        Column() {
          Image($r('app.media.no_suggestions'))
            .width(80)
            .height(80)
            .margin({ bottom: 16 })

          Text('暂无推荐好友')
            .fontSize(16)
            .fontColor('#666666')
            .margin({ bottom: 8 })

          Text('系统会根据你的兴趣推荐好友')
            .fontSize(14)
            .fontColor('#999999')
        }
        .width('100%')
        .height('100%')
        .justifyContent(FlexAlign.Center)
      }
    }
    .layoutWeight(1)
    .padding(16)
  }

  @Builder
  searchResultsView() {
    Column() {
      Text(`搜索结果 (${this.friendViewModel.searchResults.length})`)
        .fontSize(16)
        .fontWeight(FontWeight.Bold)
```

```typescript
export class MomentsViewModel extends BaseViewModel {
  @State posts: Post[] = [];
  @State currentPost: Post | null = null;
  @State userPosts: Post[] = [];
  @State trendingPosts: Post[] = [];
  @State categories: PostCategory[] = [];
  @State selectedCategory: string | null = null;
  @State isLoadingMore: boolean = false;
  @State hasMorePosts: boolean = true;

  private momentsService: MomentsService;
  private userService: UserService;
  private mediaService: MediaService;

  constructor(
    momentsService: MomentsService,
    userService: UserService,
    mediaService: MediaService
  ) {
    super();
    this.momentsService = momentsService;
    this.userService = userService;
    this.mediaService = mediaService;
  }

  onInit(): void {
    this.loadMomentsData();
  }

  onDestroy(): void {
    // 清理资源
  }

  async loadMomentsData(): Promise<void> {
    await Promise.all([
      this.loadPosts(),
      this.loadTrendingPosts(),
      this.loadCategories()
    ]);
  }

  async loadPosts(refresh: boolean = false): Promise<void> {
    if (refresh) {
      this.posts = [];
      this.hasMorePosts = true;
    }

    if (!this.hasMorePosts || this.isLoadingMore) return;

    this.isLoadingMore = true;

    await this.executeAsync(
      async () => {
        const lastPostId = refresh ? undefined : this.posts[this.posts.length - 1]?.id;
        const newPosts = await this.momentsService.getPosts({
          categoryId: this.selectedCategory,
          before: lastPostId,
          limit: 10
        });

        if (refresh) {
          this.posts = newPosts;
        } else {
          this.posts = [...this.posts, ...newPosts];
        }

        this.hasMorePosts = newPosts.length >= 10;
        this.isLoadingMore = false;
      },
      () => {
        console.info('Posts loaded successfully');
      },
      (error) => {
        this.isLoadingMore = false;
      }
    );
  }

  async loadTrendingPosts(): Promise<void> {
    await this.executeAsync(
      async () => {
        const trendingPosts = await this.momentsService.getTrendingPosts();
        this.trendingPosts = trendingPosts;
      },
      () => {
        console.info('Trending posts loaded successfully');
      }
    );
  }

  async loadCategories(): Promise<void> {
    await this.executeAsync(
      async () => {
        const categories = await this.momentsService.getCategories();
        this.categories = categories;
      },
      () => {
        console.info('Categories loaded successfully');
      }
    );
  }

  async createPost(postData: CreatePostData): Promise<Post | null> {
    const result = await this.executeAsync(
      async () => {
        // 处理媒体文件
        const processedContent = await this.processPostContent(postData.content);

        const post = await this.momentsService.createPost({
          type: postData.type,
          content: processedContent,
          visibility: postData.visibility,
          location: postData.location,
          mentions: postData.mentions,
          tags: postData.tags
        });

        this.posts.unshift(post);
        return post;
      },
      (post) => {
        console.info('Post created successfully');
        this.showSuccessMessage('动态发布成功');
      }
    );

    return result;
  }

  async updatePost(postId: string, updates: Partial<Post>): Promise<boolean> {
    const result = await this.executeAsync(
      async () => {
        if (updates.content) {
          updates.content = await this.processPostContent(updates.content);
        }

        const updatedPost = await this.momentsService.updatePost(postId, updates);

        if (updatedPost) {
          const index = this.posts.findIndex(p => p.id === postId);
          if (index !== -1) {
            this.posts[index] = updatedPost;
          }

          if (this.currentPost?.id === postId) {
            this.currentPost = updatedPost;
          }

          return true;
        }

        return false;
      },
      (success) => {
        if (success) {
          console.info('Post updated successfully');
          this.showSuccessMessage('动态更新成功');
        }
      }
    );

    return result || false;
  }

        const success = await this.momentsService.deletePost(postId);

            this.currentPost = null;

    this.loadPosts(true);

  refreshPosts(): void {
    this.loadPosts(true);
  }

  loadMorePosts(): void {
    if (!this.isLoadingMore && this.hasMorePosts) {
      this.loadPosts();
    }
  }

  // 媒体处理方法
  private async processPostContent(content: PostContent): Promise<PostContent> {
    const processedContent = { ...content };

    if (content.media && content.media.length > 0) {
      const processedMedia: MediaContent[] = [];

      for (const media of content.media) {
        if (media.url.startsWith('blob:')) {
          // 处理本地文件
          const processedMediaItem = await this.uploadMediaFile(media);
          if (processedMediaItem) {
            processedMedia.push(processedMediaItem);
          }
        } else {
          // 已经是远程URL
          processedMedia.push(media);
        }
      }

      processedContent.media = processedMedia;
    }

    return processedContent;
  }

  private async uploadMediaFile(media: MediaContent): Promise<MediaContent | null> {
    try {
      // 根据文件类型进行不同的处理
      if (media.url.startsWith('blob:')) {
        const file = await this.blobToFile(media);
        const uploadedMedia = await this.mediaService.uploadFile(file);

        return {
          ...media,
          url: uploadedMedia.url,
          thumbnail: uploadedMedia.thumbnail,
          fileSize: uploadedMedia.size
        };
      }

      return media;
    } catch (error) {
      console.error('Failed to upload media file:', error);
      return null;
    }
  }

  private async blobToFile(media: MediaContent): Promise<File> {
    // 将blob转换为File对象
    const response = await fetch(media.url);
    const blob = await response.blob();

    return new File([blob], media.fileName || 'media', {
      type: media.mimeType || 'application/octet-stream'
    });
  }

  // 统计更新方法
  private updatePostStatistics(postId: string, field: keyof PostStatistics, delta: number): void {
    const postIndex = this.posts.findIndex(p => p.id === postId);
    if (postIndex !== -1) {
      const post = this.posts[postIndex];
      const currentStatistics = post.statistics;

      this.posts[postIndex] = {
        ...post,
        statistics: {
          ...currentStatistics,
          [field]: Math.max(0, (currentStatistics[field] as number) + delta)
        }
      };
    }

    if (this.currentPost?.id === postId) {
      const currentStatistics = this.currentPost.statistics;
      this.currentPost = {
        ...this.currentPost,
        statistics: {
          ...currentStatistics,
          [field]: Math.max(0, (currentStatistics[field] as number) + delta)
        }
      };
    }
  }

  private showSuccessMessage(message: string): void {
    // 显示成功消息
    console.info(`Success: ${message}`);
  }

  // Getters
  get postsCount(): number {
    return this.posts.length;
  }

  get hasPosts(): boolean {
    return this.posts.length > 0;
  }

  get selectedCategoryName(): string {
    if (!this.selectedCategory) return '全部';
    const category = this.categories.find(c => c.id === this.selectedCategory);
    return category?.name || '全部';
  }
```

```typescript
  private async publishPost(): Promise<void> {
    if (!this.canPublish()) return;

    const content: PostContent = {};

    if (this.postContent) {
      content.text = this.postContent;
    }

    if (this.selectedImages.length > 0) {
      content.media = this.selectedImages.map(img => ({
        url: img.url,
        fileName: img.name,
        fileSize: img.size,
        width: img.width,
        height: img.height,
        format: img.format
      }));
    }

    if (this.selectedVideo) {
      content.media = [{
        url: this.selectedVideo.url,
        fileName: this.selectedVideo.name,
        fileSize: this.selectedVideo.size,
        duration: this.selectedVideo.duration,
        format: this.selectedVideo.format
      }];
    }

    if (this.location) {
      content.location = this.location;
    }

    const postData: CreatePostData = {
      type: this.postType,
      content,
      visibility: this.visibility,
      location: this.location,
      mentions: this.mentions,
      tags: this.tags
    };

    await this.momentsViewModel.createPost(postData);

    // 发布成功后返回
    router.back();
  }

    } else if (this.postType === PostType.VIDEO) {
      return this.selectedVideo !== null;
    } else if (this.postType === PostType.LOCATION) {
      return this.location !== null;

    return false;
```
